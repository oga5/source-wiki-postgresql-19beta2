size_t StartOverlap = (PageSize - ((uintptr_t)M.base() % PageSize)) % PageSize;

size_t TrimmedSize = M.allocatedSize(); TrimmedSize -= StartOverlap; TrimmedSize -= TrimmedSize % PageSize;

sys::MemoryBlock Trimmed((void *)((uintptr_t)M.base() + StartOverlap), TrimmedSize);

assert(((uintptr_t)Trimmed.base() % PageSize) == 0); assert((Trimmed.allocatedSize() % PageSize) == 0); assert(M.base() <= Trimmed.base() && Trimmed.allocatedSize() <= M.allocatedSize());

return Trimmed; } ~~~
