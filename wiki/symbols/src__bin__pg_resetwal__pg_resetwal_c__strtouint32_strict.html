~~~c static uint32 strtouint32_strict(const char *restrict s, char **restrict endptr, int base) { unsigned long val; bool is_neg;

/* skip leading whitespace */ while (isspace((unsigned char) *s)) s++;

/*

  • Is it negative? We still call strtoul() if it was, to set 'endptr'.
  • (The current callers don't care though.)

    • */ is_neg = (*s == '-');

val = strtoul(s, endptr, base);

/* reject if it was negative */ if (errno == 0 && is_neg) { errno = ERANGE; val = 0; }

/*

  • reject values larger than UINT32_MAX on platforms where long is 64 bits
  • wide.

    • */ if (errno == 0 && val != (uint32) val) { errno = ERANGE; val = UINT32_MAX; }

return (uint32) val; } ~~~
