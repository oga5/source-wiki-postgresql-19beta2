~~~c static void _selectTableAccessMethod(ArchiveHandle *AH, const char *tableam) { RestoreOptions *ropt = AH->public.ropt; PQExpBuffer cmd; const char *want, *have;

/* do nothing in --no-table-access-method mode */ if (ropt->noTableAm) return;

have = AH->currTableAm; want = tableam;

if (!want) return;

if (have && strcmp(want, have) == 0) return;

cmd = createPQExpBuffer(); appendPQExpBuffer(cmd, "SET default_table_access_method = %s;", fmtId(want));

if (RestoringToDB(AH)) { PGresult *res;

res = PQexec(AH->connection, cmd->data);

if (!res || PQresultStatus(res) != PGRES_COMMAND_OK) warn_or_exit_horribly(AH, "could not set \"default_table_access_method\": %s", PQerrorMessage(AH->connection));

PQclear(res); } else ahprintf(AH, "%s\n\n", cmd->data);

destroyPQExpBuffer(cmd);

free(AH->currTableAm); AH->currTableAm = pg_strdup(want); } ~~~
