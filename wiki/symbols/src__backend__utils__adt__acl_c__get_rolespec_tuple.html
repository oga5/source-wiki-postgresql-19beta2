switch (role->roletype) { case ROLESPEC_CSTRING: Assert(role->rolename); tuple = SearchSysCache1(AUTHNAME, CStringGetDatum(role->rolename)); if (!HeapTupleIsValid(tuple)) ereport(ERROR, (errcode(ERRCODE_UNDEFINED_OBJECT), errmsg("role \"%s\" does not exist", role->rolename))); break;

return tuple; } ~~~
