/* There is no point to wait for 2PCs if you have no 2PCs. */ if (max_prepared_xacts == 0) return true;

do { LockAcquireResult lar; LOCKTAG tag;

/* Clear state from previous iterations. */ if (more) { xid = InvalidTransactionId; more = false; }

/* If we have no xid, try to find one. */ if (!TransactionIdIsValid(xid)) xid = TwoPhaseGetXidByVirtualXID(vxid, &more); if (!TransactionIdIsValid(xid)) { Assert(!more); return true; }

/* Check or wait for XID completion. */ SET_LOCKTAG_TRANSACTION(tag, xid); lar = LockAcquire(&tag, ShareLock, false, !wait); if (lar == LOCKACQUIRE_NOT_AVAIL) return false; LockRelease(&tag, ShareLock, false); } while (more);

return true; } ~~~
