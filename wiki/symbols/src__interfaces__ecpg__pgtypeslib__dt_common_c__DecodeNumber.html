/* Special case day of year? */ if (flen == 3 && (fmask & DTK_M(YEAR)) && val >= 1 && val <= 366) { *tmask = (DTK_M(DOY) | DTK_M(MONTH) | DTK_M(DAY)); tm->tm_yday = val; j2date(date2j(tm->tm_year, 1, 1) + tm->tm_yday - 1, &tm->tm_year, &tm->tm_mon, &tm->tm_mday); }

  • year farther down. - thomas 2000-03-28
