/* If there are no PHVs anywhere, we needn't work hard */ if (root->glob->lastPHId == 0) return false;

context.relids = bms_make_singleton(varno); context.sublevels_up = 0;

/*

  • See if the jointree fragment itself contains references (in join quals)

    • */ if (find_dependent_phvs_walker(node, &context)) return true;

/*

  • Otherwise, identify the set of referenced RTEs (we can ignore joins,
  • since they should be flattened already, so their join alias lists no
  • longer matter), and tediously check each RTE. We can ignore RTEs that
  • are not marked LATERAL, though, since they couldn't possibly contain
  • any cross-references to other RTEs.

    • */ subrelids = get_relids_in_jointree(node, false, false); relid = -1; while ((relid = bms_next_member(subrelids, relid)) >= 0) { RangeTblEntry *rte = rt_fetch(relid, root->parse->rtable);

if (rte->lateral && range_table_entry_walker(rte, find_dependent_phvs_walker, &context, 0)) return true; }

return false; } ~~~
