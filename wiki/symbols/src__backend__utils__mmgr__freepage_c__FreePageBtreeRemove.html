~~~c static void FreePageBtreeRemove(FreePageManager *fpm, FreePageBtree *btp, Size index) { Assert(btp->hdr.magic == FREE_PAGE_LEAF_MAGIC); Assert(index < btp->hdr.nused);

/* When last item is removed, extirpate entire page from btree. */ if (btp->hdr.nused == 1) { FreePageBtreeRemovePage(fpm, btp); return; }

/* Physically remove the key from the page. */ --btp->hdr.nused; if (index < btp->hdr.nused) memmove(&btp->u.leaf_key[index], &btp->u.leaf_key[index + 1], sizeof(FreePageBtreeLeafKey) * (btp->hdr.nused - index));

/* If we just removed the first key, adjust ancestor keys. */ if (index == 0) FreePageBtreeAdjustAncestorKeys(fpm, btp);

/* Consider whether to consolidate this page with a sibling. */ FreePageBtreeConsolidate(fpm, btp); } ~~~
