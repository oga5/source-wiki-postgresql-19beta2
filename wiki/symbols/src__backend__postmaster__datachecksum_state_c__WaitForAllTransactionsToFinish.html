/* Retry every 3 seconds */ ResetLatch(MyLatch); rc = WaitLatch(MyLatch, WL_LATCH_SET | WL_TIMEOUT | WL_POSTMASTER_DEATH, 3000, WAIT_EVENT_CHECKSUM_ENABLE_STARTCONDITION);

/*

  • If the postmaster died, bail out. But first print a log message to
  • note that the checksumming didn't complete.
