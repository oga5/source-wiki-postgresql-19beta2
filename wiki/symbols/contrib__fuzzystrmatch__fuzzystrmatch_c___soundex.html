/* If no string left, return all-zeroes buffer */ if (!*instr) { memset(outstr, '\0', SOUNDEX_LEN + 1); return; }

/* Take the first letter as is */ *outstr++ = (char) pg_ascii_toupper((unsigned char) *instr++);

/* Fill with 0's */ while (count < SOUNDEX_LEN) { *outstr = '0'; ++outstr; ++count; }
