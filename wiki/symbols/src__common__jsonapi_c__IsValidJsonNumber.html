~~~c bool IsValidJsonNumber(const char *str, size_t len) { bool numeric_error; size_t total_len; JsonLexContext dummy_lex = {0};

if (len <= 0) return false;

  • json_lex_number expects a leading '-' to have been eaten already.

    • *

  • having to cast away the constness of str is ugly, but there's not much
  • easy alternative.

    • */ if (*str == '-') { dummy_lex.input = str + 1; dummy_lex.input_length = len - 1; } else { dummy_lex.input = str; dummy_lex.input_length = len; }

dummy_lex.token_start = dummy_lex.input;

json_lex_number(&dummy_lex, dummy_lex.input, &numeric_error, &total_len);

return (!numeric_error) && (total_len == dummy_lex.input_length); } ~~~
