~~~c static Buffer create_toy_buffer(Relation rel, BlockNumber blkno) { Buffer buf; BufferDesc *buf_hdr; uint64 buf_state; bool was_pinned = false; uint64 unset_bits = 0;

/* place buffer in shared buffers without erroring out */ buf = ReadBufferExtended(rel, MAIN_FORKNUM, blkno, RBM_ZERO_AND_LOCK, NULL); LockBuffer(buf, BUFFER_LOCK_UNLOCK);

if (RelationUsesLocalBuffers(rel)) { buf_hdr = GetLocalBufferDescriptor(-buf - 1); buf_state = pg_atomic_read_u64(&buf_hdr->state); } else { buf_hdr = GetBufferDescriptor(buf - 1); buf_state = LockBufHdr(buf_hdr); }

/*

  • We should be the only backend accessing this buffer. This is just a
  • small bit of belt-and-suspenders defense, none of this code should ever
  • run in a cluster with real data.

    • */ if (BUF_STATE_GET_REFCOUNT(buf_state) > 1) was_pinned = true; else unset_bits |= BM_VALID | BM_DIRTY;

if (RelationUsesLocalBuffers(rel)) { buf_state &= ~unset_bits; pg_atomic_unlocked_write_u64(&buf_hdr->state, buf_state); } else { UnlockBufHdrExt(buf_hdr, buf_state, 0, unset_bits, 0); }

if (was_pinned) elog(ERROR, "toy buffer %d was already pinned", buf);

return buf; } ~~~
