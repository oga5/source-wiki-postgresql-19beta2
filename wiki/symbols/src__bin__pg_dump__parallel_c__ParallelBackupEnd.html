ソース抜粋

~~~c void ParallelBackupEnd(ArchiveHandle *AH, ParallelState *pstate) { int i;

/* No work if non-parallel */ if (pstate->numWorkers == 1) return;

/* There should not be any unfinished jobs */ Assert(IsEveryWorkerIdle(pstate));

/* Close the sockets so that the workers know they can exit */ for (i = 0; i < pstate->numWorkers; i++) { closesocket(pstate->parallelSlot[i].pipeRead); closesocket(pstate->parallelSlot[i].pipeWrite); }

/* Wait for them to exit */ WaitForTerminatingWorkers(pstate);

/*

  • Unlink pstate from shutdown_info, so the exit handler will not try to
  • use it; and likewise unlink from signal_info.

    • */ shutdown_info.pstate = NULL; set_cancel_pstate(NULL);

/* Release state (mere neatnik-ism, since we're about to terminate) */ free(pstate->te); free(pstate->parallelSlot); free(pstate); } ~~~
