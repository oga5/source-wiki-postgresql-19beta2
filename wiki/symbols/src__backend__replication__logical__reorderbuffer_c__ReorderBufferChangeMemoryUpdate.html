  • compute the decoding stats.

    • */ toptxn = rbtxn_get_toptxn(txn);

if (addition) { Size oldsize = txn->size;

txn->size += sz; rb->size += sz;

/* Update the total size in the top transaction. */ toptxn->total_size += sz;

/* Update the total size in the top transaction. */ toptxn->total_size -= sz;
