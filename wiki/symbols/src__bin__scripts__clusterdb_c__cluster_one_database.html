~~~c static void cluster_one_database(const ConnParams *cparams, const char *table, const char *progname, bool verbose, bool echo) { PQExpBufferData sql;

PGconn *conn;

conn = connectDatabase(cparams, progname, echo, false, true);

initPQExpBuffer(&sql);

appendPQExpBufferStr(&sql, "CLUSTER"); if (verbose) appendPQExpBufferStr(&sql, " VERBOSE"); if (table) { appendPQExpBufferChar(&sql, ' '); appendQualifiedRelation(&sql, table, conn, echo); } appendPQExpBufferChar(&sql, ';');

if (!executeMaintenanceCommand(conn, sql.data, echo)) { if (table) pg_log_error("clustering of table \"%s\" in database \"%s\" failed: %s", table, PQdb(conn), PQerrorMessage(conn)); else pg_log_error("clustering of database \"%s\" failed: %s", PQdb(conn), PQerrorMessage(conn)); PQfinish(conn); exit(1); } PQfinish(conn); termPQExpBuffer(&sql); } ~~~
