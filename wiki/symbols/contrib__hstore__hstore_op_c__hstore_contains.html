/*

  • we exploit the fact that keys in "tmpl" are in strictly increasing
  • order to narrow the hstoreFindKey search; each search can start one
  • entry past the previous "found" entry, or at the lower bound of the
  • search

    • */

for (i = 0; res && i < tcount; ++i) { int idx = hstoreFindKey(val, &lastidx, HSTORE_KEY(te, tstr, i), HSTORE_KEYLEN(te, i));

if (idx >= 0) { bool nullval = HSTORE_VALISNULL(te, i); int vallen = HSTORE_VALLEN(te, i);

if (nullval != HSTORE_VALISNULL(ve, idx) || (!nullval && (vallen != HSTORE_VALLEN(ve, idx) || memcmp(HSTORE_VAL(te, tstr, i), HSTORE_VAL(ve, vstr, idx), vallen) != 0))) res = false; } else res = false; }

PG_RETURN_BOOL(res); } ~~~
