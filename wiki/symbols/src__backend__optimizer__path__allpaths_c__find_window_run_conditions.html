while (IsA(wfunc, RelabelType)) wfunc = (WindowFunc *) ((RelabelType *) wfunc)->arg;

/*

  • The value being compared must not change during the evaluation of the
  • window partition.

req.type = T_SupportRequestWFuncMonotonic; req.window_func = wfunc; req.window_clause = wclause;

/* call the support function */ res = (SupportRequestWFuncMonotonic *) DatumGetPointer(OidFunctionCall1(prosupport, PointerGetDatum(&req)));
