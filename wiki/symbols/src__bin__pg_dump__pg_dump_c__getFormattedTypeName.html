if (oid == 0) { if ((opts & zeroAsStar) != 0) return "*"; else if ((opts & zeroAsNone) != 0) return "NONE"; }

query = createPQExpBuffer(); appendPQExpBuffer(query, "SELECT pg_catalog.format_type('%u'::pg_catalog.oid, NULL)", oid);

/* result of format_type is already quoted */ result = pg_strdup(PQgetvalue(res, 0, 0));

PQclear(res); destroyPQExpBuffer(query);

/*

  • don't have a TypeInfo for the type, the string will be leaked once the
  • caller is done with it ... but that case really should not happen, so
  • leaking if it does seems acceptable.

    • */ if (typeInfo) typeInfo->ftypname = result;

return result; } ~~~
