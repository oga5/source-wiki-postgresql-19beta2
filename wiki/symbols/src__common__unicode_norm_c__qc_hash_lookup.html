/*

  • Compute the hash function. The hash key is the codepoint with the bytes
  • in network order.

    • */ hashkey = pg_hton32(ch); h = norminfo->hash(&hashkey);

/* An out-of-range result implies no match */ if (h < 0 || h >= norminfo->num_normprops) return NULL;

/*

  • Since it's a perfect hash, we need only match to the specific codepoint
  • it identifies.

    • */ if (ch != norminfo->normprops[h].codepoint) return NULL;

/* Success! */ return &norminfo->normprops[h]; } ~~~
