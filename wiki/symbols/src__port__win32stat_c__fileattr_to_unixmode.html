ソース抜粋

~~~c static unsigned short fileattr_to_unixmode(int attr) { unsigned short uxmode = 0;

uxmode |= (unsigned short) ((attr & FILE_ATTRIBUTE_DIRECTORY) ? (_S_IFDIR) : (_S_IFREG));

uxmode |= (unsigned short) ((attr & FILE_ATTRIBUTE_READONLY) ? (_S_IREAD) : (_S_IREAD | _S_IWRITE));

/* there is no need to simulate _S_IEXEC using CMD's PATHEXT extensions */ uxmode |= _S_IEXEC;

return uxmode; } ~~~
