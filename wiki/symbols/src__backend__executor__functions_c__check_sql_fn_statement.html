foreach(lc, queryTreeList) { Query *query = lfirst_node(Query, lc);

/*

  • Disallow calling procedures with output arguments. The current
  • implementation would just throw the output values away, unless the
  • statement is the last one. Per SQL standard, we should assign the
  • output values by name. By disallowing this here, we preserve an
  • opportunity for future improvement.

    • */ if (query->commandType == CMD_UTILITY && IsA(query->utilityStmt, CallStmt)) { CallStmt *stmt = (CallStmt *) query->utilityStmt;

if (stmt->outargs != NIL) ereport(ERROR, (errcode(ERRCODE_FEATURE_NOT_SUPPORTED), errmsg("calling procedures with output arguments is not supported in SQL functions"))); } } } ~~~
