if (OpClassCache == NULL) { /* First time through: initialize the opclass cache */ HASHCTL ctl;

opcentry = (OpClassCacheEnt *) hash_search(OpClassCache, &operatorClassOid, HASH_ENTER, &found);

if (!found) { /* Initialize new entry */ opcentry->valid = false; /* until known OK */ opcentry->numSupport = numSupport; opcentry->supportProcs = NULL; /* filled below */ } else { Assert(numSupport == opcentry->numSupport); }

  • loading logic itself, such as reliance on a non-nailed index. Given
  • the limited use-case and the fact that this adds a great deal of
  • expense, we enable it only for high values of debug_discard_caches.
