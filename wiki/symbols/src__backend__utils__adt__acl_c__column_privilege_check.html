~~~c static int column_privilege_check(Oid tableoid, AttrNumber attnum, Oid roleid, AclMode mode) { AclResult aclresult; bool is_missing = false;

/*

  • If convert_column_name failed, we can just return -1 immediately.

    • */ if (attnum == InvalidAttrNumber) return -1;

/*

  • Check for column-level privileges first. This serves in part as a check
  • on whether the column even exists, so we need to do it before checking
  • table-level privilege.

    • */ aclresult = pg_attribute_aclcheck_ext(tableoid, attnum, roleid, mode, &is_missing); if (aclresult == ACLCHECK_OK) return 1; else if (is_missing) return -1;

/* Next check if we have the privilege at the table level */ aclresult = pg_class_aclcheck_ext(tableoid, roleid, mode, &is_missing); if (aclresult == ACLCHECK_OK) return 1; else if (is_missing) return -1; else return 0; } ~~~
