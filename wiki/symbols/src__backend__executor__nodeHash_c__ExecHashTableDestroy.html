~~~c void ExecHashTableDestroy(HashJoinTable hashtable) { int i;

/*

  • Make sure all the temp files are closed. We skip batch 0, since it
  • can't have any temp files (and the arrays might not even exist if
  • nbatch is only 1). Parallel hash joins don't use these files.

    • */ if (hashtable->innerBatchFile != NULL) { for (i = 1; i < hashtable->nbatch; i++) { if (hashtable->innerBatchFile[i]) BufFileClose(hashtable->innerBatchFile[i]); if (hashtable->outerBatchFile[i]) BufFileClose(hashtable->outerBatchFile[i]); } }

/* Release working memory (batchCxt is a child, so it goes away too) */ MemoryContextDelete(hashtable->hashCxt);

/* And drop the control block */ pfree(hashtable); } ~~~
