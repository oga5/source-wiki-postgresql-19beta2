~~~c int px_gen_salt(const char *salt_type, char *buf, int rounds) { struct generator *g; char *p; char rbuf[16];

CheckBuiltinCryptoMode();

for (g = gen_list; g->name; g++) if (pg_strcasecmp(g->name, salt_type) == 0) break;

if (g->name == NULL) return PXE_UNKNOWN_SALT_ALGO;

if (g->def_rounds) { if (rounds == 0) rounds = g->def_rounds;

if (rounds < g->min_rounds || rounds > g->max_rounds) return PXE_BAD_SALT_ROUNDS; }

if (!pg_strong_random(rbuf, g->input_len)) return PXE_NO_RANDOM;

p = g->gen(rounds, rbuf, g->input_len, buf, PX_MAX_SALT_LEN); px_memset(rbuf, 0, sizeof(rbuf));

if (p == NULL) return PXE_BAD_SALT_ROUNDS;

return strlen(p); } ~~~
