~~~c int FileTruncate(File file, pgoff_t offset, uint32 wait_event_info) { int returnCode;

Assert(FileIsValid(file));

DO_DB(elog(LOG, "FileTruncate %d (%s)", file, VfdCache[file].fileName));

returnCode = FileAccess(file); if (returnCode < 0) return returnCode;

pgstat_report_wait_start(wait_event_info); returnCode = pg_ftruncate(VfdCache[file].fd, offset); pgstat_report_wait_end();

if (returnCode == 0 && VfdCache[file].fileSize > offset) { /* adjust our state for truncation of a temp file */ Assert(VfdCache[file].fdstate & FD_TEMP_FILE_LIMIT); temporary_files_size -= VfdCache[file].fileSize - offset; VfdCache[file].fileSize = offset; }

return returnCode; } ~~~
