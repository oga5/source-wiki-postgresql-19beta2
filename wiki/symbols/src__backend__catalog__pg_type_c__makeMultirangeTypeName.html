  • "multirange". Otherwise add "_multirange" to the end.

    • */ rangestr = strstr(rangeTypeName, "range"); if (rangestr) { char *prefix = pnstrdup(rangeTypeName, rangestr - rangeTypeName);

buf = psprintf("%s%s%s", prefix, "multi", rangestr); } else buf = psprintf("%s_multirange", pnstrdup(rangeTypeName, NAMEDATALEN - 12));
