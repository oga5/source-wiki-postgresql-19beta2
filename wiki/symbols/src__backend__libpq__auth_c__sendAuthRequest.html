pq_beginmessage(&buf, PqMsg_AuthenticationRequest); pq_sendint32(&buf, (int32) areq); if (extralen > 0) pq_sendbytes(&buf, extradata, extralen);

pq_endmessage(&buf);

/*

  • Flush message so client will see it, except for AUTH_REQ_OK and
  • AUTH_REQ_SASL_FIN, which need not be sent until we are ready for
  • queries.

    • */ if (areq != AUTH_REQ_OK && areq != AUTH_REQ_SASL_FIN) pq_flush();

CHECK_FOR_INTERRUPTS(); } ~~~
