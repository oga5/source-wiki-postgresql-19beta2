/* Consume all input, to be flushed later */ while (input->pos < input->size) { res = ZSTD_compressStream2(zstdcs->cstream, output, input, ZSTD_e_continue); if (ZSTD_isError(res)) pg_fatal("could not write to file: %s", ZSTD_getErrorName(res));

/* Dump output buffer if full */ if (output->pos >= output->size) { errno = 0; cnt = fwrite(output->dst, 1, output->pos, zstdcs->fp); if (cnt != output->pos) { errno = (errno) ? errno : ENOSPC; pg_fatal("could not write to file: %m"); } output->pos = 0; } } } ~~~
