subrinfo = pg_malloc_array(SubRelInfo, ntups); for (int i = 0; i < ntups; i++) { Oid cur_srsubid = atooid(PQgetvalue(res, i, i_srsubid)); Oid relid = atooid(PQgetvalue(res, i, i_srrelid)); TableInfo *tblinfo;

/*

  • exists.
