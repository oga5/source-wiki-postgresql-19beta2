if (entry->target->itarget != NULL) { appendStringInfoChar(&buf, ' '); pgpa_format_index_target(&buf, entry->target->itarget); }

if (entry->tag != PGPA_TAG_JOIN_ORDER) appendStringInfoChar(&buf, ')');

return buf.data; } ~~~
