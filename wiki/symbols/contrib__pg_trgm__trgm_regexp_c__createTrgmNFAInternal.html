~~~c static TRGM * createTrgmNFAInternal(regex_t *regex, TrgmPackedGraph **graph, MemoryContext rcontext) { TRGM *trg; TrgmNFA trgmNFA;

trgmNFA.regex = regex;

/* Collect color information from the regex */ getColorInfo(regex, &trgmNFA);

#ifdef TRGM_REGEXP_DEBUG printSourceNFA(regex, trgmNFA.colorInfo, trgmNFA.ncolors); #endif

/*

  • Stage 2: Create an expanded graph from the source NFA.

    • */ transformGraph(&trgmNFA);

#ifdef TRGM_REGEXP_DEBUG printTrgmNFA(&trgmNFA); #endif

/*

  • Fail if we were unable to make a nontrivial graph, ie it is possible to
  • get from the initial state to the final state without reading any
  • predictable trigram.

    • */ if (trgmNFA.initState->flags & TSTATE_FIN) return NULL;

/*

  • Stage 3: Select color trigrams to expand. Fail if too many trigrams.

    • */ if (!selectColorTrigrams(&trgmNFA)) return NULL;

/*

  • Stage 4: Expand color trigrams and pack graph into final
  • representation.

    • */ trg = expandColorTrigrams(&trgmNFA, rcontext);

*graph = packGraph(&trgmNFA, rcontext);

#ifdef TRGM_REGEXP_DEBUG printTrgmPackedGraph(*graph, trg); #endif

return trg; } ~~~
