~~~c static void hmac_init(PX_HMAC *h, const uint8 *key, unsigned klen) { unsigned bs, i; uint8 *keybuf; PX_MD *md = h->md;

bs = px_md_block_size(md); keybuf = palloc0(bs);

if (klen > bs) { px_md_update(md, key, klen); px_md_finish(md, keybuf); px_md_reset(md); } else memcpy(keybuf, key, klen);

for (i = 0; i < bs; i++) { h->p.ipad[i] = keybuf[i] ^ HMAC_IPAD; h->p.opad[i] = keybuf[i] ^ HMAC_OPAD; }

px_memset(keybuf, 0, bs); pfree(keybuf);

px_md_update(md, h->p.ipad, bs); } ~~~
