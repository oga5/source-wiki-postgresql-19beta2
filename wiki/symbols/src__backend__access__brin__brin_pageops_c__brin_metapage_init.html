~~~c void brin_metapage_init(Page page, BlockNumber pagesPerRange, uint16 version) { BrinMetaPageData *metadata;

brin_page_init(page, BRIN_PAGETYPE_META);

metadata = (BrinMetaPageData *) PageGetContents(page);

metadata->brinMagic = BRIN_META_MAGIC; metadata->brinVersion = version; metadata->pagesPerRange = pagesPerRange;

/*

  • Note we cheat here a little. 0 is not a valid revmap block number
  • (because it's the metapage buffer), but doing this enables the first
  • revmap page to be created when the index is.

    • */ metadata->lastRevmapPage = 0;

/*

  • Set pd_lower just past the end of the metadata. This is essential,
  • because without doing so, metadata will be lost if xlog.c compresses
  • the page.

    • */ ((PageHeader) page)->pd_lower = ((char *) metadata + sizeof(BrinMetaPageData)) - (char *) page; } ~~~
