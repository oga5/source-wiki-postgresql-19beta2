ctx->output_plugin_private = data;

foreach(option, ctx->output_plugin_options) { DefElem *elem = lfirst(option);

Assert(elem->arg == NULL || IsA(elem->arg, String));

if (elem->arg == NULL) continue; else if (!parse_bool(strVal(elem->arg), &force_binary)) ereport(ERROR, (errcode(ERRCODE_INVALID_PARAMETER_VALUE), errmsg("could not parse value \"%s\" for parameter \"%s\"", strVal(elem->arg), elem->defname))); ~~~
