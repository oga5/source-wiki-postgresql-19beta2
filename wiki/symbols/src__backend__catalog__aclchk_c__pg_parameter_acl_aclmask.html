~~~c static AclMode pg_parameter_acl_aclmask(Oid acl_oid, Oid roleid, AclMode mask, AclMaskHow how) { AclMode result; HeapTuple tuple; Datum aclDatum; bool isNull; Acl *acl;

/* Superusers bypass all permission checking. */ if (superuser_arg(roleid)) return mask;

/* Get the ACL from pg_parameter_acl */ tuple = SearchSysCache1(PARAMETERACLOID, ObjectIdGetDatum(acl_oid)); if (!HeapTupleIsValid(tuple)) ereport(ERROR, (errcode(ERRCODE_UNDEFINED_OBJECT), errmsg("parameter ACL with OID %u does not exist", acl_oid)));

aclDatum = SysCacheGetAttr(PARAMETERACLOID, tuple, Anum_pg_parameter_acl_paracl, &isNull); if (isNull) { /* No ACL, so build default ACL */ acl = acldefault(OBJECT_PARAMETER_ACL, BOOTSTRAP_SUPERUSERID); aclDatum = (Datum) 0; } else { /* detoast ACL if necessary */ acl = DatumGetAclP(aclDatum); }

result = aclmask(acl, roleid, BOOTSTRAP_SUPERUSERID, mask, how);

/* if we have a detoasted copy, free it */ if (acl && acl != DatumGetPointer(aclDatum)) pfree(acl);

ReleaseSysCache(tuple);

return result; } ~~~
