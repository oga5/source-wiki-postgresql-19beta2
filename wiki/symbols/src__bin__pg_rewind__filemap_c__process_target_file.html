~~~c void process_target_file(const char *path, file_type_t type, size_t size, const char *link_target) { file_entry_t *entry;

/*

  • Do not apply any exclusion filters here. This has advantage to remove
  • from the target data folder all paths which have been filtered out from
  • the source data folder when processing the source files.

    • */

/*

  • Like in process_source_file, pretend that pg_wal is always a directory.

    • */ if (strcmp(path, "pg_wal") == 0 && type == FILE_TYPE_SYMLINK) type = FILE_TYPE_DIRECTORY;

/* Remember this target file */ entry = insert_filehash_entry(path); if (entry->target_exists) pg_fatal("duplicate source file \"%s\"", path); entry->target_exists = true; entry->target_type = type; entry->target_size = size; entry->target_link_target = link_target ? pg_strdup(link_target) : NULL; } ~~~
