~~~c int plpgsql_location_to_lineno(int location, yyscan_t yyscanner) { const char *loc;

if (location < 0 || yyextra->scanorig == NULL) return 0; /* garbage in, garbage out */ loc = yyextra->scanorig + location;

/* be correct, but not fast, if input location goes backwards */ if (loc < yyextra->cur_line_start) location_lineno_init(yyscanner);

while (yyextra->cur_line_end != NULL && loc > yyextra->cur_line_end) { yyextra->cur_line_start = yyextra->cur_line_end + 1; yyextra->cur_line_num++; yyextra->cur_line_end = strchr(yyextra->cur_line_start, '\n'); }

return yyextra->cur_line_num; } ~~~
