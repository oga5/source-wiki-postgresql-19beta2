  • aggs. Just make a new slot every time.

root->plan_params = lappend(root->plan_params, pitem);

retval = makeNode(Param); retval->paramkind = PARAM_EXEC; retval->paramid = pitem->paramId; retval->paramtype = ptype; retval->paramtypmod = -1; retval->paramcollid = InvalidOid; retval->location = grp->location;
