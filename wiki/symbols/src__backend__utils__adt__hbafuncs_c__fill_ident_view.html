/*

  • In the unlikely event that we can't open pg_ident.conf, we throw an
  • error, rather than trying to report it via some sort of view entry.
  • (Most other error conditions should result in a message in a view
  • entry.)

    • */ file = open_auth_file(IdentFileName, ERROR, 0, NULL);

/* Now parse all the lines */ identcxt = AllocSetContextCreate(CurrentMemoryContext, "ident parser context", ALLOCSET_SMALL_SIZES); oldcxt = MemoryContextSwitchTo(identcxt); foreach(line, ident_lines) { TokenizedAuthLine *tok_line = (TokenizedAuthLine *) lfirst(line); IdentLine *identline = NULL;

/* don't parse lines that already have errors */ if (tok_line->err_msg == NULL) identline = parse_ident_line(tok_line, DEBUG3);

/* no error, set a new mapping number */ if (tok_line->err_msg == NULL) map_number++;
