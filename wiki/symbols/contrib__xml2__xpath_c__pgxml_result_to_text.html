if (res == NULL) return NULL;

/* spin some error handling */ xmlerrcxt = pg_xml_init(PG_XML_STRICTNESS_ALL);

/* Now convert this result back to text */ xpres = cstring_to_text((char *) xpresstr); } PG_CATCH(); { if (xpresstr != NULL) xmlFree(xpresstr);

pg_xml_done(xmlerrcxt, true);
