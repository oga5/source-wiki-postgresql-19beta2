if (hresult) { /* Yes, so nothing to do */ result.recent_buffer = -hresult->id - 1; } else { #ifdef USE_PREFETCH /* Not in buffers, so initiate prefetch */ if ((io_direct_flags & IO_DIRECT_DATA) == 0 && smgrprefetch(smgr, forkNum, blockNum, 1)) { result.initiated_io = true; } #endif /* USE_PREFETCH */ }
