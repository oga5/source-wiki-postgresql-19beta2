    • */ xid = UINT32_ACCESS_ONCE(proc->xmin); if (proc->databaseId == MyDatabaseId && TransactionIdIsNormal(xid) && TransactionIdPrecedesOrEquals(xid, xmin)) { /*

  • value is interpreted by vacuum.

result = true; }

LWLockRelease(ProcArrayLock);

return result; } ~~~
