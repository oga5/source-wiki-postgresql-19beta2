~~~c void pgstat_report_replslotsync(ReplicationSlot *slot) { PgStat_EntryRef *entry_ref; PgStatShared_ReplSlot *shstatent; PgStat_StatReplSlotEntry *statent;

/* Slot sync stats are valid only for synced logical slots on standby. */ Assert(slot->data.synced); Assert(RecoveryInProgress());

entry_ref = pgstat_get_entry_ref_locked(PGSTAT_KIND_REPLSLOT, InvalidOid, ReplicationSlotIndex(slot), false); Assert(entry_ref != NULL);

shstatent = (PgStatShared_ReplSlot *) entry_ref->shared_stats; statent = &shstatent->stats;

statent->slotsync_skip_count += 1; statent->slotsync_last_skip = GetCurrentTimestamp();

pgstat_unlock_entry(entry_ref); } ~~~
