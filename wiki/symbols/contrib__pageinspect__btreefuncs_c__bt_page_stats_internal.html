UnlockReleaseBuffer(buffer); relation_close(rel, AccessShareLock);

/* Build a tuple descriptor for our result type */ if (get_call_result_type(fcinfo, NULL, &tupleDesc) != TYPEFUNC_COMPOSITE) elog(ERROR, "return type must be a row type");
