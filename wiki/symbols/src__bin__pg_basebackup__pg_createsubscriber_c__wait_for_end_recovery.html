conn = connect_database(conninfo, true);

for (;;) { /* Did the recovery process finish? We're done if so. */ if (dry_run || !server_is_in_recovery(conn)) { ready = true; recovery_ended = true; break; }

/* Keep waiting */ pg_usleep(WAIT_INTERVAL * USECS_PER_SEC); timer += WAIT_INTERVAL; }

disconnect_database(conn, false);

if (!ready) pg_fatal("server did not end recovery");
