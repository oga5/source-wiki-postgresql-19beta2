~~~c void SetVariableHooks(VariableSpace space, const char *name, VariableSubstituteHook shook, VariableAssignHook ahook) { struct _variable *current, *previous;

for (previous = space, current = space->next; current; previous = current, current = current->next) { int cmp = strcmp(current->name, name);

if (cmp == 0) { /* found entry, so update */ current->substitute_hook = shook; current->assign_hook = ahook; if (shook) current->value = (*shook) (current->value); if (ahook) (void) (*ahook) (current->value); return; } if (cmp > 0) break; /* it's not there */ }
