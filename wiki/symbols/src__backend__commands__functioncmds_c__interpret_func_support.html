~~~c static Oid interpret_func_support(DefElem *defel) { List *procName = defGetQualifiedName(defel); Oid procOid; Oid argList[1];

/*

  • Support functions always take one INTERNAL argument and return
  • INTERNAL.

    • */ argList[0] = INTERNALOID;

procOid = LookupFuncName(procName, 1, argList, true); if (!OidIsValid(procOid)) ereport(ERROR, (errcode(ERRCODE_UNDEFINED_FUNCTION), errmsg("function %s does not exist", func_signature_string(procName, 1, NIL, argList))));

if (get_func_rettype(procOid) != INTERNALOID) ereport(ERROR, (errcode(ERRCODE_INVALID_OBJECT_DEFINITION), errmsg("support function %s must return type %s", NameListToString(procName), "internal")));

/*

  • Someday we might want an ACL check here; but for now, we insist that
  • you be superuser to specify a support function, so privilege on the
  • support function is moot.

    • */ if (!superuser()) ereport(ERROR, (errcode(ERRCODE_INSUFFICIENT_PRIVILEGE), errmsg("must be superuser to specify a support function")));

return procOid; } ~~~
