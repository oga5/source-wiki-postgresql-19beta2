~~~c static TrieChar * placeChar(TrieChar *node, const unsigned char *str, int lenstr, const char *replaceTo, int replacelen) { TrieChar *curnode;

if (!node) node = palloc0_array(TrieChar, 256);

Assert(lenstr > 0); /* else str[0] doesn't exist */

curnode = node + *str;

if (lenstr <= 1) { if (curnode->replaceTo) ereport(WARNING, (errcode(ERRCODE_CONFIG_FILE_ERROR), errmsg("duplicate source strings, first one will be used"))); else { curnode->replacelen = replacelen; curnode->replaceTo = (char *) palloc(replacelen); memcpy(curnode->replaceTo, replaceTo, replacelen); } } else { curnode->nextChar = placeChar(curnode->nextChar, str + 1, lenstr - 1, replaceTo, replacelen); }

return node; } ~~~
