~~~c void * shm_toc_lookup(shm_toc *toc, uint64 key, bool noError) { uint32 nentry; uint32 i;

/*

  • Read the number of entries before we examine any entry. We assume that
  • reading a uint32 is atomic.

    • */ nentry = toc->toc_nentry; pg_read_barrier();

/* Now search for a matching entry. */ for (i = 0; i < nentry; ++i) { if (toc->toc_entry[i].key == key) return ((char *) toc) + toc->toc_entry[i].offset; }

/* No matching entry was found. */ if (!noError) elog(ERROR, "could not find key " UINT64_FORMAT " in shm TOC at %p", key, toc); return NULL; } ~~~
