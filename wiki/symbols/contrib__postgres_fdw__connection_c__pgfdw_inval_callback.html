/* ConnectionHash must exist already, if we're registered */ hash_seq_init(&scan, ConnectionHash); while ((entry = (ConnCacheEntry *) hash_seq_search(&scan))) { /* Ignore invalid entries */ if (entry->conn == NULL) continue;

  • Close the connection immediately if it's not used yet in this
  • transaction. Otherwise mark it as invalid so that
  • pgfdw_xact_callback() can close it at the end of this
  • transaction.
