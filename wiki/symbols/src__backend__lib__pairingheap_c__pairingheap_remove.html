  • If this node has children, make a new subheap of the children and link
  • the subheap in place of the removed node. Otherwise just unlink this
  • node.

replacement->prev_or_parent = node->prev_or_parent; replacement->next_sibling = node->next_sibling; *prev_ptr = replacement; if (next_sibling) next_sibling->prev_or_parent = replacement; } else { *prev_ptr = next_sibling; if (next_sibling) next_sibling->prev_or_parent = node->prev_or_parent; } } ~~~
