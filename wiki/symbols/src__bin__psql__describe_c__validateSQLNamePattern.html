initPQExpBuffer(&dbbuf); added = processSQLNamePattern(pset.db, buf, pattern, have_where, force_escape, schemavar, namevar, altnamevar, visibilityrule, &dbbuf, &dotcnt); if (added_clause != NULL) *added_clause = added;

if (dotcnt >= maxparts) { pg_log_error("improper qualified name (too many dotted names): %s", pattern); goto error_return; }

if (maxparts > 1 && dotcnt == maxparts - 1) { if (PQdb(pset.db) == NULL) { pg_log_error("You are currently not connected to a database."); goto error_return; } if (strcmp(PQdb(pset.db), dbbuf.data) != 0) { pg_log_error("cross-database references are not implemented: %s", pattern); goto error_return; } } termPQExpBuffer(&dbbuf); return true;

error_return: termPQExpBuffer(&dbbuf); return false; } ~~~
