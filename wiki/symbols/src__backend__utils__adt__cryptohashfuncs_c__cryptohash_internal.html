switch (type) { case PG_SHA224: typestr = "SHA224"; digest_len = PG_SHA224_DIGEST_LENGTH; break; case PG_SHA256: typestr = "SHA256"; digest_len = PG_SHA256_DIGEST_LENGTH; break; case PG_SHA384: typestr = "SHA384"; digest_len = PG_SHA384_DIGEST_LENGTH; break; case PG_SHA512: typestr = "SHA512"; digest_len = PG_SHA512_DIGEST_LENGTH; break; case PG_MD5: case PG_SHA1: elog(ERROR, "unsupported cryptohash type %d", type); break; }
