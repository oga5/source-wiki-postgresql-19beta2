~~~c static void postquel_end(execution_state *es, SQLFunctionCachePtr fcache) { MemoryContext oldcontext;

/* Run the sub-executor in subcontext */ oldcontext = MemoryContextSwitchTo(fcache->subcontext);

/* mark status done to ensure we don't do ExecutorEnd twice */ es->status = F_EXEC_DONE;

/* Utility commands don't need Executor. */ if (es->qd->operation != CMD_UTILITY) { ExecutorFinish(es->qd); ExecutorEnd(es->qd); }

es->qd->dest->rDestroy(es->qd->dest);

FreeQueryDesc(es->qd); es->qd = NULL;

MemoryContextSwitchTo(oldcontext);

/* Delete the subcontext, if it's actually a separate context */ if (fcache->ownSubcontext) MemoryContextDelete(fcache->subcontext); fcache->subcontext = NULL; } ~~~
