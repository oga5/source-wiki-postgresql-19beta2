txid = XLogRecGetTopXid(record);

/*

  • If the top-level xid is valid, we need to assign the subxact to the
  • top-level xact. We need to do this for all records, hence we do it
  • before the switch.

    • */ if (TransactionIdIsValid(txid)) { ReorderBufferAssignChild(ctx->reorder, txid, XLogRecGetXid(record), buf.origptr); }

rmgr = GetRmgr(XLogRecGetRmid(record));

if (rmgr.rm_decode != NULL) rmgr.rm_decode(ctx, &buf); else { /* just deal with xid, and done */ ReorderBufferProcessXid(ctx->reorder, XLogRecGetXid(record), buf.origptr); } } ~~~
