~~~c static bool gistinserttuples(GISTInsertState *state, GISTInsertStack *stack, GISTSTATE *giststate, IndexTuple *tuples, int ntup, OffsetNumber oldoffnum, Buffer leftchild, Buffer rightchild, bool unlockbuf, bool unlockleftchild) { List *splitinfo; bool is_split;

/*

  • Check for any rw conflicts (in serializable isolation level) just
  • before we intend to modify the page

    • */ CheckForSerializableConflictIn(state->r, NULL, BufferGetBlockNumber(stack->buffer));

/* Insert the tuple(s) to the page, splitting the page if necessary */ is_split = gistplacetopage(state->r, state->freespace, giststate, stack->buffer, tuples, ntup, oldoffnum, NULL, leftchild, &splitinfo, true, state->heapRel, state->is_build);

/*

  • Before recursing up in case the page was split, release locks on the
  • child pages. We don't need to keep them locked when updating the
  • parent.

    • */ if (BufferIsValid(rightchild)) UnlockReleaseBuffer(rightchild); if (BufferIsValid(leftchild) && unlockleftchild) LockBuffer(leftchild, GIST_UNLOCK);

/*

  • If we had to split, insert/update the downlinks in the parent. If the
  • caller requested us to release the lock on stack->buffer, tell
  • gistfinishsplit() to do that as soon as it's safe to do so. If we
  • didn't have to split, release it ourselves.

    • */ if (splitinfo) gistfinishsplit(state, stack, giststate, splitinfo, unlockbuf); else if (unlockbuf) LockBuffer(stack->buffer, GIST_UNLOCK);

return is_split; } ~~~
