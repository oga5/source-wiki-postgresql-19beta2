  • We loop to find the bottom base type in a stack of domains.

    • */ for (;;) { HeapTuple tup; Form_pg_type typTup;

/* This test must match get_element_type */ if (IsTrueArrayType(typTup)) result = typTup->typelem; else result = InvalidOid; ReleaseSysCache(tup); return result; }

typid = typTup->typbasetype; ReleaseSysCache(tup); }

/* Like get_element_type, silently return InvalidOid for bogus input */ return InvalidOid; } ~~~
