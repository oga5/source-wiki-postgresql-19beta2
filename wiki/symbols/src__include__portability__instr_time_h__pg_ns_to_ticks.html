~~~c static inline int64 pg_ns_to_ticks(int64 ns) { #if PG_INSTR_TICKS_TO_NS int64 ticks = 0;

Assert(timing_initialized);

/*

  • If ticks_per_ns_scaled is zero, ticks are already in nanoseconds (e.g.
  • system clock on Unix).

    • */ if (ticks_per_ns_scaled == 0) return ns;

/*

  • The reverse of pg_ticks_to_ns to avoid a similar overflow problem.

    • */ if (unlikely(ns > (INT64_MAX >> TICKS_TO_NS_SHIFT))) { int64 count = ns / ticks_per_ns_scaled;

ticks = count << TICKS_TO_NS_SHIFT; ns -= count * ticks_per_ns_scaled; }

ticks += (ns << TICKS_TO_NS_SHIFT) / ticks_per_ns_scaled;

return ticks; #else Assert(timing_initialized);

return ns; #endif /* PG_INSTR_TICKS_TO_NS */ } ~~~
