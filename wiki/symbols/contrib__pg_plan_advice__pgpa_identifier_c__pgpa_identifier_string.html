~~~c const char * pgpa_identifier_string(const pgpa_identifier *rid) { const char *result;

Assert(rid->alias_name != NULL); result = quote_identifier(rid->alias_name);

Assert(rid->occurrence >= 0); if (rid->occurrence > 1) result = psprintf("%s#%d", result, rid->occurrence);

if (rid->partrel != NULL) { if (rid->partnsp == NULL) result = psprintf("%s/%s", result, quote_identifier(rid->partrel)); else result = psprintf("%s/%s.%s", result, quote_identifier(rid->partnsp), quote_identifier(rid->partrel)); }

if (rid->plan_name != NULL) result = psprintf("%s@%s", result, quote_identifier(rid->plan_name));

return result; } ~~~
