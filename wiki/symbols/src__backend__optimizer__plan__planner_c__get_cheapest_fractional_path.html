/* If all tuples will be retrieved, just return the cheapest-total path */ if (tuple_fraction <= 0.0) return best_path;

/* Convert absolute # of tuples to a fraction; no need to clamp to 0..1 */ if (tuple_fraction >= 1.0 && best_path->rows > 0) tuple_fraction /= best_path->rows;

foreach(l, rel->pathlist) { Path *path = (Path *) lfirst(l);

if (path->param_info) continue;

if (path == rel->cheapest_total_path || compare_fractional_path_costs(best_path, path, tuple_fraction) <= 0) continue;

best_path = path; }

return best_path; } ~~~
