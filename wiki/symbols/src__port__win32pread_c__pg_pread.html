~~~c ssize_t pg_pread(int fd, void *buf, size_t size, pgoff_t offset) { OVERLAPPED overlapped = {0}; HANDLE handle; DWORD result;

handle = (HANDLE) _get_osfhandle(fd); if (handle == INVALID_HANDLE_VALUE) { errno = EBADF; return -1; }

/* Avoid overflowing DWORD. */ size = Min(size, 1024 * 1024 * 1024);

/* Note that this changes the file position, despite not using it. */ overlapped.Offset = (DWORD) offset; overlapped.OffsetHigh = (DWORD) (offset >> 32);

if (!ReadFile(handle, buf, size, &result, &overlapped)) { if (GetLastError() == ERROR_HANDLE_EOF) return 0;

_dosmaperr(GetLastError()); return -1; }

return result; } ~~~
