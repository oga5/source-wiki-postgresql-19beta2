ソース抜粋

~~~c char ** pgfnames(const char *path) { DIR *dir; struct dirent *file; char **filenames; int numnames = 0; int fnsize = 200; /* enough for many small dbs */

dir = opendir(path); if (dir == NULL) { pg_log_warning("could not open directory \"%s\": %m", path); return NULL; }

filenames = palloc_array(char *, fnsize);

while (errno = 0, (file = readdir(dir)) != NULL) { if (strcmp(file->d_name, ".") != 0 && strcmp(file->d_name, "..") != 0) { if (numnames + 1 >= fnsize) { fnsize *= 2; filenames = repalloc_array(filenames, char *, fnsize); } filenames[numnames++] = pstrdup(file->d_name); } }

if (errno) pg_log_warning("could not read directory \"%s\": %m", path);

filenames[numnames] = NULL;

if (closedir(dir)) pg_log_warning("could not close directory \"%s\": %m", path);

return filenames; } ~~~
