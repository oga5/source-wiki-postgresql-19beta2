~~~c Datum hash_ltree(PG_FUNCTION_ARGS) { ltree *a = PG_GETARG_LTREE_P(0); uint32 result = 1; int an = a->numlevel; ltree_level *al = LTREE_FIRST(a);

while (an > 0) { uint32 levelHash = DatumGetUInt32(hash_any((unsigned char *) al->name, al->len));

/*

  • Combine hash values of successive elements by multiplying the
  • current value by 31 and adding on the new element's hash value.

    • *

  • This method is borrowed from hash_array(), which see for further
  • commentary.

    • */ result = (result << 5) - result + levelHash;

an--; al = LEVEL_NEXT(al); }

PG_FREE_IF_COPY(a, 0); PG_RETURN_UINT32(result); } ~~~
