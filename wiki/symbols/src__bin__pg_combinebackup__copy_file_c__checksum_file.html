~~~c static void checksum_file(const char *src, pg_checksum_context *checksum_ctx) { int src_fd; uint8 *buffer; const int buffer_size = 50 * BLCKSZ; ssize_t rb;

/* bail out if no checksum needed */ if (checksum_ctx->type == CHECKSUM_TYPE_NONE) return;

if ((src_fd = open(src, O_RDONLY | PG_BINARY, 0)) < 0) pg_fatal("could not open file \"%s\": %m", src);

buffer = pg_malloc(buffer_size);

while ((rb = read(src_fd, buffer, buffer_size)) > 0) { if (pg_checksum_update(checksum_ctx, buffer, rb) < 0) pg_fatal("could not update checksum of file \"%s\"", src); }

if (rb < 0) pg_fatal("could not read file \"%s\": %m", src);

pg_free(buffer); close(src_fd); } ~~~
