~~~c char * pg_server_to_any(const char *s, int len, int encoding) { if (len <= 0) return unconstify(char *, s); /* empty string is always valid */

if (encoding == DatabaseEncoding->encoding || encoding == PG_SQL_ASCII) return unconstify(char *, s); /* assume data is valid */

if (DatabaseEncoding->encoding == PG_SQL_ASCII) { /* No conversion is possible, but we must validate the result */ (void) pg_verify_mbstr(encoding, s, len, false); return unconstify(char *, s); }

/* Fast path if we can use cached conversion function */ if (encoding == ClientEncoding->encoding) return perform_default_encoding_conversion(s, len, false);

/* General case ... will not work outside transactions */ return (char *) pg_do_encoding_conversion((unsigned char *) unconstify(char *, s), len, DatabaseEncoding->encoding, encoding); } ~~~
