  • If the path variable is empty, don't do a path search.

    • */ if (strlen(p) == 0) return NULL;

piece = palloc(len + 1); strlcpy(piece, p, len + 1);

/* only absolute paths */ if (!is_absolute_path(mangled)) ereport(ERROR, (errcode(ERRCODE_INVALID_NAME), errmsg("component in parameter \"%s\" is not an absolute path", path_param)));
