ソース抜粋

~~~c static CombinationGenerator * generator_init(int n, int k) { CombinationGenerator *state;

Assert((n >= k) && (k > 0));

/* allocate the generator state as a single chunk of memory */ state = palloc_object(CombinationGenerator);

state->ncombinations = n_choose_k(n, k);

/* pre-allocate space for all combinations */ state->combinations = palloc_array(int, k * state->ncombinations);

state->current = 0; state->k = k; state->n = n;

/* now actually pre-generate all the combinations of K elements */ generate_combinations(state);

/* make sure we got the expected number of combinations */ Assert(state->current == state->ncombinations);

/* reset the number, so we start with the first one */ state->current = 0;

return state; } ~~~
