~~~c static bool pgfdw_exec_cleanup_query_end(PGconn *conn, const char *query, TimestampTz endtime, bool consume_input, bool ignore_errors) { PGresult *result; bool timed_out;

Assert(query != NULL);

/*

  • If requested, consume whatever data is available from the socket. (Note
  • that if all data is available, this allows pgfdw_get_cleanup_result to
  • call PQgetResult without forcing the overhead of WaitLatchOrSocket,
  • which would be large compared to the overhead of PQconsumeInput.)

    • */ if (consume_input && !PQconsumeInput(conn)) { pgfdw_report(WARNING, NULL, conn, query); return false; }

/* Get the result of the query. */ if (pgfdw_get_cleanup_result(conn, endtime, endtime, &result, &timed_out)) { if (timed_out) ereport(WARNING, (errmsg("could not get query result due to timeout"), errcontext("remote SQL command: %s", query))); else pgfdw_report(WARNING, NULL, conn, query);

return false; }

/* Issue a warning if not successful. */ if (PQresultStatus(result) != PGRES_COMMAND_OK) { pgfdw_report(WARNING, result, conn, query); return ignore_errors; } PQclear(result);

return true; } ~~~
