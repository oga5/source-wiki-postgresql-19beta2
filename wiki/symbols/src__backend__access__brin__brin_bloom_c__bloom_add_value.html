~~~c static BloomFilter * bloom_add_value(BloomFilter *filter, uint32 value, bool *updated) { int i; uint64 h1, h2;

/* compute the hashes, used for the bloom filter */ h1 = hash_bytes_uint32_extended(value, BLOOM_SEED_1) % filter->nbits; h2 = hash_bytes_uint32_extended(value, BLOOM_SEED_2) % filter->nbits;

/* compute the requested number of hashes */ for (i = 0; i < filter->nhashes; i++) { /* h1 + h2 + f(i) */ uint32 h = (h1 + i * h2) % filter->nbits; uint32 byte = (h / 8); uint32 bit = (h % 8);

/* if the bit is not set, set it and remember we did that */ if (!(filter->data[byte] & (0x01 << bit))) { filter->data[byte] |= (0x01 << bit); filter->nbits_set++; if (updated) *updated = true; } }

return filter; } ~~~
