~~~c static void set_worktable_pathlist(PlannerInfo *root, RelOptInfo *rel, RangeTblEntry *rte) { Path *ctepath; PlannerInfo *cteroot; Index levelsup; Relids required_outer;

/*

  • We need to find the non-recursive term's path, which is in the plan
  • level that's processing the recursive UNION, which is one level *below*
  • where the CTE comes from.

    • */ levelsup = rte->ctelevelsup; if (levelsup == 0) /* shouldn't happen */ elog(ERROR, "bad levelsup for CTE \"%s\"", rte->ctename); levelsup--; cteroot = root; while (levelsup-- > 0) { cteroot = cteroot->parent_root; if (!cteroot) /* shouldn't happen */ elog(ERROR, "bad levelsup for CTE \"%s\"", rte->ctename); } ctepath = cteroot->non_recursive_path; if (!ctepath) /* shouldn't happen */ elog(ERROR, "could not find path for CTE \"%s\"", rte->ctename);

/* Mark rel with estimated output rows, width, etc */ set_cte_size_estimates(root, rel, ctepath->rows);

/*

  • We don't support pushing join clauses into the quals of a worktable
  • scan, but it could still have required parameterization due to LATERAL
  • refs in its tlist. (I'm not sure this is actually possible given the
  • restrictions on recursive references, but it's easy enough to support.)

    • */ required_outer = rel->lateral_relids;

/* Generate appropriate path */ add_path(rel, create_worktablescan_path(root, rel, required_outer)); } ~~~
