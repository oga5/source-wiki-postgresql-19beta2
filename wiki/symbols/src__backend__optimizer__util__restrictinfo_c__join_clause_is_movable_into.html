~~~c bool join_clause_is_movable_into(RestrictInfo *rinfo, Relids currentrelids, Relids current_and_outer) { /* Clause must be evaluable given available context */ if (!bms_is_subset(rinfo->clause_relids, current_and_outer)) return false;

/* Clause must physically reference at least one target rel */ if (!bms_overlap(currentrelids, rinfo->clause_relids)) return false;

/* Cannot move an outer-join clause into the join's outer side */ if (bms_overlap(currentrelids, rinfo->outer_relids)) return false;

return true; } ~~~
