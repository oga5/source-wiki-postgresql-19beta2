ソース抜粋

~~~c static uint8 * text_to_bits(char *str, int len) { uint8 *bits; int off = 0; char byte = 0;

bits = palloc(len + 1);

while (off < len) { if (off % 8 == 0) byte = 0;

if ((str[off] == '0') || (str[off] == '1')) byte = byte | ((str[off] - '0') << off % 8); else ereport(ERROR, (errcode(ERRCODE_DATA_CORRUPTED), errmsg("invalid character \"%.*s\" in t_bits string", pg_mblen_cstr(str + off), str + off)));

if (off % 8 == 7) bits[off / 8] = byte;

off++; }

return bits; } ~~~
