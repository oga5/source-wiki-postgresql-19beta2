/* check for empty range */ if (pg_strncasecmp(ptr, RANGE_EMPTY_LITERAL, strlen(RANGE_EMPTY_LITERAL)) == 0) { *flags = RANGE_EMPTY; *lbound_str = NULL; *ubound_str = NULL;

ptr += strlen(RANGE_EMPTY_LITERAL);
