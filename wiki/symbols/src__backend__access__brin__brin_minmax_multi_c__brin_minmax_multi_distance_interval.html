~~~c Datum brin_minmax_multi_distance_interval(PG_FUNCTION_ARGS) { float8 delta = 0;

Interval *ia = PG_GETARG_INTERVAL_P(0); Interval *ib = PG_GETARG_INTERVAL_P(1);

int64 dayfraction; int64 days;

/*

  • Delta is (fractional) number of days between the intervals. Assume
  • months have 30 days for consistency with interval_cmp_internal. We
  • don't need to be exact, in the worst case we'll build a bit less
  • efficient ranges. But we should not contradict interval_cmp.

    • */ dayfraction = (ib->time % USECS_PER_DAY) - (ia->time % USECS_PER_DAY); days = (ib->time / USECS_PER_DAY) - (ia->time / USECS_PER_DAY); days += (int64) ib->day - (int64) ia->day; days += ((int64) ib->month - (int64) ia->month) * INT64CONST(30);

/* convert to double precision */ delta = (double) days + dayfraction / (double) USECS_PER_DAY;

Assert(delta >= 0);

PG_RETURN_FLOAT8(delta); } ~~~
