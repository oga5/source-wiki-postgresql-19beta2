~~~c RT_SCOPE RT_VALUE_TYPE * RT_FIND(RT_RADIX_TREE * tree, uint64 key) { RT_CHILD_PTR node; RT_PTR_ALLOC *slot = NULL; int shift;

#ifdef RT_SHMEM Assert(tree->ctl->magic == RT_RADIX_TREE_MAGIC); #endif

if (key > tree->ctl->max_val) return NULL;

Assert(RT_PTR_ALLOC_IS_VALID(tree->ctl->root)); node.alloc = tree->ctl->root; shift = tree->ctl->start_shift;

/* Descend the tree */ while (shift >= 0) { RT_PTR_SET_LOCAL(tree, &node); slot = RT_NODE_SEARCH(node.local, RT_GET_KEY_CHUNK(key, shift)); if (slot == NULL) return NULL;

node.alloc = *slot; shift -= RT_SPAN; }

if (RT_CHILDPTR_IS_VALUE(*slot)) return (RT_VALUE_TYPE *) slot; else { RT_PTR_SET_LOCAL(tree, &node); return (RT_VALUE_TYPE *) node.local; } } ~~~
