if (ecode == 0) return _("no SSL error reported"); errreason = ERR_reason_error_string(ecode); if (errreason != NULL) return errreason;

/*

  • In OpenSSL 3.0.0 and later, ERR_reason_error_string does not map system
  • errno values anymore. (See OpenSSL source code for the explanation.)
  • We can cover that shortcoming with this bit of code. Older OpenSSL
  • versions don't have the ERR_SYSTEM_ERROR macro, but that's okay because
  • they don't have the shortcoming either.

    • */ #ifdef ERR_SYSTEM_ERROR if (ERR_SYSTEM_ERROR(ecode)) return strerror(ERR_GET_REASON(ecode)); #endif

/* No choice but to report the numeric ecode */ snprintf(errbuf, sizeof(errbuf), _("SSL error code %lu"), ecode); return errbuf; } ~~~
