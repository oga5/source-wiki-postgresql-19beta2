~~~c static void pre_sync_fname(const char *fname, bool isdir, int elevel) { int fd;

/* Don't try to flush directories, it'll likely just fail */ if (isdir) return;

ereport_startup_progress("syncing data directory (pre-fsync), elapsed time: %ld.%02d s, current path: %s", fname);

fd = OpenTransientFile(fname, O_RDONLY | PG_BINARY);

if (fd < 0) { if (errno == EACCES) return; ereport(elevel, (errcode_for_file_access(), errmsg("could not open file \"%s\": %m", fname))); return; }

/*

  • pg_flush_data() ignores errors, which is ok because this is only a
  • hint.

    • */ pg_flush_data(fd, 0, 0);

if (CloseTransientFile(fd) != 0) ereport(elevel, (errcode_for_file_access(), errmsg("could not close file \"%s\": %m", fname))); } ~~~
