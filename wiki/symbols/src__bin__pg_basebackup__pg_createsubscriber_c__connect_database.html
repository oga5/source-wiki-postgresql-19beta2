if (exit_on_error) exit(1); return NULL; }

/* Secure search_path */ res = PQexec(conn, ALWAYS_SECURE_SEARCH_PATH_SQL); if (PQresultStatus(res) != PGRES_TUPLES_OK) { pg_log_error("could not clear \"search_path\": %s", PQresultErrorMessage(res)); PQclear(res); PQfinish(conn);

if (exit_on_error) exit(1); return NULL; } PQclear(res);

return conn; } ~~~
