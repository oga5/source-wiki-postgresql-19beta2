~~~c static int try_complete_steps(TestSpec *testspec, PermutationStep **waiting, int nwaiting, int flags) { int old_nwaiting; bool have_blocker;

do { int w = 0;

/* Reset latch; we only care about notices received within loop. */ any_new_notice = false;

/* Likewise, these variables reset for each retry. */ old_nwaiting = nwaiting; have_blocker = false;

/* Scan the array, try to complete steps. */ while (w < nwaiting) { if (try_complete_step(testspec, waiting[w], flags)) { /* Still blocked, leave it alone. */ if (waiting[w]->nblockers > 0) have_blocker = true; w++; } else { /* Done, remove it from array. */ if (w + 1 < nwaiting) memmove(&waiting[w], &waiting[w + 1], (nwaiting - (w + 1)) * sizeof(PermutationStep *)); nwaiting--; } }

/*

  • If any of the still-waiting steps have blocker conditions attached,
  • it's possible that one of the steps we examined afterwards has
  • released them (either by completing, or by sending a NOTICE). If
  • any step completions or NOTICEs happened, repeat the loop until
  • none occurs. Without this provision, completion timing could vary
  • depending on the order in which the steps appear in the array.

    • */ } while (have_blocker && (nwaiting < old_nwaiting || any_new_notice)); return nwaiting; } ~~~
