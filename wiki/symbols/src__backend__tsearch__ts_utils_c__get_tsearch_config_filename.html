~~~c char * get_tsearch_config_filename(const char *basename, const char *extension) { char sharepath[MAXPGPATH]; char *result;

/*

  • We limit the basename to contain a-z, 0-9, and underscores. This may
  • be overly restrictive, but we don't want to allow access to anything
  • outside the tsearch_data directory, so for instance '/' *must* be
  • rejected, and on some platforms '\' and ':' are risky as well. Allowing
  • uppercase might result in incompatible behavior between case-sensitive
  • and case-insensitive filesystems, and non-ASCII characters create other
  • interesting risks, so on the whole a tight policy seems best.

    • */ if (strspn(basename, "abcdefghijklmnopqrstuvwxyz0123456789_") != strlen(basename)) ereport(ERROR, (errcode(ERRCODE_INVALID_PARAMETER_VALUE), errmsg("invalid text search configuration file name \"%s\"", basename)));

get_share_path(my_exec_path, sharepath); result = palloc(MAXPGPATH); snprintf(result, MAXPGPATH, "%s/tsearch_data/%s.%s", sharepath, basename, extension);

return result; } ~~~
