~~~c void BufTableShmemRequest(void *arg) { int size;

/*

  • Request the shared buffer lookup hashtable.

    • *

  • Since we can't tolerate running out of lookup table entries, we must be
  • sure to specify an adequate table size here. The maximum steady-state
  • usage is of course NBuffers entries, but BufferAlloc() tries to insert
  • a new entry before deleting the old. In principle this could be
  • happening in each partition concurrently, so we could need as many as
  • NBuffers + NUM_BUFFER_PARTITIONS entries.

    • */ size = NBuffers + NUM_BUFFER_PARTITIONS;

ShmemRequestHash(.name = "Shared Buffer Lookup Table", .nelems = size, .ptr = &SharedBufHash, .hash_info.keysize = sizeof(BufferTag), .hash_info.entrysize = sizeof(BufferLookupEnt), .hash_info.num_partitions = NUM_BUFFER_PARTITIONS, .hash_flags = HASH_ELEM | HASH_BLOBS | HASH_PARTITION | HASH_FIXED_SIZE, ); } ~~~
