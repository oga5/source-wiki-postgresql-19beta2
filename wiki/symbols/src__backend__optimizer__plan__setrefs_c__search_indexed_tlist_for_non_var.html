/*

  • If it's a simple Const, replacing it with a Var is silly, even if there
  • happens to be an identical Const below; a Var is more expensive to
  • execute than a Const. What's more, replacing it could confuse some
  • places in the executor that expect to see simple Consts for, eg,
  • dropped columns.

    • */ if (IsA(node, Const)) return NULL;

tle = tlist_member(node, itlist->tlist); if (tle) { /* Found a matching subplan output expression */ Var *newvar;

newvar = makeVarFromTargetEntry(newvarno, tle); newvar->varnosyn = 0; /* wasn't ever a plain Var */ newvar->varattnosyn = 0; return newvar; } return NULL; /* no match */ } ~~~
