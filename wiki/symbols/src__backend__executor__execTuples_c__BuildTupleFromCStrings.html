dvalues = (Datum *) palloc(natts * sizeof(Datum)); nulls = (bool *) palloc(natts * sizeof(bool));

/*

  • Call the "in" function for each non-dropped attribute, even for nulls,
  • to support domains.

    • */ for (i = 0; i < natts; i++) { if (!TupleDescCompactAttr(tupdesc, i)->attisdropped) { /* Non-dropped attributes */ dvalues[i] = InputFunctionCall(&attinmeta->attinfuncs[i], values[i], attinmeta->attioparams[i], attinmeta->atttypmods[i]); if (values[i] != NULL) nulls[i] = false; else nulls[i] = true; } else { /* Handle dropped attributes by setting to NULL */ dvalues[i] = (Datum) 0; nulls[i] = true; } }

/*

  • Form a tuple

    • */ tuple = heap_form_tuple(tupdesc, dvalues, nulls);

/*

  • Release locally palloc'd space. XXX would probably be good to pfree
  • values of pass-by-reference datums, as well.

    • */ pfree(dvalues); pfree(nulls);

return tuple; } ~~~
