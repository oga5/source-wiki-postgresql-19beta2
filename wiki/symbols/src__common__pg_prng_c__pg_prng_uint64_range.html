~~~c uint64 pg_prng_uint64_range(pg_prng_state *state, uint64 rmin, uint64 rmax) { uint64 val;

if (likely(rmax > rmin)) { /*

  • Use bitmask rejection method to generate an offset in 0..range.
  • Each generated val is less than twice "range", so on average we
  • should not have to iterate more than twice.

    • */ uint64 range = rmax - rmin; uint32 rshift = 63 - pg_leftmost_one_pos64(range);

do { val = xoroshiro128ss(state) >> rshift; } while (val > range); } else val = 0;

return rmin + val; } ~~~
