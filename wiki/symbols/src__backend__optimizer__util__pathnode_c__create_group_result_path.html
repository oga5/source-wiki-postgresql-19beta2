pathnode->path.pathtype = T_Result; pathnode->path.parent = rel; pathnode->path.pathtarget = target; pathnode->path.param_info = NULL; /* there are no other rels... */ pathnode->path.parallel_aware = false; pathnode->path.parallel_safe = rel->consider_parallel; pathnode->path.parallel_workers = 0; pathnode->path.pathkeys = NIL; pathnode->quals = havingqual;

/*

  • We can't quite use cost_resultscan() because the quals we want to
  • account for are not baserestrict quals of the rel. Might as well just
  • hack it here.

    • */ pathnode->path.rows = 1; pathnode->path.startup_cost = target->cost.startup; pathnode->path.total_cost = target->cost.startup + cpu_tuple_cost + target->cost.per_tuple;

/*

  • Add cost of qual, if any --- but we ignore its selectivity, since our
  • rowcount estimate should be 1 no matter what the qual is.

    • */ if (havingqual) { QualCost qual_cost;

cost_qual_eval(&qual_cost, havingqual, root); /* havingqual is evaluated once at startup */ pathnode->path.startup_cost += qual_cost.startup + qual_cost.per_tuple; pathnode->path.total_cost += qual_cost.startup + qual_cost.per_tuple; }

return pathnode; } ~~~
