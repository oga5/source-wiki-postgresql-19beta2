~~~c static void accumStats(StatsData *stats, bool skipped, double lat, double lag, EStatus estatus, int64 tries) { /* Record the skipped transaction */ if (skipped) { /* no latency to record on skipped transactions */ stats->skipped++; return; }

/*

  • Record the number of retries regardless of whether the transaction was
  • successful or failed.

    • */ if (tries > 1) { stats->retries += (tries - 1); stats->retried++; }

switch (estatus) { /* Record the successful transaction */ case ESTATUS_NO_ERROR: stats->cnt++;

addToSimpleStats(&stats->latency, lat);

/* and possibly the same for schedule lag */ if (throttle_delay) addToSimpleStats(&stats->lag, lag); break;

/* Record the failed transaction */ case ESTATUS_SERIALIZATION_ERROR: stats->serialization_failures++; break; case ESTATUS_DEADLOCK_ERROR: stats->deadlock_failures++; break; case ESTATUS_OTHER_SQL_ERROR: stats->other_sql_failures++; break; default: /* internal error which should never occur */ pg_fatal("unexpected error status: %d", estatus); } } ~~~
