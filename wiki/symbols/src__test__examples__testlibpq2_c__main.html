  • conninfo string; otherwise default to setting dbname=postgres and using
  • environment variables or defaults for all other connection parameters.

    • */ if (argc > 1) conninfo = argv[1]; else conninfo = "dbname = postgres";

/* Make a connection to the database */ conn = PQconnectdb(conninfo);

/* Check to see that the backend connection was successfully made */ if (PQstatus(conn) != CONNECTION_OK) { fprintf(stderr, "%s", PQerrorMessage(conn)); exit_nicely(conn); }

/* Set always-secure search path, so malicious users can't take control. */ res = PQexec(conn, "SELECT pg_catalog.set_config('search_path', '', false)"); if (PQresultStatus(res) != PGRES_TUPLES_OK) { fprintf(stderr, "SET failed: %s", PQerrorMessage(conn)); PQclear(res); exit_nicely(conn); }

/*

  • Should PQclear PGresult whenever it is no longer needed to avoid memory
  • leaks

    • */ PQclear(res);
