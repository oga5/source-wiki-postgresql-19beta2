    • int tupArrSize; /* allocated size of tuples array */ int numParameters; PGresParamDesc *paramDescs; ExecStatusType resultStatus; char cmdStatus[CMDSTATUS_LEN]; /* cmd status from the query */ int binary; /* binary tuple values if binary == 1,

  • These fields are copied from the originating PGconn, so that operations
  • on the PGresult don't have to reference the PGconn.

    • */ PGNoticeHooks noticeHooks; PGEvent *events; int nEvents; int client_encoding; /* encoding id */

/*

  • Error information (all NULL if not an error result). errMsg is the
  • "overall" error message returned by PQresultErrorMessage. If we have
  • per-field info then it is stored in a linked list.

    • */ char *errMsg; /* error message, or NULL if no error */ PGMessageField *errFields; /* message broken into fields */ char *errQuery; /* text of triggering query, if available */

/* All NULL attributes in the query result point to this null string */ char null_field[1];

/*

  • Space management information. Note that attDescs and error stuff, if
  • not null, point into allocated blocks. But tuples points to a
  • separately malloc'd block, so that we can realloc it.

    • */ PGresult_data *curBlock; /* most recently allocated block */ int curOffset; /* start offset of free space in block */ int spaceLeft; /* number of free bytes remaining in block */
