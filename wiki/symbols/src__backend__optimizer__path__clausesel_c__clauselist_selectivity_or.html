/*

listidx++;

/*

  • Skip this clause if it's already been estimated by some other
  • statistics above.

    • */ if (bms_is_member(listidx, estimatedclauses)) continue;

s2 = clause_selectivity_ext(root, (Node *) lfirst(lc), varRelid, jointype, sjinfo, use_extended_stats);

s1 = s1 + s2 - s1 * s2; }
