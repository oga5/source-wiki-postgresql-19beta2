~~~c static int init_litdata_packet(PushFilter **pf_res, PGP_Context *ctx, PushFilter *dst) { int res; int hdrlen; uint8 hdr[6]; uint32 t; PushFilter *pkt; int type;

/*

  • Create header

    • */

if (ctx->text_mode) type = ctx->unicode_mode ? 'u' : 't'; else type = 'b';

/*

  • Store the creation time into packet. The goal is to have as few known
  • bytes as possible.

    • */ t = (uint32) time(NULL);

hdr[0] = type; hdr[1] = 0; hdr[2] = (t >> 24) & 255; hdr[3] = (t >> 16) & 255; hdr[4] = (t >> 8) & 255; hdr[5] = t & 255; hdrlen = 6;

res = write_tag_only(dst, PGP_PKT_LITERAL_DATA); if (res < 0) return res;

res = pushf_create(&pkt, &pkt_stream_filter, ctx, dst); if (res < 0) return res;

res = pushf_write(pkt, hdr, hdrlen); if (res < 0) { pushf_free(pkt); return res; }

*pf_res = pkt; return 0; } ~~~
