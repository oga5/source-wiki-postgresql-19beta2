~~~c static ObjectAddress ATExecDropInherit(Relation rel, RangeVar *parent, LOCKMODE lockmode) { ObjectAddress address; Relation parent_rel;

/*

  • AccessShareLock on the parent is probably enough, seeing that DROP
  • TABLE doesn't lock parent tables at all. We need some lock since we'll
  • be inspecting the parent's schema.

    • */ parent_rel = table_openrv(parent, AccessShareLock);

/*

  • We don't bother to check ownership of the parent table --- ownership of
  • the child is presumed enough rights.

    • */

/* Off to RemoveInheritance() where most of the work happens */ RemoveInheritance(rel, parent_rel, false);

ObjectAddressSet(address, RelationRelationId, RelationGetRelid(parent_rel));

/* keep our lock on the parent relation until commit */ table_close(parent_rel, NoLock);

return address; } ~~~
