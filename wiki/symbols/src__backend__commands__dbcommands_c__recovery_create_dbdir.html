~~~c static void recovery_create_dbdir(char *path, bool only_tblspc) { struct stat st;

Assert(RecoveryInProgress());

if (stat(path, &st) == 0) return;

if (only_tblspc && strstr(path, PG_TBLSPC_DIR_SLASH) == NULL) elog(PANIC, "requested to created invalid directory: %s", path);

if (reachedConsistency && !allow_in_place_tablespaces) ereport(PANIC, errmsg("missing directory \"%s\"", path));

elog(reachedConsistency ? WARNING : DEBUG1, "creating missing directory: %s", path);

if (pg_mkdir_p(path, pg_dir_create_mode) != 0) ereport(PANIC, errmsg("could not create missing directory \"%s\": %m", path)); } ~~~
