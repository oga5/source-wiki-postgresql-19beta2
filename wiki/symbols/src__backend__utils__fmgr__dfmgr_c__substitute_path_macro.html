~~~c char * substitute_path_macro(const char *str, const char *macro, const char *value) { const char *sep_ptr;

Assert(str != NULL); Assert(macro[0] == '$');

/* Currently, we only recognize $macro at the start of the string */ if (str[0] != '$') return pstrdup(str);

if ((sep_ptr = first_dir_separator(str)) == NULL) sep_ptr = str + strlen(str);

if (strlen(macro) != sep_ptr - str || strncmp(str, macro, strlen(macro)) != 0) ereport(ERROR, (errcode(ERRCODE_INVALID_NAME), errmsg("invalid macro name in path: %s", str)));

return psprintf("%s%s", value, sep_ptr); } ~~~
