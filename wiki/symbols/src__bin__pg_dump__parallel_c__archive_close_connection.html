~~~c static void archive_close_connection(int code, void *arg) { ShutdownInformation *si = (ShutdownInformation *) arg;

if (si->pstate) { /* In parallel mode, must figure out who we are */ ParallelSlot *slot = GetMyPSlot(si->pstate);

if (!slot) { /*

  • We're the leader. Forcibly shut down workers, then close our
  • own database connection, if any.

    • */ ShutdownWorkersHard(si->pstate);

if (si->AHX) DisconnectDatabase(si->AHX); } else { /*

  • We're a worker. Shut down our own DB connection if any. On
  • Windows, we also have to close our communication sockets, to
  • emulate what will happen on Unix when the worker process exits.
  • (Without this, if this is a premature exit, the leader would
  • fail to detect it because there would be no EOF condition on
  • the other end of the pipe.)

    • */ if (slot->AH) DisconnectDatabase(&(slot->AH->public));

#ifdef WIN32 closesocket(slot->pipeRevRead); closesocket(slot->pipeRevWrite); #endif } } else { /* Non-parallel operation: just kill the leader DB connection */ if (si->AHX) DisconnectDatabase(si->AHX); } } ~~~
