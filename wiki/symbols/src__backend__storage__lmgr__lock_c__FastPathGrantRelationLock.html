/* Scan for existing entry for this relid, remembering empty slot. */ for (i = 0; i < FP_LOCK_SLOTS_PER_GROUP; i++) { /* index into the whole per-backend array */ uint32 f = FAST_PATH_SLOT(group, i);

/* If no existing entry, use any empty slot. */ if (unused_slot < FastPathLockSlotsPerBackend()) { MyProc->fpRelId[unused_slot] = relid; FAST_PATH_SET_LOCKMODE(MyProc, unused_slot, lockmode); ++FastPathLocalUseCounts[group]; return true; }
