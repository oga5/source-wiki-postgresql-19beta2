~~~c void RelationDropStorage(Relation rel) { PendingRelDelete *pending;

/* Add the relation to the list of stuff to delete at commit */ pending = (PendingRelDelete *) MemoryContextAlloc(TopMemoryContext, sizeof(PendingRelDelete)); pending->rlocator = rel->rd_locator; pending->procNumber = rel->rd_backend; pending->atCommit = true; /* delete if commit */ pending->nestLevel = GetCurrentTransactionNestLevel(); pending->next = pendingDeletes; pendingDeletes = pending;

/*

  • NOTE: if the relation was created in this transaction, it will now be
  • present in the pending-delete list twice, once with atCommit true and
  • once with atCommit false. Hence, it will be physically deleted at end
  • of xact in either case (and the other entry will be ignored by
  • smgrDoPendingDeletes, so no error will occur). We could instead remove
  • the existing list entry and delete the physical file immediately, but
  • for now I'll keep the logic simple.

    • */

RelationCloseSmgr(rel); } ~~~
