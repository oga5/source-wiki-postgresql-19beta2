/* Open the directory unless the caller did it. */ if (dir == NULL && ((dir = opendir(fullpath)) == NULL)) { report_backup_error(context, "could not open directory \"%s\": %m", fullpath); simple_string_list_append(&context->ignore_list, relpath);

if (relpath == NULL) newrelpath = pstrdup(filename); else newrelpath = psprintf("%s/%s", relpath, filename);
