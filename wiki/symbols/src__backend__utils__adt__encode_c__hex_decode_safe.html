/*

  • We must process 2 vectors at a time since the output will be half the
  • length of the input.

    • */ for (i = 0; i < tail_idx; i += sizeof(Vector8) * 2) { Vector8 srcv; Vector8 dstv1; Vector8 dstv2;

vector8_load(&srcv, (const uint8 *) &src[i]); success &= hex_decode_simd_helper(srcv, &dstv1);

vector8_load(&srcv, (const uint8 *) &src[i + sizeof(Vector8)]); success &= hex_decode_simd_helper(srcv, &dstv2);

vector8_store((uint8 *) &dst[i / 2], vector8_pack_16(dstv1, dstv2)); }

/*

  • If something didn't look right in the vector path, try again in the
  • scalar path so that we can handle it correctly.

    • */ if (!success) i = 0;

return i / 2 + hex_decode_safe_scalar(src + i, len - i, dst + i / 2, escontext); #endif } ~~~
