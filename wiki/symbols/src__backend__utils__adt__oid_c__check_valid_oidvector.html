~~~c void check_valid_oidvector(const oidvector *oidArray) { /*

  • We insist on ndim == 1 and dataoffset == 0 (that is, no nulls) because
  • otherwise the array's layout will not be what calling code expects. We
  • needn't be picky about the index lower bound though. Checking elemtype
  • is just paranoia.

    • */ if (oidArray->ndim != 1 || oidArray->dataoffset != 0 || oidArray->elemtype != OIDOID) ereport(ERROR, (errcode(ERRCODE_DATATYPE_MISMATCH), errmsg("array is not a valid oidvector"))); } ~~~
