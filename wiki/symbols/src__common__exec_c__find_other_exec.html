if (find_my_exec(argv0, retpath) < 0) return -1;

/* Trim off program name and keep just directory */ *last_dir_separator(retpath) = '\0'; canonicalize_path(retpath);

/* Now append the other program's name */ snprintf(retpath + strlen(retpath), MAXPGPATH - strlen(retpath), "/%s%s", target, EXE);

if (validate_exec(retpath) != 0) return -1;

snprintf(cmd, sizeof(cmd), "\"%s\" -V", retpath);

if ((line = pipe_read_line(cmd)) == NULL) return -1;

if (strcmp(line, versionstr) != 0) { pfree(line); return -2; }

pfree(line); return 0; } ~~~
