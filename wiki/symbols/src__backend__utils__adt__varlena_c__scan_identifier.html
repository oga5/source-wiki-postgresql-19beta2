~~~c char * scan_identifier(char **endp, char **nextp, char separator, bool downcase_unquoted) { char *token;

if (**nextp == '"') return scan_quoted_identifier(endp, nextp);

/* Unquoted identifier --- extends to separator or whitespace */ token = *nextp;

while (**nextp && **nextp != separator && !scanner_isspace(**nextp)) (*nextp)++;

if (*nextp == token) return NULL; /* empty token */

*endp = *nextp;

if (downcase_unquoted) { /*

  • Downcase the identifier, using same code as main lexer does.

    • *

  • XXX because we want to overwrite the input in-place, we cannot
  • support a downcasing transformation that increases the string
  • length. This is not a problem given the current implementation of
  • downcase_truncate_identifier, but we'll probably have to do
  • something about this someday.

    • */ int len = *endp - token; char *downname = downcase_truncate_identifier(token, len, false);

Assert(strlen(downname) <= len); strncpy(token, downname, len); /* strncpy is required here */ pfree(downname); }

return token; } ~~~
