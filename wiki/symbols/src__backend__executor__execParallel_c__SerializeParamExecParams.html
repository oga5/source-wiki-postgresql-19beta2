/* First write the number of parameters as a 4-byte integer. */ nparams = bms_num_members(params); memcpy(start_address, &nparams, sizeof(int)); start_address += sizeof(int);

/* Write details for each parameter in turn. */ paramid = -1; while ((paramid = bms_next_member(params, paramid)) >= 0) { Oid typeOid; int16 typLen; bool typByVal;

prm = &(estate->es_param_exec_vals[paramid]); typeOid = list_nth_oid(estate->es_plannedstmt->paramExecTypes, paramid);

/* Write paramid. */ memcpy(start_address, &paramid, sizeof(int)); start_address += sizeof(int);

/* Write datum/isnull */ if (OidIsValid(typeOid)) get_typlenbyval(typeOid, &typLen, &typByVal); else { /* If no type OID, assume by-value, like copyParamList does. */ typLen = sizeof(Datum); typByVal = true; } datumSerialize(prm->value, prm->isnull, typByVal, typLen, &start_address); }

return handle; } ~~~
