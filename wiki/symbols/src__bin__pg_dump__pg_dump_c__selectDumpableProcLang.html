~~~c static void selectDumpableProcLang(ProcLangInfo *plang, Archive *fout) { if (checkExtensionMembership(&plang->dobj, fout)) return; /* extension membership overrides all else */

/*

  • Only include procedural languages when we are dumping everything.

    • *

  • For from-initdb procedural languages, only include ACLs, as we do for
  • the pg_catalog namespace. We need this because procedural languages do
  • not live in any namespace.

    • */ if (!fout->dopt->include_everything) plang->dobj.dump = DUMP_COMPONENT_NONE; else { if (plang->dobj.catId.oid <= g_last_builtin_oid) plang->dobj.dump = fout->remoteVersion < 90600 ? DUMP_COMPONENT_NONE : DUMP_COMPONENT_ACL; else plang->dobj.dump = DUMP_COMPONENT_ALL; } } ~~~
