/* Save our memory context for later use by client API calls. */ ValidatorMemoryContext = CurrentMemoryContext;

ctx->state = OAUTH_STATE_INIT; ctx->port = port;

Assert(port->hba); ctx->issuer = port->hba->oauth_issuer; ctx->scope = port->hba->oauth_scope;

load_validator_library(port->hba->oauth_validator);

return ctx; } ~~~
