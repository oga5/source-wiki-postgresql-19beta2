/* Nothing to do if no extensions */ if (numExtensions == 0) return;

query = createPQExpBuffer();

/* refclassid constraint is redundant but may speed the search */ appendPQExpBufferStr(query, "SELECT " "classid, objid, refobjid " "FROM pg_depend " "WHERE refclassid = 'pg_extension'::regclass " "AND deptype = 'e' " "ORDER BY 3");

res = ExecuteSqlQuery(fout, query->data, PGRES_TUPLES_OK);

ntups = PQntuples(res);

i_classid = PQfnumber(res, "classid"); i_objid = PQfnumber(res, "objid"); i_refobjid = PQfnumber(res, "refobjid");

/*

  • Since we ordered the SELECT by referenced ID, we can expect that

  • saves on searches.
