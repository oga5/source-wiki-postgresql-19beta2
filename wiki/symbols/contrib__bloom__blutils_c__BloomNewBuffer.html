/* First, try to get a page from FSM */ for (;;) { BlockNumber blkno = GetFreeIndexPage(index);

if (blkno == InvalidBlockNumber) break;

buffer = ReadBuffer(index, blkno);

/*

  • We have to guard against the possibility that someone else already
  • recycled this page; the buffer may be locked if so.

    • */ if (ConditionalLockBuffer(buffer)) { Page page = BufferGetPage(buffer);

if (PageIsNew(page)) return buffer; /* OK to use, if never initialized */

if (BloomPageIsDeleted(page)) return buffer; /* OK to use */

LockBuffer(buffer, BUFFER_LOCK_UNLOCK); }

/* Can't use it, so release buffer and try again */ ReleaseBuffer(buffer); }
