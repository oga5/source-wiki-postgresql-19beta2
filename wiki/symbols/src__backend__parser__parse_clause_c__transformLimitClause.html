~~~c Node * transformLimitClause(ParseState *pstate, Node *clause, ParseExprKind exprKind, const char *constructName, LimitOption limitOption) { Node *qual;

if (clause == NULL) return NULL;

qual = transformExpr(pstate, clause, exprKind);

qual = coerce_to_specific_type(pstate, qual, INT8OID, constructName);

/* LIMIT can't refer to any variables of the current query */ checkExprIsVarFree(pstate, qual, constructName);

/*

  • Don't allow NULLs in FETCH FIRST .. WITH TIES. This test is ugly and
  • extremely simplistic, in that you can pass a NULL anyway by hiding it
  • inside an expression -- but this protects ruleutils against emitting an
  • unadorned NULL that's not accepted back by the grammar.

    • */ if (exprKind == EXPR_KIND_LIMIT && limitOption == LIMIT_OPTION_WITH_TIES && IsA(clause, A_Const) && castNode(A_Const, clause)->isnull) ereport(ERROR, (errcode(ERRCODE_INVALID_ROW_COUNT_IN_LIMIT_CLAUSE), errmsg("row count cannot be null in FETCH FIRST ... WITH TIES clause")));

return qual; } ~~~
