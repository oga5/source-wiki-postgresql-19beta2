  • We allow a NULL tupledesc for relations not expected to have missing
  • values, such as catalog relations and indexes.

    • */ Assert(!tupleDesc || attnum <= tupleDesc->natts); if (attnum > (int) HeapTupleHeaderGetNatts(tup->t_data)) { if (tupleDesc && TupleDescCompactAttr(tupleDesc, attnum - 1)->atthasmissing) return false; else return true; }
