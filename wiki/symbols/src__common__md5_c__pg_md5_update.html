if (len >= gap) { memmove(ctx->md5_buf + ctx->md5_i, data, gap); md5_calc(ctx->md5_buf, ctx);

ctx->md5_i = len - i; memmove(ctx->md5_buf, data + i, ctx->md5_i); } else { memmove(ctx->md5_buf + ctx->md5_i, data, len); ctx->md5_i += len; } } ~~~
