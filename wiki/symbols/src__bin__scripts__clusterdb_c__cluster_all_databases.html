if (tables->head != NULL) { SimpleStringListCell *cell;

for (cell = tables->head; cell; cell = cell->next) cluster_one_database(cparams, cell->val, progname, verbose, echo); } else cluster_one_database(cparams, NULL, progname, verbose, echo); }
