~~~c static int get_min_scale(NumericVar *var) { int min_scale; int last_digit_pos;

/*

  • Ordinarily, the input value will be "stripped" so that the last
  • NumericDigit is nonzero. But we don't want to get into an infinite
  • loop if it isn't, so explicitly find the last nonzero digit.

    • */ last_digit_pos = var->ndigits - 1; while (last_digit_pos >= 0 && var->digits[last_digit_pos] == 0) last_digit_pos--;

if (last_digit_pos >= 0) { /* compute min_scale assuming that last ndigit has no zeroes */ min_scale = (last_digit_pos - var->weight) * DEC_DIGITS;

/*

  • We could get a negative result if there are no digits after the
  • decimal point. In this case the min_scale must be zero.

    • */ if (min_scale > 0) { /*

  • Reduce min_scale if trailing digit(s) in last NumericDigit are
  • zero.

    • */ NumericDigit last_digit = var->digits[last_digit_pos];

while (last_digit % 10 == 0) { min_scale--; last_digit /= 10; } } else min_scale = 0; } else min_scale = 0; /* result if input is zero */

return min_scale; } ~~~
