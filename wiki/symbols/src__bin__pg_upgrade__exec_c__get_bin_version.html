~~~c static void get_bin_version(ClusterInfo *cluster) { char cmd[MAXPGPATH], cmd_output[MAX_STRING]; FILE *output; int rc; int v1 = 0, v2 = 0;

snprintf(cmd, sizeof(cmd), "\"%s/pg_ctl\" --version", cluster->bindir); fflush(NULL);

if ((output = popen(cmd, "r")) == NULL || fgets(cmd_output, sizeof(cmd_output), output) == NULL) pg_fatal("could not get pg_ctl version data using %s: %m", cmd);

rc = pclose(output); if (rc != 0) pg_fatal("could not get pg_ctl version data using %s: %s", cmd, wait_result_to_str(rc));

if (sscanf(cmd_output, "%*s %*s %d.%d", &v1, &v2) < 1) pg_fatal("could not get pg_ctl version output from %s", cmd);

if (v1 < 10) { /* old style, e.g. 9.6.1 */ cluster->bin_version = v1 * 10000 + v2 * 100; } else { /* new style, e.g. 10.1 */ cluster->bin_version = v1 * 10000; } } ~~~
