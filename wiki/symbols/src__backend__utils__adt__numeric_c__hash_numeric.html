/*

  • Omit any leading or trailing zeros from the input to the hash. The
  • numeric implementation *should* guarantee that leading and trailing
  • zeros are suppressed, but we're paranoid. Note that we measure the
  • starting and ending offsets in units of NumericDigits, not bytes.

    • */ digits = NUMERIC_DIGITS(key); for (i = 0; i < NUMERIC_NDIGITS(key); i++) { if (digits[i] != (NumericDigit) 0) break;

start_offset++;

/*

  • The weight is effectively the # of digits before the decimal point,
  • so decrement it for each leading zero we skip.

    • */ weight--; }

/*

  • If there are no non-zero digits, then the value of the number is zero,
  • regardless of any other fields.

    • */ if (NUMERIC_NDIGITS(key) == start_offset) PG_RETURN_UINT32(-1);

for (i = NUMERIC_NDIGITS(key) - 1; i >= 0; i--) { if (digits[i] != (NumericDigit) 0) break;

end_offset++; }

/* If we get here, there should be at least one non-zero digit */ Assert(start_offset + end_offset < NUMERIC_NDIGITS(key));
