/* Find the query level the ReturningExpr belongs to */ for (levelsup = rexpr->retlevelsup; levelsup > 0; levelsup--) root = root->parent_root;

/*

  • ReturningExprs. Just make a new slot every time.

retval = makeNode(Param); retval->paramkind = PARAM_EXEC; retval->paramid = pitem->paramId; retval->paramtype = ptype; retval->paramtypmod = exprTypmod((Node *) rexpr->retexpr); retval->paramcollid = exprCollation((Node *) rexpr->retexpr); retval->location = exprLocation((Node *) rexpr->retexpr);
