already_attached = dsa_is_attached(handle); if (already_attached) segment = dsm_find_mapping(handle); else segment = dsm_attach(handle);

if (segment == NULL) ereport(ERROR, (errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE), errmsg("could not attach to dynamic shared area")));

control = (dsa_area_control *) dsm_segment_address(segment);

LWLockAcquire(&control->lock, LW_SHARED); size = control->total_segment_size; LWLockRelease(&control->lock);

if (!already_attached) dsm_detach(segment);

return size; } ~~~
