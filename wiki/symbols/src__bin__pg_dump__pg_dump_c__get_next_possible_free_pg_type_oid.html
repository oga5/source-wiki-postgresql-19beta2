ソース抜粋

~~~c static Oid get_next_possible_free_pg_type_oid(Archive *fout, PQExpBuffer upgrade_query) { /*

  • If the old version didn't assign an array type, but the new version
  • does, we must select an unused type OID to assign. This currently only
  • happens for domains, when upgrading pre-v11 to v11 and up.

    • *

  • Note: local state here is kind of ugly, but we must have some, since we
  • mustn't choose the same unused OID more than once.

    • */ static Oid next_possible_free_oid = FirstNormalObjectId; PGresult *res; bool is_dup;

do { ++next_possible_free_oid; printfPQExpBuffer(upgrade_query, "SELECT EXISTS(SELECT 1 " "FROM pg_catalog.pg_type " "WHERE oid = '%u'::pg_catalog.oid);", next_possible_free_oid); res = ExecuteSqlQueryForSingleRow(fout, upgrade_query->data); is_dup = (PQgetvalue(res, 0, 0)[0] == 't'); PQclear(res); } while (is_dup);

return next_possible_free_oid; } ~~~
