if (num_slots) { int i_slotname; int i_plugin; int i_twophase; int i_failover; int i_caught_up; int i_invalid;

slotinfos = pg_malloc_array(LogicalSlotInfo, num_slots);

i_slotname = PQfnumber(res, "slot_name"); i_plugin = PQfnumber(res, "plugin"); i_twophase = PQfnumber(res, "two_phase"); i_failover = PQfnumber(res, "failover"); i_caught_up = PQfnumber(res, "caught_up"); i_invalid = PQfnumber(res, "invalid");

for (int slotnum = 0; slotnum < num_slots; slotnum++) { LogicalSlotInfo *curr = &slotinfos[slotnum];

dbinfo->slot_arr.slots = slotinfos; dbinfo->slot_arr.nslots = num_slots; } ~~~
