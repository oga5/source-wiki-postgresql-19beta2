~~~c void vac_open_indexes(Relation relation, LOCKMODE lockmode, int *nindexes, Relation **Irel) { List *indexoidlist; ListCell *indexoidscan; int i;

Assert(lockmode != NoLock);

indexoidlist = RelationGetIndexList(relation);

/* allocate enough memory for all indexes */ i = list_length(indexoidlist);

if (i > 0) *Irel = (Relation *) palloc(i * sizeof(Relation)); else *Irel = NULL;

/* collect just the ready indexes */ i = 0; foreach(indexoidscan, indexoidlist) { Oid indexoid = lfirst_oid(indexoidscan); Relation indrel;

indrel = index_open(indexoid, lockmode); if (indrel->rd_index->indisready) (*Irel)[i++] = indrel; else index_close(indrel, lockmode); }

*nindexes = i;

list_free(indexoidlist); } ~~~
