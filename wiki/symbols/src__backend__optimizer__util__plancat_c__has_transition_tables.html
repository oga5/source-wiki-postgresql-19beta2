~~~c bool has_transition_tables(PlannerInfo *root, Index rti, CmdType event) { RangeTblEntry *rte = planner_rt_fetch(rti, root); Relation relation; TriggerDesc *trigDesc; bool result = false;

Assert(rte->rtekind == RTE_RELATION);

/* Currently foreign tables cannot have transition tables */ if (rte->relkind == RELKIND_FOREIGN_TABLE) return result;

/* Assume we already have adequate lock */ relation = table_open(rte->relid, NoLock);

trigDesc = relation->trigdesc; switch (event) { case CMD_INSERT: if (trigDesc && trigDesc->trig_insert_new_table) result = true; break; case CMD_UPDATE: if (trigDesc && (trigDesc->trig_update_old_table || trigDesc->trig_update_new_table)) result = true; break; case CMD_DELETE: if (trigDesc && trigDesc->trig_delete_old_table) result = true; break; /* There is no separate event for MERGE, only INSERT/UPDATE/DELETE */ case CMD_MERGE: result = false; break; default: elog(ERROR, "unrecognized CmdType: %d", (int) event); break; }

table_close(relation, NoLock); return result; } ~~~
