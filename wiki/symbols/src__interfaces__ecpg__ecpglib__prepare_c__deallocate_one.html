/* first deallocate the statement in the backend */ if (this->prepared) { char *text; PGresult *query;

if (text) { sprintf(text, "deallocate \"%s\"", this->name); query = PQexec(this->stmt->connection->connection, text); ecpg_free(text); if (ecpg_check_PQresult(query, lineno, this->stmt->connection->connection, this->stmt->compat)) { PQclear(query); r = true; } } }

/*

  • Just ignore all errors since we do not know the list of cursors we are
  • allowed to free. We have to trust the software.
