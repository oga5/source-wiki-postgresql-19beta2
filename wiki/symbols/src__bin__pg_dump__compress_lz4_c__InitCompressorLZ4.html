cs->compression_spec = compression_spec;

/*

  • Read operations have access to the whole input. No state needs to be
  • carried between calls.

    • */ if (cs->readF) return;

if (!LZ4State_compression_init(state)) pg_fatal("could not initialize LZ4 compression: %s", LZ4F_getErrorName(state->errcode));

cs->private_data = state; } ~~~
