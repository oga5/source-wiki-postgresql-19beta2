~~~c static void gist_indexsortbuild(GISTBuildState *state) { IndexTuple itup; GistSortedBuildLevelState *levelstate; BulkWriteBuffer rootbuf;

/* Reserve block 0 for the root page */ state->pages_allocated = 1;

state->bulkstate = smgr_bulk_start_rel(state->indexrel, MAIN_FORKNUM);

/* Allocate a temporary buffer for the first leaf page batch. */ levelstate = palloc0_object(GistSortedBuildLevelState); levelstate->pages[0] = palloc(BLCKSZ); levelstate->parent = NULL; gistinitpage(levelstate->pages[0], F_LEAF);

/*

  • Fill index pages with tuples in the sorted order.

    • */ while ((itup = tuplesort_getindextuple(state->sortstate, true)) != NULL) { gist_indexsortbuild_levelstate_add(state, levelstate, itup); MemoryContextReset(state->giststate->tempCxt); }

/*

  • Write out the partially full non-root pages.

    • *

  • Keep in mind that flush can build a new root. If number of pages is > 1
  • then new root is required.

    • */ while (levelstate->parent != NULL || levelstate->current_page != 0) { GistSortedBuildLevelState *parent;

gist_indexsortbuild_levelstate_flush(state, levelstate); parent = levelstate->parent; for (int i = 0; i < GIST_SORTED_BUILD_PAGE_NUM; i++) if (levelstate->pages[i]) pfree(levelstate->pages[i]); pfree(levelstate); levelstate = parent; }

/* Write out the root */ PageSetLSN(levelstate->pages[0], GistBuildLSN); rootbuf = smgr_bulk_get_buf(state->bulkstate); memcpy(rootbuf, levelstate->pages[0], BLCKSZ); smgr_bulk_write(state->bulkstate, GIST_ROOT_BLKNO, rootbuf, true);

pfree(levelstate);

smgr_bulk_finish(state->bulkstate); } ~~~
