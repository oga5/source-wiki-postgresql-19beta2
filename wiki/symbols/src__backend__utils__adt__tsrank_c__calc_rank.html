~~~c static float calc_rank(const float *w, TSVector t, TSQuery q, int32 method) { QueryItem *item = GETQUERY(q); float res = 0.0; int len;

if (!t->size || !q->size) return 0.0;

/* XXX: What about NOT? */ res = (item->type == QI_OPR && (item->qoperator.oper == OP_AND || item->qoperator.oper == OP_PHRASE)) ? calc_rank_and(w, t, q) : calc_rank_or(w, t, q);

if (res < 0) res = 1e-20f;

if ((method & RANK_NORM_LOGLENGTH) && t->size > 0) res /= log((double) (cnt_length(t) + 1)) / log(2.0);

if (method & RANK_NORM_LENGTH) { len = cnt_length(t); if (len > 0) res /= (float) len; }

/* RANK_NORM_EXTDIST not applicable */

if ((method & RANK_NORM_UNIQ) && t->size > 0) res /= (float) (t->size);

if ((method & RANK_NORM_LOGUNIQ) && t->size > 0) res /= log((double) (t->size + 1)) / log(2.0);

if (method & RANK_NORM_RDIVRPLUS1) res /= (res + 1);

return res; } ~~~
