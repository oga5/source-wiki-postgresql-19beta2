~~~c void PGSemaphoreInit(int maxSemas) { struct stat statbuf;

/*

  • We use the data directory's inode number to seed the search for free
  • semaphore keys. This minimizes the odds of collision with other
  • postmasters, while maximizing the odds that we will detect and clean up
  • semaphores left over from a crashed postmaster in our own directory.

    • */ if (stat(DataDir, &statbuf) < 0) ereport(FATAL, (errcode_for_file_access(), errmsg("could not stat data directory \"%s\": %m", DataDir)));

numSharedSemas = 0; maxSharedSemas = maxSemas;

maxSemaSets = (maxSemas + SEMAS_PER_SET - 1) / SEMAS_PER_SET; mySemaSets = (IpcSemaphoreId *) malloc(maxSemaSets * sizeof(IpcSemaphoreId)); if (mySemaSets == NULL) elog(PANIC, "out of memory"); numSemaSets = 0; nextSemaKey = statbuf.st_ino; nextSemaNumber = SEMAS_PER_SET; /* force sema set alloc on 1st call */

on_shmem_exit(ReleaseSemaphores, 0); } ~~~
