GetPublicationsStr(publications, &cmd, true); appendStringInfoString(&cmd, ")\n");

  • In case of ALTER SUBSCRIPTION ... REFRESH PUBLICATION,

  • not be re-synced.

appendStringInfo(&cmd, "AND NOT (N.nspname = %s AND C.relname = %s)\n", schemaname_lit, seqname_lit);

pfree(schemaname_lit); pfree(seqname_lit); }
