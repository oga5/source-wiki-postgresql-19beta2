~~~c static bool checkLevel(lquery_level *curq, ltree_level *curt) { lquery_variant *curvar = LQL_FIRST(curq); bool success;

success = (curq->flag & LQL_NOT) ? false : true;

/* numvar == 0 means '*' which matches anything */ if (curq->numvar == 0) return success;

for (int i = 0; i < curq->numvar; i++) { bool prefix = (curvar->flag & LVAR_ANYEND); bool ci = (curvar->flag & LVAR_INCASE);

if (curvar->flag & LVAR_SUBLEXEME) { if (compare_subnode(curt, curvar->name, curvar->len, prefix, ci)) return success; } else if (ltree_label_match(curvar->name, curvar->len, curt->name, curt->len, prefix, ci)) return success;

curvar = LVAR_NEXT(curvar); } return !success; } ~~~
