/* Identify all the rel's columns that appear in grouping_columns */ groupbyattnos = NULL; foreach(gl, grouping_columns) { Var *gvar = (Var *) lfirst(gl);

if (bms_is_subset(pkattnos, groupbyattnos)) { /* The PK is a subset of grouping_columns, so we win */ *constraintDeps = lappend_oid(*constraintDeps, constraintOid); return true; }
