~~~c SH_SCOPE size_t SH_ESTIMATE_SPACE(double nentries) { uint64 size; uint64 space;

/* scale request by SH_FILLFACTOR, as SH_CREATE does */ nentries = nentries / SH_FILLFACTOR;

/* fail if we'd overrun SH_MAX_SIZE entries */ if (nentries >= SH_MAX_SIZE) return SIZE_MAX;

/* should be safe to convert to uint64 */ size = (uint64) nentries;

/* supporting zero sized hashes would complicate matters */ size = Max(size, 2);

/* round up size to the next power of 2, that's how bucketing works */ size = pg_nextpower2_64(size);

/* calculate space needed for ->data */ space = ((uint64) sizeof(SH_ELEMENT_TYPE)) * size;

/* verify that allocation of ->data is possible on this platform */ if (space >= SIZE_MAX / 2) return SIZE_MAX;

return (size_t) space + sizeof(SH_TYPE); } ~~~
