ソース抜粋

~~~c static bool apply_typmod_special(Numeric num, int32 typmod, Node *escontext) { int precision; int scale;

Assert(NUMERIC_IS_SPECIAL(num)); /* caller error if not */

/*

  • NaN is allowed regardless of the typmod; that's rather dubious perhaps,
  • but it's a longstanding behavior. Inf is rejected if we have any
  • typmod restriction, since an infinity shouldn't be claimed to fit in
  • any finite number of digits.

    • */ if (NUMERIC_IS_NAN(num)) return true;

/* Do nothing if we have a default typmod (-1) */ if (!is_valid_numeric_typmod(typmod)) return true;

precision = numeric_typmod_precision(typmod); scale = numeric_typmod_scale(typmod);

ereturn(escontext, false, (errcode(ERRCODE_NUMERIC_VALUE_OUT_OF_RANGE), errmsg("numeric field overflow"), errdetail("A field with precision %d, scale %d cannot hold an infinite value.", precision, scale))); } ~~~
