/* Fail if not a binary opclause (probably shouldn't happen) */ if (list_length(args) != 2) return false;

left = (Node *) linitial(args); right = (Node *) lsecond(args);

/*

  • Examine both sides. Note that when varRelid is nonzero, Vars of other
  • relations will be treated as pseudoconstants.

    • */ examine_variable(root, left, varRelid, vardata); examine_variable(root, right, varRelid, &rdata);

/*

  • If one side is a variable and the other not, we win.

    • */ if (vardata->rel && rdata.rel == NULL) { *varonleft = true; *other = estimate_expression_value(root, rdata.var); /* Assume we need no ReleaseVariableStats(rdata) here */ return true; }

if (vardata->rel == NULL && rdata.rel) { *varonleft = false; *other = estimate_expression_value(root, vardata->var); /* Assume we need no ReleaseVariableStats(*vardata) here */ *vardata = rdata; return true; }

/* Oops, clause has wrong structure (probably var op var) */ ReleaseVariableStats(*vardata); ReleaseVariableStats(rdata);

return false; } ~~~
