~~~c IndexBulkDeleteResult * gistvacuumcleanup(IndexVacuumInfo *info, IndexBulkDeleteResult *stats) { /* No-op in ANALYZE ONLY mode */ if (info->analyze_only) return stats;

/*

  • If gistbulkdelete was called, we need not do anything, just return the
  • stats from the latest gistbulkdelete call. If it wasn't called, we
  • still need to do a pass over the index, to obtain index statistics.

    • */ if (stats == NULL) { stats = palloc0_object(IndexBulkDeleteResult); gistvacuumscan(info, stats, NULL, NULL); }

/*

  • It's quite possible for us to be fooled by concurrent page splits into
  • double-counting some index tuples, so disbelieve any total that exceeds
  • the underlying heap's count ... if we know that accurately. Otherwise
  • this might just make matters worse.

    • */ if (!info->estimated_count) { if (stats->num_index_tuples > info->num_heap_tuples) stats->num_index_tuples = info->num_heap_tuples; }

return stats; } ~~~
