~~~c static void read_dictionary(DictSyn *d, const char *filename) { char *real_filename = get_tsearch_config_filename(filename, "rules"); tsearch_readline_state trst; char *line; int cur = 0;

if (!tsearch_readline_begin(&trst, real_filename)) ereport(ERROR, (errcode(ERRCODE_CONFIG_FILE_ERROR), errmsg("could not open synonym file \"%s\": %m", real_filename)));

while ((line = tsearch_readline(&trst)) != NULL) { char *value; char *key; char *pos; char *end;

if (*line == '\0') continue;

value = str_tolower(line, strlen(line), DEFAULT_COLLATION_OID); pfree(line);

pos = value; while ((key = find_word(pos, &end)) != NULL) { /* Enlarge syn structure if full */ if (cur == d->len) { d->len = (d->len > 0) ? 2 * d->len : 16; if (d->syn) d->syn = repalloc_array(d->syn, Syn, d->len); else d->syn = palloc_array(Syn, d->len); }

/* Save first word only if we will match it */ if (pos != value || d->matchorig) { d->syn[cur].key = pnstrdup(key, end - key); d->syn[cur].value = pstrdup(value);

cur++; }

pos = end;

/* Don't bother scanning synonyms if we will not match them */ if (!d->matchsynonyms) break; }

pfree(value); }

tsearch_readline_end(&trst); ~~~
