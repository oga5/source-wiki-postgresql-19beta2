~~~c static char * get_base_conninfo(const char *conninfo, char **dbname) { PQExpBuffer buf; PQconninfoOption *conn_opts; PQconninfoOption *conn_opt; char *errmsg = NULL; char *ret;

conn_opts = PQconninfoParse(conninfo, &errmsg); if (conn_opts == NULL) { pg_log_error("could not parse connection string: %s", errmsg); PQfreemem(errmsg); return NULL; }

buf = createPQExpBuffer(); for (conn_opt = conn_opts; conn_opt->keyword != NULL; conn_opt++) { if (conn_opt->val != NULL && conn_opt->val[0] != '\0') { if (strcmp(conn_opt->keyword, "dbname") == 0) { if (dbname) *dbname = pg_strdup(conn_opt->val); continue; } appendConnStrItem(buf, conn_opt->keyword, conn_opt->val); } }

ret = pg_strdup(buf->data);

destroyPQExpBuffer(buf); PQconninfoFree(conn_opts);

return ret; } ~~~
