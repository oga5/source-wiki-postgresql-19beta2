/*

  • The secret is of form:

    • */ v = pstrdup(secret); scheme_str = strsep(&v, "$"); if (v == NULL) goto invalid_secret; iterations_str = strsep(&v, ":"); if (v == NULL) goto invalid_secret; salt_str = strsep(&v, "$"); if (v == NULL) goto invalid_secret; storedkey_str = strsep(&v, ":"); if (v == NULL) goto invalid_secret; serverkey_str = v;
