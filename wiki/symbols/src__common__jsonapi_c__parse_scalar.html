/* if no semantic function, just consume the token */ if (sfunc == NULL) return json_lex(lex);

val = ALLOC(len + 1); if (val == NULL) return JSON_OUT_OF_MEMORY;

memcpy(val, lex->token_start, len); val[len] = '\0'; }

/* consume the token */ result = json_lex(lex); if (result != JSON_SUCCESS) { FREE(val); return result; }
