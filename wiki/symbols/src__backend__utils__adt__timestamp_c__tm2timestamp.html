~~~c int tm2timestamp(struct pg_tm *tm, fsec_t fsec, int *tzp, Timestamp *result) { TimeOffset date; TimeOffset time;

/* Prevent overflow in Julian-day routines */ if (!IS_VALID_JULIAN(tm->tm_year, tm->tm_mon, tm->tm_mday)) { *result = 0; /* keep compiler quiet */ return -1; }

date = date2j(tm->tm_year, tm->tm_mon, tm->tm_mday) - POSTGRES_EPOCH_JDATE; time = time2t(tm->tm_hour, tm->tm_min, tm->tm_sec, fsec);

if (unlikely(pg_mul_s64_overflow(date, USECS_PER_DAY, result) || pg_add_s64_overflow(*result, time, result))) { *result = 0; /* keep compiler quiet */ return -1; } if (tzp != NULL) *result = dt2local(*result, -(*tzp));

/* final range check catches just-out-of-range timestamps */ if (!IS_VALID_TIMESTAMP(*result)) { *result = 0; /* keep compiler quiet */ return -1; }

return 0; } ~~~
