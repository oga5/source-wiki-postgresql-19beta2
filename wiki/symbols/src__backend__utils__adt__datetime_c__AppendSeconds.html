~~~c static char * AppendSeconds(char *cp, int sec, fsec_t fsec, int precision, bool fillzeros) { Assert(precision >= 0);

if (fillzeros) cp = pg_ultostr_zeropad(cp, abs(sec), 2); else cp = pg_ultostr(cp, abs(sec));

/* fsec_t is just an int32 */ if (fsec != 0) { int32 value = abs(fsec); char *end = &cp[precision + 1]; bool gotnonzero = false;

*cp++ = '.';

/*

  • Append the fractional seconds part. Note that we don't want any
  • trailing zeros here, so since we're building the number in reverse
  • we'll skip appending zeros until we've output a non-zero digit.

    • */ while (precision--) { int32 oldval = value; int32 remainder;

value /= 10; remainder = oldval - value * 10;

/* check if we got a non-zero */ if (remainder) gotnonzero = true;

if (gotnonzero) cp[precision] = '0' + remainder; else end = &cp[precision]; }

/*

  • If we still have a non-zero value then precision must have not been
  • enough to print the number. We punt the problem to pg_ultostr(),
  • which will generate a correct answer in the minimum valid width.

    • */ if (value) return pg_ultostr(cp, abs(fsec));

return end; } else return cp; } ~~~
