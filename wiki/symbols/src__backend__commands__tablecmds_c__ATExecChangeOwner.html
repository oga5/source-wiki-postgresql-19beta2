/*

  • Get exclusive lock till end of transaction on the target table. Use
  • relation_open so that we can work on indexes and sequences.

    • */ target_rel = relation_open(relationOid, lockmode);

/* Get its pg_class tuple, too */ class_rel = table_open(RelationRelationId, RowExclusiveLock);

/* Can we change the ownership of this tuple? */ switch (tuple_class->relkind) { case RELKIND_RELATION: case RELKIND_VIEW: case RELKIND_MATVIEW: case RELKIND_FOREIGN_TABLE: case RELKIND_PARTITIONED_TABLE: case RELKIND_PROPGRAPH: /* ok to change owner */ break; case RELKIND_INDEX: if (!recursing) { /*

  • Because ALTER INDEX OWNER used to be allowed, and in fact
  • is generated by old versions of pg_dump, we give a warning

  • nothing at all if the command would be a no-op anyway.
