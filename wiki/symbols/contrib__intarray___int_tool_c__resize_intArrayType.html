~~~c ArrayType * resize_intArrayType(ArrayType *a, int num) { int nbytes; int i;

/* if no elements, return a zero-dimensional array */ if (num <= 0) { Assert(num == 0); a = construct_empty_array(INT4OID); return a; }

if (num == ARRNELEMS(a)) return a;

nbytes = ARR_DATA_OFFSET(a) + sizeof(int) * num;

a = (ArrayType *) repalloc(a, nbytes);

SET_VARSIZE(a, nbytes); /* usually the array should be 1-D already, but just in case ... */ for (i = 0; i < ARR_NDIM(a); i++) { ARR_DIMS(a)[i] = num; num = 1; } return a; } ~~~
