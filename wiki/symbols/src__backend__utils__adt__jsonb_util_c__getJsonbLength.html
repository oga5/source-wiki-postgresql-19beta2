~~~c uint32 getJsonbLength(const JsonbContainer *jc, int index) { uint32 off; uint32 len;

/*

  • If the length is stored directly in the JEntry, just return it.
  • Otherwise, get the begin offset of the entry, and subtract that from
  • the stored end+1 offset.

    • */ if (JBE_HAS_OFF(jc->children[index])) { off = getJsonbOffset(jc, index); len = JBE_OFFLENFLD(jc->children[index]) - off; } else len = JBE_OFFLENFLD(jc->children[index]);

return len; } ~~~
