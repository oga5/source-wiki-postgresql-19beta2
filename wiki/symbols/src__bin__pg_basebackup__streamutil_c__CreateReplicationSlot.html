/* Add any requested options */ if (use_new_option_syntax) appendPQExpBufferStr(query, " ("); if (is_physical) { if (reserve_wal) AppendPlainCommandOption(query, use_new_option_syntax, "RESERVE_WAL"); } else { if (failover && PQserverVersion(conn) >= 170000) AppendPlainCommandOption(query, use_new_option_syntax, "FAILOVER");

if (two_phase && PQserverVersion(conn) >= 150000) AppendPlainCommandOption(query, use_new_option_syntax, "TWO_PHASE");

if (PQserverVersion(conn) >= 100000) { /* pg_recvlogical doesn't use an exported snapshot, so suppress */ if (use_new_option_syntax) AppendStringCommandOption(query, use_new_option_syntax, "SNAPSHOT", "nothing"); else AppendPlainCommandOption(query, use_new_option_syntax, "NOEXPORT_SNAPSHOT"); } } ~~~
