ソース抜粋

~~~c int PQenterPipelineMode(PGconn *conn) { if (!conn) return 0;

/* succeed with no action if already in pipeline mode */ if (conn->pipelineStatus != PQ_PIPELINE_OFF) return 1;

if (conn->asyncStatus != PGASYNC_IDLE) { libpq_append_conn_error(conn, "cannot enter pipeline mode, connection not idle"); return 0; }

conn->pipelineStatus = PQ_PIPELINE_ON;

return 1; } ~~~
