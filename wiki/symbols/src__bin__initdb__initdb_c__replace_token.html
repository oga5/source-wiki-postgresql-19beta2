~~~c static char ** replace_token(char **lines, const char *token, const char *replacement) { int toklen, replen, diff;

toklen = strlen(token); replen = strlen(replacement); diff = replen - toklen;

for (int i = 0; lines[i]; i++) { char *where; char *endwhere; char *newline; int pre;

/* nothing to do if no change needed */ if ((where = strstr(lines[i], token)) == NULL) continue;

/*

  • Reject false match. Note a blind spot: we don't check for a valid
  • match following a false match. That case can't occur at present,
  • so not worth complicating this code for it.

    • */ if (!(where == lines[i] || isspace((unsigned char) where[-1]))) continue; endwhere = where + strlen(token); if (!(*endwhere == '\0' || isspace((unsigned char) *endwhere))) continue;

/* if we get here a change is needed - set up new line */

newline = (char *) pg_malloc(strlen(lines[i]) + diff + 1);

pre = where - lines[i];

memcpy(newline, lines[i], pre);

memcpy(newline + pre, replacement, replen);

strcpy(newline + pre + replen, lines[i] + pre + toklen);

free(lines[i]); lines[i] = newline; }

return lines; } ~~~
