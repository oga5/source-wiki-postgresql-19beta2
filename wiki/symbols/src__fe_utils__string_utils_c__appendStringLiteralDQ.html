~~~c void appendStringLiteralDQ(PQExpBuffer buf, const char *str, const char *dqprefix) { static const char suffixes[] = "_XXXXXXX"; int nextchar = 0; PQExpBuffer delimBuf = createPQExpBuffer();

/* start with $ + dqprefix if not NULL */ appendPQExpBufferChar(delimBuf, '$'); if (dqprefix) appendPQExpBufferStr(delimBuf, dqprefix);

/*

  • Make sure we choose a delimiter which (without the trailing $) is not
  • present in the string being quoted. We don't check with the trailing $
  • because a string ending in $foo must not be quoted with $foo$.

    • */ while (strstr(str, delimBuf->data) != NULL) { appendPQExpBufferChar(delimBuf, suffixes[nextchar++]); nextchar %= sizeof(suffixes) - 1; }

/* add trailing $ */ appendPQExpBufferChar(delimBuf, '$');

/* quote it and we are all done */ appendPQExpBufferStr(buf, delimBuf->data); appendPQExpBufferStr(buf, str); appendPQExpBufferStr(buf, delimBuf->data);

destroyPQExpBuffer(delimBuf); } ~~~
