~~~c Selectivity mcv_combine_selectivities(Selectivity simple_sel, Selectivity mcv_sel, Selectivity mcv_basesel, Selectivity mcv_totalsel) { Selectivity other_sel; Selectivity sel;

/* estimated selectivity of values not covered by MCV matches */ other_sel = simple_sel - mcv_basesel; CLAMP_PROBABILITY(other_sel);

/* this non-MCV selectivity cannot exceed 1 - mcv_totalsel */ if (other_sel > 1.0 - mcv_totalsel) other_sel = 1.0 - mcv_totalsel;

/* overall selectivity is the sum of the MCV and non-MCV parts */ sel = mcv_sel + other_sel; CLAMP_PROBABILITY(sel);

return sel; } ~~~
