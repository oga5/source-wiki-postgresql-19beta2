/* Compute hash value for scalarVal */ switch (scalarVal->type) { case jbvNull: tmp = 0x01; break; case jbvString: tmp = DatumGetUInt32(hash_any((const unsigned char *) scalarVal->val.string.val, scalarVal->val.string.len)); break; case jbvNumeric: /* Must hash equal numerics to equal hash codes */ tmp = DatumGetUInt32(DirectFunctionCall1(hash_numeric, NumericGetDatum(scalarVal->val.numeric))); break; case jbvBool: tmp = scalarVal->val.boolean ? 0x02 : 0x04;

break; default: elog(ERROR, "invalid jsonb scalar type"); tmp = 0; /* keep compiler quiet */ break; }
