  • First byte indicates standard SSL handshake message

  • that would be a startup packet hundreds of megabytes long)

    • */

#ifdef USE_SSL if (!LoadedSSL || port->laddr.addr.ss_family == AF_UNIX) { /* SSL not supported */ goto reject; }

  • already
