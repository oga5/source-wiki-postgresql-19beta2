~~~c static void dumpAccessMethod(Archive *fout, const AccessMethodInfo *aminfo) { DumpOptions *dopt = fout->dopt; PQExpBuffer q; PQExpBuffer delq; char *qamname;

/* Do nothing if not dumping schema */ if (!dopt->dumpSchema) return;

q = createPQExpBuffer(); delq = createPQExpBuffer();

qamname = pg_strdup(fmtId(aminfo->dobj.name));

appendPQExpBuffer(q, "CREATE ACCESS METHOD %s ", qamname);

switch (aminfo->amtype) { case AMTYPE_INDEX: appendPQExpBufferStr(q, "TYPE INDEX "); break; case AMTYPE_TABLE: appendPQExpBufferStr(q, "TYPE TABLE "); break; default: pg_log_warning("invalid type \"%c\" of access method \"%s\"", aminfo->amtype, qamname); destroyPQExpBuffer(q); destroyPQExpBuffer(delq); free(qamname); return; }

appendPQExpBuffer(q, "HANDLER %s;\n", aminfo->amhandler);

appendPQExpBuffer(delq, "DROP ACCESS METHOD %s;\n", qamname);

if (dopt->binary_upgrade) binary_upgrade_extension_member(q, &aminfo->dobj, "ACCESS METHOD", qamname, NULL);

if (aminfo->dobj.dump & DUMP_COMPONENT_DEFINITION) ArchiveEntry(fout, aminfo->dobj.catId, aminfo->dobj.dumpId, ARCHIVE_OPTS(.tag = aminfo->dobj.name, .description = "ACCESS METHOD", .section = SECTION_PRE_DATA, .createStmt = q->data, .dropStmt = delq->data));

/* Dump Access Method Comments */ if (aminfo->dobj.dump & DUMP_COMPONENT_COMMENT) dumpComment(fout, "ACCESS METHOD", qamname, NULL, "", aminfo->dobj.catId, 0, aminfo->dobj.dumpId);

destroyPQExpBuffer(q); ~~~
