~~~c static bool needs_toast_table(Relation rel) { /*

  • No need to create a TOAST table for partitioned tables.

    • */ if (rel->rd_rel->relkind == RELKIND_PARTITIONED_TABLE) return false;

/*

  • We cannot allow toasting a shared relation after initdb (because
  • there's no way to mark it toasted in other databases' pg_class).

    • */ if (rel->rd_rel->relisshared && !IsBootstrapProcessingMode()) return false;

/*

  • Ignore attempts to create toast tables on catalog tables after initdb.
  • Which catalogs get toast tables is explicitly chosen in catalog/pg_*.h.
  • (We could get here via some ALTER TABLE command if the catalog doesn't
  • have a toast table.)

    • */ if (IsCatalogRelation(rel) && !IsBootstrapProcessingMode()) return false;

/* Otherwise, let the AM decide. */ return table_relation_needs_toast_table(rel); } ~~~
