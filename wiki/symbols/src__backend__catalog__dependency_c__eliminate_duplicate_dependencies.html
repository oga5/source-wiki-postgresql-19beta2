~~~c static void eliminate_duplicate_dependencies(ObjectAddresses *addrs) { ObjectAddress *priorobj; int oldref, newrefs;

/*

  • We can't sort if the array has "extra" data, because there's no way to
  • keep it in sync. Fortunately that combination of features is not
  • needed.

    • */ Assert(!addrs->extras);

if (addrs->numrefs <= 1) return; /* nothing to do */

/* Sort the refs so that duplicates are adjacent */ qsort(addrs->refs, addrs->numrefs, sizeof(ObjectAddress), object_address_comparator);

/* Remove dups */ priorobj = addrs->refs; newrefs = 1; for (oldref = 1; oldref < addrs->numrefs; oldref++) { ObjectAddress *thisobj = addrs->refs + oldref;

if (priorobj->classId == thisobj->classId && priorobj->objectId == thisobj->objectId) { if (priorobj->objectSubId == thisobj->objectSubId) continue; /* identical, so drop thisobj */

/*

  • If we have a whole-object reference and a reference to a part
  • of the same object, we don't need the whole-object reference
  • (for example, we don't need to reference both table foo and
  • column foo.bar). The whole-object reference will always appear
  • first in the sorted list.

    • */ if (priorobj->objectSubId == 0) { /* replace whole ref with partial */ priorobj->objectSubId = thisobj->objectSubId; continue; } } /* Not identical, so add thisobj to output set */ priorobj++; *priorobj = *thisobj; newrefs++; }

addrs->numrefs = newrefs; } ~~~
