use_recovery_conf = PQserverVersion(pgconn) < MINIMUM_VERSION_FOR_RECOVERY_GUC;

snprintf(filename, MAXPGPATH, "%s/%s", target_dir, use_recovery_conf ? "recovery.conf" : "postgresql.auto.conf");

cf = fopen(filename, use_recovery_conf ? "w" : "a"); if (cf == NULL) pg_fatal("could not open file \"%s\": %m", filename);

if (fwrite(contents->data, contents->len, 1, cf) != 1) pg_fatal("could not write to file \"%s\": %m", filename);

fclose(cf);

if (!use_recovery_conf) { snprintf(filename, MAXPGPATH, "%s/%s", target_dir, "standby.signal"); cf = fopen(filename, "w"); if (cf == NULL) pg_fatal("could not create file \"%s\": %m", filename);

fclose(cf); } } ~~~
