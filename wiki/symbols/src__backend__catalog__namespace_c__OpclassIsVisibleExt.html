opctup = SearchSysCache1(CLAOID, ObjectIdGetDatum(opcid)); if (!HeapTupleIsValid(opctup)) { if (is_missing != NULL) { *is_missing = true; return false; } elog(ERROR, "cache lookup failed for opclass %u", opcid); } opcform = (Form_pg_opclass) GETSTRUCT(opctup);

recomputeNamespacePath();

/*

  • Quick check: if it ain't in the path at all, it ain't visible. Items in
  • the system namespace are surely in the path and so we needn't even do
  • list_member_oid() for them.

    • */ opcnamespace = opcform->opcnamespace; if (opcnamespace != PG_CATALOG_NAMESPACE && !list_member_oid(activeSearchPath, opcnamespace)) visible = false; else { /*

  • If it is in the path, it might still not be visible; it could be
  • hidden by another opclass of the same name earlier in the path. So
  • we must do a slow check to see if this opclass would be found by
  • OpclassnameGetOpcid.

    • */ char *opcname = NameStr(opcform->opcname);

visible = (OpclassnameGetOpcid(opcform->opcmethod, opcname) == opcid); }

ReleaseSysCache(opctup);

return visible; } ~~~
