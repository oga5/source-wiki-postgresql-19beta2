~~~c int ReadOffset(ArchiveHandle *AH, pgoff_t *o) { int i; int off; int offsetFlg;

/* Initialize to zero */ *o = 0;

/* Check for old version */ if (AH->version < K_VERS_1_7) { /* Prior versions wrote offsets using WriteInt */ i = ReadInt(AH); /* -1 means not set */ if (i < 0) return K_OFFSET_POS_NOT_SET; else if (i == 0) return K_OFFSET_NO_DATA;

/* Cast to pgoff_t because it was written as an int. */ *o = (pgoff_t) i; return K_OFFSET_POS_SET; }

/*

  • Read the flag indicating the state of the data pointer. Check if valid
  • and die if not.

    • *

  • This used to be handled by a negative or zero pointer, now we use an
  • extra byte specifically for the state.

    • */ offsetFlg = AH->ReadBytePtr(AH) & 0xFF;

switch (offsetFlg) { case K_OFFSET_POS_NOT_SET: case K_OFFSET_NO_DATA: case K_OFFSET_POS_SET:

break;

default: pg_fatal("unexpected data offset flag %d", offsetFlg); }

/*

  • Read the bytes

    • */ for (off = 0; off < AH->offSize; off++) { if (off < sizeof(pgoff_t)) *o |= ((pgoff_t) (AH->ReadBytePtr(AH))) << (off * 8); else { if (AH->ReadBytePtr(AH) != 0) pg_fatal("file offset in dump file is too large"); } } ~~~
