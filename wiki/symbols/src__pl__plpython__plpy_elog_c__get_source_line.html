~~~c static char * get_source_line(const char *src, int lineno) { const char *s = NULL; const char *next = src; int current = 0;

/* sanity check */ if (lineno <= 0) return NULL;

while (current < lineno) { s = next; next = strchr(s + 1, '\n'); current++; if (next == NULL) break; }

if (current != lineno) return NULL;

while (*s && isspace((unsigned char) *s)) s++;

if (next == NULL) return pstrdup(s);

/*

  • Sanity check, next < s if the line was all-whitespace, which should
  • never happen if Python reported a frame created on that line, but check
  • anyway.

    • */ if (next < s) return NULL;

return pnstrdup(s, next - s); } ~~~
