resultlen = enc->encode_len(dataptr, datalen);

/*

  • resultlen possibly overflows uint32, therefore on 32-bit machines it's
  • unsafe to rely on palloc's internal check.

    • */ if (resultlen > MaxAllocSize - VARHDRSZ) ereport(ERROR, (errcode(ERRCODE_PROGRAM_LIMIT_EXCEEDED), errmsg("result of encoding conversion is too large")));

result = palloc(VARHDRSZ + resultlen);

res = enc->encode(dataptr, datalen, VARDATA(result));

/* Make this FATAL 'cause we've trodden on memory ... */ if (res > resultlen) elog(FATAL, "overflow - encode estimate too small");
