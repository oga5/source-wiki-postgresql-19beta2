/* Expand "something.*" in SELECT and RETURNING, but not UPDATE */ expand_star = (exprKind != EXPR_KIND_UPDATE_SOURCE);

foreach(o_target, targetlist) { ResTarget *res = (ResTarget *) lfirst(o_target);

/*

  • Check for "something.*". Depending on the complexity of the
  • "something", the star could appear as the last field in ColumnRef,
  • or as the last indirection item in A_Indirection.

    • */ if (expand_star) { if (IsA(res->val, ColumnRef)) { ColumnRef *cref = (ColumnRef *) res->val;

if (IsA(llast(ind->indirection), A_Star)) { /* It is something.*, expand into multiple items */ p_target = list_concat(p_target, ExpandIndirectionStar(pstate, ind, true, exprKind)); continue; } } }
