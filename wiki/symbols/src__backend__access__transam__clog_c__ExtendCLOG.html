~~~c void ExtendCLOG(TransactionId newestXact) { int64 pageno; LWLock *lock;

/*

  • No work except at first XID of a page. But beware: just after
  • wraparound, the first XID of page zero is FirstNormalTransactionId.

    • */ if (TransactionIdToPgIndex(newestXact) != 0 && !TransactionIdEquals(newestXact, FirstNormalTransactionId)) return;

pageno = TransactionIdToPage(newestXact); lock = SimpleLruGetBankLock(XactCtl, pageno);

LWLockAcquire(lock, LW_EXCLUSIVE);

/* Zero the page and make a WAL entry about it */ SimpleLruZeroPage(XactCtl, pageno); XLogSimpleInsertInt64(RM_CLOG_ID, CLOG_ZEROPAGE, pageno);

LWLockRelease(lock); } ~~~
