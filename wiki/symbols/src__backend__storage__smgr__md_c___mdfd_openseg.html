~~~c static MdfdVec * _mdfd_openseg(SMgrRelation reln, ForkNumber forknum, BlockNumber segno, int oflags) { MdfdVec *v; File fd; MdPathStr fullpath;

fullpath = _mdfd_segpath(reln, forknum, segno);

/* open the file */ fd = PathNameOpenFile(fullpath.str, _mdfd_open_flags() | oflags);

if (fd < 0) return NULL;

/*

  • Segments are always opened in order from lowest to highest, so we must
  • be adding a new one at the end.

    • */ Assert(segno == reln->md_num_open_segs[forknum]);

_fdvec_resize(reln, forknum, segno + 1);

/* fill the entry */ v = &reln->md_seg_fds[forknum][segno]; v->mdfd_vfd = fd; v->mdfd_segno = segno;

Assert(_mdnblocks(reln, forknum, v) <= ((BlockNumber) RELSEG_SIZE));

/* all done */ return v; } ~~~
