~~~c static void bloom_filter_size(int ndistinct, double false_positive_rate, int *nbytesp, int *nbitsp, int *nhashesp) { double k; int nbits, nbytes;

/* sizing bloom filter: -(n * ln(p)) / (ln(2))^2 */ nbits = ceil(-(ndistinct * log(false_positive_rate)) / pow(log(2.0), 2));

/* round m to whole bytes */ nbytes = ((nbits + 7) / 8); nbits = nbytes * 8;

/*

  • round(log(2.0) * m / ndistinct), but assume round() may not be
  • available on Windows

    • */ k = log(2.0) * nbits / ndistinct; k = (k - floor(k) >= 0.5) ? ceil(k) : floor(k);

if (nbytesp) *nbytesp = nbytes;

if (nbitsp) *nbitsp = nbits;

if (nhashesp) *nhashesp = (int) k; } ~~~
