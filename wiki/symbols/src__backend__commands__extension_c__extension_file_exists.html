/*

  • If the control directory doesn't exist, we want to silently return
  • false. Any other error will be reported by ReadDir.

    • */ if (dir == NULL && errno == ENOENT) { /* do nothing */ } else { while ((de = ReadDir(dir, location->loc)) != NULL) { char *extname;

if (!is_extension_control_filename(de->d_name)) continue;

/* extract extension name from 'name.control' filename */ extname = pstrdup(de->d_name); *strrchr(extname, '.') = '\0';
