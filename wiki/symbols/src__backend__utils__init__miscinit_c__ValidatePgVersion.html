file = AllocateFile(full_path, "r"); if (!file) { if (errno == ENOENT) ereport(FATAL, (errcode(ERRCODE_INVALID_PARAMETER_VALUE), errmsg("\"%s\" is not a valid data directory", path), errdetail("File \"%s\" is missing.", full_path))); else ereport(FATAL, (errcode_for_file_access(), errmsg("could not open file \"%s\": %m", full_path))); }

file_version_string[0] = '\0'; ret = fscanf(file, "%63s", file_version_string); file_major = strtol(file_version_string, &endptr, 10);

FreeFile(file);
