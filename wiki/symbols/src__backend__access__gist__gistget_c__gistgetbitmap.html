if (!so->qual_ok) return 0;

pgstat_count_index_scan(scan->indexRelation); if (scan->instrument) scan->instrument->nsearches++;

/* Begin the scan by processing the root page */ so->curPageData = so->nPageData = 0; scan->xs_hitup = NULL; if (so->pageDataCxt) MemoryContextReset(so->pageDataCxt);

fakeItem.blkno = GIST_ROOT_BLKNO; memset(&fakeItem.data.parentlsn, 0, sizeof(GistNSN)); gistScanPage(scan, &fakeItem, NULL, tbm, &ntids);

/*

  • While scanning a leaf page, ItemPointers of matching heap tuples will
  • be stored directly into tbm, so we don't need to deal with them here.

    • */ for (;;) { GISTSearchItem *item = getNextGISTSearchItem(so);

if (!item) break;

CHECK_FOR_INTERRUPTS();

gistScanPage(scan, item, item->distances, tbm, &ntids);

pfree(item); }

return ntids; } ~~~
