~~~c static int CheckPasswordAuth(Port *port, const char **logdetail) { char *passwd; int result; char *shadow_pass; bool md5_password = false;

sendAuthRequest(port, AUTH_REQ_PASSWORD, NULL, 0);

passwd = recv_password_packet(port); if (passwd == NULL) return STATUS_EOF; /* client wouldn't send password */

shadow_pass = get_role_password(port->user_name, logdetail); if (shadow_pass) { result = plain_crypt_verify(port->user_name, shadow_pass, passwd, logdetail); md5_password = (get_password_type(shadow_pass) == PASSWORD_TYPE_MD5); } else result = STATUS_ERROR;

if (shadow_pass) pfree(shadow_pass); pfree(passwd);

if (result == STATUS_OK) { if (md5_password) queue_md5_password_warning(); set_authn_id(port, port->user_name); }

return result; } ~~~
