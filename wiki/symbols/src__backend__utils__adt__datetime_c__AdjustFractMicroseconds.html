~~~c static bool AdjustFractMicroseconds(double frac, int64 scale, struct pg_itm_in *itm_in) { int64 usec;

/* Fast path for common case */ if (frac == 0) return true;

/*

  • We assume the input frac has abs value less than 1, so overflow of frac
  • or usec is not an issue for interesting values of scale.

    • */ frac *= scale; usec = (int64) frac;

/* Round off any fractional microsecond */ frac -= usec; if (frac > 0.5) usec++; else if (frac < -0.5) usec--;

return !pg_add_s64_overflow(itm_in->tm_usec, usec, &itm_in->tm_usec); } ~~~
