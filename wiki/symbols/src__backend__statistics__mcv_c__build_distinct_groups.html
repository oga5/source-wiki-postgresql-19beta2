~~~c static SortItem * build_distinct_groups(int numrows, SortItem *items, MultiSortSupport mss, int *ndistinct) { int i, j; int ngroups = count_distinct_groups(numrows, items, mss);

SortItem *groups = (SortItem *) palloc(ngroups * sizeof(SortItem));

j = 0; groups[0] = items[0]; groups[0].count = 1;

for (i = 1; i < numrows; i++) { /* Assume sorted in ascending order. */ Assert(multi_sort_compare(&items[i], &items[i - 1], mss) >= 0);

/* New distinct group detected. */ if (multi_sort_compare(&items[i], &items[i - 1], mss) != 0) { groups[++j] = items[i]; groups[j].count = 0; }

groups[j].count++; }

/* ensure we filled the expected number of distinct groups */ Assert(j + 1 == ngroups);

/* Sort the distinct groups by frequency (in descending order). */ qsort_interruptible(groups, ngroups, sizeof(SortItem), compare_sort_item_count, NULL);

*ndistinct = ngroups; return groups; } ~~~
