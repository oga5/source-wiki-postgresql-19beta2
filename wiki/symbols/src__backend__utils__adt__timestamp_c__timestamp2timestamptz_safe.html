/* timestamp2tm should not fail on valid timestamps, but cope */ if (timestamp2tm(timestamp, NULL, tm, &fsec, NULL, NULL) == 0) { tz = DetermineTimeZoneOffset(tm, session_timezone);

if (IS_VALID_TIMESTAMP(result)) return result; }

if (timestamp < 0) TIMESTAMP_NOBEGIN(result); else TIMESTAMP_NOEND(result);
