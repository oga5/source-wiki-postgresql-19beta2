/* inserting transaction aborted */ if (HeapTupleHeaderXminInvalid(tuple)) { Assert(!TransactionIdDidCommit(xmin)); return false; } /* check if it's one of our txids, toplevel is also in there */ else if (TransactionIdInArray(xmin, snapshot->subxip, snapshot->subxcnt)) { bool resolved; CommandId cmin = HeapTupleHeaderGetRawCommandId(tuple); CommandId cmax = InvalidCommandId;

  • cmin/cmax was stored in a combo CID. So we need to lookup the
  • actual values externally.

    • */ resolved = ResolveCminCmaxDuringDecoding(HistoricSnapshotGetTupleCids(), snapshot, htup, buffer, &cmin, &cmax);

/*

  • If we haven't resolved the combo CID to cmin/cmax, that means we

    • *

  • regular logical decoding we only execute this code at commit time,

  • elog inside ResolveCminCmaxDuringDecoding.

  • assigned, and error out based on this (when unable to resolve combo
  • CID below that observed maximum value).

    • */ if (!resolved) return false;

Assert(cmin != InvalidCommandId);
