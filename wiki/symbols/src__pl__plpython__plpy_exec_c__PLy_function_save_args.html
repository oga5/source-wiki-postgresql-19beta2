/* saved args are always allocated in procedure's context */ result = (PLySavedArgs *) MemoryContextAllocZero(proc->mcxt, offsetof(PLySavedArgs, namedargs) + proc->nargs * sizeof(PyObject *)); result->nargs = proc->nargs;

/* Fetch the "args" list */ result->args = PyDict_GetItemString(proc->globals, "args"); Py_XINCREF(result->args);

/* If it's a trigger, also save "TD" */ if (proc->is_trigger == PLPY_TRIGGER) { result->td = PyDict_GetItemString(proc->globals, "TD"); Py_XINCREF(result->td); }

/* Fetch all the named arguments */ if (proc->argnames) { int i;

for (i = 0; i < result->nargs; i++) { if (proc->argnames[i]) { result->namedargs[i] = PyDict_GetItemString(proc->globals, proc->argnames[i]); Py_XINCREF(result->namedargs[i]); } } }

return result; } ~~~
