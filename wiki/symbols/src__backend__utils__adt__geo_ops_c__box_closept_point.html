~~~c static float8 box_closept_point(Point *result, BOX *box, Point *pt) { float8 dist, d; Point point, closept; LSEG lseg;

if (box_contain_point(box, pt)) { if (result != NULL) *result = *pt;

return 0.0; }

/* pairwise check lseg distances */ point.x = box->low.x; point.y = box->high.y; statlseg_construct(&lseg, &box->low, &point); dist = lseg_closept_point(result, &lseg, pt);

statlseg_construct(&lseg, &box->high, &point); d = lseg_closept_point(&closept, &lseg, pt); if (float8_lt(d, dist)) { dist = d; if (result != NULL) *result = closept; }

point.x = box->high.x; point.y = box->low.y; statlseg_construct(&lseg, &box->low, &point); d = lseg_closept_point(&closept, &lseg, pt); if (float8_lt(d, dist)) { dist = d; if (result != NULL) *result = closept; }

return dist; } ~~~
