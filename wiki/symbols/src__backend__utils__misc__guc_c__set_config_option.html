ソース抜粋

~~~c int set_config_option(const char *name, const char *value, GucContext context, GucSource source, GucAction action, bool changeVal, int elevel, bool is_reload) { Oid srole;

/*

  • Non-interactive sources should be treated as having all privileges,
  • except for PGC_S_CLIENT. Note in particular that this is true for
  • pg_db_role_setting sources (PGC_S_GLOBAL etc): we assume a suitable
  • privilege check was done when the pg_db_role_setting entry was made.

    • */ if (source >= PGC_S_INTERACTIVE || source == PGC_S_CLIENT) srole = GetUserId(); else srole = BOOTSTRAP_SUPERUSERID;

return set_config_with_handle(name, NULL, value, context, source, srole, action, changeVal, elevel, is_reload); } ~~~
