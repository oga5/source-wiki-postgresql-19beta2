/* match at the node itself */ root = findeq(root, ex, subs, isfind);

/* unless we matched here, consider matches at child nodes */ if (root && (root->flags & QTN_NOCHANGE) == 0 && root->valnode->type == QI_OPR) { int i, j = 0;

/*

  • tree.

    • */ for (i = 0; i < root->nchild; i++) { root->child[j] = dofindsubquery(root->child[i], ex, subs, isfind); if (root->child[j]) j++; }

root->nchild = j;

/*

  • If we have just zero or one remaining child node, simplify out this
  • operator node.
