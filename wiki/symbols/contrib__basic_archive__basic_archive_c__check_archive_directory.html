~~~c static bool check_archive_directory(char newval, void extra, GucSource source) { /*

  • The default value is an empty string, so we have to accept that value.
  • Our check_configured callback also checks for this and prevents
  • archiving from proceeding if it is still empty.

    • */ if (*newval == NULL || (*newval)[0] == '\0') return true;

/*

  • Make sure the file paths won't be too long. The docs indicate that the
  • file names to be archived can be up to 64 characters long.

    • */ if (strlen(*newval) + 64 + 2 >= MAXPGPATH) { GUC_check_errdetail("Archive directory too long."); return false; }

return true; } ~~~
