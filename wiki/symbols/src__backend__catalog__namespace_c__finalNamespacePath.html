~~~c static List * finalNamespacePath(List *oidlist, Oid *firstNS) { List *finalPath = NIL; ListCell *lc;

foreach(lc, oidlist) { Oid namespaceId = lfirst_oid(lc);

if (!list_member_oid(finalPath, namespaceId)) { if (InvokeNamespaceSearchHook(namespaceId, false)) finalPath = lappend_oid(finalPath, namespaceId); } }

/*

  • Remember the first member of the explicit list. (Note: this is
  • nominally wrong if temp_missing, but we need it anyway to distinguish
  • explicit from implicit mention of pg_catalog.)

    • */ if (finalPath == NIL) *firstNS = InvalidOid; else *firstNS = linitial_oid(finalPath);

/*

  • Add any implicitly-searched namespaces to the list. Note these go on
  • the front, not the back; also notice that we do not check USAGE
  • permissions for these.

    • */ if (!list_member_oid(finalPath, PG_CATALOG_NAMESPACE)) finalPath = lcons_oid(PG_CATALOG_NAMESPACE, finalPath);

if (OidIsValid(myTempNamespace) && !list_member_oid(finalPath, myTempNamespace)) finalPath = lcons_oid(myTempNamespace, finalPath);

return finalPath; } ~~~
