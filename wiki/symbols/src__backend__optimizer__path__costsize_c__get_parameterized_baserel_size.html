~~~c double get_parameterized_baserel_size(PlannerInfo *root, RelOptInfo *rel, List *param_clauses) { List *allclauses; double nrows;

/*

  • Estimate the number of rows returned by the parameterized scan, knowing
  • that it will apply all the extra join clauses as well as the rel's own
  • restriction clauses. Note that we force the clauses to be treated as
  • non-join clauses during selectivity estimation.

    • */ allclauses = list_concat_copy(param_clauses, rel->baserestrictinfo); nrows = rel->tuples * clauselist_selectivity(root, allclauses, rel->relid, /* do not use 0! */ JOIN_INNER, NULL); nrows = clamp_row_est(nrows); /* For safety, make sure result is not more than the base estimate */ if (nrows > rel->rows) nrows = rel->rows; return nrows; } ~~~
