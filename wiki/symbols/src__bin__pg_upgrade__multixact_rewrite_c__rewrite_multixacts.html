/*

  • Convert old multixids, if needed, by reading them one-by-one from the
  • old cluster.

old_reader = AllocOldMultiXactRead(old_cluster.pgdata, old_cluster.controldata.chkpnt_nxtmulti, old_cluster.controldata.chkpnt_nxtmxoff);
