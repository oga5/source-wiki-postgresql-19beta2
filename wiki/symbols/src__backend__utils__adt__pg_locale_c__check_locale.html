~~~c bool check_locale(int category, const char *locale, char **canonname) { char *save; char *res;

/* Don't let Windows' non-ASCII locale names in. */ if (!pg_is_ascii(locale)) { ereport(WARNING, (errcode(ERRCODE_INVALID_PARAMETER_VALUE), errmsg("locale name \"%s\" contains non-ASCII characters", locale))); return false; }

if (canonname) *canonname = NULL; /* in case of failure */

save = setlocale(category, NULL); if (!save) return false; /* won't happen, we hope */

/* save may be pointing at a modifiable scratch variable, see above. */ save = pstrdup(save);

/* set the locale with setlocale, to see if it accepts it. */ res = setlocale(category, locale);

/* save canonical name if requested. */ if (res && canonname) *canonname = pstrdup(res);

/* restore old value. */ if (!setlocale(category, save)) elog(WARNING, "failed to restore old locale \"%s\"", save); pfree(save);

/* Don't let Windows' non-ASCII locale names out. */ if (canonname && *canonname && !pg_is_ascii(*canonname)) { ereport(WARNING, (errcode(ERRCODE_INVALID_PARAMETER_VALUE), errmsg("locale name \"%s\" contains non-ASCII characters", *canonname))); pfree(*canonname); *canonname = NULL; return false; }

return (res != NULL); } ~~~
