~~~c static void shell_finish_command(bbsink_shell *sink) { int pclose_rc;

/* There should be a command running. */ Assert(sink->current_command != NULL); Assert(sink->pipe != NULL);

/* Close down the pipe we opened. */ pclose_rc = ClosePipeStream(sink->pipe); if (pclose_rc == -1) ereport(ERROR, (errcode_for_file_access(), errmsg("could not close pipe to external command: %m"))); else if (pclose_rc != 0) { ereport(ERROR, (errcode(ERRCODE_EXTERNAL_ROUTINE_EXCEPTION), errmsg("shell command \"%s\" failed", sink->current_command), errdetail_internal("%s", wait_result_to_str(pclose_rc)))); }

/* Clean up. */ sink->pipe = NULL; pfree(sink->current_command); sink->current_command = NULL; } ~~~
