while (remaining > 0) { size_t chunk; size_t required; size_t status;

/* If not enough space, must flush buffer */ required = LZ4F_compressBound(chunk, &state->prefs); if (required > state->buflen - state->bufdata) { cs->writeF(AH, state->buffer, state->bufdata); state->bufdata = 0; }

status = LZ4F_compressUpdate(state->ctx, state->buffer + state->bufdata, state->buflen - state->bufdata, data, chunk, NULL);

if (LZ4F_isError(status)) pg_fatal("could not compress data: %s", LZ4F_getErrorName(status));

state->bufdata += status;

data = ((const char *) data) + chunk; remaining -= chunk; } } ~~~
