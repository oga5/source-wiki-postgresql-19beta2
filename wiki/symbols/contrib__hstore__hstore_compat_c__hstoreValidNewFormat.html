if (!HSE_ISFIRST(entries[0])) return 0;

if (vsize > VARSIZE(hs)) return 0;

/* entry position must be nondecreasing */

for (i = 1; i < 2 * count; ++i) { if (HSE_ISFIRST(entries[i]) || (HSE_ENDPOS(entries[i]) < HSE_ENDPOS(entries[i - 1]))) return 0; }

/* key length must be nondecreasing and keys must not be null */

for (i = 1; i < count; ++i) { if (HSTORE_KEYLEN(entries, i) < HSTORE_KEYLEN(entries, i - 1)) return 0; if (HSE_ISNULL(entries[2 * i])) return 0; }

if (vsize != VARSIZE(hs)) return 1;

return 2; } ~~~
