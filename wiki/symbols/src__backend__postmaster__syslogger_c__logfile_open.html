~~~c static FILE * logfile_open(const char *filename, const char *mode, bool allow_errors) { FILE *fh; mode_t oumask;

/*

  • Note we do not let Log_file_mode disable IWUSR, since we certainly want
  • to be able to write the files ourselves.

    • */ oumask = umask((mode_t) ((~(Log_file_mode | S_IWUSR)) & (S_IRWXU | S_IRWXG | S_IRWXO))); fh = fopen(filename, mode); umask(oumask);

if (fh) { setvbuf(fh, NULL, PG_IOLBF, 0);

#ifdef WIN32 /* use CRLF line endings on Windows */ _setmode(_fileno(fh), _O_TEXT); #endif } else { int save_errno = errno;

ereport(allow_errors ? LOG : FATAL, (errcode_for_file_access(), errmsg("could not open log file \"%s\": %m", filename))); errno = save_errno; }

return fh; } ~~~
