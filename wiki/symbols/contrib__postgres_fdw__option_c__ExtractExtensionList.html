~~~c List * ExtractExtensionList(const char *extensionsString, bool warnOnMissing) { List *extensionOids = NIL; List *extlist; ListCell *lc;

/* SplitIdentifierString scribbles on its input, so pstrdup first */ if (!SplitIdentifierString(pstrdup(extensionsString), ',', &extlist)) { /* syntax error in name list */ ereport(ERROR, (errcode(ERRCODE_INVALID_PARAMETER_VALUE), errmsg("parameter \"%s\" must be a list of extension names", "extensions"))); }

foreach(lc, extlist) { const char *extension_name = (const char *) lfirst(lc); Oid extension_oid = get_extension_oid(extension_name, true);

if (OidIsValid(extension_oid)) { extensionOids = lappend_oid(extensionOids, extension_oid); } else if (warnOnMissing) { ereport(WARNING, (errcode(ERRCODE_UNDEFINED_OBJECT), errmsg("extension \"%s\" is not installed", extension_name))); } }

list_free(extlist); return extensionOids; } ~~~
