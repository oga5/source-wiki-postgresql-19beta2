~~~c static void AcquireExecutorLocks(List *stmt_list, bool acquire) { ListCell *lc1;

foreach(lc1, stmt_list) { PlannedStmt *plannedstmt = lfirst_node(PlannedStmt, lc1); ListCell *lc2;

if (plannedstmt->commandType == CMD_UTILITY) { /*

  • Ignore utility statements, except those (such as EXPLAIN) that
  • contain a parsed-but-not-planned query. Note: it's okay to use
  • ScanQueryForLocks, even though the query hasn't been through
  • rule rewriting, because rewriting doesn't change the query
  • representation.

    • */ Query *query = UtilityContainsQuery(plannedstmt->utilityStmt);

if (query) ScanQueryForLocks(query, acquire); continue; }

foreach(lc2, plannedstmt->rtable) { RangeTblEntry *rte = (RangeTblEntry *) lfirst(lc2);

if (!(rte->rtekind == RTE_RELATION || (rte->rtekind == RTE_SUBQUERY && OidIsValid(rte->relid)))) continue;

/*

  • Acquire the appropriate type of lock on each relation OID. Note
  • that we don't actually try to open the rel, and hence will not
  • fail if it's been dropped entirely --- we'll just transiently
  • acquire a non-conflicting lock.

    • */ if (acquire) LockRelationOid(rte->relid, rte->rellockmode); else UnlockRelationOid(rte->relid, rte->rellockmode); } } } ~~~
