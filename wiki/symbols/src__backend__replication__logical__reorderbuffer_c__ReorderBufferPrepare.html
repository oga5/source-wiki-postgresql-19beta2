~~~c void ReorderBufferPrepare(ReorderBuffer *rb, TransactionId xid, char *gid) { ReorderBufferTXN *txn;

txn = ReorderBufferTXNByXid(rb, xid, false, NULL, InvalidXLogRecPtr, false);

/* unknown transaction, nothing to replay */ if (txn == NULL) return;

/*

  • txn must have been marked as a prepared transaction and must have
  • neither been skipped nor sent a prepare. Also, the prepare info must
  • have been updated in it by now.

    • */ Assert((txn->txn_flags & RBTXN_PREPARE_STATUS_MASK) == RBTXN_IS_PREPARED); Assert(XLogRecPtrIsValid(txn->final_lsn));

txn->gid = pstrdup(gid);

ReorderBufferReplay(txn, rb, xid, txn->final_lsn, txn->end_lsn, txn->prepare_time, txn->origin_id, txn->origin_lsn);

/*

  • Send a prepare if not already done so. This might occur if we have
  • detected a concurrent abort while replaying the non-streaming
  • transaction.

    • */ if (!rbtxn_sent_prepare(txn)) { rb->prepare(rb, txn, txn->final_lsn); txn->txn_flags |= RBTXN_SENT_PREPARE; } } ~~~
