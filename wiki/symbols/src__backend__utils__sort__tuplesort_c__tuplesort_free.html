if (state->tapeset) spaceUsed = LogicalTapeSetBlocks(state->tapeset); else spaceUsed = (state->allowedMem - state->availMem + 1023) / 1024;

/*

  • Delete temporary "tape" files, if any.

    • *

  • We don't bother to destroy the individual tapes here. They will go away
  • with the sortcontext. (In TSS_FINALMERGE state, we have closed
  • finished tapes already.)

    • */ if (state->tapeset) LogicalTapeSetClose(state->tapeset);

if (trace_sort) { if (state->tapeset) elog(LOG, "%s of worker %d ended, %" PRId64 " disk blocks used: %s", SERIAL(state) ? "external sort" : "parallel external sort", state->worker, spaceUsed, pg_rusage_show(&state->ru_start)); else elog(LOG, "%s of worker %d ended, %" PRId64 " KB used: %s", SERIAL(state) ? "internal sort" : "unperformed parallel sort", state->worker, spaceUsed, pg_rusage_show(&state->ru_start)); }

TRACE_POSTGRESQL_SORT_DONE(state->tapeset != NULL, spaceUsed);

FREESTATE(state); MemoryContextSwitchTo(oldcontext);

/*

  • Free the per-sort memory context, thereby releasing all working memory.

    • */ MemoryContextReset(state->base.sortcontext); } ~~~
