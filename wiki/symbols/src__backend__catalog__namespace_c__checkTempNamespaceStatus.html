~~~c TempNamespaceStatus checkTempNamespaceStatus(Oid namespaceId) { PGPROC *proc; ProcNumber procNumber;

Assert(OidIsValid(MyDatabaseId));

procNumber = GetTempNamespaceProcNumber(namespaceId);

/* No such namespace, or its name shows it's not temp? */ if (procNumber == INVALID_PROC_NUMBER) return TEMP_NAMESPACE_NOT_TEMP;

/* Is the backend alive? */ proc = ProcNumberGetProc(procNumber); if (proc == NULL) return TEMP_NAMESPACE_IDLE;

/* Is the backend connected to the same database we are looking at? */ if (proc->databaseId != MyDatabaseId) return TEMP_NAMESPACE_IDLE;

/* Does the backend own the temporary namespace? */ if (proc->tempNamespaceId != namespaceId) return TEMP_NAMESPACE_IDLE;

/* Yup, so namespace is busy */ return TEMP_NAMESPACE_IN_USE; } ~~~
