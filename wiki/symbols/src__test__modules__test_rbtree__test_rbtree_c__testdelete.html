/* Choose unique ids to delete */ deleteIds = (int *) palloc(delsize * sizeof(int)); chosen = (bool *) palloc0(size * sizeof(bool));

for (i = 0; i < delsize; i++) { int k = pg_prng_uint64_range(&pg_global_prng_state, 0, size - 1);

while (chosen[k]) k = (k + 1) % size; deleteIds[i] = k; chosen[k] = true; }

/* Delete elements */ for (i = 0; i < delsize; i++) { IntRBTreeNode find; IntRBTreeNode *node;

/* Check that deleted elements are deleted */ for (i = 0; i < size; i++) { IntRBTreeNode node; IntRBTreeNode *result;
