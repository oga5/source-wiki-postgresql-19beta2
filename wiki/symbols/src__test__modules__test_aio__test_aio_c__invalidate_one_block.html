~~~c static void invalidate_one_block(Relation rel, ForkNumber forknum, BlockNumber blkno) { PrefetchBufferResult pr; Buffer buf;

/*

  • This is a gross hack, but there's no other API exposed that allows to
  • get a buffer ID without actually reading the block in.

    • */ pr = PrefetchBuffer(rel, forknum, blkno); buf = pr.recent_buffer;

if (BufferIsValid(buf)) { /* if the buffer contents aren't valid, this'll return false */ if (ReadRecentBuffer(rel->rd_locator, forknum, blkno, buf)) { BufferDesc *buf_hdr = BufferIsLocal(buf) ? GetLocalBufferDescriptor(-buf - 1) : GetBufferDescriptor(buf - 1); bool flushed;

LockBuffer(buf, BUFFER_LOCK_EXCLUSIVE);

if (pg_atomic_read_u64(&buf_hdr->state) & BM_DIRTY) { if (BufferIsLocal(buf)) FlushLocalBuffer(buf_hdr, NULL); else FlushOneBuffer(buf); } UnlockReleaseBuffer(buf);

if (BufferIsLocal(buf)) InvalidateLocalBuffer(GetLocalBufferDescriptor(-buf - 1), true); else if (!EvictUnpinnedBuffer(buf, &flushed)) elog(ERROR, "couldn't evict"); } }

} ~~~
