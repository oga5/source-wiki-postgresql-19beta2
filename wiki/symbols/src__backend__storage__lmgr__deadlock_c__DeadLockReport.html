/* reset locktagbuf to hold next object description */ resetStringInfo(&locktagbuf);

/* Duplicate all the above for the server ... */ appendBinaryStringInfo(&logbuf, clientbuf.data, clientbuf.len);

appendStringInfoChar(&logbuf, '\n');
