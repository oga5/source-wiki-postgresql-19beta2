  • scan everything between pstr and pstr_end. This is not including the
  • last character so we might set it to '\0' for the parsing

    • */

char last_char; int err = 0; char *pstr_end; char *strtol_end = NULL;

while (**pstr == ' ') pstr++; pstr_end = find_end_token(*pstr, pfmt); if (!pstr_end) { /* there was an error, no match */ return 1; } last_char = *pstr_end; *pstr_end = '\0';

switch (scan_type) { case PGTYPES_TYPE_UINT:

/*

  • numbers may be blank-padded, this is the only deviation from
  • the fmt-string we accept
