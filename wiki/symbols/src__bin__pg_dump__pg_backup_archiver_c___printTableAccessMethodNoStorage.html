~~~c static void _printTableAccessMethodNoStorage(ArchiveHandle *AH, TocEntry *te) { RestoreOptions *ropt = AH->public.ropt; const char *tableam = te->tableam; PQExpBuffer cmd;

/* do nothing in --no-table-access-method mode */ if (ropt->noTableAm) return;

if (!tableam) return;

Assert(te->relkind == RELKIND_PARTITIONED_TABLE);

cmd = createPQExpBuffer();

appendPQExpBufferStr(cmd, "ALTER TABLE "); appendPQExpBuffer(cmd, "%s ", fmtQualifiedId(te->namespace, te->tag)); appendPQExpBuffer(cmd, "SET ACCESS METHOD %s;", fmtId(tableam));

if (RestoringToDB(AH)) { PGresult *res;

res = PQexec(AH->connection, cmd->data);

if (!res || PQresultStatus(res) != PGRES_COMMAND_OK) warn_or_exit_horribly(AH, "could not alter table access method: %s", PQerrorMessage(AH->connection)); PQclear(res); } else ahprintf(AH, "%s\n\n", cmd->data);

destroyPQExpBuffer(cmd); } ~~~
