1LOAD 'pg_plan_advice';
2SET max_parallel_workers_per_gather = 0;
3
4CREATE TABLE join_dim (id serial primary key, dim text)
5	WITH (autovacuum_enabled = false);
6INSERT INTO join_dim (dim) SELECT random()::text FROM generate_series(1,100) g;
7VACUUM ANALYZE join_dim;
8
9CREATE TABLE join_fact (
10	id int primary key,
11	dim_id integer not null references join_dim (id)
12) WITH (autovacuum_enabled = false);
13INSERT INTO join_fact
14	SELECT g, (g%3)+1 FROM generate_series(1,100000) g;
15CREATE INDEX join_fact_dim_id ON join_fact (dim_id);
16VACUUM ANALYZE join_fact;
17
18-- We expect a hash join by default.
19EXPLAIN (COSTS OFF, PLAN_ADVICE)
20	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
21
22-- Try forcing each join method in turn with join_dim as the inner table.
23-- All of these should work except for MERGE_JOIN_MATERIALIZE; that will
24-- fail, because the planner knows that join_dim (id) is unique, and will
25-- refuse to add mark/restore overhead.
26BEGIN;
27SET LOCAL pg_plan_advice.advice = 'HASH_JOIN(d)';
28EXPLAIN (COSTS OFF, PLAN_ADVICE)
29	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
30SET LOCAL pg_plan_advice.advice = 'MERGE_JOIN_MATERIALIZE(d)';
31EXPLAIN (COSTS OFF, PLAN_ADVICE)
32	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
33SET LOCAL pg_plan_advice.advice = 'MERGE_JOIN_PLAIN(d)';
34EXPLAIN (COSTS OFF, PLAN_ADVICE)
35	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
36SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_MATERIALIZE(d)';
37EXPLAIN (COSTS OFF, PLAN_ADVICE)
38	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
39SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_MEMOIZE(d)';
40EXPLAIN (COSTS OFF, PLAN_ADVICE)
41	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
42SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_PLAIN(d)';
43EXPLAIN (COSTS OFF, PLAN_ADVICE)
44	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
45COMMIT;
46
47-- Now try forcing each join method in turn with join_fact as the inner
48-- table. All of these should work.
49BEGIN;
50SET LOCAL pg_plan_advice.advice = 'HASH_JOIN(f)';
51EXPLAIN (COSTS OFF, PLAN_ADVICE)
52	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
53SET LOCAL pg_plan_advice.advice = 'MERGE_JOIN_MATERIALIZE(f)';
54EXPLAIN (COSTS OFF, PLAN_ADVICE)
55	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
56SET LOCAL pg_plan_advice.advice = 'MERGE_JOIN_PLAIN(f)';
57EXPLAIN (COSTS OFF, PLAN_ADVICE)
58	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
59SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_MATERIALIZE(f)';
60EXPLAIN (COSTS OFF, PLAN_ADVICE)
61	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
62SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_MEMOIZE(f)';
63EXPLAIN (COSTS OFF, PLAN_ADVICE)
64	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
65SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_PLAIN(f)';
66EXPLAIN (COSTS OFF, PLAN_ADVICE)
67	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
68COMMIT;
69
70-- Non-working cases. We can't force a foreign join between these tables,
71-- because they aren't foreign tables. We also can't use two different
72-- strategies on the same table, nor can we put both tables on the inner
73-- side of the same join.
74BEGIN;
75SET LOCAL pg_plan_advice.advice = 'FOREIGN_JOIN((f d))';
76EXPLAIN (COSTS OFF, PLAN_ADVICE)
77	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
78SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_PLAIN(f) NESTED_LOOP_MATERIALIZE(f)';
79EXPLAIN (COSTS OFF, PLAN_ADVICE)
80	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
81SET LOCAL pg_plan_advice.advice = 'NESTED_LOOP_PLAIN(f d)';
82EXPLAIN (COSTS OFF, PLAN_ADVICE)
83	SELECT * FROM join_fact f JOIN join_dim d ON f.dim_id = d.id;
84COMMIT;
