1LOAD 'pg_plan_advice';
2
3-- An empty string is allowed. Empty target lists are allowed for most advice
4-- tags, but not for JOIN_ORDER. "Supplied Plan Advice" should be omitted in
5-- text format when there is no actual advice, but not in non-text format.
6SET pg_plan_advice.advice = '';
7EXPLAIN (COSTS OFF) SELECT 1;
8SET pg_plan_advice.advice = 'SEQ_SCAN()';
9EXPLAIN (COSTS OFF) SELECT 1;
10SET pg_plan_advice.advice = 'NESTED_LOOP_PLAIN()';
11EXPLAIN (COSTS OFF, FORMAT JSON) SELECT 1;
12SET pg_plan_advice.advice = 'JOIN_ORDER()';
13
14-- Test assorted variations in capitalization, whitespace, and which parts of
15-- the relation identifier are included. These should all work.
16SET pg_plan_advice.advice = 'SEQ_SCAN(x)';
17EXPLAIN (COSTS OFF) SELECT 1;
18SET pg_plan_advice.advice = 'seq_scan(x@y)';
19EXPLAIN (COSTS OFF) SELECT 1;
20SET pg_plan_advice.advice = 'SEQ_scan(x#2)';
21EXPLAIN (COSTS OFF) SELECT 1;
22SET pg_plan_advice.advice = 'SEQ_SCAN (x/y)';
23EXPLAIN (COSTS OFF) SELECT 1;
24SET pg_plan_advice.advice = '  SEQ_SCAN ( x / y . z )  ';
25EXPLAIN (COSTS OFF) SELECT 1;
26SET pg_plan_advice.advice = 'SEQ_SCAN("x"#2/"y"."z"@"t")';
27EXPLAIN (COSTS OFF) SELECT 1;
28
29-- Syntax errors.
30SET pg_plan_advice.advice = 'SEQUENTIAL_SCAN(x)';
31SET pg_plan_advice.advice = 'SEQ_SCAN';
32SET pg_plan_advice.advice = 'SEQ_SCAN(';
33SET pg_plan_advice.advice = 'SEQ_SCAN("';
34SET pg_plan_advice.advice = 'SEQ_SCAN("")';
35SET pg_plan_advice.advice = 'SEQ_SCAN("a"';
36SET pg_plan_advice.advice = 'SEQ_SCAN(#';
37SET pg_plan_advice.advice = '()';
38SET pg_plan_advice.advice = '123';
39
40-- Tags like SEQ_SCAN and NO_GATHER don't allow sublists at all; other tags,
41-- except for JOIN_ORDER, allow at most one level of sublist. Hence, these
42-- examples should error out.
43SET pg_plan_advice.advice = 'SEQ_SCAN((x))';
44SET pg_plan_advice.advice = 'DO_NOT_SCAN((x))';
45SET pg_plan_advice.advice = 'GATHER(((x)))';
46
47-- Legal comments.
48SET pg_plan_advice.advice = '/**/';
49EXPLAIN (COSTS OFF) SELECT 1;
50SET pg_plan_advice.advice = 'HASH_JOIN(_)/***/';
51EXPLAIN (COSTS OFF) SELECT 1;
52SET pg_plan_advice.advice = '/* comment */ HASH_JOIN(/*x*/y)';
53EXPLAIN (COSTS OFF) SELECT 1;
54SET pg_plan_advice.advice = '/* comment */ HASH_JOIN(y//*x*/z)';
55EXPLAIN (COSTS OFF) SELECT 1;
56
57-- Unterminated comments.
58SET pg_plan_advice.advice = '/*';
59SET pg_plan_advice.advice = 'JOIN_ORDER("fOO") /* oops';
60
61-- Nested comments are not supported, so the first of these is legal and
62-- the second is not.
63SET pg_plan_advice.advice = '/*/*/';
64EXPLAIN (COSTS OFF) SELECT 1;
65SET pg_plan_advice.advice = '/*/* stuff */*/';
66
67-- Foreign join requires multiple relation identifiers.
68SET pg_plan_advice.advice = 'FOREIGN_JOIN(a)';
69SET pg_plan_advice.advice = 'FOREIGN_JOIN((a))';
70
71-- Tag keywords used as alias names work fine, because the 'identifier'
72-- nonterminal accepts all token types.
73SET pg_plan_advice.advice = 'SEQ_SCAN(hash_join)';
74EXPLAIN (COSTS OFF) SELECT 1;
75SET pg_plan_advice.advice = 'SEQ_SCAN(seq_scan)';
76EXPLAIN (COSTS OFF) SELECT 1;
77SET pg_plan_advice.advice = 'SEQ_SCAN(gather)';
78EXPLAIN (COSTS OFF) SELECT 1;
79SET pg_plan_advice.advice = 'SEQ_SCAN(join_order)';
80EXPLAIN (COSTS OFF) SELECT 1;

82-- Tag keywords used as partition names or plan names should also work,
83-- since pgpa_identifier_string() can generate these from real partition
84-- and subquery names.
