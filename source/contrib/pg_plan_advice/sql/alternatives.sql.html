contrib/pg_plan_advice/sql/alternatives.sql
1LOAD 'pg_plan_advice';
2SET max_parallel_workers_per_gather = 0;
3
4CREATE TABLE alt_t1 (a int) WITH (autovacuum_enabled = false);
5CREATE TABLE alt_t2 (a int) WITH (autovacuum_enabled = false);
6CREATE INDEX ON alt_t2(a);
7INSERT INTO alt_t1 SELECT generate_series(1, 1000);
8INSERT INTO alt_t2 SELECT generate_series(1, 100000);
9VACUUM ANALYZE alt_t1;
10VACUUM ANALYZE alt_t2;
11
12-- This query uses an OR to prevent the EXISTS from being converted to a
13-- semi-join, forcing the planner through the AlternativeSubPlan path.
14EXPLAIN (COSTS OFF, PLAN_ADVICE)
15SELECT * FROM alt_t1
16WHERE EXISTS (SELECT 1 FROM alt_t2 WHERE alt_t2.a = alt_t1.a) OR alt_t1.a < 0;
17
18-- We should be able to force either AlternativeSubPlan by advising against
19-- scanning the other relation.
20BEGIN;
21SET LOCAL pg_plan_advice.advice = 'DO_NOT_SCAN(alt_t2@exists_1)';
22EXPLAIN (COSTS OFF, PLAN_ADVICE)
23SELECT * FROM alt_t1
24WHERE EXISTS (SELECT 1 FROM alt_t2 WHERE alt_t2.a = alt_t1.a) OR alt_t1.a < 0;
25SET LOCAL pg_plan_advice.advice = 'DO_NOT_SCAN(alt_t2@exists_2)';
26EXPLAIN (COSTS OFF, PLAN_ADVICE)
27SELECT * FROM alt_t1
28WHERE EXISTS (SELECT 1 FROM alt_t2 WHERE alt_t2.a = alt_t1.a) OR alt_t1.a < 0;
29COMMIT;
30
31-- Now let's test a case involving MinMaxAggPath, which we treat similarly
32-- to the AlternativeSubPlan case.
33CREATE TABLE alt_minmax (a int) WITH (autovacuum_enabled = false);
34CREATE INDEX ON alt_minmax(a);
35INSERT INTO alt_minmax SELECT generate_series(1, 10000);
36VACUUM ANALYZE alt_minmax;
37
38-- Using an Index Scan inside of an InitPlan should win over a full table
39-- scan.
40EXPLAIN (COSTS OFF, PLAN_ADVICE)
41SELECT min(a), max(a) FROM alt_minmax;
42
43-- Advising against the scan of alt_minmax at the root query level should
44-- change nothing, but if we say we don't want either of or both of the
45-- minmax-variant scans, the plan should switch to a full table scan.
46BEGIN;
47SET LOCAL pg_plan_advice.advice = 'DO_NOT_SCAN(alt_minmax)';
48EXPLAIN (COSTS OFF, PLAN_ADVICE)
49SELECT min(a), max(a) FROM alt_minmax;
50SET LOCAL pg_plan_advice.advice = 'DO_NOT_SCAN(alt_minmax@minmax_1)';
51EXPLAIN (COSTS OFF, PLAN_ADVICE)
52SELECT min(a), max(a) FROM alt_minmax;
53SET LOCAL pg_plan_advice.advice = 'DO_NOT_SCAN(alt_minmax@minmax_1) DO_NOT_SCAN(alt_minmax@minmax_2)';
54EXPLAIN (COSTS OFF, PLAN_ADVICE)
55SELECT min(a), max(a) FROM alt_minmax;
56COMMIT;
57
58DROP TABLE alt_t1, alt_t2, alt_minmax;
