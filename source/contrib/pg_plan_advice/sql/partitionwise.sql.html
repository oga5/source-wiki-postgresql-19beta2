contrib/pg_plan_advice/sql/partitionwise.sql
1LOAD 'pg_plan_advice';
2SET max_parallel_workers_per_gather = 0;
3SET enable_partitionwise_join = true;
4
5CREATE TABLE pt1 (id integer primary key, dim1 text, val1 int)
6	PARTITION BY RANGE (id);
7CREATE TABLE pt1a PARTITION OF pt1 FOR VALUES FROM (1) to (1001)
8	WITH (autovacuum_enabled = false);
9CREATE TABLE pt1b PARTITION OF pt1 FOR VALUES FROM (1001) to (2001)
10	WITH (autovacuum_enabled = false);
11CREATE TABLE pt1c PARTITION OF pt1 FOR VALUES FROM (2001) to (3001)
12	WITH (autovacuum_enabled = false);
13INSERT INTO pt1 (id, dim1, val1)
14	SELECT g, 'some filler text ' || g, (g % 3) + 1
15	  FROM generate_series(1,3000) g;
16VACUUM ANALYZE pt1;
17
18CREATE TABLE pt2 (id integer primary key, dim2 text, val2 int)
19	PARTITION BY RANGE (id);
20CREATE TABLE pt2a PARTITION OF pt2 FOR VALUES FROM (1) to (1001)
21	WITH (autovacuum_enabled = false);
22CREATE TABLE pt2b PARTITION OF pt2 FOR VALUES FROM (1001) to (2001)
23	WITH (autovacuum_enabled = false);
24CREATE TABLE pt2c PARTITION OF pt2 FOR VALUES FROM (2001) to (3001)
25	WITH (autovacuum_enabled = false);
26INSERT INTO pt2 (id, dim2, val2)
27	SELECT g, 'some other text ' || g, (g % 5) + 1
28	  FROM generate_series(1,3000,2) g;
29VACUUM ANALYZE pt2;
30
31CREATE TABLE pt3 (id integer primary key, dim3 text, val3 int)
32	PARTITION BY RANGE (id);
33CREATE TABLE pt3a PARTITION OF pt3 FOR VALUES FROM (1) to (1001)
34	WITH (autovacuum_enabled = false);
35CREATE TABLE pt3b PARTITION OF pt3 FOR VALUES FROM (1001) to (2001)
36	WITH (autovacuum_enabled = false);
37CREATE TABLE pt3c PARTITION OF pt3 FOR VALUES FROM (2001) to (3001)
38	WITH (autovacuum_enabled = false);
39INSERT INTO pt3 (id, dim3, val3)
40	SELECT g, 'a third random text ' || g, (g % 7) + 1
41	  FROM generate_series(1,3000,3) g;
42VACUUM ANALYZE pt3;
43
44CREATE TABLE ptmismatch (id integer primary key, dimm text, valm int)
45	PARTITION BY RANGE (id);
46CREATE TABLE ptmismatcha PARTITION OF ptmismatch
47    FOR VALUES FROM (1) to (1501)
48	WITH (autovacuum_enabled = false);
49CREATE TABLE ptmismatchb PARTITION OF ptmismatch
50    FOR VALUES FROM (1501) to (3001)
51	WITH (autovacuum_enabled = false);
52INSERT INTO ptmismatch (id, dimm, valm)
53	SELECT g, 'yet another text ' || g, (g % 2) + 1
54	  FROM generate_series(1,3000) g;
55VACUUM ANALYZE ptmismatch;
56
57EXPLAIN (PLAN_ADVICE, COSTS OFF)
58SELECT * FROM pt1, pt2, pt3 WHERE pt1.id = pt2.id AND pt2.id = pt3.id
59   AND val1 = 1 AND val2 = 1 AND val3 = 1;
60
61-- Suppress partitionwise join, or do it just partially.
62BEGIN;
63SET LOCAL pg_plan_advice.advice = 'PARTITIONWISE(pt1 pt2 pt3)';
64EXPLAIN (PLAN_ADVICE, COSTS OFF)
65SELECT * FROM pt1, pt2, pt3 WHERE pt1.id = pt2.id AND pt2.id = pt3.id
66   AND val1 = 1 AND val2 = 1 AND val3 = 1;
67SET LOCAL pg_plan_advice.advice = 'PARTITIONWISE((pt1 pt2) pt3)';
68EXPLAIN (PLAN_ADVICE, COSTS OFF)
69SELECT * FROM pt1, pt2, pt3 WHERE pt1.id = pt2.id AND pt2.id = pt3.id
70   AND val1 = 1 AND val2 = 1 AND val3 = 1;
71COMMIT;
72
73-- Test conflicting advice.
74BEGIN;
75SET LOCAL pg_plan_advice.advice = 'PARTITIONWISE((pt1 pt2) (pt1 pt3))';
76EXPLAIN (PLAN_ADVICE, COSTS OFF)
77SELECT * FROM pt1, pt2, pt3 WHERE pt1.id = pt2.id AND pt2.id = pt3.id
78   AND val1 = 1 AND val2 = 1 AND val3 = 1;
79COMMIT;
80
81-- Can't force a partitionwise join with a mismatched table.
82BEGIN;
83SET LOCAL pg_plan_advice.advice = 'PARTITIONWISE((pt1 ptmismatch))';
84EXPLAIN (PLAN_ADVICE, COSTS OFF)
85SELECT * FROM pt1, ptmismatch WHERE pt1.id = ptmismatch.id;
86COMMIT;
87
88-- Force join order for a particular branch of the partitionwise join with
89-- and without mentioning the schema name.
90BEGIN;
91SET LOCAL pg_plan_advice.advice = 'JOIN_ORDER(pt3/public.pt3a pt2/public.pt2a pt1/public.pt1a)';
92EXPLAIN (PLAN_ADVICE, COSTS OFF)
93SELECT * FROM pt1, pt2, pt3 WHERE pt1.id = pt2.id AND pt2.id = pt3.id
94   AND val1 = 1 AND val2 = 1 AND val3 = 1;
95SET LOCAL pg_plan_advice.advice = 'JOIN_ORDER(pt3/pt3a pt2/pt2a pt1/pt1a)';
96EXPLAIN (PLAN_ADVICE, COSTS OFF)
97SELECT * FROM pt1, pt2, pt3 WHERE pt1.id = pt2.id AND pt2.id = pt3.id
98   AND val1 = 1 AND val2 = 1 AND val3 = 1;
99COMMIT;
