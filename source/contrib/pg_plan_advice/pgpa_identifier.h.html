28/* Convenience function for comparing possibly-NULL strings. */
29static inline bool
30strings_equal_or_both_null(const char *a, const char *b)
31{
32	if (a == b)
33		return true;
34	else if (a == NULL || b == NULL)
35		return false;
36	else
37		return strcmp(a, b) == 0;
38}
