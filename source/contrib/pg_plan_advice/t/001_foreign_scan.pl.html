3# Verify plan advice for foreign scans.

5use strict;
6use warnings FATAL => 'all';
7
8use PostgreSQL::Test::Cluster;
9use PostgreSQL::Test::Utils;
10use Test::More;
11
12my $node = PostgreSQL::Test::Cluster->new('node');
13$node->init;
14$node->append_conf('postgresql.conf',
15	"session_preload_libraries = 'pg_plan_advice'");
16$node->start;
17
18my $host = $node->host;
19my $port = $node->port;
20
21$node->safe_psql(
22	'postgres', qq{
23	CREATE EXTENSION postgres_fdw;
24	CREATE SERVER loopback FOREIGN DATA WRAPPER postgres_fdw
25		OPTIONS (host '$host', port '$port', dbname 'postgres');
26	CREATE USER MAPPING FOR CURRENT_USER SERVER loopback;
27	CREATE TABLE base_tab (a int);
28	CREATE TABLE base_tab2 (a int);
29	CREATE FOREIGN TABLE ftab (a int)
30		SERVER loopback OPTIONS (table_name 'base_tab');
31	CREATE FOREIGN TABLE ftab2 (a int)
32		SERVER loopback OPTIONS (table_name 'base_tab2');
33});
34
35# Get the generated advice from EXPLAIN output.
36sub extract_generated_advice
37{
38	my ($explain) = @_;
39	my $generated_advice = '';
40	my $collecting = 0;
41	foreach my $line (split /\n/, $explain)
42	{
43		if ($line =~ /Generated Plan Advice:/)
44		{
45			$collecting = 1;
46			next;
47		}
48		if ($collecting)
49		{
50			$line =~ s/^\s+//;
51			$line =~ s/\s+$//;
52			$generated_advice .= ' ' if $generated_advice ne '';
53			$generated_advice .= $line;
54		}
55	}
56	return $generated_advice;
57}
58
59# A pushed-down aggregate over a single foreign table yields a ForeignScan
60# that names exactly one relation.  No FOREIGN_JOIN advice should be generated.
61my $agg_explain = $node->safe_psql('postgres',
62	"EXPLAIN (COSTS OFF, PLAN_ADVICE) SELECT count(*) FROM ftab;");
63my $agg_pat = qr/\QRelations: Aggregate on (ftab)\E/;
64like($agg_explain, $agg_pat, 'single-table aggregate is pushed down');
65my $agg_advice = extract_generated_advice($agg_explain);
66is($agg_advice, 'NO_GATHER(ftab)', 'advice for single-table aggregate');
67
68# A foreign join should generate FOREIGN_JOIN advice. Here we force this by
69# disabling local join methods.
70my $join_explain = $node->safe_psql(
71	'postgres', q{
72	SET enable_mergejoin = off;
73	SET enable_hashjoin = off;
74	SET enable_nestloop = off;
75	EXPLAIN (COSTS OFF, PLAN_ADVICE) SELECT * FROM ftab JOIN ftab2 USING (a);
76});
77my $ja_expected = 'FOREIGN_JOIN((ftab ftab2)) NO_GATHER(ftab ftab2)';
78my $ja_actual = extract_generated_advice($join_explain);
79is($ja_actual, $ja_expected, 'advice for foreign join');
80
81$node->stop;
82
83done_testing();
