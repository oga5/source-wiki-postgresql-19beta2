19/*
20 * Advice items generally take the form SOME_TAG(item [...]), where an item
21 * can take various forms. The simplest case is a relation identifier, but
22 * some tags allow sublists, and JOIN_ORDER() allows both ordered and unordered
23 * sublists.
24 */
25typedef enum
26{
27	PGPA_TARGET_IDENTIFIER,		/* relation identifier */
28	PGPA_TARGET_ORDERED_LIST,	/* (item ...) */
29	PGPA_TARGET_UNORDERED_LIST	/* {item ...} */
30} pgpa_target_type;
31
32/*
33 * An index specification.
34 */
35typedef struct pgpa_index_target
36{
37	/* Index schema and name */
38	char	   *indnamespace;
39	char	   *indname;
40} pgpa_index_target;
41
42/*
43 * A single item about which advice is being given, which could be either
44 * a relation identifier that we want to break out into its constituent fields,
45 * or a sublist of some kind.
46 */
47typedef struct pgpa_advice_target
48{
49	pgpa_target_type ttype;
50
51	/*
52	 * This field is meaningful when ttype is PGPA_TARGET_IDENTIFIER.
53	 *
54	 * All identifiers must have an alias name and an occurrence number; the
55	 * remaining fields can be NULL. Note that it's possible to specify a
56	 * partition name without a partition schema, but not the reverse.
57	 */
58	pgpa_identifier rid;
59
60	/*
61	 * This field is set when ttype is PGPA_TARGET_IDENTIFIER and the advice
62	 * tag is PGPA_TAG_INDEX_SCAN or PGPA_TAG_INDEX_ONLY_SCAN.
63	 */
64	pgpa_index_target *itarget;
65
66	/*
67	 * When the ttype is PGPA_TARGET_<anything>_LIST, this field contains a
68	 * list of additional pgpa_advice_target objects. Otherwise, it is unused.
69	 */
70	List	   *children;
71} pgpa_advice_target;

73/*
74 * These are all the kinds of advice that we know how to parse. If a keyword
75 * is found at the top level, it must be in this list.
76 *
77 * If you change anything here, also update pgpa_parse_advice_tag and
78 * pgpa_cstring_advice_tag.
79 */
80typedef enum pgpa_advice_tag_type
81{
82	PGPA_TAG_BITMAP_HEAP_SCAN,
83	PGPA_TAG_DO_NOT_SCAN,
84	PGPA_TAG_FOREIGN_JOIN,
85	PGPA_TAG_GATHER,
86	PGPA_TAG_GATHER_MERGE,
87	PGPA_TAG_HASH_JOIN,
88	PGPA_TAG_INDEX_ONLY_SCAN,
89	PGPA_TAG_INDEX_SCAN,
90	PGPA_TAG_JOIN_ORDER,
91	PGPA_TAG_MERGE_JOIN_MATERIALIZE,
92	PGPA_TAG_MERGE_JOIN_PLAIN,
93	PGPA_TAG_NESTED_LOOP_MATERIALIZE,
94	PGPA_TAG_NESTED_LOOP_MEMOIZE,
95	PGPA_TAG_NESTED_LOOP_PLAIN,
96	PGPA_TAG_NO_GATHER,
97	PGPA_TAG_PARTITIONWISE,
98	PGPA_TAG_SEMIJOIN_NON_UNIQUE,
99	PGPA_TAG_SEMIJOIN_UNIQUE,
100	PGPA_TAG_SEQ_SCAN,
101	PGPA_TAG_TID_SCAN
102} pgpa_advice_tag_type;
103
104/*
105 * An item of advice, meaning a tag and the list of all targets to which
106 * it is being applied.
107 *
108 * "targets" is a list of pgpa_advice_target objects.
109 *
110 * The List returned from pgpa_yyparse is list of pgpa_advice_item objects.
111 */
112typedef struct pgpa_advice_item
113{
114	pgpa_advice_tag_type tag;
115	List	   *targets;
116} pgpa_advice_item;
117
118/*
119 * Result of comparing an array of pgpa_identifier objects to a
120 * pgpa_advice_target.
121 *
122 * PGPA_ITM_EQUAL means all targets are matched by some identifier, and
123 * all identifiers were matched to a target.
124 *
125 * PGPA_ITM_KEYS_ARE_SUBSET means that all identifiers matched to a target,
126 * but there were leftover targets. Generally, this means that the advice is
127 * looking to apply to all of the rels we have plus some additional ones that
128 * we don't have.
129 *
130 * PGPA_ITM_TARGETS_ARE_SUBSET means that all targets are matched by
131 * identifiers, but there were leftover identifiers. Generally, this means
132 * that the advice is looking to apply to some but not all of the rels we have.
133 *
134 * PGPA_ITM_INTERSECTING means that some identifiers and targets were matched,
135 * but neither all identifiers nor all targets could be matched to items in
136 * the other set.
137 *
138 * PGPA_ITM_DISJOINT means that no matches between identifiers and targets were
139 * found.
140 */
141typedef enum
142{
143	PGPA_ITM_EQUAL,
144	PGPA_ITM_KEYS_ARE_SUBSET,
145	PGPA_ITM_TARGETS_ARE_SUBSET,
146	PGPA_ITM_INTERSECTING,
147	PGPA_ITM_DISJOINT
148} pgpa_itm_type;
