1LOAD 'pg_plan_advice';
2SET max_parallel_workers_per_gather = 0;
3CREATE TABLE alt_t1 (a int) WITH (autovacuum_enabled = false);
4CREATE TABLE alt_t2 (a int) WITH (autovacuum_enabled = false);
5CREATE INDEX ON alt_t2(a);
6INSERT INTO alt_t1 SELECT generate_series(1, 1000);
7INSERT INTO alt_t2 SELECT generate_series(1, 100000);
8VACUUM ANALYZE alt_t1;
9VACUUM ANALYZE alt_t2;
10-- This query uses an OR to prevent the EXISTS from being converted to a
11-- semi-join, forcing the planner through the AlternativeSubPlan path.
12EXPLAIN (COSTS OFF, PLAN_ADVICE)
13SELECT * FROM alt_t1
14WHERE EXISTS (SELECT 1 FROM alt_t2 WHERE alt_t2.a = alt_t1.a) OR alt_t1.a < 0;
15                            QUERY PLAN                             
16-------------------------------------------------------------------
17 Seq Scan on alt_t1
18   Filter: ((ANY (a = (hashed SubPlan exists_2).col1)) OR (a < 0))
19   SubPlan exists_2
20     ->  Seq Scan on alt_t2
21 Generated Plan Advice:
22   SEQ_SCAN(alt_t1 alt_t2@exists_2)
23   NO_GATHER(alt_t1 alt_t2@exists_2)
24   DO_NOT_SCAN(alt_t2@exists_1)
