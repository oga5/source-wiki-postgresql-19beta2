83Datum
84hstore_to_plpython(PG_FUNCTION_ARGS)
85{
86	HStore	   *in = PG_GETARG_HSTORE_P(0);
87	int			i;
88	int			count = HS_COUNT(in);
89	char	   *base = STRPTR(in);
90	HEntry	   *entries = ARRPTR(in);
91	PyObject   *dict;
92
93	dict = PyDict_New();
94	if (!dict)
95		ereport(ERROR,
96				(errcode(ERRCODE_OUT_OF_MEMORY),
97				 errmsg("out of memory")));
98
99	for (i = 0; i < count; i++)
100	{
101		PyObject   *key;
102
103		key = PLyUnicode_FromStringAndSize(HSTORE_KEY(entries, base, i),
104										   HSTORE_KEYLEN(entries, i));
105		if (HSTORE_VALISNULL(entries, i))
106			PyDict_SetItem(dict, key, Py_None);
107		else
108		{
109			PyObject   *value;
110
111			value = PLyUnicode_FromStringAndSize(HSTORE_VAL(entries, base, i),
112												 HSTORE_VALLEN(entries, i));
113			PyDict_SetItem(dict, key, value);
114			Py_XDECREF(value);
115		}
116		Py_XDECREF(key);
117	}
118
119	return PointerGetDatum(dict);
120}

123PG_FUNCTION_INFO_V1(plpython_to_hstore);
124
125Datum
126plpython_to_hstore(PG_FUNCTION_ARGS)
127{
128	PyObject   *dict;
129	PyObject   *volatile items;
130	Py_ssize_t	pcount;
131	HStore	   *volatile out;
132
133	dict = (PyObject *) PG_GETARG_POINTER(0);
134
135	/*
136	 * As of Python 3, PyMapping_Check() is unreliable unless one first checks
137	 * that the object isn't a sequence.  (Cleaner solutions exist, but not
138	 * before Python 3.10, which we're not prepared to require yet.)
139	 */
140	if (PySequence_Check(dict) || !PyMapping_Check(dict))
141		ereport(ERROR,
142				(errcode(ERRCODE_WRONG_OBJECT_TYPE),
143				 errmsg("not a Python mapping")));
144
145	pcount = PyMapping_Size(dict);
146	if (pcount < 0)
147		ereport(ERROR,
148				(errcode(ERRCODE_DATATYPE_MISMATCH),
149				 errmsg("could not get size of Python mapping")));
150
151	items = PyMapping_Items(dict);
152	if (items == NULL)
153		ereport(ERROR,
154				(errcode(ERRCODE_DATATYPE_MISMATCH),
155				 errmsg("could not get items from Python mapping")));
156
157	PG_TRY();
158	{
159		int32		buflen;
160		Py_ssize_t	i;
161		Pairs	   *pairs;
162
163		pairs = palloc_array(Pairs, pcount);
164
165		for (i = 0; i < pcount; i++)
166		{
167			PyObject   *tuple;
168			PyObject   *key;
169			PyObject   *value;
170
171			tuple = PyList_GetItem(items, i);
172
173			/* The mapping's items() must yield key/value pairs */
174			if (tuple == NULL || !PyTuple_Check(tuple) || PyTuple_Size(tuple) < 2)
175				ereport(ERROR,
176						(errcode(ERRCODE_DATATYPE_MISMATCH),
177						 errmsg("items() of a Python mapping must return key/value pairs")));
178
179			key = PyTuple_GetItem(tuple, 0);
180			value = PyTuple_GetItem(tuple, 1);
181
182			pairs[i].key = PLyObject_AsString(key);
183			pairs[i].keylen = hstoreCheckKeyLen(strlen(pairs[i].key));
184			pairs[i].needfree = true;
185
186			if (value == Py_None)
187			{
188				pairs[i].val = NULL;
189				pairs[i].vallen = 0;
190				pairs[i].isnull = true;
191			}
192			else
193			{
194				pairs[i].val = PLyObject_AsString(value);
195				pairs[i].vallen = hstoreCheckValLen(strlen(pairs[i].val));
196				pairs[i].isnull = false;
197			}
198		}
199
200		pcount = hstoreUniquePairs(pairs, pcount, &buflen);
201		out = hstorePairs(pairs, pcount, buflen);
202	}
203	PG_FINALLY();
204	{
205		Py_DECREF(items);
206	}
207	PG_END_TRY();
208
209	PG_RETURN_POINTER(out);
210}
