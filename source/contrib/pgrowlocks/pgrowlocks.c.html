51PG_FUNCTION_INFO_V1(pgrowlocks);
52
53/* ----------
54 * pgrowlocks:
55 * returns tids of rows being locked
56 * ----------
57 */
58
59#define NCHARS 32
60
61#define		Atnum_tid		0
62#define		Atnum_xmax		1
63#define		Atnum_ismulti	2
64#define		Atnum_xids		3
65#define		Atnum_modes		4
66#define		Atnum_pids		5
67
68Datum
69pgrowlocks(PG_FUNCTION_ARGS)
70{
71	text	   *relname = PG_GETARG_TEXT_PP(0);
72	ReturnSetInfo *rsinfo = (ReturnSetInfo *) fcinfo->resultinfo;
73	AttInMetadata *attinmeta;
74	Relation	rel;
75	RangeVar   *relrv;
76	TableScanDesc scan;
77	HeapScanDesc hscan;
78	HeapTuple	tuple;
79	AclResult	aclresult;
80	char	  **values;
81
82	InitMaterializedSRF(fcinfo, 0);
83
84	/* Access the table */
85	relrv = makeRangeVarFromNameList(textToQualifiedNameList(relname));
86	rel = relation_openrv(relrv, AccessShareLock);
87
88	if (rel->rd_rel->relkind == RELKIND_PARTITIONED_TABLE)
89		ereport(ERROR,
90				(errcode(ERRCODE_WRONG_OBJECT_TYPE),
91				 errmsg("\"%s\" is a partitioned table",
92						RelationGetRelationName(rel)),
93				 errdetail("Partitioned tables do not contain rows.")));
94	else if (rel->rd_rel->relkind != RELKIND_RELATION)
95		ereport(ERROR,
96				(errcode(ERRCODE_WRONG_OBJECT_TYPE),
97				 errmsg("\"%s\" is not a table",
98						RelationGetRelationName(rel))));
99	else if (rel->rd_rel->relam != HEAP_TABLE_AM_OID)
100		ereport(ERROR,
101				(errcode(ERRCODE_FEATURE_NOT_SUPPORTED),
102				 errmsg("only heap AM is supported")));
103
104	/*
105	 * check permissions: must have SELECT on table or be in
106	 * pg_stat_scan_tables
107	 */
108	aclresult = pg_class_aclcheck(RelationGetRelid(rel), GetUserId(),
109								  ACL_SELECT);
110	if (aclresult != ACLCHECK_OK)
111		aclresult = has_privs_of_role(GetUserId(), ROLE_PG_STAT_SCAN_TABLES) ? ACLCHECK_OK : ACLCHECK_NO_PRIV;
112
113	if (aclresult != ACLCHECK_OK)
114		aclcheck_error(aclresult, get_relkind_objtype(rel->rd_rel->relkind),
115					   RelationGetRelationName(rel));
116
117	/* Scan the relation */
118	scan = table_beginscan(rel, GetActiveSnapshot(), 0, NULL, SO_NONE);
119	hscan = (HeapScanDesc) scan;
120
121	attinmeta = TupleDescGetAttInMetadata(rsinfo->setDesc);
122
123	values = (char **) palloc(rsinfo->setDesc->natts * sizeof(char *));
124
125	while ((tuple = heap_getnext(scan, ForwardScanDirection)) != NULL)
126	{
127		TM_Result	htsu;
128		TransactionId xmax;
129		uint16		infomask;
130
131		/* must hold a buffer lock to call HeapTupleSatisfiesUpdate */
132		LockBuffer(hscan->rs_cbuf, BUFFER_LOCK_SHARE);
133
134		htsu = HeapTupleSatisfiesUpdate(tuple,
135										GetCurrentCommandId(false),
136										hscan->rs_cbuf);
137		xmax = HeapTupleHeaderGetRawXmax(tuple->t_data);
138		infomask = tuple->t_data->t_infomask;
139
140		/*
141		 * A tuple is locked if HTSU returns BeingModified.
142		 */
143		if (htsu == TM_BeingModified)
144		{
145			values[Atnum_tid] = DatumGetCString(DirectFunctionCall1(tidout,
146																	PointerGetDatum(&tuple->t_self)));
147
148			values[Atnum_xmax] = palloc(NCHARS * sizeof(char));
149			snprintf(values[Atnum_xmax], NCHARS, "%u", xmax);
150			if (infomask & HEAP_XMAX_IS_MULTI)
151			{
152				MultiXactMember *members;
153				int			nmembers;
154				bool		first = true;
155				bool		allow_old;
156
157				values[Atnum_ismulti] = pstrdup("true");
158
159				allow_old = HEAP_LOCKED_UPGRADED(infomask);
160				nmembers = GetMultiXactIdMembers(xmax, &members, allow_old,
161												 false);
162				if (nmembers == -1)
163				{
164					values[Atnum_xids] = "{0}";
165					values[Atnum_modes] = "{transient upgrade status}";
166					values[Atnum_pids] = "{0}";
167				}
168				else
169				{
170					int			j;
171
172					values[Atnum_xids] = palloc(NCHARS * nmembers);
173					values[Atnum_modes] = palloc(NCHARS * nmembers);
174					values[Atnum_pids] = palloc(NCHARS * nmembers);
175
176					strcpy(values[Atnum_xids], "{");
177					strcpy(values[Atnum_modes], "{");
178					strcpy(values[Atnum_pids], "{");
179
180					for (j = 0; j < nmembers; j++)
181					{
182						char		buf[NCHARS];
183
184						if (!first)
185						{
186							strcat(values[Atnum_xids], ",");
187							strcat(values[Atnum_modes], ",");
188							strcat(values[Atnum_pids], ",");
189						}
190						snprintf(buf, NCHARS, "%u", members[j].xid);
191						strcat(values[Atnum_xids], buf);
192						switch (members[j].status)
193						{
194							case MultiXactStatusUpdate:
195								snprintf(buf, NCHARS, "Update");
196								break;
197							case MultiXactStatusNoKeyUpdate:
198								snprintf(buf, NCHARS, "No Key Update");
199								break;
200							case MultiXactStatusForUpdate:
201								snprintf(buf, NCHARS, "For Update");
202								break;
203							case MultiXactStatusForNoKeyUpdate:
204								snprintf(buf, NCHARS, "For No Key Update");
205								break;
206							case MultiXactStatusForShare:
207								snprintf(buf, NCHARS, "For Share");
208								break;
209							case MultiXactStatusForKeyShare:
210								snprintf(buf, NCHARS, "For Key Share");
211								break;
212						}
213						strcat(values[Atnum_modes], buf);
214						snprintf(buf, NCHARS, "%d",
215								 BackendXidGetPid(members[j].xid));
216						strcat(values[Atnum_pids], buf);
217
218						first = false;
219					}
220
221					strcat(values[Atnum_xids], "}");
222					strcat(values[Atnum_modes], "}");
223					strcat(values[Atnum_pids], "}");
224				}
225			}
226			else
227			{
228				values[Atnum_ismulti] = pstrdup("false");
229
230				values[Atnum_xids] = palloc(NCHARS * sizeof(char));
231				snprintf(values[Atnum_xids], NCHARS, "{%u}", xmax);
232
233				values[Atnum_modes] = palloc(NCHARS);
234				if (infomask & HEAP_XMAX_LOCK_ONLY)
235				{
236					if (HEAP_XMAX_IS_SHR_LOCKED(infomask))
237						snprintf(values[Atnum_modes], NCHARS, "{For Share}");
238					else if (HEAP_XMAX_IS_KEYSHR_LOCKED(infomask))
239						snprintf(values[Atnum_modes], NCHARS, "{For Key Share}");
240					else if (HEAP_XMAX_IS_EXCL_LOCKED(infomask))
241					{
242						if (tuple->t_data->t_infomask2 & HEAP_KEYS_UPDATED)
243							snprintf(values[Atnum_modes], NCHARS, "{For Update}");
244						else
245							snprintf(values[Atnum_modes], NCHARS, "{For No Key Update}");
246					}
247					else
248						/* neither keyshare nor exclusive bit it set */
249						snprintf(values[Atnum_modes], NCHARS,
250								 "{transient upgrade status}");
251				}
252				else
253				{
254					if (tuple->t_data->t_infomask2 & HEAP_KEYS_UPDATED)
255						snprintf(values[Atnum_modes], NCHARS, "{Update}");
256					else
257						snprintf(values[Atnum_modes], NCHARS, "{No Key Update}");
258				}
259
260				values[Atnum_pids] = palloc(NCHARS * sizeof(char));
261				snprintf(values[Atnum_pids], NCHARS, "{%d}",
262						 BackendXidGetPid(xmax));
263			}
264
265			LockBuffer(hscan->rs_cbuf, BUFFER_LOCK_UNLOCK);
266
267			/* build a tuple */
268			tuple = BuildTupleFromCStrings(attinmeta, values);
269			tuplestore_puttuple(rsinfo->setResult, tuple);
270		}
271		else
272		{
273			LockBuffer(hscan->rs_cbuf, BUFFER_LOCK_UNLOCK);
274		}
275	}
276
277	table_endscan(scan);
278	table_close(rel, AccessShareLock);
279	return (Datum) 0;
280}
