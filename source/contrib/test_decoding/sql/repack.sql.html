1-- Test REPACK (CONCURRENTLY).
2-- This test isn't strictly about logical decoding per se, but
3-- REPACK (CONCURRENTLY) involves logical decoding and therefore requires
4-- to be run under higher than minimal wal_level, so we can't have them in
5-- the main regression test suite.
6
7-- Ownership of partitions is checked
8CREATE TABLE ptnowner(i int unique not null) PARTITION BY LIST (i);
9CREATE INDEX ptnowner_i_idx ON ptnowner(i);
10CREATE TABLE ptnowner1 PARTITION OF ptnowner FOR VALUES IN (1);
11CREATE ROLE regress_ptnowner;
12CREATE TABLE ptnowner2 PARTITION OF ptnowner FOR VALUES IN (2);
13ALTER TABLE ptnowner1 OWNER TO regress_ptnowner;
14SET SESSION AUTHORIZATION regress_ptnowner;
15ALTER TABLE ptnowner1 REPLICA IDENTITY USING INDEX ptnowner1_i_key;
16REPACK (CONCURRENTLY) ptnowner1;
17RESET SESSION AUTHORIZATION;
18ALTER TABLE ptnowner OWNER TO regress_ptnowner;
19CREATE TEMP TABLE ptnowner_oldnodes AS
20  SELECT oid, relname, relfilenode FROM pg_partition_tree('ptnowner') AS tree
21  JOIN pg_class AS c ON c.oid=tree.relid;
22SELECT a.relname, a.relfilenode=b.relfilenode FROM pg_class a
23  JOIN ptnowner_oldnodes b USING (oid) ORDER BY a.relname COLLATE "C";
24DROP TABLE ptnowner;
25DROP ROLE regress_ptnowner;
26
27-- Verify that REPACK (CONCURRENTLY) doesn't lose "attmissingval" columns
28CREATE TABLE rpk_missing (id int PRIMARY KEY);
29INSERT INTO rpk_missing SELECT generate_series(1, 3);
30ALTER TABLE rpk_missing ADD COLUMN a int DEFAULT 42;
31SELECT * FROM rpk_missing;
32REPACK (CONCURRENTLY) rpk_missing;
33SELECT * FROM rpk_missing;
34DROP TABLE rpk_missing;

36-- Error cases for concurrent mode
37
38-- Doesn't like partitioned tables
39CREATE TABLE clstrpart (a int) PARTITION BY RANGE (a);
40REPACK (CONCURRENTLY) clstrpart;
41
42-- Disallowed in catalogs
43REPACK (CONCURRENTLY) pg_class;
44
45-- Doesn't support TOAST tables directly
46CREATE TABLE repack_conc_toast (t text);
47SELECT reltoastrelid::regclass AS toast_rel
48FROM pg_class WHERE oid = 'repack_conc_toast'::regclass \gset
49\set VERBOSITY sqlstate
50REPACK (CONCURRENTLY) :toast_rel;
51\set VERBOSITY default
52DROP TABLE repack_conc_toast;
53
54-- Only support permanent tables, temp and unlogged tables are not supported
55CREATE TEMP TABLE repack_conc_temp (i int PRIMARY KEY);
56REPACK (CONCURRENTLY) repack_conc_temp;
57DROP TABLE repack_conc_temp;
58CREATE UNLOGGED TABLE repack_conc_unlogged (i int PRIMARY KEY);
59REPACK (CONCURRENTLY) repack_conc_unlogged;
60DROP TABLE repack_conc_unlogged;
61
62-- Doesn't support tables with REPLICA IDENTITY NOTHING, even if they have a primary key
63CREATE TABLE repack_conc_replident (i int PRIMARY KEY);
64ALTER TABLE repack_conc_replident REPLICA IDENTITY NOTHING;
65REPACK (CONCURRENTLY) repack_conc_replident;
66
67-- Doesn't support tables without a primary key or replica identity index
68ALTER TABLE repack_conc_replident DROP CONSTRAINT repack_conc_replident_pkey;
69ALTER TABLE repack_conc_replident REPLICA IDENTITY DEFAULT;
70REPACK (CONCURRENTLY) repack_conc_replident;
71
72-- Doesn't support tables with deferrable primary keys
73ALTER TABLE repack_conc_replident ADD PRIMARY KEY (i) DEFERRABLE;
74REPACK (CONCURRENTLY) repack_conc_replident;
75
76-- clean up
77DROP TABLE repack_conc_replident, clstrpart;
