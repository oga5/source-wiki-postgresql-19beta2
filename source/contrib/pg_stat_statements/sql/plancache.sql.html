5-- Setup
6CREATE OR REPLACE FUNCTION select_one_func(int) RETURNS VOID AS $$
7DECLARE
8  ret INT;
9BEGIN
10  SELECT $1 INTO ret;
11END;
12$$ LANGUAGE plpgsql;
13CREATE OR REPLACE PROCEDURE select_one_proc(int) AS $$
14DECLARE
15  ret INT;
16BEGIN
17  SELECT $1 INTO ret;
18END;
19$$ LANGUAGE plpgsql;
20
21-- Prepared statements
22SELECT pg_stat_statements_reset() IS NOT NULL AS t;
23PREPARE p1 AS SELECT $1 AS a;
24SET plan_cache_mode TO force_generic_plan;
25EXECUTE p1(1);
26SET plan_cache_mode TO force_custom_plan;
27EXECUTE p1(1);
28SELECT calls, generic_plan_calls, custom_plan_calls, query FROM pg_stat_statements
29  ORDER BY query COLLATE "C";
30DEALLOCATE p1;
31
32-- Extended query protocol
33SELECT pg_stat_statements_reset() IS NOT NULL AS t;
34SELECT $1 AS a \parse p1
35SET plan_cache_mode TO force_generic_plan;
36\bind_named p1 1
37;
38SET plan_cache_mode TO force_custom_plan;
39\bind_named p1 1
40;
41SELECT calls, generic_plan_calls, custom_plan_calls, query FROM pg_stat_statements
42  ORDER BY query COLLATE "C";
43\close_prepared p1
44
45-- EXPLAIN [ANALYZE] EXECUTE
46SET pg_stat_statements.track = 'all';
47SELECT pg_stat_statements_reset() IS NOT NULL AS t;
48PREPARE p1 AS SELECT $1;
49SET plan_cache_mode TO force_generic_plan;
50EXPLAIN (COSTS OFF) EXECUTE p1(1);
51EXPLAIN (ANALYZE, COSTS OFF, SUMMARY OFF, TIMING OFF, BUFFERS OFF) EXECUTE p1(1);
52SET plan_cache_mode TO force_custom_plan;
53EXPLAIN (COSTS OFF) EXECUTE p1(1);
54EXPLAIN (ANALYZE, COSTS OFF, SUMMARY OFF, TIMING OFF, BUFFERS OFF) EXECUTE p1(1);
55SELECT calls, generic_plan_calls, custom_plan_calls, toplevel, query FROM pg_stat_statements
56  ORDER BY query COLLATE "C";
57RESET pg_stat_statements.track;
58DEALLOCATE p1;
59
60-- Functions/procedures
61SET pg_stat_statements.track = 'all';
62SELECT pg_stat_statements_reset() IS NOT NULL AS t;
63SET plan_cache_mode TO force_generic_plan;
64SELECT select_one_func(1);
65CALL select_one_proc(1);
66SET plan_cache_mode TO force_custom_plan;
67SELECT select_one_func(1);
68CALL select_one_proc(1);
69SELECT calls, generic_plan_calls, custom_plan_calls, toplevel, query FROM pg_stat_statements
70  ORDER BY query COLLATE "C";
71
72--
73-- EXPLAIN [ANALYZE] EXECUTE + functions/procedures
74--
75SET pg_stat_statements.track = 'all';
76SELECT pg_stat_statements_reset() IS NOT NULL AS t;
77SET plan_cache_mode TO force_generic_plan;
78EXPLAIN (ANALYZE, COSTS OFF, SUMMARY OFF, TIMING OFF, BUFFERS OFF) SELECT select_one_func(1);
79EXPLAIN (COSTS OFF) SELECT select_one_func(1);
80CALL select_one_proc(1);
81SET plan_cache_mode TO force_custom_plan;
82EXPLAIN (ANALYZE, COSTS OFF, SUMMARY OFF, TIMING OFF, BUFFERS OFF) SELECT select_one_func(1);
83EXPLAIN (COSTS OFF) SELECT select_one_func(1);
84CALL select_one_proc(1);
85SELECT calls, generic_plan_calls, custom_plan_calls, toplevel, query FROM pg_stat_statements
86  ORDER BY query COLLATE "C", toplevel;
87
88RESET pg_stat_statements.track;

90--
91-- Procedure with internal ROLLBACK and the extended query protocol.
92-- The PlannedStmt used in pgss_ProcessUtility() is freed by the internal
93-- ROLLBACK.
94--
95CREATE OR REPLACE PROCEDURE rollback_proc(a INOUT int) AS $$
96BEGIN
97  ROLLBACK;
98END;
99$$ LANGUAGE plpgsql;
100SELECT pg_stat_statements_reset() IS NOT NULL AS t;
101CALL rollback_proc($1) \parse stmt_rollback
102\bind_named stmt_rollback 1 \g
103\bind_named stmt_rollback 2 \g
104SELECT calls, query FROM pg_stat_statements
105  WHERE query LIKE '%rollback_proc%'
106  ORDER BY query COLLATE "C";
107DROP PROCEDURE rollback_proc;

112DROP FUNCTION select_one_func(int);
113DROP PROCEDURE select_one_proc(int);
