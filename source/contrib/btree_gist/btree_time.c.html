90static int
91gbt_timekey_cmp(const void *a, const void *b, FmgrInfo *flinfo)
92{
93	timeKEY    *ia = (timeKEY *) (((const Nsrt *) a)->t);
94	timeKEY    *ib = (timeKEY *) (((const Nsrt *) b)->t);
95	int			res;
96
97	res = DatumGetInt32(DirectFunctionCall2(time_cmp, TimeADTGetDatum(ia->lower), TimeADTGetDatum(ib->lower)));
98	if (res == 0)
99		return DatumGetInt32(DirectFunctionCall2(time_cmp, TimeADTGetDatum(ia->upper), TimeADTGetDatum(ib->upper)));
100
101	return res;
102}

104static float8
105gbt_time_dist(const void *a, const void *b, FmgrInfo *flinfo)
106{
107	const TimeADT *aa = (const TimeADT *) a;
108	const TimeADT *bb = (const TimeADT *) b;
109	Interval   *i;
110
111	i = DatumGetIntervalP(DirectFunctionCall2(time_mi_time,
112											  TimeADTGetDatum(*aa),
113											  TimeADTGetDatum(*bb)));
114	return fabs(INTERVAL_TO_SEC(i));
115}

157Datum
158gbt_timetz_compress(PG_FUNCTION_ARGS)
159{
160	GISTENTRY  *entry = (GISTENTRY *) PG_GETARG_POINTER(0);
161	GISTENTRY  *retval;
162
163	if (entry->leafkey)
164	{
165		timeKEY    *r = palloc_object(timeKEY);
166		TimeTzADT  *tz = DatumGetTimeTzADTP(entry->key);
167		TimeADT		tmp;
168
169		retval = palloc_object(GISTENTRY);
170
171		/* We are using the time + zone only to compress */
172		tmp = tz->time + (tz->zone * INT64CONST(1000000));
173		r->lower = r->upper = tmp;
174		gistentryinit(*retval, PointerGetDatum(r),
175					  entry->rel, entry->page,
176					  entry->offset, false);
177	}
178	else
179		retval = entry;
180	PG_RETURN_POINTER(retval);
181}

232Datum
233gbt_timetz_consistent(PG_FUNCTION_ARGS)
234{
235	GISTENTRY  *entry = (GISTENTRY *) PG_GETARG_POINTER(0);
236	TimeTzADT  *query = PG_GETARG_TIMETZADT_P(1);
237	StrategyNumber strategy = (StrategyNumber) PG_GETARG_UINT16(2);
238#ifdef NOT_USED
239	Oid			subtype = PG_GETARG_OID(3);
240#endif
241	bool	   *recheck = (bool *) PG_GETARG_POINTER(4);
242	timeKEY    *kkk = (timeKEY *) DatumGetPointer(entry->key);
243	TimeADT		qqq;
244	GBT_NUMKEY_R key;
245
246	/* All cases served by this function are inexact */
247	*recheck = true;
248
249	qqq = query->time + (query->zone * INT64CONST(1000000));
250
251	key.lower = (GBT_NUMKEY *) &kkk->lower;
252	key.upper = (GBT_NUMKEY *) &kkk->upper;
253
254	PG_RETURN_BOOL(gbt_num_consistent(&key, &qqq, &strategy,
255									  GIST_LEAF(entry), &tinfo, fcinfo->flinfo));
256}

268Datum
269gbt_time_penalty(PG_FUNCTION_ARGS)
270{
271	timeKEY    *origentry = (timeKEY *) DatumGetPointer(((GISTENTRY *) PG_GETARG_POINTER(0))->key);
272	timeKEY    *newentry = (timeKEY *) DatumGetPointer(((GISTENTRY *) PG_GETARG_POINTER(1))->key);
273	float	   *result = (float *) PG_GETARG_POINTER(2);
274	Interval   *intr;
275	double		res;
276	double		res2;
277
278	intr = DatumGetIntervalP(DirectFunctionCall2(time_mi_time,
279												 TimeADTGetDatum(newentry->upper),
280												 TimeADTGetDatum(origentry->upper)));
281	res = INTERVAL_TO_SEC(intr);
282	res = Max(res, 0);
283
284	intr = DatumGetIntervalP(DirectFunctionCall2(time_mi_time,
285												 TimeADTGetDatum(origentry->lower),
286												 TimeADTGetDatum(newentry->lower)));
287	res2 = INTERVAL_TO_SEC(intr);
288	res2 = Max(res2, 0);
289
290	res += res2;
291
292	*result = 0.0;
293
294	if (res > 0)
295	{
296		intr = DatumGetIntervalP(DirectFunctionCall2(time_mi_time,
297													 TimeADTGetDatum(origentry->upper),
298													 TimeADTGetDatum(origentry->lower)));
299		*result += FLT_MIN;
300		*result += (float) (res / (res + INTERVAL_TO_SEC(intr)));
301		*result *= (FLT_MAX / (((GISTENTRY *) PG_GETARG_POINTER(0))->rel->rd_att->natts + 1));
302	}
303
304	PG_RETURN_POINTER(result);
305}

326static int
327gbt_timekey_ssup_cmp(Datum x, Datum y, SortSupport ssup)
328{
329	timeKEY    *arg1 = (timeKEY *) DatumGetPointer(x);
330	timeKEY    *arg2 = (timeKEY *) DatumGetPointer(y);
331
332	/* for leaf items we expect lower == upper, so only compare lower */
333	return DatumGetInt32(DirectFunctionCall2(time_cmp,
334											 TimeADTGetDatum(arg1->lower),
335											 TimeADTGetDatum(arg2->lower)));
336}
