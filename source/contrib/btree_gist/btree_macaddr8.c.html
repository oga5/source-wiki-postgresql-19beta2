60static int
61gbt_macad8key_cmp(const void *a, const void *b, FmgrInfo *flinfo)
62{
63	mac8KEY    *ia = (mac8KEY *) (((const Nsrt *) a)->t);
64	mac8KEY    *ib = (mac8KEY *) (((const Nsrt *) b)->t);
65	int			res;
66
67	res = DatumGetInt32(DirectFunctionCall2(macaddr8_cmp, Macaddr8PGetDatum(&ia->lower), Macaddr8PGetDatum(&ib->lower)));
68	if (res == 0)
69		return DatumGetInt32(DirectFunctionCall2(macaddr8_cmp, Macaddr8PGetDatum(&ia->upper), Macaddr8PGetDatum(&ib->upper)));
70
71	return res;
72}

94static uint64
95mac8_2_uint64(macaddr8 *m)
96{
97	unsigned char *mi = (unsigned char *) m;
98	uint64		res = 0;
99	int			i;
100
101	for (i = 0; i < 8; i++)
102		res += (((uint64) mi[i]) << ((uint64) ((7 - i) * 8)));
103	return res;
104}

122Datum
123gbt_macad8_consistent(PG_FUNCTION_ARGS)
124{
125	GISTENTRY  *entry = (GISTENTRY *) PG_GETARG_POINTER(0);
126	macaddr8   *query = (macaddr8 *) PG_GETARG_POINTER(1);
127	StrategyNumber strategy = (StrategyNumber) PG_GETARG_UINT16(2);
128#ifdef NOT_USED
129	Oid			subtype = PG_GETARG_OID(3);
130#endif
131	bool	   *recheck = (bool *) PG_GETARG_POINTER(4);
132	mac8KEY    *kkk = (mac8KEY *) DatumGetPointer(entry->key);
133	GBT_NUMKEY_R key;
134
135	/* All cases served by this function are exact */
136	*recheck = false;
137
138	key.lower = (GBT_NUMKEY *) &kkk->lower;
139	key.upper = (GBT_NUMKEY *) &kkk->upper;
140
141	PG_RETURN_BOOL(gbt_num_consistent(&key, query, &strategy,
142									  GIST_LEAF(entry), &tinfo, fcinfo->flinfo));
143}

155Datum
156gbt_macad8_penalty(PG_FUNCTION_ARGS)
157{
158	mac8KEY    *origentry = (mac8KEY *) DatumGetPointer(((GISTENTRY *) PG_GETARG_POINTER(0))->key);
159	mac8KEY    *newentry = (mac8KEY *) DatumGetPointer(((GISTENTRY *) PG_GETARG_POINTER(1))->key);
160	float	   *result = (float *) PG_GETARG_POINTER(2);
161	uint64		iorg[2],
162				inew[2];
163
164	iorg[0] = mac8_2_uint64(&origentry->lower);
165	iorg[1] = mac8_2_uint64(&origentry->upper);
166	inew[0] = mac8_2_uint64(&newentry->lower);
167	inew[1] = mac8_2_uint64(&newentry->upper);
168
169	penalty_num(result, iorg[0], iorg[1], inew[0], inew[1]);
170
171	PG_RETURN_POINTER(result);
172}

193static int
194gbt_macaddr8_ssup_cmp(Datum x, Datum y, SortSupport ssup)
195{
196	mac8KEY    *arg1 = (mac8KEY *) DatumGetPointer(x);
197	mac8KEY    *arg2 = (mac8KEY *) DatumGetPointer(y);
198
199	/* for leaf items we expect lower == upper, so only compare lower */
200	return DatumGetInt32(DirectFunctionCall2(macaddr8_cmp,
201											 Macaddr8PGetDatum(&arg1->lower),
202											 Macaddr8PGetDatum(&arg2->lower)));
203}
