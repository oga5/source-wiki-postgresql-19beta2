6--
7-- Mark gist_inet_ops and gist_cidr_ops opclasses as non-default.
8-- This is the first step on the way to eventually removing them.
9--
10-- There's no SQL command for this, so fake it with a manual update on
11-- pg_opclass.
12--
13DO LANGUAGE plpgsql
14$$
15DECLARE
16  my_schema pg_catalog.text := pg_catalog.quote_ident(pg_catalog.current_schema());
17  old_path pg_catalog.text := pg_catalog.current_setting('search_path');
18BEGIN
19-- for safety, transiently set search_path to just pg_catalog+pg_temp
20PERFORM pg_catalog.set_config('search_path', 'pg_catalog, pg_temp', true);
21
22UPDATE pg_catalog.pg_opclass
23SET opcdefault = false
24WHERE opcmethod = (SELECT oid FROM pg_catalog.pg_am WHERE amname = 'gist') AND
25      opcname IN ('gist_inet_ops', 'gist_cidr_ops') AND
26      opcnamespace = my_schema::pg_catalog.regnamespace;
27
28PERFORM pg_catalog.set_config('search_path', old_path, true);
29END
30$$;
