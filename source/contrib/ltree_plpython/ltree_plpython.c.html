37
38Datum
39ltree_to_plpython(PG_FUNCTION_ARGS)
40{
41	ltree	   *in = PG_GETARG_LTREE_P(0);
42	int			i;
43	PyObject   *list;
44	ltree_level *curlevel;
45
46	list = PyList_New(in->numlevel);
47	if (!list)
48		ereport(ERROR,
49				(errcode(ERRCODE_OUT_OF_MEMORY),
50				 errmsg("out of memory")));
51
52	curlevel = LTREE_FIRST(in);
53	for (i = 0; i < in->numlevel; i++)
54	{
55		PyList_SetItem(list, i, PLyUnicode_FromStringAndSize(curlevel->name, curlevel->len));
56		curlevel = LEVEL_NEXT(curlevel);
57	}
58
59	PG_FREE_IF_COPY(in, 0);
60
61	return PointerGetDatum(list);
62}
