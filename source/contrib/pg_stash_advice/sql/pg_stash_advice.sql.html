1CREATE EXTENSION pg_stash_advice;
2SET compute_query_id = on;
3SET max_parallel_workers_per_gather = 0;

5-- Helper: extract query identifier from EXPLAIN VERBOSE output.
6CREATE OR REPLACE FUNCTION get_query_id(query_text text) RETURNS bigint
7LANGUAGE plpgsql AS $$
8DECLARE
9    line text;
10    qid bigint;
11BEGIN
12    FOR line IN EXECUTE 'EXPLAIN (VERBOSE, FORMAT TEXT) ' || query_text
13    LOOP
14        IF line ~ 'Query Identifier:' THEN
15            qid := regexp_replace(line, '.*Query Identifier:\s*(-?\d+).*', '\1')::bigint;
16            RETURN qid;
17        END IF;
18    END LOOP;
19    RAISE EXCEPTION 'Query Identifier not found in EXPLAIN output';
20END;
21$$;

23CREATE TABLE aa_dim1 (id integer primary key, dim1 text, val1 int)
24	WITH (autovacuum_enabled = false);
25INSERT INTO aa_dim1 (id, dim1, val1)
26	SELECT g, 'some filler text ' || g, (g % 3) + 1
27	  FROM generate_series(1,100) g;
28VACUUM ANALYZE aa_dim1;
29
30CREATE TABLE aa_dim2 (id integer primary key, dim2 text, val2 int)
31	WITH (autovacuum_enabled = false);
32INSERT INTO aa_dim2 (id, dim2, val2)
33	SELECT g, 'some filler text ' || g, (g % 7) + 1
34	  FROM generate_series(1,1000) g;
35VACUUM ANALYZE aa_dim2;
36
37CREATE TABLE aa_fact (
38	id int primary key,
39	dim1_id integer not null references aa_dim1 (id),
40	dim2_id integer not null references aa_dim2 (id)
41) WITH (autovacuum_enabled = false);
42INSERT INTO aa_fact
43	SELECT g, (g%100)+1, (g%100)+1 FROM generate_series(1,100000) g;
44VACUUM ANALYZE aa_fact;

46-- Get the query identifier.
47SELECT get_query_id($$
48SELECT * FROM aa_fact f LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
49	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
50	WHERE val1 = 1 AND val2 = 1;
51$$) AS qid \gset

53-- Create an advice stash and point pg_stash_advice at it.
54SELECT pg_create_advice_stash('regress_stash');
55SET pg_stash_advice.stash_name = 'regress_stash';
56
57-- Run our test query for the first time with no stashed advice.
58EXPLAIN (COSTS OFF)
59SELECT * FROM aa_fact f LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
60	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
61	WHERE val1 = 1 AND val2 = 1;
62
63-- Force an index scan on dim1
64SELECT pg_set_stashed_advice('regress_stash', :'qid',
65	'INDEX_SCAN(d1 aa_dim1_pkey)');
66EXPLAIN (COSTS OFF) SELECT * FROM aa_fact f
67	LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
68	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
69	WHERE val1 = 1 AND val2 = 1;
70
71-- Force an alternative join order
72SELECT pg_set_stashed_advice('regress_stash', :'qid',
73	'join_order(f d1 d2)');
74EXPLAIN (COSTS OFF) SELECT * FROM aa_fact f
75	LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
76	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
77	WHERE val1 = 1 AND val2 = 1;
78
79-- Force an alternative join strategy
80SELECT pg_set_stashed_advice('regress_stash', :'qid',
81	'NESTED_LOOP_PLAIN(d1)');
82EXPLAIN (COSTS OFF) SELECT * FROM aa_fact f
83	LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
84	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
85	WHERE val1 = 1 AND val2 = 1;
86
87-- Add a useless extra entry to our test stash. Shouldn't change the result
88-- from the previous test.
89-- (If we're unlucky enough that this ever fails due to query ID actually
90-- being 1, then just put some other constant here. Seems unlikely.)
91SELECT pg_set_stashed_advice('regress_stash', 1, 'SEQ_SCAN(d1)');
92EXPLAIN (COSTS OFF) SELECT * FROM aa_fact f
93	LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
94	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
95	WHERE val1 = 1 AND val2 = 1;
96
97-- Try an empty stash to be sure it does nothing
98SELECT pg_create_advice_stash('regress_empty_stash');
99SET pg_stash_advice.stash_name = 'regress_empty_stash';
100EXPLAIN (COSTS OFF) SELECT * FROM aa_fact f
101	LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
102	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
103	WHERE val1 = 1 AND val2 = 1;
104
105-- Test that we can list each stash individually and all of them together,
106-- but not a nonexistent stash.
107SELECT * FROM pg_get_advice_stashes() ORDER BY stash_name;
108SELECT stash_name, advice_string
109	FROM pg_get_advice_stash_contents('regress_stash') ORDER BY advice_string;
110SELECT stash_name, advice_string
111	 FROM pg_get_advice_stash_contents('regress_empty_stash')
112	 ORDER BY advice_string;
113SELECT stash_name, advice_string
114	 FROM pg_get_advice_stash_contents(NULL) ORDER BY advice_string;
115SELECT stash_name, advice_string
116	 FROM pg_get_advice_stash_contents('no_such_stash')
117	 ORDER BY advice_string;
118
119-- Test that we can remove advice.
120SELECT pg_set_stashed_advice('regress_stash', :'qid', null);
121SET pg_stash_advice.stash_name = 'regress_stash';
122EXPLAIN (COSTS OFF) SELECT * FROM aa_fact f
123	LEFT JOIN aa_dim1 d1 ON f.dim1_id = d1.id
124	LEFT JOIN aa_dim2 d2 ON f.dim2_id = d2.id
125	WHERE val1 = 1 AND val2 = 1;
126SELECT * FROM pg_get_advice_stashes() ORDER BY stash_name;
127SELECT stash_name, advice_string
128	FROM pg_get_advice_stash_contents('regress_stash') ORDER BY advice_string;
129
130-- Can't create a stash that already exists, or drop one that doesn't.
131SELECT pg_create_advice_stash('regress_stash');
132SELECT pg_drop_advice_stash('no_such_stash');
133
134-- Can't add to or remove from a stash that does not exist.
135SELECT pg_set_stashed_advice('no_such_stash', 1, 'SEQ_SCAN(t)');
136SELECT pg_set_stashed_advice('no_such_stash', 1, null);
137
138-- Can't use query ID 0.
139SELECT pg_set_stashed_advice('regress_stash', 0, 'SEQ_SCAN(t)');
140
141-- Stash names must be non-empty, ASCII, and not too long, and must look
142-- like identifiers.
143SELECT pg_create_advice_stash('');
144SELECT pg_create_advice_stash('aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa');
145SELECT pg_create_advice_stash('   ');
146SET pg_stash_advice.stash_name = '99bottles';
147
148-- Clean up state in dynamic shared memory.
149SELECT pg_drop_advice_stash('regress_stash');
150SELECT pg_drop_advice_stash('regress_empty_stash');
