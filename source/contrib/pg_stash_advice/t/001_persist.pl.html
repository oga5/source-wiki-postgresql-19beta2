4use strict;
5use warnings FATAL => 'all';
6
7use PostgreSQL::Test::Cluster;
8use PostgreSQL::Test::Utils;
9use Test::More;
10
11my $node = PostgreSQL::Test::Cluster->new('main');
12
13$node->init;
14$node->append_conf(
15	'postgresql.conf',
16	qq{shared_preload_libraries = 'pg_plan_advice, pg_stash_advice'
17pg_stash_advice.persist = true
18pg_stash_advice.persist_interval = 0});
19$node->start;
20
21$node->safe_psql("postgres", "CREATE EXTENSION pg_stash_advice;\n");
22
23# Create two stashes: one with 2 entries, one with 1 entry.
24$node->safe_psql(
25	"postgres", qq{
26	SELECT pg_create_advice_stash('stash_a');
27	SELECT pg_set_stashed_advice('stash_a', 1001, 'IndexScan(t)');
28	SELECT pg_set_stashed_advice('stash_a', 1002, E'line1\\nline2\\ttab\\\\backslash');
29	SELECT pg_create_advice_stash('stash_b');
30	SELECT pg_set_stashed_advice('stash_b', 2001, 'SeqScan(t)');
31});
32
33# Verify before restart.
34my $result = $node->safe_psql("postgres",
35	"SELECT stash_name, num_entries FROM pg_get_advice_stashes() ORDER BY stash_name"
36);
37is($result, "stash_a|2\nstash_b|1", 'stashes present before restart');
38
39# Restart and verify the data survived.
40$node->restart;
41$node->wait_for_log("loaded 2 advice stashes and 3 entries");
42
43$result = $node->safe_psql("postgres",
44	"SELECT stash_name, num_entries FROM pg_get_advice_stashes() ORDER BY stash_name"
45);
46is($result, "stash_a|2\nstash_b|1", 'stashes survived restart');
47
48# Verify entry contents, including the one with special characters.
49$result = $node->safe_psql("postgres",
50	"SELECT stash_name, query_id, advice_string FROM pg_get_advice_stash_contents(NULL) ORDER BY stash_name, query_id"
51);
52is( $result,
53	"stash_a|1001|IndexScan(t)\nstash_a|1002|line1\nline2\ttab\\backslash\nstash_b|2001|SeqScan(t)",
54	'entry contents survived restart with special characters intact');
55
56# Add a third stash with 0 entries.
57$node->safe_psql(
58	"postgres", qq{
59	SELECT pg_create_advice_stash('stash_c');
60});
61
62# Restart again and verify all three stashes are present.
63$node->restart;
64$node->wait_for_log("loaded 3 advice stashes and 3 entries");
65
66$result = $node->safe_psql("postgres",
67	"SELECT stash_name, num_entries FROM pg_get_advice_stashes() ORDER BY stash_name"
68);
69is( $result,
70	"stash_a|2\nstash_b|1\nstash_c|0",
71	'all three stashes survived second restart');
72
73# Drop all stashes and verify the dump file is removed after restart.
74$node->safe_psql(
75	"postgres", qq{
76	SELECT pg_drop_advice_stash('stash_a');
77	SELECT pg_drop_advice_stash('stash_b');
78	SELECT pg_drop_advice_stash('stash_c');
79});
80
81$node->restart;
82
83$result = $node->safe_psql("postgres",
84	"SELECT count(*) FROM pg_get_advice_stashes()");
85is($result, "0", 'no stashes after dropping all and restarting');
86
87ok( !-f $node->data_dir . '/pg_stash_advice.tsv',
88	'dump file removed after all stashes dropped');
89
90$node->stop;
91
92done_testing();
