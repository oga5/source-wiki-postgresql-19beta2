29typedef struct pgsa_stash_count
30{
31	uint32		status;
32	uint64		pgsa_stash_id;
33	int64		num_entries;
34} pgsa_stash_count;

47/*
48 * SQL-callable function to create an advice stash
49 */
50Datum
51pg_create_advice_stash(PG_FUNCTION_ARGS)
52{
53	char	   *stash_name = text_to_cstring(PG_GETARG_TEXT_PP(0));
54
55	pgsa_check_stash_name(stash_name);
56	if (unlikely(pgsa_entry_dshash == NULL))
57		pgsa_attach();
58	pgsa_check_lockout();
59	LWLockAcquire(&pgsa_state->lock, LW_EXCLUSIVE);
60	pgsa_create_stash(stash_name);
61	LWLockRelease(&pgsa_state->lock);
62	PG_RETURN_VOID();
63}

83/*
84 * SQL-callable function to provide a list of advice stashes
85 */
86Datum
87pg_get_advice_stashes(PG_FUNCTION_ARGS)
88{
89	ReturnSetInfo *rsinfo = (ReturnSetInfo *) fcinfo->resultinfo;
90	dshash_seq_status iterator;
91	pgsa_entry *entry;
92	pgsa_stash *stash;
93	pgsa_stash_count_table_hash *chash;
94
95	InitMaterializedSRF(fcinfo, 0);
96
97	/* Attach to dynamic shared memory if not already done. */
98	if (unlikely(pgsa_entry_dshash == NULL))
99		pgsa_attach();
100
101	/* If stash data is still being restored from disk, ignore. */
102	if (pg_atomic_unlocked_test_flag(&pgsa_state->stashes_ready))
103		return (Datum) 0;
104
105	/* Tally up the number of entries per stash. */
106	chash = pgsa_stash_count_table_create(CurrentMemoryContext, 64, NULL);
107	dshash_seq_init(&iterator, pgsa_entry_dshash, true);
108	while ((entry = dshash_seq_next(&iterator)) != NULL)
109	{
110		pgsa_stash_count *c;
111		bool		found;
112
113		c = pgsa_stash_count_table_insert(chash,
114										  entry->key.pgsa_stash_id,
115										  &found);
116		if (!found)
117			c->num_entries = 1;
118		else
119			c->num_entries++;
120	}
121	dshash_seq_term(&iterator);
122
123	/* Emit results. */
124	dshash_seq_init(&iterator, pgsa_stash_dshash, true);
125	while ((stash = dshash_seq_next(&iterator)) != NULL)
126	{
127		Datum		values[2];
128		bool		nulls[2];
129		pgsa_stash_count *c;
130
131		values[0] = CStringGetTextDatum(stash->name);
132		nulls[0] = false;
133
134		c = pgsa_stash_count_table_lookup(chash, stash->pgsa_stash_id);
135		values[1] = Int64GetDatum(c == NULL ? 0 : c->num_entries);
136		nulls[1] = false;
137
138		tuplestore_putvalues(rsinfo->setResult, rsinfo->setDesc, values,
139							 nulls);
140	}
141	dshash_seq_term(&iterator);
142
143	return (Datum) 0;
144}

146/*
147 * SQL-callable function to provide advice stash contents
148 */
149Datum
150pg_get_advice_stash_contents(PG_FUNCTION_ARGS)
151{
152	ReturnSetInfo *rsinfo = (ReturnSetInfo *) fcinfo->resultinfo;
153	dshash_seq_status iterator;
154	char	   *stash_name = NULL;
155	pgsa_stash_name_table_hash *nhash = NULL;
156	uint64		stash_id = 0;
157	pgsa_entry *entry;
158
159	InitMaterializedSRF(fcinfo, 0);
160
161	/* Attach to dynamic shared memory if not already done. */
162	if (unlikely(pgsa_entry_dshash == NULL))
163		pgsa_attach();
164
165	/* If stash data is still being restored from disk, ignore. */
166	if (pg_atomic_unlocked_test_flag(&pgsa_state->stashes_ready))
167		return (Datum) 0;
168
169	/* User can pass NULL for all stashes, or the name of a specific stash. */
170	if (!PG_ARGISNULL(0))
171	{
172		stash_name = text_to_cstring(PG_GETARG_TEXT_PP(0));
173		pgsa_check_stash_name(stash_name);
174		stash_id = pgsa_lookup_stash_id(stash_name);
175
176		/* If the user specified a stash name, it should exist. */
177		if (stash_id == 0)
178			ereport(ERROR,
179					errcode(ERRCODE_INVALID_PARAMETER_VALUE),
180					errmsg("advice stash \"%s\" does not exist", stash_name));
181	}
182	else
183	{
184		pgsa_stash *stash;
185
186		/*
187		 * If we're dumping data about all stashes, we need an ID->name lookup
188		 * table.
189		 */
190		nhash = pgsa_stash_name_table_create(CurrentMemoryContext, 64, NULL);
191		dshash_seq_init(&iterator, pgsa_stash_dshash, true);
192		while ((stash = dshash_seq_next(&iterator)) != NULL)
193		{
194			pgsa_stash_name *n;
195			bool		found;
196
197			n = pgsa_stash_name_table_insert(nhash,
198											 stash->pgsa_stash_id,
199											 &found);
200			Assert(!found);
201			n->name = pstrdup(stash->name);
202		}
203		dshash_seq_term(&iterator);
204	}
205
206	/* Now iterate over all the entries. */
207	dshash_seq_init(&iterator, pgsa_entry_dshash, false);
208	while ((entry = dshash_seq_next(&iterator)) != NULL)
209	{
210		Datum		values[3];
211		bool		nulls[3];
212		char	   *this_stash_name;
213		char	   *advice_string;
214
215		/* Skip incomplete entries where the advice string was never set. */
216		if (entry->advice_string == InvalidDsaPointer)
217			continue;
218
219		if (stash_id != 0)
220		{
221			/*
222			 * We're only dumping data for one particular stash, so skip
223			 * entries for any other stash and use the stash name specified by
224			 * the user.
225			 */
226			if (stash_id != entry->key.pgsa_stash_id)
227				continue;
228			this_stash_name = stash_name;
229		}
230		else
231		{
232			pgsa_stash_name *n;
233
234			/*
235			 * We're dumping data for all stashes, so look up the correct name
236			 * to use in the hash table. If nothing is found, which is
237			 * possible due to race conditions, make up a string to use.
238			 */
239			n = pgsa_stash_name_table_lookup(nhash, entry->key.pgsa_stash_id);
240			if (n != NULL)
241				this_stash_name = n->name;
242			else
243				this_stash_name = psprintf("<stash %" PRIu64 ">",
244										   entry->key.pgsa_stash_id);
245		}
246
247		/* Work out tuple values. */
248		values[0] = CStringGetTextDatum(this_stash_name);
249		nulls[0] = false;
250		values[1] = Int64GetDatum(entry->key.queryId);
251		nulls[1] = false;
252		advice_string = dsa_get_address(pgsa_dsa_area, entry->advice_string);
253		values[2] = CStringGetTextDatum(advice_string);
254		nulls[2] = false;
255
256		/* Emit the tuple. */
257		tuplestore_putvalues(rsinfo->setResult, rsinfo->setDesc, values,
258							 nulls);
259	}
260	dshash_seq_term(&iterator);
261
262	return (Datum) 0;
263}

265/*
266 * SQL-callable function to update an advice stash entry for a particular
267 * query ID
268 *
269 * If the advice string (the third argument) is NULL, we delete any existing
270 * advice stash entry; otherwise, we either create an entry or update it with
271 * the new advice string.
272 */
273Datum
274pg_set_stashed_advice(PG_FUNCTION_ARGS)
275{
276	char	   *stash_name;
277	int64		queryId;
278
279	if (PG_ARGISNULL(0) || PG_ARGISNULL(1))
280		PG_RETURN_NULL();
281
282	/* Get and check advice stash name. */
283	stash_name = text_to_cstring(PG_GETARG_TEXT_PP(0));
284	pgsa_check_stash_name(stash_name);
285
286	/*
287	 * Get and check query ID.
288	 *
289	 * Query ID 0 means no query ID was computed, so reject that.
290	 */
291	queryId = PG_GETARG_INT64(1);
292	if (queryId == 0)
293		ereport(ERROR,
294				errcode(ERRCODE_INVALID_PARAMETER_VALUE),
295				errmsg("cannot set advice string for query ID 0"));
296
297	/* Attach to dynamic shared memory if not already done. */
298	if (unlikely(pgsa_entry_dshash == NULL))
299		pgsa_attach();
300
301	/* Don't allow writes if stash data is still being restored from disk. */
302	pgsa_check_lockout();
303
304	/* Now call the appropriate function to do the real work. */
305	if (PG_ARGISNULL(2))
306	{
307		LWLockAcquire(&pgsa_state->lock, LW_SHARED);
308		pgsa_clear_advice_string(stash_name, queryId);
309		LWLockRelease(&pgsa_state->lock);
310	}
311	else
312	{
313		char	   *advice_string = text_to_cstring(PG_GETARG_TEXT_PP(2));
314
315		LWLockAcquire(&pgsa_state->lock, LW_SHARED);
316		pgsa_set_advice_string(stash_name, queryId, advice_string);
317		LWLockRelease(&pgsa_state->lock);
318	}
319
320	PG_RETURN_VOID();
321}

323/*
324 * SQL-callable function to start the persistence background worker.
325 */
326Datum
327pg_start_stash_advice_worker(PG_FUNCTION_ARGS)
328{
329	pid_t		pid;
330
331	if (unlikely(pgsa_entry_dshash == NULL))
332		pgsa_attach();
333
334	LWLockAcquire(&pgsa_state->lock, LW_SHARED);
335	pid = pgsa_state->bgworker_pid;
336	LWLockRelease(&pgsa_state->lock);
337
338	if (pid != InvalidPid)
339		ereport(ERROR,
340				(errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
341				 errmsg("pg_stash_advice worker is already running under PID %d",
342						(int) pid)));
343
344	pgsa_start_worker();
345
346	PG_RETURN_VOID();
