66/*
67 * PLyUnicode_FromJsonbValue
68 *
69 * Transform string JsonbValue to Python string.
70 */
71static PyObject *
72PLyUnicode_FromJsonbValue(JsonbValue *jbv)
73{
74	Assert(jbv->type == jbvString);
75
76	return PLyUnicode_FromStringAndSize(jbv->val.string.val, jbv->val.string.len);
77}

79/*
80 * PLyUnicode_ToJsonbValue
81 *
82 * Transform Python string to JsonbValue.
83 */
84static void
85PLyUnicode_ToJsonbValue(PyObject *obj, JsonbValue *jbvElem)
86{
87	jbvElem->type = jbvString;
88	jbvElem->val.string.val = PLyObject_AsString(obj);
89	jbvElem->val.string.len = strlen(jbvElem->val.string.val);
90}

92/*
93 * PLyObject_FromJsonbValue
94 *
95 * Transform JsonbValue to PyObject.
96 */
97static PyObject *
98PLyObject_FromJsonbValue(JsonbValue *jsonbValue)
99{
100	switch (jsonbValue->type)
101	{
102		case jbvNull:
103			Py_RETURN_NONE;
104
105		case jbvBinary:
106			return PLyObject_FromJsonbContainer(jsonbValue->val.binary.data);
107
108		case jbvNumeric:
109			{
110				Datum		num;
111				char	   *str;
112
113				num = NumericGetDatum(jsonbValue->val.numeric);
114				str = DatumGetCString(DirectFunctionCall1(numeric_out, num));
115
116				return PyObject_CallFunction(decimal_constructor, "s", str);
117			}
118
119		case jbvString:
120			return PLyUnicode_FromJsonbValue(jsonbValue);
121
122		case jbvBool:
123			if (jsonbValue->val.boolean)
124				Py_RETURN_TRUE;
125			else
126				Py_RETURN_FALSE;
127
128		default:
129			elog(ERROR, "unexpected jsonb value type: %d", jsonbValue->type);
130			return NULL;
131	}
132}

134/*
135 * PLyObject_FromJsonbContainer
136 *
137 * Transform JsonbContainer to PyObject.
138 */
139static PyObject *
140PLyObject_FromJsonbContainer(JsonbContainer *jsonb)
141{
142	JsonbIteratorToken r;
143	JsonbValue	v;
144	JsonbIterator *it;
145	PyObject   *result;
146
147	/* this can recurse via PLyObject_FromJsonbValue() */
148	check_stack_depth();
149
150	it = JsonbIteratorInit(jsonb);
151	r = JsonbIteratorNext(&it, &v, true);
152
153	switch (r)
154	{
155		case WJB_BEGIN_ARRAY:
156			if (v.val.array.rawScalar)
157			{
158				JsonbValue	tmp;
159
160				if ((r = JsonbIteratorNext(&it, &v, true)) != WJB_ELEM ||
161					(r = JsonbIteratorNext(&it, &tmp, true)) != WJB_END_ARRAY ||
162					(r = JsonbIteratorNext(&it, &tmp, true)) != WJB_DONE)
163					elog(ERROR, "unexpected jsonb token: %d", r);
164
165				result = PLyObject_FromJsonbValue(&v);
166			}
167			else
168			{
169				PyObject   *volatile elem = NULL;
170
171				result = PyList_New(0);
172				if (!result)
173					return NULL;
174
175				PG_TRY();
176				{
177					while ((r = JsonbIteratorNext(&it, &v, true)) != WJB_DONE)
178					{
179						if (r != WJB_ELEM)
180							continue;
181
182						elem = PLyObject_FromJsonbValue(&v);
183
184						PyList_Append(result, elem);
185						Py_XDECREF(elem);
186						elem = NULL;
187					}
188				}
189				PG_CATCH();
190				{
191					Py_XDECREF(elem);
192					Py_XDECREF(result);
193					PG_RE_THROW();
194				}
195				PG_END_TRY();
196			}
197			break;
198
199		case WJB_BEGIN_OBJECT:
200			{
201				PyObject   *volatile result_v = PyDict_New();
202				PyObject   *volatile key = NULL;
203				PyObject   *volatile val = NULL;
204
205				if (!result_v)
206					return NULL;
207
208				PG_TRY();
209				{
210					while ((r = JsonbIteratorNext(&it, &v, true)) != WJB_DONE)
211					{
212						if (r != WJB_KEY)
213							continue;
214
215						key = PLyUnicode_FromJsonbValue(&v);
216						if (!key)
217						{
218							Py_XDECREF(result_v);
219							result_v = NULL;
220							break;
221						}
222
223						if ((r = JsonbIteratorNext(&it, &v, true)) != WJB_VALUE)
224							elog(ERROR, "unexpected jsonb token: %d", r);
225
226						val = PLyObject_FromJsonbValue(&v);
227						if (!val)
228						{
229							Py_XDECREF(key);
230							key = NULL;
231							Py_XDECREF(result_v);
232							result_v = NULL;
233							break;
234						}
235
236						PyDict_SetItem(result_v, key, val);
237
238						Py_XDECREF(key);
239						key = NULL;
240						Py_XDECREF(val);
241						val = NULL;
242					}
243				}
244				PG_CATCH();
245				{
246					Py_XDECREF(result_v);
247					Py_XDECREF(key);
248					Py_XDECREF(val);
249					PG_RE_THROW();
250				}
251				PG_END_TRY();
252
253				result = result_v;
254			}
255			break;
256
257		default:
258			elog(ERROR, "unexpected jsonb token: %d", r);
259			return NULL;
260	}
261
262	return result;
263}

265/*
266 * PLyMapping_ToJsonbValue
267 *
268 * Transform Python dict to JsonbValue.
269 */
270static void
271PLyMapping_ToJsonbValue(PyObject *obj, JsonbInState *jsonb_state)
272{
273	Py_ssize_t	pcount;
274	PyObject   *volatile items;
275
276	pcount = PyMapping_Size(obj);
277	if (pcount < 0)
278		PLy_elog(ERROR, "could not get size of Python mapping");
279
280	items = PyMapping_Items(obj);
281	if (items == NULL)
282		PLy_elog(ERROR, "could not get items from Python mapping");
283
284	PG_TRY();
285	{
286		Py_ssize_t	i;
287
288		pushJsonbValue(jsonb_state, WJB_BEGIN_OBJECT, NULL);
289
290		for (i = 0; i < pcount; i++)
291		{
292			JsonbValue	jbvKey;
293			PyObject   *item = PyList_GetItem(items, i);
294			PyObject   *key;
295			PyObject   *value;
296
297			/* The mapping's items() must yield key/value pairs */
298			if (item == NULL || !PyTuple_Check(item) || PyTuple_Size(item) < 2)
299				PLy_elog(ERROR, "items() of a Python mapping must return key/value pairs");
300
301			key = PyTuple_GetItem(item, 0);
302			value = PyTuple_GetItem(item, 1);
303
304			/* Python dictionary can have None as key */
305			if (key == Py_None)
306			{
307				jbvKey.type = jbvString;
308				jbvKey.val.string.len = 0;
309				jbvKey.val.string.val = "";
310			}
311			else
312			{
313				/* All others types of keys we serialize to string */
314				PLyUnicode_ToJsonbValue(key, &jbvKey);
315			}
316
317			pushJsonbValue(jsonb_state, WJB_KEY, &jbvKey);
318			PLyObject_ToJsonbValue(value, jsonb_state, false);
319		}
320
321		pushJsonbValue(jsonb_state, WJB_END_OBJECT, NULL);
322	}
323	PG_FINALLY();
324	{
325		Py_DECREF(items);
326	}
327	PG_END_TRY();
328}

330/*
331 * PLySequence_ToJsonbValue
332 *
333 * Transform python list to JsonbValue. Expects transformed PyObject and
334 * a state required for jsonb construction.
335 */
336static void
337PLySequence_ToJsonbValue(PyObject *obj, JsonbInState *jsonb_state)
338{
339	Py_ssize_t	i;
340	Py_ssize_t	pcount;
341	PyObject   *volatile value = NULL;
342
343	pcount = PySequence_Size(obj);
344
345	pushJsonbValue(jsonb_state, WJB_BEGIN_ARRAY, NULL);
346
347	PG_TRY();
348	{
349		for (i = 0; i < pcount; i++)
350		{
351			value = PySequence_GetItem(obj, i);
352
353			/* PySequence_GetItem() can return NULL, with an exception set */
354			if (value == NULL)
355				PLy_elog(ERROR, "could not get element %d from sequence", (int) i);
356
357			PLyObject_ToJsonbValue(value, jsonb_state, true);
358			Py_XDECREF(value);
359			value = NULL;
360		}
361	}
362	PG_CATCH();
363	{
364		Py_XDECREF(value);
365		PG_RE_THROW();
366	}
367	PG_END_TRY();
368
369	pushJsonbValue(jsonb_state, WJB_END_ARRAY, NULL);
370}

372/*
373 * PLyNumber_ToJsonbValue(PyObject *obj)
374 *
375 * Transform python number to JsonbValue.
376 */
377static JsonbValue *
378PLyNumber_ToJsonbValue(PyObject *obj, JsonbValue *jbvNum)
379{
380	Numeric		num;
381	char	   *str = PLyObject_AsString(obj);
382
383	PG_TRY();
384	{
385		Datum		numd;
386
387		numd = DirectFunctionCall3(numeric_in,
388								   CStringGetDatum(str),
389								   ObjectIdGetDatum(InvalidOid),
390								   Int32GetDatum(-1));
391		num = DatumGetNumeric(numd);
392	}
393	PG_CATCH();
394	{
395		ereport(ERROR,
396				(errcode(ERRCODE_DATATYPE_MISMATCH),
397				 errmsg("could not convert value \"%s\" to jsonb", str)));
398	}
399	PG_END_TRY();
400
401	pfree(str);
402
403	/*
404	 * jsonb doesn't allow NaN or infinity (per JSON specification), so we
405	 * have to reject those here explicitly.
406	 */
407	if (numeric_is_nan(num))
408		ereport(ERROR,
409				(errcode(ERRCODE_NUMERIC_VALUE_OUT_OF_RANGE),
410				 errmsg("cannot convert NaN to jsonb")));
411	if (numeric_is_inf(num))
412		ereport(ERROR,
413				(errcode(ERRCODE_NUMERIC_VALUE_OUT_OF_RANGE),
414				 errmsg("cannot convert infinity to jsonb")));
415
416	jbvNum->type = jbvNumeric;
417	jbvNum->val.numeric = num;
418
419	return jbvNum;
420}

422/*
423 * PLyObject_ToJsonbValue(PyObject *obj)
424 *
425 * Transform python object to JsonbValue.
426 */
427static void
428PLyObject_ToJsonbValue(PyObject *obj, JsonbInState *jsonb_state, bool is_elem)
429{
430	JsonbValue *out;
431
432	/* this can recurse via PLyMapping_ToJsonbValue() */
433	check_stack_depth();
434
435	if (!PyUnicode_Check(obj))
436	{
437		if (PySequence_Check(obj))
438		{
439			PLySequence_ToJsonbValue(obj, jsonb_state);
440			return;
441		}
442		else if (PyMapping_Check(obj))
443		{
444			PLyMapping_ToJsonbValue(obj, jsonb_state);
445			return;
446		}
447	}
448
449	out = palloc_object(JsonbValue);
450
451	if (obj == Py_None)
452		out->type = jbvNull;
453	else if (PyUnicode_Check(obj))
454		PLyUnicode_ToJsonbValue(obj, out);
455
456	/*
457	 * PyNumber_Check() returns true for booleans, so boolean check should
458	 * come first.
459	 */
460	else if (PyBool_Check(obj))
461	{
462		out->type = jbvBool;
463		out->val.boolean = (obj == Py_True);
464	}
465	else if (PyNumber_Check(obj))
466		out = PLyNumber_ToJsonbValue(obj, out);
467	else
468		ereport(ERROR,
469				(errcode(ERRCODE_FEATURE_NOT_SUPPORTED),
470				 errmsg("Python type \"%s\" cannot be transformed to jsonb",
471						PLyObject_AsString((PyObject *) obj->ob_type))));
472
473	if (jsonb_state->parseState)
474	{
475		/* We're in an array or object, so push value as element or field. */
476		pushJsonbValue(jsonb_state, is_elem ? WJB_ELEM : WJB_VALUE, out);
477	}
478	else
479	{
480		/*
481		 * We are at top level, so it's a raw scalar.  If we just shove the
482		 * scalar value into jsonb_state->result, JsonbValueToJsonb will take
483		 * care of wrapping it into a dummy array.
484		 */
485		jsonb_state->result = out;
486	}
487}

489/*
490 * plpython_to_jsonb
491 *
492 * Transform python object to Jsonb datum
493 */
494PG_FUNCTION_INFO_V1(plpython_to_jsonb);
495Datum
496plpython_to_jsonb(PG_FUNCTION_ARGS)
497{
498	PyObject   *obj = (PyObject *) PG_GETARG_POINTER(0);
499	JsonbInState jsonb_state = {0};
500
501	PLyObject_ToJsonbValue(obj, &jsonb_state, true);
502	PG_RETURN_POINTER(JsonbValueToJsonb(jsonb_state.result));
503}

505/*
506 * jsonb_to_plpython
507 *
508 * Transform Jsonb datum to PyObject and return it as internal.
509 */
510PG_FUNCTION_INFO_V1(jsonb_to_plpython);
511Datum
512jsonb_to_plpython(PG_FUNCTION_ARGS)
513{
514	PyObject   *result;
515	Jsonb	   *in = PG_GETARG_JSONB_P(0);
516
517	/*
518	 * Initialize pointer to Decimal constructor. First we try "cdecimal", C
519	 * version of decimal library. In case of failure we use slower "decimal"
520	 * module.
521	 */
522	if (!decimal_constructor)
523	{
524		PyObject   *decimal_module = PyImport_ImportModule("cdecimal");
525
526		if (!decimal_module)
527		{
528			PyErr_Clear();
529			decimal_module = PyImport_ImportModule("decimal");
530		}
531		Assert(decimal_module);
532		decimal_constructor = PyObject_GetAttrString(decimal_module, "Decimal");
533	}
534
535	result = PLyObject_FromJsonbContainer(&in->root);
536	if (!result)
537		PLy_elog(ERROR, "transformation from jsonb to Python failed");
538
539	return PointerGetDatum(result);
