59# Helper routine to convert a space-separated list of Unicode characters to
60# hexadecimal list format, suitable for outputting in a C array.
61sub codepoint_string_to_hex
62{
63	my $codepoint_string = shift;
64
65	my $result;
66
67	foreach (split(' ', $codepoint_string))
68	{
69		my $cp = $_;
70		my $utf8 = "0x$cp, ";
71		$result .= $utf8;
72	}
73	$result .= '0';    # null-terminated the array
74	return $result;
75}

77# Process the input file line by line
78my $linenum = 0;
79while (my $line = <$INPUT>)
80{
81	$linenum = $linenum + 1;
82	if ($line =~ /^\s*#/) { next; }    # ignore comments
83
84	if ($line =~ /^@/) { next; }       # ignore @Part0 like headers
85
86	# Split the line wanted and get the fields needed:
87	#
88	# source; NFC; NFD; NFKC; NFKD
89	my ($source, $nfc, $nfd, $nfkc, $nfkd) = split(';', $line);
90
91	my $source_utf8 = codepoint_string_to_hex($source);
92	my $nfc_utf8 = codepoint_string_to_hex($nfc);
93	my $nfd_utf8 = codepoint_string_to_hex($nfd);
94	my $nfkc_utf8 = codepoint_string_to_hex($nfkc);
95	my $nfkd_utf8 = codepoint_string_to_hex($nfkd);
96
97	print $OUTPUT
98	  "\t{ $linenum, { $source_utf8 }, { { $nfc_utf8 }, { $nfd_utf8 }, { $nfkc_utf8 }, { $nfkd_utf8 } } },\n";
99}

101# Output terminator entry
102print $OUTPUT "\t{ 0, { 0 }, { { 0 }, { 0 }, { 0 }, { 0 } } }";
103print $OUTPUT "\n};\n";

105close $OUTPUT;
106close $INPUT;
