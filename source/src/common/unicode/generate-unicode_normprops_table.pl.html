src/common/unicode/generate-unicode_normprops_table.pl
1#!/usr/bin/perl
2#
3# Generate table of Unicode normalization "quick check" properties
4# (see UAX #15).  Pass DerivedNormalizationProps.txt as argument.  The
5# output is on stdout.
6#
7# Copyright (c) 2020-2026, PostgreSQL Global Development Group
8
9use strict;
10use warnings FATAL => 'all';
11
12use FindBin;
13use lib "$FindBin::RealBin/../../tools/";
14use PerfectHash;
15
16my %data;
17
18print
19  "/* generated by src/common/unicode/generate-unicode_normprops_table.pl, do not edit */\n\n";
20
21print <<EOS;
22#include "common/unicode_norm.h"
23
24/*
25 * Normalization quick check entry for codepoint.  We use a bit field
26 * here to save space.
27 */
28typedef struct
29{
30	unsigned int codepoint:21;
31	signed int	quickcheck:4;	/* really UnicodeNormalizationQC */
32} pg_unicode_normprops;
33
34/* Typedef for hash function on quick check table */
35typedef int (*qc_hash_func) (const void *key);
36
37/* Information for quick check lookup with perfect hash function */
38typedef struct
39{
40	const pg_unicode_normprops *normprops;
41	qc_hash_func	hash;
42	int		num_normprops;
43} pg_unicode_norminfo;
44EOS
45
46foreach my $line (<ARGV>)
47{
48	chomp $line;
49	$line =~ s/\s*#.*$//;
50	next if $line eq '';
51	my ($codepoint, $prop, $value) = split /\s*;\s*/, $line;
52	next if $prop !~ /_QC/;
53
54	my ($first, $last);
55	if ($codepoint =~ /\.\./)
56	{
57		($first, $last) = split /\.\./, $codepoint;
58	}
59	else
60	{
61		$first = $last = $codepoint;
62	}
63
64	foreach my $cp (hex($first) .. hex($last))
65	{
66		$data{$prop}{$cp} = $value;
67	}
68}
69
70# We create a separate array for each normalization form rather than,
71# say, a two-dimensional array, because that array would be very
72# sparse and would create unnecessary overhead especially for the NFC
73# lookup.
74foreach my $prop (sort keys %data)
75{
76	# Don't build the tables for the "D" forms because they are too
77	# big.  See also unicode_is_normalized_quickcheck().
78	next if $prop eq "NFD_QC" || $prop eq "NFKD_QC";
79
80	print "\n";
81	print
82	  "static const pg_unicode_normprops UnicodeNormProps_${prop}[] = {\n";
83
84	my %subdata = %{ $data{$prop} };
85	my @cp_packed;
86	foreach my $cp (sort { $a <=> $b } keys %subdata)
87	{
88		my $qc;
89		if ($subdata{$cp} eq 'N')
90		{
91			$qc = 'UNICODE_NORM_QC_NO';
92		}
93		elsif ($subdata{$cp} eq 'M')
94		{
95			$qc = 'UNICODE_NORM_QC_MAYBE';
96		}
97		else
98		{
99			die;
100		}
101		printf "\t{0x%04X, %s},\n", $cp, $qc;
102
103		# Save the bytes as a string in network order.
104		push @cp_packed, pack('N', $cp);
105	}
106
107	print "};\n";
108
109	# Emit the definition of the perfect hash function.
110	my $funcname = $prop . '_hash_func';
111	my $f = PerfectHash::generate_hash_function(\@cp_packed, $funcname,
112		fixed_key_length => 4);
113	printf "\n/* Perfect hash function for %s */", $prop;
114	print "\nstatic $f\n";
115
116	# Emit the structure that wraps the hash lookup information into
117	# one variable.
118	printf "/* Hash lookup information for %s */", $prop;
119	printf "\nstatic const pg_unicode_norminfo ";
120	printf "UnicodeNormInfo_%s = {\n", $prop;
121	printf "\tUnicodeNormProps_%s,\n", $prop;
122	printf "\t%s,\n", $funcname;
123	printf "\t%d\n", scalar @cp_packed;
124	printf "};\n";
125}
