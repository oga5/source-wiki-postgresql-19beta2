11CREATE FUNCTION subtransaction_ctx_test(what_error text = NULL) RETURNS text
12AS $$
13with plpy.subtransaction():
14    plpy.execute("INSERT INTO subtransaction_tbl VALUES (1)")
15    plpy.execute("INSERT INTO subtransaction_tbl VALUES (2)")
16    if what_error == "SPI":
17        plpy.execute("INSERT INTO subtransaction_tbl VALUES ('oops')")
18    elif what_error == "Python":
19        raise Exception("Python exception")
20$$ LANGUAGE plpython3u;

32-- Nested subtransactions
33
34CREATE FUNCTION subtransaction_nested_test(swallow boolean = 'f') RETURNS text
35AS $$
36plpy.execute("INSERT INTO subtransaction_tbl VALUES (1)")
37with plpy.subtransaction():
38    plpy.execute("INSERT INTO subtransaction_tbl VALUES (2)")
39    try:
40        with plpy.subtransaction():
41            plpy.execute("INSERT INTO subtransaction_tbl VALUES (3)")
42            plpy.execute("error")
43    except plpy.SPIError as e:
44        if not swallow:
45            raise
46        plpy.notice("Swallowed %s(%r)" % (e.__class__.__name__, e.args[0]))
47return "ok"
48$$ LANGUAGE plpython3u;
49
50SELECT subtransaction_nested_test();
51SELECT * FROM subtransaction_tbl;
52TRUNCATE subtransaction_tbl;

54SELECT subtransaction_nested_test('t');
55SELECT * FROM subtransaction_tbl;
56TRUNCATE subtransaction_tbl;

146AS $$

148try:

150        plpy.execute("INSERT INTO subtransaction_tbl VALUES (1)")
151        plpy.execute(p, [2])

154    plpy.warning("Caught a SPI error from an explicit subtransaction")

160except plpy.SPIError:

162$$ LANGUAGE plpython3u;

212-- cursor/subtransactions interactions
213
214CREATE FUNCTION cursor_in_subxact() RETURNS int AS $$
215with plpy.subtransaction():
216    cur = plpy.cursor("select * from generate_series(1, 20) as gen(i)")
217    cur.fetch(10)
218fetched = cur.fetch(10);
219return int(fetched[5]["i"])
220$$ LANGUAGE plpython3u;
221
222CREATE FUNCTION cursor_aborted_subxact() RETURNS int AS $$
223try:
224    with plpy.subtransaction():
225        cur = plpy.cursor("select * from generate_series(1, 20) as gen(i)")
226        cur.fetch(10);
227        plpy.execute("select no_such_function()")
228except plpy.SPIError:
229    fetched = cur.fetch(10)
230    return int(fetched[5]["i"])
231return 0 # not reached
232$$ LANGUAGE plpython3u;
233
234CREATE FUNCTION cursor_plan_aborted_subxact() RETURNS int AS $$
235try:
236    with plpy.subtransaction():
237        plpy.execute('create temporary table tmp(i) '
238                     'as select generate_series(1, 10)')
239        plan = plpy.prepare("select i from tmp")
240        cur = plpy.cursor(plan)
241        plpy.execute("select no_such_function()")
242except plpy.SPIError:
243    fetched = cur.fetch(5)
244    return fetched[2]["i"]
245return 0 # not reached
246$$ LANGUAGE plpython3u;
247
248CREATE FUNCTION cursor_close_aborted_subxact() RETURNS boolean AS $$
249try:
250    with plpy.subtransaction():
251        cur = plpy.cursor('select 1')
252        plpy.execute("select no_such_function()")
253except plpy.SPIError:
254    cur.close()
255    return True
256return False # not reached
257$$ LANGUAGE plpython3u;
