4use strict;
5use warnings FATAL => 'all';
6
7use PostgreSQL::Test::Utils;
8use Test::More;
9
10#########################################
11# Basic checks
12
13program_help_ok('pg_test_timing');
14program_version_ok('pg_test_timing');
15program_options_handling_ok('pg_test_timing');
16
17#########################################
18# Test invalid option combinations
19
20command_fails_like(
21	[ 'pg_test_timing', '--duration' => 'a' ],
22	qr/\Qpg_test_timing: invalid argument for option --duration\E/,
23	'pg_test_timing: invalid argument for option --duration');
24command_fails_like(
25	[ 'pg_test_timing', '--duration' => '0' ],
26	qr/\Qpg_test_timing: --duration must be in range 1..4294967295\E/,
27	'pg_test_timing: --duration must be in range');
28command_fails_like(
29	[ 'pg_test_timing', '--cutoff' => '101' ],
30	qr/\Qpg_test_timing: --cutoff must be in range 0..100\E/,
31	'pg_test_timing: --cutoff must be in range');
32
33#########################################
34# We obviously can't check for specific output, but we can
35# do a simple run and make sure it produces something.
36# Also, note the output in the log for data collection purposes.
37
38my $cmd = [ 'pg_test_timing', '--duration' => '1' ];
39my ($stdout, $stderr);
40print("# Running: " . join(" ", @{$cmd}) . "\n");
41my $result = IPC::Run::run $cmd, '>' => \$stdout, '2>' => \$stderr;
42is($result, 1, "pg_test_timing: exit code 0");
43is($stderr, '', "pg_test_timing: no stderr");
44like(
45	$stdout,
46	qr/
47\QTesting timing overhead for 1 second.\E.*
48\QHistogram of timing durations:\E.*
49\QObserved timing durations up to 99.9900%:\E
50/sx,
51	'pg_test_timing: stdout passes sanity check');
52
53note "pg_test_timing results:\n$stdout\n";
54
55done_testing();
