3# Test pg_upgrade with the extension_control_path GUC active.
4
5use strict;
6use warnings FATAL => 'all';
7
8use File::Copy;
9use File::Path;
10use PostgreSQL::Test::Cluster;
11use PostgreSQL::Test::Utils;
12use Test::More;
13
14# Make sure the extension file .so path is provided
15my $ext_lib_so = $ENV{TEST_EXT_LIB}
16  or die "couldn't get the extension so path";
17
18# Create the custom extension directory layout:
19#   $ext_dir/extension/  -- .control and .sql files
20#   $ext_dir/lib/        -- .so file
21my $ext_dir = PostgreSQL::Test::Utils::tempdir();
22mkpath("$ext_dir/extension");
23mkpath("$ext_dir/lib");
24my $ext_lib = $ext_dir . '/lib';
25
26# Copy the .so file into the lib/ subdirectory.
27copy($ext_lib_so, $ext_lib)
28  or die "could not copy '$ext_lib_so' to '$ext_lib': $!";
29
30create_extension_files('test_ext', $ext_dir);
31
32my $sep = $windows_os ? ";" : ":";
33my $ext_path = $windows_os ? ($ext_dir =~ s/\\/\\\\/gr) : $ext_dir;
34my $ext_lib_path = $windows_os ? ($ext_lib =~ s/\\/\\\\/gr) : $ext_lib;
35
36my $extension_control_path_conf = qq(
37extension_control_path = '\$system$sep$ext_path'
38dynamic_library_path = '\$libdir$sep$ext_lib_path'
39);
40
41my $old =
42  PostgreSQL::Test::Cluster->new('old', install_path => $ENV{oldinstall});
43$old->init;
44
45# Configure extension_control_path so the .control file is found in our
46# extension/ directory, and dynamic_library_path so the .so is found in lib/.
47$old->append_conf('postgresql.conf', $extension_control_path_conf);
48
49$old->start;
50
51# CREATE EXTENSION 'test_ext'
52$old->safe_psql('postgres', 'CREATE EXTENSION test_ext');
53
54# Verify the extension works before the upgrade.
55my ($code, $stdout, $stderr) = $old->psql('postgres', 'SELECT test_ext()');
56is($code, 0, 'extension works before upgrade');
57like($stderr, qr/NOTICE:  running successful/, 'extension working');
58
59$old->stop;
60
61my $new = PostgreSQL::Test::Cluster->new('new');
62$new->init;
63
64# Pre-configure the new cluster with dynamic_library_path and
65# extension_control_path before running pg_upgrade.
66$new->append_conf('postgresql.conf', $extension_control_path_conf);
67
68# In a VPATH build, we'll be started in the source directory, but we want
69# to run pg_upgrade in the build directory so that any files generated finish
70# in it, like delete_old_cluster.{sh,bat}.
71chdir ${PostgreSQL::Test::Utils::tmp_check};
72
73command_ok(
74	[
75		'pg_upgrade', '--no-sync',
76		'--old-datadir' => $old->data_dir,
77		'--new-datadir' => $new->data_dir,
78		'--old-bindir' => $old->config_data('--bindir'),
79		'--new-bindir' => $new->config_data('--bindir'),
80		'--socketdir' => $new->host,
81		'--old-port' => $old->port,
82		'--new-port' => $new->port,
83		'--copy',
84	],
85	'pg_upgrade succeeds with extension installed via extension_control_path'
86);
87
88$new->start;
89
90# Verify the extension still works after the upgrade.
91($code, $stdout, $stderr) = $new->psql('postgres', 'SELECT test_ext()');
92is($code, 0, 'extension works after upgrade');
93like($stderr, qr/NOTICE:  running successful/, 'extension working');
94
95$new->stop;
96
97# Write .control and .sql files into $ext_dir/extension/
98# `module_pathname` contains the `$libdir/` to simulate most of the extensions
99# that use it as a prefix in the `module_pathname` by default
100sub create_extension_files
101{
102	my ($ext_name, $ext_dir) = @_;
103
104	open my $cf, '>', "$ext_dir/extension/$ext_name.control"
105	  or die "could not create control file: $!";
106	print $cf
107	  "comment = 'Test C extension for pg_upgrade + extension_control_path'\n";
108	print $cf "default_version = '1.0'\n";
109	print $cf "module_pathname = '\$libdir/$ext_name'\n";
110	print $cf "relocatable = true\n";
111	close $cf;
112
113	open my $sqlf, '>', "$ext_dir/extension/$ext_name--1.0.sql"
114	  or die "could not create SQL file: $!";
115	print $sqlf "/* $ext_name--1.0.sql */\n";
116	print $sqlf
117	  "-- complain if script is sourced in psql, rather than via CREATE EXTENSION\n";
118	print $sqlf
119	  qq'\\echo Use "CREATE EXTENSION $ext_name" to load this file. \\quit\n';
120	print $sqlf "CREATE FUNCTION test_ext()\n";
121	print $sqlf "RETURNS void AS 'MODULE_PATHNAME'\n";
122	print $sqlf "LANGUAGE C;\n";
123	close $sqlf;
124}
125
126done_testing();
