3# Tests for file transfer modes

5use strict;
6use warnings FATAL => 'all';
7
8use PostgreSQL::Test::Cluster;
9use PostgreSQL::Test::Utils;
10use Test::More;
11
12sub test_mode
13{
14	my ($mode) = @_;
15
16	my $old =
17	  PostgreSQL::Test::Cluster->new('old', install_path => $ENV{oldinstall});
18	my $new = PostgreSQL::Test::Cluster->new('new');
19
20	# --swap can't be used to upgrade from versions older than 10, so just skip
21	# the test if the old cluster version is too old.
22	if ($old->pg_version < 10 && $mode eq "--swap")
23	{
24		$old->clean_node();
25		$new->clean_node();
26		return;
27	}
28
29	if (defined($ENV{oldinstall}))
30	{
31		# Checksums are now enabled by default, but weren't before 18, so pass
32		# '-k' to initdb on older versions so that upgrades work.
33		$old->init(extra => ['-k']);
34	}
35	else
36	{
37		$old->init();
38	}
39	$new->init();
40
41	# allow_in_place_tablespaces is available as far back as v10.
42	if ($old->pg_version >= 10)
43	{
44		$new->append_conf('postgresql.conf',
45			"allow_in_place_tablespaces = true");
46		$old->append_conf('postgresql.conf',
47			"allow_in_place_tablespaces = true");
48	}
49
50	# We can only test security labels if both the old and new installations
51	# have dummy_seclabel.
52	my $test_seclabel = 1;
53	$old->start;
54	if (!$old->check_extension('dummy_seclabel'))
55	{
56		$test_seclabel = 0;
57	}
58	$old->stop;
59	$new->start;
60	if (!$new->check_extension('dummy_seclabel'))
61	{
62		$test_seclabel = 0;
63	}
64	$new->stop;
65
66	# Create a small variety of simple test objects in the old cluster.  We'll
67	# check that these reach the new version after upgrading.
68	$old->start;
69	$old->safe_psql('postgres',
70		"CREATE TABLE test1 AS SELECT generate_series(1, 100)");
71	$old->safe_psql('postgres', "CREATE DATABASE testdb1");
72	$old->safe_psql('testdb1',
73		"CREATE TABLE test2 AS SELECT generate_series(200, 300)");
74	$old->safe_psql('testdb1', "VACUUM FULL test2");
75	$old->safe_psql('testdb1', "CREATE SEQUENCE testseq START 5432");
76
77	# If an old installation is provided, we can test non-in-place tablespaces.
78	if (defined($ENV{oldinstall}))
79	{
80		my $tblspc = PostgreSQL::Test::Utils::tempdir_short();
81		$old->safe_psql('postgres',
82			"CREATE TABLESPACE test_tblspc LOCATION '$tblspc'");
83		$old->safe_psql('postgres',
84			"CREATE DATABASE testdb2 TABLESPACE test_tblspc");
85		$old->safe_psql('postgres',
86			"CREATE TABLE test3 TABLESPACE test_tblspc AS SELECT generate_series(300, 401)"
87		);
88		$old->safe_psql('testdb2',
89			"CREATE TABLE test4 AS SELECT generate_series(400, 502)");
90	}
91
92	# If the old cluster is >= v10, we can test in-place tablespaces.
93	if ($old->pg_version >= 10)
94	{
95		$old->safe_psql('postgres',
96			"CREATE TABLESPACE inplc_tblspc LOCATION ''");
97		$old->safe_psql('postgres',
98			"CREATE DATABASE testdb3 TABLESPACE inplc_tblspc");
99		$old->safe_psql('postgres',
100			"CREATE TABLE test5 TABLESPACE inplc_tblspc AS SELECT generate_series(503, 606)"
101		);
102		$old->safe_psql('testdb3',
103			"CREATE TABLE test6 AS SELECT generate_series(607, 711)");
104	}
105
106	# While we are here, test handling of large objects.
107	$old->safe_psql(
108		'postgres', q|
109		CREATE ROLE regress_lo_1;
110		CREATE ROLE regress_lo_2;
111
112		SELECT lo_from_bytea(4532, '\xffffff00');
113		COMMENT ON LARGE OBJECT 4532 IS 'test';
114
115		SELECT lo_from_bytea(4533, '\x0f0f0f0f');
116		ALTER LARGE OBJECT 4533 OWNER TO regress_lo_1;
117		GRANT SELECT ON LARGE OBJECT 4533 TO regress_lo_2;
118	|);
119
120	if ($test_seclabel)
121	{
122		$old->safe_psql(
123			'postgres', q|
124			CREATE EXTENSION dummy_seclabel;
125
126			SELECT lo_from_bytea(4534, '\x00ffffff');
127			SECURITY LABEL ON LARGE OBJECT 4534 IS 'classified';
128		|);
129	}
130	$old->stop;
131
132	my $result = command_ok_or_fails_like(
133		[
134			'pg_upgrade', '--no-sync',
135			'--old-datadir' => $old->data_dir,
136			'--new-datadir' => $new->data_dir,
137			'--old-bindir' => $old->config_data('--bindir'),
138			'--new-bindir' => $new->config_data('--bindir'),
139			'--socketdir' => $new->host,
140			'--old-port' => $old->port,
141			'--new-port' => $new->port,
142			$mode
143		],
144		qr/.* not supported on this platform|could not .* between old and new data directories: .*/,
145		qr/^$/,
146		"pg_upgrade with transfer mode $mode");
147
148	# If pg_upgrade was successful, check that all of our test objects reached
149	# the new version.
150	if ($result)
151	{
152		$new->start;
153		$result = $new->safe_psql('postgres', "SELECT COUNT(*) FROM test1");
154		is($result, '100', "test1 data after pg_upgrade $mode");
155		$result = $new->safe_psql('testdb1', "SELECT COUNT(*) FROM test2");
156		is($result, '101', "test2 data after pg_upgrade $mode");
157		$result = $new->safe_psql('testdb1', "SELECT nextval('testseq')");
158		is($result, '5432', "sequence data after pg_upgrade $mode");
159
160		# Tests for non-in-place tablespaces.
161		if (defined($ENV{oldinstall}))
162		{
163			$result =
164			  $new->safe_psql('postgres', "SELECT COUNT(*) FROM test3");
165			is($result, '102', "test3 data after pg_upgrade $mode");
166			$result =
167			  $new->safe_psql('testdb2', "SELECT COUNT(*) FROM test4");
168			is($result, '103', "test4 data after pg_upgrade $mode");
169		}
170
171		# Tests for in-place tablespaces.
172		if ($old->pg_version >= 10)
173		{
174			$result =
175			  $new->safe_psql('postgres', "SELECT COUNT(*) FROM test5");
176			is($result, '104', "test5 data after pg_upgrade $mode");
177			$result =
178			  $new->safe_psql('testdb3', "SELECT COUNT(*) FROM test6");
179			is($result, '105', "test6 data after pg_upgrade $mode");
180		}
181
182		# Tests for large objects
183		$result = $new->safe_psql('postgres', "SELECT lo_get(4532)");
184		is($result, '\xffffff00', "LO contents after upgrade");
185		$result = $new->safe_psql('postgres',
186			"SELECT obj_description(4532, 'pg_largeobject')");
187		is($result, 'test', "comment on LO after pg_upgrade");
188
189		$result = $new->safe_psql('postgres', "SELECT lo_get(4533)");
190		is($result, '\x0f0f0f0f', "LO contents after upgrade");
191		$result = $new->safe_psql('postgres',
192			"SELECT lomowner::regrole FROM pg_largeobject_metadata WHERE oid = 4533"
193		);
194		is($result, 'regress_lo_1', "LO owner after upgrade");
195		$result = $new->safe_psql('postgres',
196			"SELECT lomacl FROM pg_largeobject_metadata WHERE oid = 4533");
197		is( $result,
198			'{regress_lo_1=rw/regress_lo_1,regress_lo_2=r/regress_lo_1}',
199			"LO ACL after upgrade");
200
201		if ($test_seclabel)
202		{
203			$result = $new->safe_psql('postgres', "SELECT lo_get(4534)");
204			is($result, '\x00ffffff', "LO contents after upgrade");
205			$result = $new->safe_psql(
206				'postgres', q|
207				SELECT label FROM pg_seclabel WHERE objoid = 4534
208				AND classoid = 'pg_largeobject'::regclass
209			|);
210			is($result, 'classified', "seclabel on LO after pg_upgrade");
211		}
212		$new->stop;
213	}
214
215	$old->clean_node();
216	$new->clean_node();
217}
218
219# Run pg_upgrade in tmp_check to avoid leaving files like
220# delete_old_cluster.{sh,bat} in the source directory for VPATH and meson
221# builds.
222chdir ${PostgreSQL::Test::Utils::tmp_check};
223
224test_mode('--clone');
225test_mode('--copy');
226test_mode('--copy-file-range');
227test_mode('--link');
228test_mode('--swap');
229
230done_testing();
