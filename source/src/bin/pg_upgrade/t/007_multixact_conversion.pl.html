3# Version 19 expanded MultiXactOffset from 32 to 64 bits.  Upgrading
4# across that requires rewriting the SLRU files to the new format.
5# This file contains tests for the conversion.
6#
7# To run, set 'oldinstall' ENV variable to point to a pre-v19
8# installation.  If it's not set, or if it points to a v19 or above
9# installation, this still performs a very basic test, upgrading a
10# cluster with some multixacts.  It's not very interesting, however,
11# because there's no conversion involved in that case.
12

22my $tempdir = PostgreSQL::Test::Utils::tempdir;

24# A workload that consumes multixids.  The purpose of this is to
25# generate some multixids in the old cluster, so that we can test
26# upgrading them.  The workload is a mix of KEY SHARE locking queries
27# and UPDATEs, and commits and aborts, to generate a mix of multixids
28# with different statuses.  It consumes around 3000 multixids with
29# 60000 members in total.  That's enough to span more than one
30# multixids 'offsets' page, and more than one 'members' segment with
31# the default block size.
32#
33# The workload leaves behind a table called 'mxofftest' containing a
34# small number of rows referencing some of the generated multixids.
35#
36# Because this function is used to generate test data on the old
37# installation, it needs to work with older PostgreSQL server
38# versions.
39#
40# The first argument is the cluster to connect to, the second argument
41# is a cluster using the new version.  We need the 'psql' binary from
42# the new version, the new cluster is otherwise unused.  (We need to
43# use the new 'psql' because some of the more advanced background psql
44# perl module features depend on a fairly recent psql version.)
45sub mxact_workload
46{
47	my $node = shift;       # Cluster to connect to
48	my $binnode = shift;    # Use the psql binary from this cluster
49
50	my $connstr = $node->connstr('postgres');
51
52	$node->start;
53	$node->safe_psql(
54		'postgres', qq[
55		CREATE TABLE mxofftest (id INT PRIMARY KEY, n_updated INT)
56		  WITH (AUTOVACUUM_ENABLED=FALSE);
57		INSERT INTO mxofftest SELECT G, 0 FROM GENERATE_SERIES(1, 50) G;
58	]);
59
60	my $nclients = 20;
61	my $update_every = 13;
62	my $abort_every = 11;
63	my @connections = ();
64
65	# Silence the logging of the statements we run to avoid
66	# unnecessarily bloating the test logs.  This runs before the
67	# upgrade we're testing, so the details should not be very
68	# interesting for debugging.  But if needed, you can make it more
69	# verbose by setting this.
70	my $verbose = 0;
71
72	# Bump the timeout on the connections to avoid false negatives on
73	# slow test systems. The timeout covers the whole duration that
74	# the connections are open rather than the individual queries.
75	my $connection_timeout_secs =
76	  4 * $PostgreSQL::Test::Utils::timeout_default;
77
78	# Open multiple connections to the database.  Start a transaction
79	# in each connection.
80	for (0 .. $nclients)
81	{
82		# Use the psql binary from the new installation.  The
83		# BackgroundPsql functionality doesn't work with older psql
84		# versions.
85		my $conn = $binnode->background_psql(
86			'',
87			connstr => $node->connstr('postgres'),
88			timeout => $connection_timeout_secs);
89
90		$conn->query_safe("SET log_statement=none", verbose => $verbose)
91		  unless $verbose;
92		$conn->query_safe("SET enable_seqscan=off", verbose => $verbose);
93		$conn->query_safe("BEGIN", verbose => $verbose);
94
95		push(@connections, $conn);
96	}
97
98	# Run queries using cycling through the connections in a
99	# round-robin fashion.  We keep a transaction open in each
100	# connection at all times, and lock/update the rows.  With 20
101	# connections, each SELECT FOR KEY SHARE query generates a new
102	# multixid, containing the XIDs of all the transactions running at
103	# the time, ie. around 20 XIDs.
104	for (my $i = 0; $i < 3000; $i++)
105	{
106		note "generating multixids $i / 3000\n" if ($i % 100 == 0);
107
108		my $conn = $connections[ $i % $nclients ];
109
110		my $sql = ($i % $abort_every == 0) ? "ABORT" : "COMMIT";
111		$conn->query_safe($sql, verbose => $verbose);
112
113		$conn->query_safe("BEGIN", verbose => $verbose);
114		if ($i % $update_every == 0)
115		{
116			$sql = qq[
117			  UPDATE mxofftest SET n_updated = n_updated + 1 WHERE id = ${i} % 50;
118			];
119		}
120		else
121		{
122			my $threshold = int($i / 3000 * 50);
123			$sql = qq[
124			  select count(*) from (
125				SELECT * FROM mxofftest WHERE id >= $threshold FOR KEY SHARE
126			  ) as x
127			];
128		}
129		$conn->query_safe($sql, verbose => $verbose);
130	}
131
132	for my $conn (@connections)
133	{
134		$conn->quit();
135	}
136
137	$node->stop;
138	return;
139}

141# Return contents of the 'mxofftest' table, created by mxact_workload
142sub get_test_table_contents
143{
144	my ($node, $filename) = @_;
145
146	my $contents = $node->safe_psql('postgres',
147		"SELECT ctid, xmin, xmax, * FROM mxofftest");
148
149	my $path = $tempdir . '/' . $filename;
150	open(my $fh, '>', $path)
151	  || die "could not open $path for writing $!";
152	print $fh $contents;
153	close($fh);
154
155	return $path;
156}

158# Return the members of all updating multixids in the given range
159sub get_updating_multixact_members
160{
161	my ($node, $from, $to, $filename) = @_;
162
163	my $path = $tempdir . '/' . $filename;
164	open(my $fh, '>', $path)
165	  || die "could not open $path for writing $!";
166
167	if ($to >= $from)
168	{
169		my $res = $node->safe_psql(
170			'postgres', qq[
171			SELECT multi, mode, xid
172			FROM generate_series($from, $to - 1) as multi,
173				 pg_get_multixact_members(multi::text::xid)
174			WHERE mode not in ('keysh', 'sh');
175		]);
176		print $fh $res;
177	}
178	else
179	{
180		# Multixids wrapped around.  Split the query into two parts,
181		# before and after the wraparound.
182		my $res = $node->safe_psql(
183			'postgres', qq[
184			SELECT multi, mode, xid
185			FROM generate_series($from, 4294967295) as multi,
186				 pg_get_multixact_members(multi::text::xid)
187			WHERE mode not in ('keysh', 'sh');
188		]);
189		print $fh $res;
190		$res = $node->safe_psql(
191			'postgres', qq[
192			SELECT multi, mode, xid
193			FROM generate_series(1, $to - 1) as multi,
194				 pg_get_multixact_members(multi::text::xid)
195			WHERE mode not in ('keysh', 'sh');
196		]);
197		print $fh $res;
198	}
199
200	close($fh);
201	return $path;
202}

204# Read multixid related fields from the control file
205#
206# Note: This is used on both the old and the new installation, so the
207# command arguments and the output parsing used here must work with
208# all PostgreSQL versions supported by the test.
209sub read_multixid_fields
210{
211	my $node = shift;
212
213	my $pg_controldata_path = $node->installed_command('pg_controldata');
214	my ($stdout, $stderr) =
215	  run_command([ $pg_controldata_path, $node->data_dir ]);
216	$stdout =~ /^Latest checkpoint's oldestMultiXid:\s*(.*)$/m
217	  or die "could not read oldestMultiXid from pg_controldata";
218	my $oldest_multi_xid = $1;
219	$stdout =~ /^Latest checkpoint's NextMultiXactId:\s*(.*)$/m
220	  or die "could not read NextMultiXactId from pg_controldata";
221	my $next_multi_xid = $1;
222	$stdout =~ /^Latest checkpoint's NextMultiOffset:\s*(.*)$/m
223	  or die "could not read NextMultiOffset from pg_controldata";
224	my $next_multi_offset = $1;
225
226	return ($oldest_multi_xid, $next_multi_xid, $next_multi_offset);
227}

229# Reset a cluster's next multixid and mxoffset to given values.
230#
231# Note: This is used on the old installation, so the command arguments
232# and the output parsing used here must work with all pre-v19
233# PostgreSQL versions supported by the test.
234sub reset_mxid_mxoffset_pre_v19
235{
236	my $node = shift;
237	my $mxid = shift;
238	my $mxoffset = shift;
239
240	my $pg_resetwal_path = $node->installed_command('pg_resetwal');
241	# Get block size
242	my ($out, $err) =
243	  run_command([ $pg_resetwal_path, '--dry-run', $node->data_dir ]);
244	$out =~ /^Database block size: *(\d+)$/m or die;
245
246	# Verify that no multixids are currently in use.  Resetting would
247	# destroy them.  (A freshly initialized cluster has no multixids.)
248	$out =~ /^Latest checkpoint's NextMultiXactId: *(\d+)$/m or die;
249	my $next_mxid = $1;
250	$out =~ /^Latest checkpoint's oldestMultiXid: *(\d+)$/m or die;
251	my $oldest_mxid = $1;
252	die "cluster has some multixids in use" unless $next_mxid == $oldest_mxid;
253
254	# Extract a few other values from pg_resetwal --dry-run output
255	# that we need for the calculations below
256	$out =~ /^Database block size: *(\d+)$/m or die;
257	my $blcksz = $1;
258	# SLRU_PAGES_PER_SEGMENT is always 32 on pre-19 versions
259	my $slru_pages_per_segment = 32;
260
261	# Do the reset
262	my @cmd = (
263		$pg_resetwal_path,
264		'--pgdata' => $node->data_dir,
265		'--multixact-offset' => $mxoffset,
266		'--multixact-ids' => "$mxid,$mxid");
267	command_ok(\@cmd, 'reset multixids and offset');
268
269	# pg_resetwal just updates the control file.  The cluster will
270	# refuse to start up, if the SLRU segments corresponding to the
271	# next multixid and offset does not exist.  Create a segments that
272	# covers the given values, filled with zeros.  But first remove
273	# any old segments.
274	unlink glob $node->data_dir . "/pg_multixact/offsets/*";
275	unlink glob $node->data_dir . "/pg_multixact/members/*";
276
277	# Initialize the 'offsets' SLRU file containing the new next multixid
278	# with zeros
279	#
280	# sizeof(MultiXactOffset) == 4 in PostgreSQL versions before 19
281	my $multixact_offsets_per_page = $blcksz / 4;
282	my $segno =
283	  int($mxid / $multixact_offsets_per_page / $slru_pages_per_segment);
284	my $path =
285	  sprintf('%s/pg_multixact/offsets/%04X', $node->data_dir, $segno);
286	open my $fh, ">", $path
287	  or die "could not open \"$path\": $!";
288	binmode $fh;
289	my $bytes_per_seg = $slru_pages_per_segment * $blcksz;
290	syswrite($fh, "\0" x $bytes_per_seg) == $bytes_per_seg
291	  or die "could not write to \"$path\": $!";
292	close $fh;
293
294	# Same for the 'members' SLRU
295	my $multixact_members_per_page = int($blcksz / 20) * 4;
296	$segno =
297	  int($mxoffset / $multixact_members_per_page / $slru_pages_per_segment);
298	$path = sprintf "%s/pg_multixact/members/%04X", $node->data_dir, $segno;
299	open $fh, ">", $path
300	  or die "could not open \"$path\": $!";
301	binmode $fh;
302	syswrite($fh, "\0" x $bytes_per_seg) == $bytes_per_seg
303	  or die "could not write to \"$path\": $!";
304	close($fh);
305}

307# Main test workhorse routine.  Dump data on old version, run
308# pg_upgrade, compare data after upgrade.
309sub upgrade_and_compare
310{
311	my $tag = shift;
312	my $oldnode = shift;
313	my $newnode = shift;
314
315	command_ok(
316		[
317			'pg_upgrade', '--no-sync',
318			'--old-datadir' => $oldnode->data_dir,
319			'--new-datadir' => $newnode->data_dir,
320			'--old-bindir' => $oldnode->config_data('--bindir'),
321			'--new-bindir' => $newnode->config_data('--bindir'),
322			'--socketdir' => $newnode->host,
323			'--old-port' => $oldnode->port,
324			'--new-port' => $newnode->port,
325		],
326		'run of pg_upgrade for new instance');
327
328	# Dump contents of the test table, and the status of all updating
329	# multixids from the old cluster.  (Locking-only multixids don't
330	# need to be preserved so we ignore those)
331	#
332	# Note: we do this *after* running pg_upgrade, to ensure that we
333	# don't set all the hint bits before upgrade by doing the SELECT
334	# on the table.
335	my ($multixids_start, $multixids_end, undef) =
336	  read_multixid_fields($oldnode);
337	$oldnode->start;
338	my $old_table_contents =
339	  get_test_table_contents($oldnode, "oldnode_${tag}_table_contents");
340	my $old_multixacts =
341	  get_updating_multixact_members($oldnode, $multixids_start,
342		$multixids_end, "oldnode_${tag}_multixacts");
343	$oldnode->stop;
344
345	# Compare them with the upgraded cluster
346	$newnode->start;
347	my $new_table_contents =
348	  get_test_table_contents($newnode, "newnode_${tag}_table_contents");
349	my $new_multixacts =
350	  get_updating_multixact_members($newnode, $multixids_start,
351		$multixids_end, "newnode_${tag}_multixacts");
352	$newnode->stop;
353
354	compare_files($old_table_contents, $new_table_contents,
355		'test table contents from original and upgraded clusters match');
356	compare_files($old_multixacts, $new_multixacts,
357		'multixact members from original and upgraded clusters match');
358}

360# In a VPATH build, we'll be started in the source directory, but we want
361# to run pg_upgrade in the build directory so that any files generated finish
362# in it, like delete_old_cluster.{sh,bat}.
363chdir ${PostgreSQL::Test::Utils::tmp_check};

367# Basic scenario: Create a cluster using old installation, run
368# multixid-creating workload on it, then upgrade.
369#
370# This works even even if the old and new version is the same,
371# although it's not very interesting as the conversion routines only
372# run when upgrading from a pre-v19 cluster.
373{
374	my $tag = 'basic';
375	my $old =
376	  PostgreSQL::Test::Cluster->new("${tag}_oldnode",
377		install_path => $ENV{oldinstall});
378	my $new = PostgreSQL::Test::Cluster->new("${tag}_newnode");
379
380	$old->init(extra => ['-k']);
381
382	$old_version = $old->pg_version;
383	note "old installation is version $old_version\n";
384
385	# Run the workload
386	my (undef, $start_mxid, $start_mxoff) = read_multixid_fields($old);
387	mxact_workload($old, $new);
388	my (undef, $finish_mxid, $finish_mxoff) = read_multixid_fields($old);
389
390	note "Testing upgrade, ${tag} scenario\n"
391	  . " mxid from ${start_mxid} to ${finish_mxid}\n"
392	  . " oldnode mxoff from ${start_mxoff} to ${finish_mxoff}\n";
393
394	$new->init;
395	upgrade_and_compare($tag, $old, $new);
396}

398# Wraparound scenario: This is the same as the basic scenario, but the
399# old cluster goes through multixid and offset wraparound.
400#
401# This requires the old installation to be version 18 or older,
402# because the hacks we use to reset the old cluster to a state just
403# before the wraparound rely on the pre-v19 file format.  If the old
404# cluster is of v19 or above, multixact SLRU conversion is not needed
405# anyway.
406SKIP:
407{
408	skip
409	  "skipping mxoffset conversion tests because upgrading from the old version does not require conversion"
410	  if ($old_version >= '19devel');
411
412	my $tag = 'wraparound';
413	my $old =
414	  PostgreSQL::Test::Cluster->new("${tag}_oldnode",
415		install_path => $ENV{oldinstall});
416	my $new = PostgreSQL::Test::Cluster->new("${tag}_newnode");
417
418	$old->init(extra => ['-k']);
419
420	# Reset the old cluster to just before multixid and 32-bit offset
421	# wraparound.
422	reset_mxid_mxoffset_pre_v19($old, 0xFFFFFA00, 0xFFFFEC00);
423
424	# Run the workload.  This crosses multixid and offset wraparound.
425	my (undef, $start_mxid, $start_mxoff) = read_multixid_fields($old);
426	mxact_workload($old, $new);
427	my (undef, $finish_mxid, $finish_mxoff) = read_multixid_fields($old);
428
429	note "Testing upgrade, ${tag} scenario\n"
430	  . " mxid from ${start_mxid} to ${finish_mxid}\n"
431	  . " oldnode mxoff from ${start_mxoff} to ${finish_mxoff}\n";
432
433	# Verify that wraparounds happened.
434	cmp_ok($finish_mxid, '<', $start_mxid,
435		"multixid wrapped around in old cluster");
436	cmp_ok($finish_mxoff, '<', $start_mxoff,
437		"mxoff wrapped around in old cluster");
438
439	$new->init;
440	upgrade_and_compare($tag, $old, $new);
441}
