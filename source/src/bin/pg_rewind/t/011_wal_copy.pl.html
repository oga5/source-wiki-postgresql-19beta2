src/bin/pg_rewind/t/011_wal_copy.pl
1# Copyright (c) 2025-2026, PostgreSQL Global Development Group
2#
3# Check how the copy of WAL segments is handled from the source to
4# the target server.
5
6use strict;
7use warnings FATAL => 'all';
8use PostgreSQL::Test::Utils;
9use Test::More;
10use File::stat qw(stat);
11
12use FindBin;
13use lib $FindBin::RealBin;
14use RewindTest;
15
16RewindTest::setup_cluster();
17RewindTest::start_primary();
18RewindTest::create_standby();
19
20# Advance WAL on primary
21RewindTest::primary_psql("CREATE TABLE t(a int)");
22RewindTest::primary_psql("INSERT INTO t VALUES(0)");
23
24# Segment that is not copied from the source to the target, being
25# generated before the servers have diverged.
26my $wal_seg_skipped = $node_primary->safe_psql('postgres',
27	'SELECT pg_walfile_name(pg_current_wal_lsn())');
28
29RewindTest::primary_psql("SELECT pg_switch_wal()");
30
31# Follow-up segment, that will include corrupted contents, and will be
32# copied from the source to the target even if generated before the point
33# of divergence.
34RewindTest::primary_psql("INSERT INTO t VALUES(0)");
35my $corrupt_wal_seg = $node_primary->safe_psql('postgres',
36	'SELECT pg_walfile_name(pg_current_wal_lsn())');
37RewindTest::primary_psql("SELECT pg_switch_wal()");
38
39RewindTest::primary_psql("CHECKPOINT");
40RewindTest::promote_standby;
41
42# New segment on a new timeline, expected to be copied.
43my $new_timeline_wal_seg = $node_standby->safe_psql('postgres',
44	'SELECT pg_walfile_name(pg_current_wal_lsn())');
45
46# Corrupt a WAL segment on target that has been generated before the
47# divergence point.  We will check that it is copied from the source.
48my $corrupt_wal_seg_in_target_path =
49  $node_primary->data_dir . '/pg_wal/' . $corrupt_wal_seg;
50open my $fh, ">>", $corrupt_wal_seg_in_target_path
51  or die "could not open $corrupt_wal_seg_in_target_path";
52
53print $fh 'a';
54close $fh;
55
56my $corrupt_wal_seg_stat_before_rewind =
57  stat($corrupt_wal_seg_in_target_path);
58ok(defined($corrupt_wal_seg_stat_before_rewind),
59	"segment $corrupt_wal_seg exists in target before rewind");
60
61# Verify that the WAL segment on the new timeline does not exist in target
62# before the rewind.
63my $new_timeline_wal_seg_path =
64  $node_primary->data_dir . '/pg_wal/' . $new_timeline_wal_seg;
65my $new_timeline_wal_seg_stat = stat($new_timeline_wal_seg_path);
66ok(!defined($new_timeline_wal_seg_stat),
67	"segment $new_timeline_wal_seg does not exist in target before rewind");
68
69$node_standby->stop();
70$node_primary->stop();
71
72# Cross-check how WAL segments are handled:
73# - The "corrupted" segment generated before the point of divergence is
74#   copied.
75# - The "clean" segment generated before the point of divergence is skipped.
76# - The segment of the new timeline is copied.
77command_checks_all(
78	[
79		'pg_rewind', '--debug',
80		'--source-pgdata' => $node_standby->data_dir,
81		'--target-pgdata' => $node_primary->data_dir,
82		'--no-sync',
83	],
84	0,
85	[qr//],
86	[
87		qr/pg_wal\/$wal_seg_skipped \(NONE\)/,
88		qr/pg_wal\/$corrupt_wal_seg \(COPY\)/,
89		qr/pg_wal\/$new_timeline_wal_seg \(COPY\)/,
90	],
91	'run pg_rewind');
92
93# Verify that the first WAL segment of the new timeline now exists in
94# target.
95$new_timeline_wal_seg_stat = stat($new_timeline_wal_seg_path);
96ok(defined($new_timeline_wal_seg_stat),
97	"new timeline segment $new_timeline_wal_seg exists in target after rewind"
98);
99
100# Validate that the WAL segment with the same file name as the
101# corrupted WAL segment in target has been copied from source
102# where it was still intact.
103my $corrupt_wal_seg_in_source_path =
104  $node_standby->data_dir . '/pg_wal/' . $corrupt_wal_seg;
105my $corrupt_wal_seg_source_stat = stat($corrupt_wal_seg_in_source_path);
106ok(defined($corrupt_wal_seg_source_stat),
107	"corrupted $corrupt_wal_seg exists in source after rewind");
108
109my $corrupt_wal_seg_stat_after_rewind = stat($corrupt_wal_seg_in_target_path);
110ok(defined($corrupt_wal_seg_stat_after_rewind),
111	"corrupted $corrupt_wal_seg exists in target after rewind");
112isnt(
113	$corrupt_wal_seg_stat_before_rewind->size,
114	$corrupt_wal_seg_source_stat->size,
115	"different size of corrupted $corrupt_wal_seg in source vs target before rewind"
116);
117is( $corrupt_wal_seg_stat_after_rewind->size,
118	$corrupt_wal_seg_source_stat->size,
119	"same size of corrupted $corrupt_wal_seg in source and target after rewind"
120);
121
122done_testing();
