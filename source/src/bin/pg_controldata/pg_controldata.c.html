32static void
33usage(const char *progname)
34{
35	printf(_("%s displays control information of a PostgreSQL database cluster.\n\n"), progname);
36	printf(_("Usage:\n"));
37	printf(_("  %s [OPTION] [DATADIR]\n"), progname);
38	printf(_("\nOptions:\n"));
39	printf(_(" [-D, --pgdata=]DATADIR  data directory\n"));
40	printf(_("  -V, --version          output version information, then exit\n"));
41	printf(_("  -?, --help             show this help, then exit\n"));
42	printf(_("\nIf no data directory (DATADIR) is specified, "
43			 "the environment variable PGDATA\nis used.\n\n"));
44	printf(_("Report bugs to <%s>.\n"), PACKAGE_BUGREPORT);
45	printf(_("%s home page: <%s>\n"), PACKAGE_NAME, PACKAGE_URL);
46}

49static const char *
50dbState(DBState state)
51{
52	switch (state)
53	{
54		case DB_STARTUP:
55			return _("starting up");
56		case DB_SHUTDOWNED:
57			return _("shut down");
58		case DB_SHUTDOWNED_IN_RECOVERY:
59			return _("shut down in recovery");
60		case DB_SHUTDOWNING:
61			return _("shutting down");
62		case DB_IN_CRASH_RECOVERY:
63			return _("in crash recovery");
64		case DB_IN_ARCHIVE_RECOVERY:
65			return _("in archive recovery");
66		case DB_IN_PRODUCTION:
67			return _("in production");
68	}
69	return _("unrecognized status code");
70}
71
72static const char *
73wal_level_str(WalLevel wal_level)
74{
75	switch (wal_level)
76	{
77		case WAL_LEVEL_MINIMAL:
78			return "minimal";
79		case WAL_LEVEL_REPLICA:
80			return "replica";
81		case WAL_LEVEL_LOGICAL:
82			return "logical";
83	}
84	return _("unrecognized \"wal_level\"");
85}

88int
89main(int argc, char *argv[])
90{
91	static struct option long_options[] = {
92		{"pgdata", required_argument, NULL, 'D'},
93		{NULL, 0, NULL, 0}
94	};
95
96	ControlFileData *ControlFile;
97	bool		crc_ok;
98	char	   *DataDir = NULL;
99	time_t		time_tmp;
100	struct tm  *tm_tmp;
101	char		pgctime_str[128];
102	char		ckpttime_str[128];
103	char		mock_auth_nonce_str[MOCK_AUTH_NONCE_LEN * 2 + 1];
104	const char *strftime_fmt = "%c";
105	const char *progname;
106	char		xlogfilename[MAXFNAMELEN];
107	int			c;
108	int			i;
109	int			WalSegSz;
110
111	pg_logging_init(argv[0]);
112	set_pglocale_pgservice(argv[0], PG_TEXTDOMAIN("pg_controldata"));
113	progname = get_progname(argv[0]);
114
115	if (argc > 1)
116	{
117		if (strcmp(argv[1], "--help") == 0 || strcmp(argv[1], "-?") == 0)
118		{
119			usage(progname);
120			exit(0);
121		}
122		if (strcmp(argv[1], "--version") == 0 || strcmp(argv[1], "-V") == 0)
123		{
124			puts("pg_controldata (PostgreSQL) " PG_VERSION);
125			exit(0);
126		}
127	}
128
129	while ((c = getopt_long(argc, argv, "D:", long_options, NULL)) != -1)
130	{
131		switch (c)
132		{
133			case 'D':
134				DataDir = optarg;
135				break;
136
137			default:
138				/* getopt_long already emitted a complaint */
139				pg_log_error_hint("Try \"%s --help\" for more information.", progname);
140				exit(1);
141		}
142	}
143
144	if (DataDir == NULL)
145	{
146		if (optind < argc)
147			DataDir = argv[optind++];
148		else
149			DataDir = getenv("PGDATA");
150	}
151
152	/* Complain if any arguments remain */
153	if (optind < argc)
154	{
155		pg_log_error("too many command-line arguments (first is \"%s\")",
156					 argv[optind]);
157		pg_log_error_hint("Try \"%s --help\" for more information.", progname);
158		exit(1);
159	}
160
161	if (DataDir == NULL)
162	{
163		pg_log_error("no data directory specified");
164		pg_log_error_hint("Try \"%s --help\" for more information.", progname);
165		exit(1);
166	}
167
168	/* get a copy of the control file */
169	ControlFile = get_controlfile(DataDir, &crc_ok);
170	if (ControlFile->pg_control_version != PG_CONTROL_VERSION)
171	{
172		pg_log_warning("control file version (%u) does not match the version understood by this program (%u)",
173					   ControlFile->pg_control_version, PG_CONTROL_VERSION);
174		pg_log_warning_detail("Either the control file has been created with a different version of PostgreSQL, "
175							  "or it is corrupt.  The results below are untrustworthy.");
176	}
177	else if (!crc_ok)
178	{
179		pg_log_warning("calculated CRC checksum does not match value stored in control file");
180		pg_log_warning_detail("Either the control file is corrupt, or it has a different layout than this program "
181							  "is expecting.  The results below are untrustworthy.");
182	}
183
184	/* set wal segment size */
185	WalSegSz = ControlFile->xlog_seg_size;
186
187	if (!IsValidWalSegSize(WalSegSz))
188	{
189		pg_log_warning(ngettext("invalid WAL segment size in control file (%d byte)",
190								"invalid WAL segment size in control file (%d bytes)",
191								WalSegSz),
192					   WalSegSz);
193		pg_log_warning_detail("The WAL segment size must be a power of two between 1 MB and 1 GB.");
194		pg_log_warning_detail("The file is corrupt and the results below are untrustworthy.");
195	}
196
197	/*
198	 * This slightly-chintzy coding will work as long as the control file
199	 * timestamps are within the range of time_t; that should be the case in
200	 * all foreseeable circumstances, so we don't bother importing the
201	 * backend's timezone library into pg_controldata.
202	 *
203	 * Use variable for format to suppress overly-anal-retentive gcc warning
204	 * about %c
205	 */
206	time_tmp = (time_t) ControlFile->time;
207	tm_tmp = localtime(&time_tmp);
208
209	if (tm_tmp != NULL)
210		strftime(pgctime_str, sizeof(pgctime_str), strftime_fmt, tm_tmp);
211	else
212		snprintf(pgctime_str, sizeof(pgctime_str), _("???"));
213
214	time_tmp = (time_t) ControlFile->checkPointCopy.time;
215	tm_tmp = localtime(&time_tmp);
216
217	if (tm_tmp != NULL)
218		strftime(ckpttime_str, sizeof(ckpttime_str), strftime_fmt, tm_tmp);
219	else
220		snprintf(ckpttime_str, sizeof(ckpttime_str), _("???"));
221
222	/*
223	 * Calculate name of the WAL file containing the latest checkpoint's REDO
224	 * start point.
225	 *
226	 * A corrupted control file could report a WAL segment size of 0 or
227	 * negative value, and to guard against division by zero, we need to treat
228	 * that specially.
229	 */
230	if (WalSegSz > 0)
231	{
232		XLogSegNo	segno;
233
234		XLByteToSeg(ControlFile->checkPointCopy.redo, segno, WalSegSz);
235		XLogFileName(xlogfilename, ControlFile->checkPointCopy.ThisTimeLineID,
236					 segno, WalSegSz);
237	}
238	else
239		strcpy(xlogfilename, _("???"));
240
241	for (i = 0; i < MOCK_AUTH_NONCE_LEN; i++)
242		snprintf(&mock_auth_nonce_str[i * 2], 3, "%02x",
243				 (unsigned char) ControlFile->mock_authentication_nonce[i]);
244
245	printf(_("pg_control version number:            %u\n"),
246		   ControlFile->pg_control_version);
247	printf(_("Catalog version number:               %u\n"),
248		   ControlFile->catalog_version_no);
249	printf(_("Database system identifier:           %" PRIu64 "\n"),
250		   ControlFile->system_identifier);
251	printf(_("Database cluster state:               %s\n"),
252		   dbState(ControlFile->state));
253	printf(_("pg_control last modified:             %s\n"),
254		   pgctime_str);
255	printf(_("Latest checkpoint location:           %X/%08X\n"),
256		   LSN_FORMAT_ARGS(ControlFile->checkPoint));
257	printf(_("Latest checkpoint's REDO location:    %X/%08X\n"),
258		   LSN_FORMAT_ARGS(ControlFile->checkPointCopy.redo));
259	printf(_("Latest checkpoint's REDO WAL file:    %s\n"),
260		   xlogfilename);
261	printf(_("Latest checkpoint's TimeLineID:       %u\n"),
262		   ControlFile->checkPointCopy.ThisTimeLineID);
263	printf(_("Latest checkpoint's PrevTimeLineID:   %u\n"),
264		   ControlFile->checkPointCopy.PrevTimeLineID);
265	printf(_("Latest checkpoint's full_page_writes: %s\n"),
266		   ControlFile->checkPointCopy.fullPageWrites ? _("on") : _("off"));
267	printf(_("Latest checkpoint's NextXID:          %u:%u\n"),
268		   EpochFromFullTransactionId(ControlFile->checkPointCopy.nextXid),
269		   XidFromFullTransactionId(ControlFile->checkPointCopy.nextXid));
270	printf(_("Latest checkpoint's NextOID:          %u\n"),
271		   ControlFile->checkPointCopy.nextOid);
272	printf(_("Latest checkpoint's NextMultiXactId:  %u\n"),
273		   ControlFile->checkPointCopy.nextMulti);
274	printf(_("Latest checkpoint's NextMultiOffset:  %" PRIu64 "\n"),
275		   ControlFile->checkPointCopy.nextMultiOffset);
276	printf(_("Latest checkpoint's oldestXID:        %u\n"),
277		   ControlFile->checkPointCopy.oldestXid);
278	printf(_("Latest checkpoint's oldestXID's DB:   %u\n"),
279		   ControlFile->checkPointCopy.oldestXidDB);
280	printf(_("Latest checkpoint's oldestActiveXID:  %u\n"),
281		   ControlFile->checkPointCopy.oldestActiveXid);
282	printf(_("Latest checkpoint's oldestMultiXid:   %u\n"),
283		   ControlFile->checkPointCopy.oldestMulti);
284	printf(_("Latest checkpoint's oldestMulti's DB: %u\n"),
285		   ControlFile->checkPointCopy.oldestMultiDB);
286	printf(_("Latest checkpoint's oldestCommitTsXid:%u\n"),
287		   ControlFile->checkPointCopy.oldestCommitTsXid);
288	printf(_("Latest checkpoint's newestCommitTsXid:%u\n"),
289		   ControlFile->checkPointCopy.newestCommitTsXid);
290	printf(_("Latest checkpoint's data_checksum_version:%u\n"),
291		   ControlFile->checkPointCopy.dataChecksumState);
292	printf(_("Time of latest checkpoint:            %s\n"),
293		   ckpttime_str);
294	printf(_("Fake LSN counter for unlogged rels:   %X/%08X\n"),
295		   LSN_FORMAT_ARGS(ControlFile->unloggedLSN));
296	printf(_("Minimum recovery ending location:     %X/%08X\n"),
297		   LSN_FORMAT_ARGS(ControlFile->minRecoveryPoint));
298	printf(_("Min recovery ending loc's timeline:   %u\n"),
299		   ControlFile->minRecoveryPointTLI);
300	printf(_("Backup start location:                %X/%08X\n"),
301		   LSN_FORMAT_ARGS(ControlFile->backupStartPoint));
302	printf(_("Backup end location:                  %X/%08X\n"),
303		   LSN_FORMAT_ARGS(ControlFile->backupEndPoint));
304	printf(_("End-of-backup record required:        %s\n"),
305		   ControlFile->backupEndRequired ? _("yes") : _("no"));
306	printf(_("wal_level setting:                    %s\n"),
307		   wal_level_str(ControlFile->wal_level));
308	printf(_("wal_log_hints setting:                %s\n"),
309		   ControlFile->wal_log_hints ? _("on") : _("off"));
310	printf(_("max_connections setting:              %d\n"),
311		   ControlFile->MaxConnections);
312	printf(_("max_worker_processes setting:         %d\n"),
313		   ControlFile->max_worker_processes);
314	printf(_("max_wal_senders setting:              %d\n"),
315		   ControlFile->max_wal_senders);
316	printf(_("max_prepared_xacts setting:           %d\n"),
317		   ControlFile->max_prepared_xacts);
318	printf(_("max_locks_per_xact setting:           %d\n"),
319		   ControlFile->max_locks_per_xact);
320	printf(_("track_commit_timestamp setting:       %s\n"),
321		   ControlFile->track_commit_timestamp ? _("on") : _("off"));
322	printf(_("Maximum data alignment:               %u\n"),
323		   ControlFile->maxAlign);
324	/* we don't print floatFormat since can't say much useful about it */
325	printf(_("Database block size:                  %u\n"),
326		   ControlFile->blcksz);
327	printf(_("Blocks per segment of large relation: %u\n"),
328		   ControlFile->relseg_size);
329	printf(_("Pages per SLRU segment:               %u\n"),
330		   ControlFile->slru_pages_per_segment);
331	printf(_("WAL block size:                       %u\n"),
332		   ControlFile->xlog_blcksz);
333	printf(_("Bytes per WAL segment:                %u\n"),
334		   ControlFile->xlog_seg_size);
335	printf(_("Maximum length of identifiers:        %u\n"),
336		   ControlFile->nameDataLen);
337	printf(_("Maximum columns in an index:          %u\n"),
338		   ControlFile->indexMaxKeys);
339	printf(_("Maximum size of a TOAST chunk:        %u\n"),
340		   ControlFile->toast_max_chunk_size);
341	printf(_("Size of a large-object chunk:         %u\n"),
342		   ControlFile->loblksize);
343	/* This is no longer configurable, but users may still expect to see it: */
344	printf(_("Date/time type storage:               %s\n"),
345		   _("64-bit integers"));
346	printf(_("Float8 argument passing:              %s\n"),
347		   (ControlFile->float8ByVal ? _("by value") : _("by reference")));
348	printf(_("Data page checksum version:           %u\n"),
349		   ControlFile->data_checksum_version);
350	printf(_("Default char data signedness:         %s\n"),
351		   (ControlFile->default_char_signedness ? _("signed") : _("unsigned")));
352	printf(_("Mock authentication nonce:            %s\n"),
353		   mock_auth_nonce_str);
354	return 0;
355}
