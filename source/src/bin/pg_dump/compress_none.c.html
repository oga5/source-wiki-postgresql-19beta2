25/*
26 * We buffer outgoing data, just to ensure that data blocks written to the
27 * archive file are of reasonable size.  The read side could use this struct,
28 * but there's no need because it does not retain data across calls.
29 */
30typedef struct NoneCompressorState
31{
32	char	   *buffer;			/* buffer for unwritten data */
33	size_t		buflen;			/* allocated size of buffer */
34	size_t		bufdata;		/* amount of valid data currently in buffer */
35} NoneCompressorState;
36
37/*
38 * Private routines
39 */
40
41static void
42ReadDataFromArchiveNone(ArchiveHandle *AH, CompressorState *cs)
43{
44	size_t		cnt;
45	char	   *buf;
46	size_t		buflen;
47
48	buflen = DEFAULT_IO_BUFFER_SIZE;
49	buf = pg_malloc(buflen);
50
51	while ((cnt = cs->readF(AH, &buf, &buflen)))
52	{
53		ahwrite(buf, 1, cnt, AH);
54	}
55
56	free(buf);
57}
58
59
60static void
61WriteDataToArchiveNone(ArchiveHandle *AH, CompressorState *cs,
62					   const void *data, size_t dLen)
63{
64	NoneCompressorState *nonecs = (NoneCompressorState *) cs->private_data;
65	size_t		remaining = dLen;
66
67	while (remaining > 0)
68	{
69		size_t		chunk;
70
71		/* Dump buffer if full */
72		if (nonecs->bufdata >= nonecs->buflen)
73		{
74			cs->writeF(AH, nonecs->buffer, nonecs->bufdata);
75			nonecs->bufdata = 0;
76		}
77		/* And fill it */
78		chunk = nonecs->buflen - nonecs->bufdata;
79		if (chunk > remaining)
80			chunk = remaining;
81		memcpy(nonecs->buffer + nonecs->bufdata, data, chunk);
82		nonecs->bufdata += chunk;
83		data = ((const char *) data) + chunk;
84		remaining -= chunk;
85	}
86}
87
88static void
89EndCompressorNone(ArchiveHandle *AH, CompressorState *cs)
90{
91	NoneCompressorState *nonecs = (NoneCompressorState *) cs->private_data;
92
93	if (nonecs)
94	{
95		/* Dump buffer if nonempty */
96		if (nonecs->bufdata > 0)
97			cs->writeF(AH, nonecs->buffer, nonecs->bufdata);
98		/* Free working state */
99		pg_free(nonecs->buffer);
100		pg_free(nonecs);
101		cs->private_data = NULL;
102	}
103}
104
105/*
106 * Public interface
107 */
108
109void
110InitCompressorNone(CompressorState *cs,
111				   const pg_compress_specification compression_spec)
112{
113	cs->readData = ReadDataFromArchiveNone;
114	cs->writeData = WriteDataToArchiveNone;
115	cs->end = EndCompressorNone;
116
117	cs->compression_spec = compression_spec;
118
119	/*
120	 * If the caller has defined a write function, prepare the necessary
121	 * buffer.
122	 */
123	if (cs->writeF)
124	{
125		NoneCompressorState *nonecs;
126
127		nonecs = pg_malloc_object(NoneCompressorState);
128		nonecs->buflen = DEFAULT_IO_BUFFER_SIZE;
129		nonecs->buffer = pg_malloc(nonecs->buflen);
130		nonecs->bufdata = 0;
131
132		cs->private_data = nonecs;
133	}
134}

146static size_t
147read_none(void *ptr, size_t size, CompressFileHandle *CFH)
148{
149	FILE	   *fp = (FILE *) CFH->private_data;
150	size_t		ret;
151
152	ret = fread(ptr, 1, size, fp);
153	if (ferror(fp))
154		pg_fatal("could not read from input file: %m");
155
156	return ret;
157}
158
159static void
160write_none(const void *ptr, size_t size, CompressFileHandle *CFH)
161{
162	size_t		ret;
163
164	errno = 0;
165	ret = fwrite(ptr, 1, size, (FILE *) CFH->private_data);
166	if (ret != size)
167	{
168		errno = (errno) ? errno : ENOSPC;
169		pg_fatal("could not write to file: %m");
170	}
171}
172
173static const char *
174get_error_none(CompressFileHandle *CFH)
175{
176	return strerror(errno);
177}
178
179static char *
180gets_none(char *ptr, int size, CompressFileHandle *CFH)
181{
182	return fgets(ptr, size, (FILE *) CFH->private_data);
183}
184
185static int
186getc_none(CompressFileHandle *CFH)
187{
188	FILE	   *fp = (FILE *) CFH->private_data;
189	int			ret;
190
191	ret = fgetc(fp);
192	if (ret == EOF)
193	{
194		if (!feof(fp))
195			pg_fatal("could not read from input file: %m");
196		else
197			pg_fatal("could not read from input file: end of file");
198	}
199
200	return ret;
201}
202
203static bool
204close_none(CompressFileHandle *CFH)
205{
206	FILE	   *fp = (FILE *) CFH->private_data;
207	int			ret = 0;
208
209	CFH->private_data = NULL;
210
211	if (fp)
212	{
213		errno = 0;
214		ret = fclose(fp);
215		if (ret != 0)
216			pg_log_error("could not close file: %m");
217	}
218
219	return ret == 0;
220}
221
222static bool
223eof_none(CompressFileHandle *CFH)
224{
225	return feof((FILE *) CFH->private_data) != 0;
226}
227
228static bool
229open_none(const char *path, int fd, const char *mode, CompressFileHandle *CFH)
230{
231	Assert(CFH->private_data == NULL);
232
233	if (fd >= 0)
234	{
235		int			dup_fd = dup(fd);
236
237		if (dup_fd < 0)
238			return false;
239		CFH->private_data = fdopen(dup_fd, mode);
240		if (CFH->private_data == NULL)
241		{
242			close(dup_fd);
243			return false;
244		}
245	}
246	else
247	{
248		CFH->private_data = fopen(path, mode);
249		if (CFH->private_data == NULL)
250			return false;
251	}
252
253	return true;
254}
255
256static bool
257open_write_none(const char *path, const char *mode, CompressFileHandle *CFH)
258{
259	Assert(CFH->private_data == NULL);
260
261	CFH->private_data = fopen(path, mode);
262	if (CFH->private_data == NULL)
263		return false;
264
265	return true;
266}

272void
273InitCompressFileHandleNone(CompressFileHandle *CFH,
274						   const pg_compress_specification compression_spec)
275{
276	CFH->open_func = open_none;
277	CFH->open_write_func = open_write_none;
278	CFH->read_func = read_none;
279	CFH->write_func = write_none;
280	CFH->gets_func = gets_none;
281	CFH->getc_func = getc_none;
282	CFH->close_func = close_none;
283	CFH->eof_func = eof_none;
284	CFH->get_error_func = get_error_none;
285
286	CFH->private_data = NULL;
