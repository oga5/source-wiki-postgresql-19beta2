172# Test: Decode a continuation record (contrecord) that spans multiple WAL
173# segments.
174#
175# Now consume all remaining room in the current WAL segment, leaving
176# space enough only for the start of a largish record.
177$node->safe_psql(
178	'postgres', q{
179DO $$
180DECLARE
181    wal_segsize int := setting::int FROM pg_settings WHERE name = 'wal_segment_size';
182    remain int;
183    iters  int := 0;
184BEGIN
185    LOOP
186        INSERT into t1(b)
187        select repeat(encode(sha256(g::text::bytea), 'hex'), (random() * 15 + 1)::int)
188        from generate_series(1, 10) g;
189
190        remain := wal_segsize - (pg_current_wal_insert_lsn() - '0/0') % wal_segsize;
191        IF remain < 2 * setting::int from pg_settings where name = 'block_size' THEN
192            RAISE log 'exiting after % iterations, % bytes to end of WAL segment', iters, remain;
193            EXIT;
194        END IF;
195        iters := iters + 1;
196    END LOOP;
197END
198$$;
199});

253# Test that pg_waldump reports a detailed error message when dumping
254# a WAL file with an invalid magic number (0000).
255#
256# The broken WAL file is created by copying a valid WAL file and
257# overwriting its magic number with 0000.
258my $broken_wal_dir = PostgreSQL::Test::Utils::tempdir_short();
259my $broken_wal = "$broken_wal_dir/$start_walfile";
260copy($node->data_dir . '/pg_wal/' . $start_walfile, $broken_wal)
261  || die "copying $start_walfile $!";
262
263my $fh;
264open($fh, '+<', $broken_wal)
265  or BAIL_OUT("open failed: $!");
266binmode $fh;
267
268sysseek($fh, 0, 0)
269  or BAIL_OUT("sysseek failed: $!");
270syswrite($fh, pack("S", 0))
271  or BAIL_OUT("syswrite failed: $!");
272close($fh)
273  or BAIL_OUT("close failed: $!");

280# Test for: Display a message that we're skipping data if `from`
281# wasn't a pointer to the start of a record.
282sub test_pg_waldump_skip_bytes
283{
284	my ($path, $startlsn, $endlsn) = @_;
285
286	# Construct a new LSN that is one byte past the original
287	# start_lsn.
288	my ($part1, $part2) = split qr{/}, $startlsn;
289	my $lsn2 = hex $part2;
290	$lsn2++;
291	my $new_start = sprintf("%s/%X", $part1, $lsn2);
292
293	my ($stdout, $stderr);
294
295	my $result = IPC::Run::run [
296		'pg_waldump',
297		'--start' => $new_start,
298		'--end' => $endlsn,
299		'--path' => $path,
300	  ],
301	  '>' => \$stdout,
302	  '2>' => \$stderr;
303	ok($result, "runs with start segment and start LSN specified");
304	like($stderr, qr/first record is after/, 'info message printed');
305}

308# Helper function to test various options.  Pass options as arguments.
309# Output lines are returned as array.
310sub test_pg_waldump
311{
312	local $Test::Builder::Level = $Test::Builder::Level + 1;
313	my ($path, $startlsn, $endlsn, @opts) = @_;
314
315	my ($stdout, $stderr);
316
317	my $result = IPC::Run::run [
318		'pg_waldump',
319		'--start' => $startlsn,
320		'--end' => $endlsn,
321		'--path' => $path,
322		@opts
323	  ],
324	  '>' => \$stdout,
325	  '2>' => \$stderr;
326	ok($result, "pg_waldump @opts: runs ok");
327	is($stderr, '', "pg_waldump @opts: no stderr");
328	my @lines = split /\n/, $stdout;
329	ok(@lines > 0, "pg_waldump @opts: some lines are output");
330	return @lines;
331}

333# Create a tar archive, shuffle the file order
334sub generate_archive
335{
336	my ($archive, $directory, $compression_flags) = @_;
337
338	my @files;
339	opendir my $dh, $directory or die "opendir: $!";
340	while (my $entry = readdir $dh)
341	{
342		# Skip '.' and '..'
343		next if $entry eq '.' || $entry eq '..';
344		push @files, $entry;
345	}
346	closedir $dh;
347
348	@files = shuffle @files;
349
350	# move into the WAL directory before archiving files
351	my $cwd = getcwd;
352	chdir($directory) || die "chdir: $!";
353	command_ok([ $tar, @tar_p_flags, $compression_flags, $archive, @files ]);
354	chdir($cwd) || die "chdir: $!";
355}
