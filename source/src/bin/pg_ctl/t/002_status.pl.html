4use strict;
5use warnings FATAL => 'all';
6
7use PostgreSQL::Test::Cluster;
8use PostgreSQL::Test::Utils;
9use Test::More;
10
11my $tempdir = PostgreSQL::Test::Utils::tempdir;
12
13command_exit_is([ 'pg_ctl', 'status', '--pgdata' => "$tempdir/nonexistent" ],
14	4, 'pg_ctl status with nonexistent directory');
15
16my $node = PostgreSQL::Test::Cluster->new('main');
17$node->init;
18
19command_exit_is([ 'pg_ctl', 'status', '--pgdata' => $node->data_dir ],
20	3, 'pg_ctl status with server not running');
21
22system_or_bail(
23	'pg_ctl',
24	'--log' => "$tempdir/logfile",
25	'--pgdata' => $node->data_dir,
26	'--wait', 'start');
27command_exit_is([ 'pg_ctl', 'status', '--pgdata' => $node->data_dir ],
28	0, 'pg_ctl status with server running');
29
30system_or_bail 'pg_ctl', 'stop', '--pgdata' => $node->data_dir;
31
32done_testing();
