14# Returns the filesystem path for the named relation.
15#
16# Assumes the test node is running
17sub relation_filepath
18{
19	my ($dbname, $relname) = @_;
20
21	my $pgdata = $node->data_dir;
22	my $rel =
23	  $node->safe_psql($dbname, qq(SELECT pg_relation_filepath('$relname')));
24	die "path not found for relation $relname" unless defined $rel;
25	return "$pgdata/$rel";
26}
27
28# Returns the name of the toast relation associated with the named relation.
29#
30# Assumes the test node is running
31sub relation_toast
32{
33	my ($dbname, $relname) = @_;
34
35	my $rel = $node->safe_psql(
36		$dbname, qq(
37		SELECT c.reltoastrelid::regclass
38			FROM pg_catalog.pg_class c
39			WHERE c.oid = '$relname'::regclass
40			  AND c.reltoastrelid != 0
41			));
42	return $rel;
43}

45# Adds the relation file for the given (dbname, relname) to the list
46# to be corrupted by means of overwriting junk in the first page.
47#
48# Assumes the test node is running.
49sub plan_to_corrupt_first_page
50{
51	my ($dbname, $relname) = @_;
52	my $relpath = relation_filepath($dbname, $relname);
53	$corrupt_page{$relpath} = 1;
54}

56# Adds the relation file for the given (dbname, relname) to the list
57# to be corrupted by means of removing the file..
58#
59# Assumes the test node is running
60sub plan_to_remove_relation_file
61{
62	my ($dbname, $relname) = @_;
63	my $relpath = relation_filepath($dbname, $relname);
64	$remove_relation{$relpath} = 1;
65}

67# For the given (dbname, relname), if a corresponding toast table
68# exists, adds that toast table's relation file to the list to be
69# corrupted by means of removing the file.
70#
71# Assumes the test node is running.
72sub plan_to_remove_toast_file
73{
74	my ($dbname, $relname) = @_;
75	my $toastname = relation_toast($dbname, $relname);
76	plan_to_remove_relation_file($dbname, $toastname) if ($toastname);
77}

79# Corrupts the first page of the given file path
80sub corrupt_first_page
81{
82	my ($relpath) = @_;
83
84	my $fh;
85	open($fh, '+<', $relpath)
86	  or BAIL_OUT("open failed: $!");
87	binmode $fh;
88
89	# Corrupt some line pointers.  The values are chosen to hit the
90	# various line-pointer-corruption checks in verify_heapam.c
91	# on both little-endian and big-endian architectures.
92	sysseek($fh, 32, 0)
93	  or BAIL_OUT("sysseek failed: $!");
94	syswrite(
95		$fh,
96		pack("L*",
97			0xAAA15550, 0xAAA0D550, 0x00010000, 0x00008000,
98			0x0000800F, 0x001e8000, 0xFFFFFFFF)
99	) or BAIL_OUT("syswrite failed: $!");
100	close($fh)
101	  or BAIL_OUT("close failed: $!");
102}

104# Stops the node, performs all the corruptions previously planned, and
105# starts the node again.
106#
107sub perform_all_corruptions()
108{
109	$node->stop();
110	for my $relpath (keys %corrupt_page)
111	{
112		corrupt_first_page($relpath);
113	}
114	for my $relpath (keys %remove_relation)
115	{
116		unlink($relpath);
117	}
118	$node->start;
119}
