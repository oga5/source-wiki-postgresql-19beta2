20# Check that "wc -l" does what we expect, else forget it
21my $wcstdin = "foo bar\nbaz\n";
22my ($wcstdout, $wcstderr);
23my $result = IPC::Run::run [ 'wc', '-l' ],
24  '<' => \$wcstdin,
25  '>' => \$wcstdout,
26  '2>' => \$wcstderr;
27chomp $wcstdout;
28if ($wcstdout !~ /^ *2$/ || $wcstderr ne '')

73# fire up an interactive psql session and configure it such that each query

80for my $pty (values %{ $h->{run}->{PTYS} })

85# Simple test case: type something and see if psql responds as expected
86sub do_command
87{
88	my ($send, $pattern, $annotation) = @_;
89
90	# report test failures from caller location
91	local $Test::Builder::Level = $Test::Builder::Level + 1;
92
93	# send the data to be sent and wait for its result
94	my $out = $h->query_until($pattern, $send);
95	my $okay = ($out =~ $pattern && !$h->{timeout}->is_expired);
96	ok($okay, $annotation);
97	# for debugging, log actual output if it didn't match
98	local $Data::Dumper::Terse = 1;
99	local $Data::Dumper::Useqq = 1;
100	diag 'Actual output was ' . Dumper($out) . "Did not match \"$pattern\"\n"
101	  if !$okay;
102	return;
103}

105# Test invocation of the pager
106#
107# Note that interactive_psql starts psql with --no-align --tuples-only,
108# and that the output string will include psql's prompts and command echo.
109# So we have to test for patterns that can't match the command itself,
110# and we can't assume the match will extend across a whole line (there
111# might be a prompt ahead of it in the output).
112
113do_command(
114	"SELECT 'test' AS t FROM generate_series(1,23);\n",
115	qr/test\r?$/m,
116	"execute SELECT query that needs no pagination");
117
118do_command(
119	"SELECT 'test' AS t FROM generate_series(1,24);\n",
120	qr/24\r?$/m,
121	"execute SELECT query that needs pagination");
122
123do_command(
124	"\\pset expanded\nSELECT generate_series(1,20) as g;\n",
125	qr/39\r?$/m,
126	"execute SELECT query that needs pagination in expanded mode");
127
128do_command(
129	"\\pset tuples_only off\n\\d+ public.view_030_pager\n",
130	qr/55\r?$/m,
131	"execute command with footer that needs pagination");
132
133do_command(
134	"\\pset expanded on\n\\pset format wrapped\nSELECT repeat('x',80) AS payload FROM generate_series(1,11);\n",
135	qr/34\r?$/m,
136	"execute SELECT query that needs pagination in expanded wrapped mode");
137
138# send psql an explicit \q to shut it down, else pty won't close properly
139$h->quit or die "psql returned $?";
140
141# done
142$node->stop;
143done_testing();
