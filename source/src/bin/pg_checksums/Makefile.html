11PGFILEDESC = "pg_checksums - verify data checksums in an offline cluster"
12PGAPPICON=win32
13
14subdir = src/bin/pg_checksums
15top_builddir = ../../..
16include $(top_builddir)/src/Makefile.global
17
18# We need libpq only because fe_utils does.
19LDFLAGS_INTERNAL += -L$(top_builddir)/src/fe_utils -lpgfeutils $(libpq_pgport)
20
21OBJS = \
22	$(WIN32RES) \
23	pg_checksums.o
24
25all: pg_checksums
26
27pg_checksums: $(OBJS) | submake-libpgport
28	$(CC) $(CFLAGS) $^ $(LDFLAGS) $(LDFLAGS_EX) $(LIBS) -o $@$(X)
29
30install: all installdirs
31	$(INSTALL_PROGRAM) pg_checksums$(X) '$(DESTDIR)$(bindir)/pg_checksums$(X)'
32
33installdirs:
34	$(MKDIR_P) '$(DESTDIR)$(bindir)'
35
36uninstall:
37	rm -f '$(DESTDIR)$(bindir)/pg_checksums$(X)'
38
39clean distclean:
40	rm -f pg_checksums$(X) $(OBJS)
41	rm -rf tmp_check
42
43check:
44	$(prove_check)
45
46installcheck:
47	$(prove_installcheck)
