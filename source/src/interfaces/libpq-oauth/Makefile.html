16PGFILEDESC = "libpq-oauth - device authorization OAuth support"
17
18# This is an internal module; we don't want an SONAME and therefore do not set
19# SO_MAJOR_VERSION. This requires an explicit override for the shared library
20# name.
21NAME = pq-oauth
22override shlib := lib$(NAME)$(DLSUFFIX)
23override stlib := libpq-oauth.a
24
25override CPPFLAGS := -I$(libpq_srcdir) -I$(top_builddir)/src/port $(CPPFLAGS) $(LIBCURL_CPPFLAGS)
26override CFLAGS += $(PTHREAD_CFLAGS)
27
28override CPPFLAGS_SHLIB := -DUSE_DYNAMIC_OAUTH
29
30# A bit of forward-looking paranoia: don't allow libpq-oauth.so to accidentally
31# depend on the encoding IDs coming from libpq. They're not guaranteed to match
32# the IDs in use by our version of pgcommon, now that we allow the major version
33# of libpq to differ from the major version of libpq-oauth.
34override CPPFLAGS_SHLIB += -DUSE_PRIVATE_ENCODING_FUNCS
35
36OBJS = \
37	$(WIN32RES)
38
39OBJS_STATIC = oauth-curl.o
40
41# The shared library needs additional glue symbols.

44	oauth-utils.o \

46oauth-utils.o: override CPPFLAGS += $(CPPFLAGS_SHLIB)

48# Add shlib-/stlib-specific objects.

77#
78# Top-Level Targets
79#
80# The existence of a t/ folder induces the buildfarm to run Make directly on
81# this subdirectory, bypassing the recursion skip in src/interfaces/Makefile.
82# Wrap the standard build targets in a with_libcurl conditional to avoid
83# building OAuth code on platforms that haven't requested it. (The "clean"-style
84# targets remain available.)
85#
86
87ifeq ($(with_libcurl), yes)
88
89# By default, a library without an SONAME doesn't get a static library, so we
90# add it to the build explicitly.
91all: all-lib all-static-lib
92
93# Ignore the standard rules for SONAME-less installation; we want both the
94# static and shared libraries to go into libdir.
95install: all installdirs $(stlib) $(shlib)
96	$(INSTALL_SHLIB) $(shlib) '$(DESTDIR)$(libdir)/$(shlib)'
97	$(INSTALL_STLIB) $(stlib) '$(DESTDIR)$(libdir)/$(stlib)'
98
99installdirs:
100	$(MKDIR_P) '$(DESTDIR)$(libdir)'
101
102check: all-tests
103	$(prove_check)
104
105installcheck: all-tests
106	$(prove_installcheck)
107
108endif # with_libcurl
109
110uninstall:
111	rm -f '$(DESTDIR)$(libdir)/$(stlib)'
112	rm -f '$(DESTDIR)$(libdir)/$(shlib)'
113
114clean distclean: clean-lib
115	rm -f $(OBJS) $(OBJS_STATIC) $(OBJS_SHLIB)
116	rm -f test-oauth-curl.o oauth_tests$(X)
117	rm -rf tmp_check
