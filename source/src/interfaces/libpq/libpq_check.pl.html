1#!/usr/bin/perl
2#
3# src/interfaces/libpq/libpq_check.pl
4#
5# Copyright (c) 2025-2026, PostgreSQL Global Development Group
6#
7# Check the state of a libpq library.  Currently, this script checks that
8# exit() is not called, because client libraries must not terminate the
9# host application.
10#
11# This script is called by both Makefile and Meson.
12
13use strict;
14use warnings FATAL => 'all';
15
16use Getopt::Long;
17use Config;
18
19my $nm_path;
20my $input_file;
21my $stamp_file;
22my @problematic_lines;
23
24Getopt::Long::GetOptions(
25	'nm:s' => \$nm_path,
26	'input_file:s' => \$input_file,
27	'stamp_file:s' => \$stamp_file) or die "$0: wrong arguments\n";
28
29die "$0: --input_file must be specified\n" unless defined $input_file;
30die "$0: --nm must be specified\n" unless defined $nm_path and -x $nm_path;
31
32sub create_stamp_file
33{
34	open my $fh, '>', $stamp_file
35	  or die "can't open $stamp_file: $!";
36	close $fh;
37}
38
39# Skip on Windows and Solaris
40if (   $Config{osname} =~ /MSWin32|cygwin|msys/i
41	|| $Config{osname} =~ /solaris/i)
42{
43	exit 0;
44}
45
46# Run nm to scan for symbols.  If nm fails at runtime, skip the check.
47open my $fh, '-|', "$nm_path -A -u $input_file 2>/dev/null"
48  or exit 0;
49
50while (<$fh>)
51{
52	# Set of symbols allowed.
53
54	# The exclusion of __cxa_atexit is necessary on OpenBSD, which seems
55	# to insert references to that even in pure C code.
56	next if /__cxa_atexit/;
57
58	# Excluding __tsan_func_exit is necessary when using ThreadSanitizer data
59	# race detector which uses this function for instrumentation of function
60	# exit.
61	next if /__tsan_func_exit/;
62
63	# Excluding pthread_exit allows legitimate thread terminations in some
64	# builds.
65	next if /pthread_exit/;
66
67	# Anything containing "exit" is suspicious.
68	# (Ideally we should reject abort() too, but there are various scenarios
69	# where build toolchains insert abort() calls, e.g. to implement
70	# assert().)
71	if (/exit/)
72	{
73		push @problematic_lines, $_;
74	}
75}
76close $fh;
77
78if (@problematic_lines)
79{
80	print "libpq must not be calling any function which invokes exit\n";
81	print "Problematic symbol references:\n";
82	print @problematic_lines;
83
84	exit 1;
85}
86# Create stamp file, if required
87if (defined($stamp_file))
88{
89	create_stamp_file();
90}
91
92exit 0;
