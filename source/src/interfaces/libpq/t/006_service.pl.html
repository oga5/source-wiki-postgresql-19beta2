2use strict;
3use warnings FATAL => 'all';
4use File::Copy;
5use PostgreSQL::Test::Utils;
6use PostgreSQL::Test::Cluster;
7use Test::More;
8
9# This tests scenarios related to the service name and the service file,
10# for the connection options and their environment variables.
11
12my $node = PostgreSQL::Test::Cluster->new('node');
13$node->init;
14$node->start;
15
16# Set up a dummy node used for the connection tests, but do not start it.
17# This ensures that the environment variables used for the connection do
18# not interfere with the connection attempts, and that the service file's
19# contents are used.
20my $dummy_node = PostgreSQL::Test::Cluster->new('dummy_node');
21$dummy_node->init;
22
23my $td = PostgreSQL::Test::Utils::tempdir;
24
25# Create the set of service files used in the tests.
26# File that includes a valid service name, and uses a decomposed connection
27# string for its contents, split on spaces.
28my $srvfile_valid = "$td/pg_service_valid.conf";
29append_to_file($srvfile_valid, "[my_srv]\n");
30foreach my $param (split(/\s+/, $node->connstr))
31{
32	append_to_file($srvfile_valid, $param . "\n");
33}
34
35# File defined with no contents, used as default value for PGSERVICEFILE,
36# so as no lookup is attempted in the user's home directory.
37my $srvfile_empty = "$td/pg_service_empty.conf";
38append_to_file($srvfile_empty, '');
39
40# Default service file in PGSYSCONFDIR.
41my $srvfile_default = "$td/pg_service.conf";
42
43# Missing service file.
44my $srvfile_missing = "$td/pg_service_missing.conf";
45
46# Service file with nested "service" defined.
47my $srvfile_nested = "$td/pg_service_nested.conf";
48copy($srvfile_valid, $srvfile_nested)
49  or die "Could not copy $srvfile_valid to $srvfile_nested: $!";
50append_to_file($srvfile_nested, "service=invalid_srv\n");
51
52# Service file with nested "servicefile" defined.
53my $srvfile_nested_2 = "$td/pg_service_nested_2.conf";
54copy($srvfile_valid, $srvfile_nested_2)
55  or die "Could not copy $srvfile_valid to $srvfile_nested_2: $!";
56append_to_file($srvfile_nested_2, 'servicefile=' . $srvfile_default . "\n");
57
58# Set the fallback directory lookup of the service file to the temporary
59# directory of this test.  PGSYSCONFDIR is used if the service file
60# defined in PGSERVICEFILE cannot be found, or when a service file is
61# found but not the service name.
62local $ENV{PGSYSCONFDIR} = $td;
63# Force PGSERVICEFILE to a default location, so as this test never
64# tries to look at a home directory.  This value needs to remain
65# at the top of this script before running any tests, and should never
66# be changed.
67local $ENV{PGSERVICEFILE} = "$srvfile_empty";
68
69# Checks combinations of service name and a valid service file.
70{
71	local $ENV{PGSERVICEFILE} = $srvfile_valid;
72	$dummy_node->connect_ok(
73		'service=my_srv',
74		'connection with correct "service" string and PGSERVICEFILE',
75		sql => "SELECT 'connect1_1'",
76		expected_stdout => qr/connect1_1/);
77
78	$dummy_node->connect_ok(
79		'postgres://?service=my_srv',
80		'connection with correct "service" URI and PGSERVICEFILE',
81		sql => "SELECT 'connect1_2'",
82		expected_stdout => qr/connect1_2/);
83
84	$dummy_node->connect_fails(
85		'service=undefined-service',
86		'connection with incorrect "service" string and PGSERVICEFILE',
87		expected_stderr =>
88		  qr/definition of service "undefined-service" not found/);
89
90	local $ENV{PGSERVICE} = 'my_srv';
91	$dummy_node->connect_ok(
92		'',
93		'connection with correct PGSERVICE and PGSERVICEFILE',
94		sql => "SELECT 'connect1_3'",
95		expected_stdout => qr/connect1_3/);
96
97	local $ENV{PGSERVICE} = 'undefined-service';
98	$dummy_node->connect_fails(
99		'',
100		'connection with incorrect PGSERVICE and PGSERVICEFILE',
101		expected_stdout =>
102		  qr/definition of service "undefined-service" not found/);
103}
104
105# Checks case of incorrect service file.
106{
107	local $ENV{PGSERVICEFILE} = $srvfile_missing;
108	$dummy_node->connect_fails(
109		'service=my_srv',
110		'connection with correct "service" string and incorrect PGSERVICEFILE',
111		expected_stderr =>
112		  qr/service file ".*pg_service_missing.conf" not found/);
113}
114
115# Checks case of service file named "pg_service.conf" in PGSYSCONFDIR.
116{
117	# Create copy of valid file
118	my $srvfile_default = "$td/pg_service.conf";
119	copy($srvfile_valid, $srvfile_default);
120
121	$dummy_node->connect_ok(
122		'service=my_srv',
123		'connection with correct "service" string and pg_service.conf',
124		sql => "SELECT 'connect2_1'",
125		expected_stdout => qr/connect2_1/);
126
127	$dummy_node->connect_ok(
128		'postgres://?service=my_srv',
129		'connection with correct "service" URI and default pg_service.conf',
130		sql => "SELECT 'connect2_2'",
131		expected_stdout => qr/connect2_2/);
132
133	$dummy_node->connect_fails(
134		'service=undefined-service',
135		'connection with incorrect "service" string and default pg_service.conf',
136		expected_stderr =>
137		  qr/definition of service "undefined-service" not found/);
138
139	local $ENV{PGSERVICE} = 'my_srv';
140	$dummy_node->connect_ok(
141		'',
142		'connection with correct PGSERVICE and default pg_service.conf',
143		sql => "SELECT 'connect2_3'",
144		expected_stdout => qr/connect2_3/);
145
146	my $srvfile_empty_win_cared = $srvfile_empty;
147	$srvfile_empty_win_cared =~ s/\\/\\\\/g;
148	$dummy_node->connect_ok(
149		q{service=my_srv servicefile='} . $srvfile_empty_win_cared . q{'},
150		'SERVICEFILE updated when service is found in default pg_service.conf',
151		sql => '\echo :SERVICEFILE',
152		expected_stdout => qr/^\Q$srvfile_default\E$/);
153
154	local $ENV{PGSERVICE} = 'undefined-service';
155	$dummy_node->connect_fails(
156		'',
157		'connection with incorrect PGSERVICE and default pg_service.conf',
158		expected_stdout =>
159		  qr/definition of service "undefined-service" not found/);
160
161	# Remove default pg_service.conf.
162	unlink($srvfile_default);
163}
164
165# Checks nested service file contents.
166{
167	local $ENV{PGSERVICEFILE} = $srvfile_nested;
168
169	$dummy_node->connect_fails(
170		'service=my_srv',
171		'connection with "service" in nested service file',
172		expected_stderr =>
173		  qr/nested "service" specifications not supported in service file/);
174
175	local $ENV{PGSERVICEFILE} = $srvfile_nested_2;
176
177	$dummy_node->connect_fails(
178		'service=my_srv',
179		'connection with "servicefile" in nested service file',
180		expected_stderr =>
181		  qr/nested "servicefile" specifications not supported in service file/
182	);
183}
184
185# Properly escape backslashes in the path, to ensure the generation of
186# correct connection strings.
187my $srvfile_win_cared = $srvfile_valid;
188$srvfile_win_cared =~ s/\\/\\\\/g;
189
190# Checks that the "servicefile" option works as expected
191{
192	$dummy_node->connect_ok(
193		q{service=my_srv servicefile='} . $srvfile_win_cared . q{'},
194		'connection with valid servicefile in connection string',
195		sql => "SELECT 'connect3_1'",
196		expected_stdout => qr/connect3_1/);
197
198	# Encode slashes and backslash
199	my $encoded_srvfile = $srvfile_valid =~ s{([\\/])}{
200		$1 eq '/' ? '%2F' : '%5C'
201	}ger;
202
203	# Additionally encode a colon in servicefile path of Windows
204	$encoded_srvfile =~ s/:/%3A/g;
205
206	$dummy_node->connect_ok(
207		'postgresql:///?service=my_srv&servicefile=' . $encoded_srvfile,
208		'connection with valid servicefile in URI',
209		sql => "SELECT 'connect3_2'",
210		expected_stdout => qr/connect3_2/);
211
212	local $ENV{PGSERVICE} = 'my_srv';
213	$dummy_node->connect_ok(
214		q{servicefile='} . $srvfile_win_cared . q{'},
215		'connection with PGSERVICE and servicefile in connection string',
216		sql => "SELECT 'connect3_3'",
217		expected_stdout => qr/connect3_3/);
218
219	$dummy_node->connect_ok(
220		'postgresql://?servicefile=' . $encoded_srvfile,
221		'connection with PGSERVICE and servicefile in URI',
222		sql => "SELECT 'connect3_4'",
223		expected_stdout => qr/connect3_4/);
224}
225
226# Check that the "servicefile" option takes priority over the PGSERVICEFILE
227# environment variable.
228{
229	local $ENV{PGSERVICEFILE} = 'non-existent-file.conf';
230
231	$dummy_node->connect_fails(
232		'service=my_srv',
233		'connection with invalid PGSERVICEFILE',
234		expected_stderr =>
235		  qr/service file "non-existent-file\.conf" not found/);
236
237	$dummy_node->connect_ok(
238		q{service=my_srv servicefile='} . $srvfile_win_cared . q{'},
239		'connection with both servicefile and PGSERVICEFILE',
240		sql => "SELECT 'connect4_1'",
241		expected_stdout => qr/connect4_1/);
242}
243
244$node->teardown_node;
245
246done_testing();
