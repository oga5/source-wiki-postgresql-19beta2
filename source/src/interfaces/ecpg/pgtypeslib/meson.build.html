3ecpg_pgtypes_sources = files(
4  'common.c',
5  'datetime.c',
6  'dt_common.c',
7  'interval.c',
8  'numeric.c',
9  'timestamp.c',
10)
11ecpg_pgtypes_so_sources = [] # for shared lib, in addition to the above
12
13export_file = custom_target('libpgtypes.exports', kwargs: gen_export_kwargs)
14ecpg_pgtypes_inc = [include_directories('.'), ecpg_inc]
15ecpg_pgtypes_c_args = ['-DSO_MAJOR_VERSION=3']
16
17if host_system == 'windows'
18  ecpg_pgtypes_so_sources += rc_lib_gen.process(win32ver_rc, extra_args: [
19    '--NAME', 'pgtypes',
20    '--FILEDESC', 'pgtypes - library for data type mapping',])
21endif
22
23# see src/interfaces/libpq/meson.build
24if build_static_lib
25ecpg_pgtypes_st = static_library('libpgtypes',
26  ecpg_pgtypes_sources,
27  include_directories: ecpg_pgtypes_inc,
28  c_args: ecpg_pgtypes_c_args,
29  c_pch: pch_postgres_fe_h,
30  dependencies: frontend_stlib_code,
31  kwargs: default_lib_args,
32)
33ecpg_targets += ecpg_pgtypes_st
34endif
35
36if build_shared_lib
37ecpg_pgtypes_so = shared_library('libpgtypes',
38  ecpg_pgtypes_sources + ecpg_pgtypes_so_sources,
39  include_directories: ecpg_pgtypes_inc,
40  c_args: ecpg_pgtypes_c_args,
41  c_pch: pch_postgres_fe_h,
42  dependencies: frontend_shlib_code,
43  version: '3.' + pg_version_major.to_string(),
44  soversion: host_system not in ['aix', 'windows'] ? '3' : '',
45  darwin_versions: ['3', '3.' + pg_version_major.to_string()],
46  link_args: export_fmt.format(export_file.full_path()),
47  link_depends: export_file,
48  kwargs: default_lib_args,
49)
50ecpg_targets += ecpg_pgtypes_so
51endif
52
53pkgconfig.generate(
54  name: 'libpgtypes',
55  description: 'PostgreSQL libpgtypes library',
56  url: pg_url,
57  libraries: ecpg_pgtypes_so,
58  libraries_private: [frontend_stlib_code],
59)
