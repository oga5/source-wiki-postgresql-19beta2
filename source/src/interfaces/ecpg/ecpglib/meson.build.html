3ecpglib_sources = files(
4  'connect.c',
5  'data.c',
6  'descriptor.c',
7  'error.c',
8  'execute.c',
9  'memory.c',
10  'misc.c',
11  'prepare.c',
12  'sqlda.c',
13  'typename.c',
14)
15ecpglib_so_sources = [] # for shared lib, in addition to the above
16
17ecpglib_inc = [include_directories('.'), ecpg_inc]
18ecpglib_c_args = ['-DSO_MAJOR_VERSION=6']
19export_file = custom_target('libecpg.exports', kwargs: gen_export_kwargs)
20
21if host_system == 'windows'
22  ecpglib_so_sources += rc_lib_gen.process(win32ver_rc, extra_args: [
23    '--NAME', 'libecpg',
24    '--FILEDESC', 'ECPG - embedded SQL in C',])
25endif
26
27# see src/interfaces/libpq/meson.build
28if build_static_lib
29ecpglib_st = static_library('libecpg',
30  ecpglib_sources,
31  include_directories: ecpglib_inc,
32  c_args: ecpglib_c_args,
33  c_pch: pch_postgres_fe_h,
34  dependencies: [frontend_stlib_code, thread_dep, libpq],
35  link_with: [ecpg_pgtypes_st],
36  kwargs: default_lib_args,
37)
38ecpg_targets += ecpglib_st
39endif
40
41if build_shared_lib
42ecpglib_so = shared_library('libecpg',
43  ecpglib_sources + ecpglib_so_sources,
44  include_directories: ecpglib_inc,
45  c_args: ecpglib_c_args,
46  c_pch: pch_postgres_fe_h,
47  dependencies: [frontend_shlib_code, libpq, thread_dep],
48  link_with: ecpg_pgtypes_so,
49  soversion: host_system not in ['aix', 'windows'] ? '6' : '',
50  darwin_versions: ['6', '6.' + pg_version_major.to_string()],
51  version: '6.' + pg_version_major.to_string(),
52  link_args: export_fmt.format(export_file.full_path()),
53  link_depends: export_file,
54  kwargs: default_lib_args,
55)
56ecpg_targets += ecpglib_so
57endif
58
59pkgconfig.generate(
60  name: 'libecpg',
61  description: 'PostgreSQL libecpg library',
62  url: pg_url,
63  libraries: ecpglib_so,
64  libraries_private: [frontend_stlib_code, thread_dep],
65  requires_private: ['libpgtypes', 'libpq'],
66)

68subdir('po', if_found: libintl)
