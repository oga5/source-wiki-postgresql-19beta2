63/*
64 * Shared-memory state for WAL recovery.
65 */
66typedef struct XLogRecoveryCtlData
67{
68	/*
69	 * SharedHotStandbyActive indicates if we allow hot standby queries to be
70	 * run.  Protected by info_lck.
71	 */
72	bool		SharedHotStandbyActive;
73
74	/*
75	 * SharedPromoteIsTriggered indicates if a standby promotion has been
76	 * triggered.  Protected by info_lck.
77	 */
78	bool		SharedPromoteIsTriggered;
79
80	/*
81	 * recoveryWakeupLatch is used to wake up the startup process to continue
82	 * WAL replay, if it is waiting for WAL to arrive or promotion to be
83	 * requested.
84	 *
85	 * Note that the startup process also uses another latch, its procLatch,
86	 * to wait for recovery conflict. If we get rid of recoveryWakeupLatch for
87	 * signaling the startup process in favor of using its procLatch, which
88	 * comports better with possible generic signal handlers using that latch.
89	 * But we should not do that because the startup process doesn't assume
90	 * that it's waken up by walreceiver process or SIGHUP signal handler
91	 * while it's waiting for recovery conflict. The separate latches,
92	 * recoveryWakeupLatch and procLatch, should be used for inter-process
93	 * communication for WAL replay and recovery conflict, respectively.
94	 */
95	Latch		recoveryWakeupLatch;
96
97	/*
98	 * Last record successfully replayed.
99	 */
100	XLogRecPtr	lastReplayedReadRecPtr; /* start position */
101	XLogRecPtr	lastReplayedEndRecPtr;	/* end+1 position */
102	TimeLineID	lastReplayedTLI;	/* timeline */
103
104	/*
105	 * When we're currently replaying a record, ie. in a redo function,
106	 * replayEndRecPtr points to the end+1 of the record being replayed,
107	 * otherwise it's equal to lastReplayedEndRecPtr.
108	 */
109	XLogRecPtr	replayEndRecPtr;
110	TimeLineID	replayEndTLI;
111	/* timestamp of last COMMIT/ABORT record replayed (or being replayed) */
112	TimestampTz recoveryLastXTime;
113
114	/*
115	 * timestamp of when we started replaying the current chunk of WAL data,
116	 * only relevant for replication or archive recovery
117	 */
118	TimestampTz currentChunkStartTime;
119	/* Recovery pause state */
120	RecoveryPauseState recoveryPauseState;
121	ConditionVariable recoveryNotPausedCV;
122
123	slock_t		info_lck;		/* locks shared variables shown above */
124} XLogRecoveryCtlData;

161/*
162 * FinishWalRecovery() returns this.  It contains information about the point
163 * where recovery ended, and why it ended.
164 */
165typedef struct
166{
167	/*
168	 * Information about the last valid or applied record, after which new WAL
169	 * can be appended.  'lastRec' is the position where the last record
170	 * starts, and 'endOfLog' is its end.  'lastPage' is a copy of the last
171	 * partial page that contains endOfLog (or NULL if endOfLog is exactly at
172	 * page boundary).  'lastPageBeginPtr' is the position where the last page
173	 * begins.
174	 *
175	 * endOfLogTLI is the TLI in the filename of the XLOG segment containing
176	 * the last applied record.  It could be different from lastRecTLI, if
177	 * there was a timeline switch in that segment, and we were reading the
178	 * old WAL from a segment belonging to a higher timeline.
179	 */
180	XLogRecPtr	lastRec;		/* start of last valid or applied record */
181	TimeLineID	lastRecTLI;
182	XLogRecPtr	endOfLog;		/* end of last valid or applied record */
183	TimeLineID	endOfLogTLI;
184
185	XLogRecPtr	lastPageBeginPtr;	/* LSN of page that contains endOfLog */
186	char	   *lastPage;		/* copy of the last page, up to endOfLog */
187
188	/*
189	 * abortedRecPtr is the start pointer of a broken record at end of WAL
190	 * when recovery completes; missingContrecPtr is the location of the first
191	 * contrecord that went missing.  See CreateOverwriteContrecordRecord for
192	 * details.
193	 */
194	XLogRecPtr	abortedRecPtr;
195	XLogRecPtr	missingContrecPtr;
196
197	/* short human-readable string describing why recovery ended */
198	char	   *recoveryStopReason;
199
200	/*
201	 * If standby or recovery signal file was found, these flags are set
202	 * accordingly.
203	 */
204	bool		standby_signal_file_found;
205	bool		recovery_signal_file_found;
206} EndOfWalRecoveryInfo;
