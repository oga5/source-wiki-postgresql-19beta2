31CATALOG(pg_conversion,2607,ConversionRelationId)
32{
33	/* oid */
34	Oid			oid;
35
36	/* name of the conversion */
37	NameData	conname;
38
39	/* namespace that the conversion belongs to */
40	Oid			connamespace BKI_DEFAULT(pg_catalog) BKI_LOOKUP(pg_namespace);
41
42	/* owner of the conversion */
43	Oid			conowner BKI_DEFAULT(POSTGRES) BKI_LOOKUP(pg_authid);
44
45	/* FOR encoding id */
46	int32		conforencoding BKI_LOOKUP(encoding);
47
48	/* TO encoding id */
49	int32		contoencoding BKI_LOOKUP(encoding);
50
51	/* OID of the conversion proc */
52	regproc		conproc BKI_LOOKUP(pg_proc);
53
54	/* true if this is a default conversion */
55	bool		condefault BKI_DEFAULT(t);
56} FormData_pg_conversion;
57
58END_CATALOG_STRUCT

60/* ----------------
61 *		Form_pg_conversion corresponds to a pointer to a tuple with
62 *		the format of pg_conversion relation.
