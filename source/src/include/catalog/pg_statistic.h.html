31CATALOG(pg_statistic,2619,StatisticRelationId)
32{
33	/* These fields form the unique key for the entry: */
34	Oid			starelid BKI_LOOKUP(pg_class);	/* relation containing
35												 * attribute */
36	int16		staattnum;		/* attribute (column) stats are for */
37	bool		stainherit;		/* true if inheritance children are included */
38
39	/* the fraction of the column's entries that are NULL: */
40	float4		stanullfrac;
41
42	/*
43	 * stawidth is the average width in bytes of non-null entries.  For
44	 * fixed-width datatypes this is of course the same as the typlen, but for
45	 * var-width types it is more useful.  Note that this is the average width
46	 * of the data as actually stored, post-TOASTing (eg, for a
47	 * moved-out-of-line value, only the size of the pointer object is
48	 * counted).  This is the appropriate definition for the primary use of
49	 * the statistic, which is to estimate sizes of in-memory hash tables of
50	 * tuples.
51	 */
52	int32		stawidth;
53
54	/* ----------------
55	 * stadistinct indicates the (approximate) number of distinct non-null
56	 * data values in the column.  The interpretation is:
57	 *		0		unknown or not computed
58	 *		> 0		actual number of distinct values
59	 *		< 0		negative of multiplier for number of rows
60	 * The special negative case allows us to cope with columns that are
61	 * unique (stadistinct = -1) or nearly so (for example, a column in which
62	 * non-null values appear about twice on the average could be represented
63	 * by stadistinct = -0.5 if there are no nulls, or -0.4 if 20% of the
64	 * column is nulls).  Because the number-of-rows statistic in pg_class may
65	 * be updated more frequently than pg_statistic is, it's important to be
66	 * able to describe such situations as a multiple of the number of rows,
67	 * rather than a fixed number of distinct values.  But in other cases a
68	 * fixed number is correct (eg, a boolean column).
69	 * ----------------
70	 */
71	float4		stadistinct;
72
73	/* ----------------
74	 * To allow keeping statistics on different kinds of datatypes,
75	 * we do not hard-wire any particular meaning for the remaining
76	 * statistical fields.  Instead, we provide several "slots" in which
77	 * statistical data can be placed.  Each slot includes:
78	 *		kind			integer code identifying kind of data (see below)
79	 *		op				OID of associated operator, if needed
80	 *		coll			OID of relevant collation, or 0 if none
81	 *		numbers			float4 array (for statistical values)
82	 *		values			anyarray (for representations of data values)
83	 * The ID, operator, and collation fields are never NULL; they are zeroes
84	 * in an unused slot.  The numbers and values fields are NULL in an
85	 * unused slot, and might also be NULL in a used slot if the slot kind
86	 * has no need for one or the other.
87	 * ----------------
88	 */
89
90	int16		stakind1;
91	int16		stakind2;
92	int16		stakind3;
93	int16		stakind4;
94	int16		stakind5;
95
96	Oid			staop1 BKI_LOOKUP_OPT(pg_operator);
97	Oid			staop2 BKI_LOOKUP_OPT(pg_operator);
98	Oid			staop3 BKI_LOOKUP_OPT(pg_operator);
99	Oid			staop4 BKI_LOOKUP_OPT(pg_operator);
100	Oid			staop5 BKI_LOOKUP_OPT(pg_operator);
101
102	Oid			stacoll1 BKI_LOOKUP_OPT(pg_collation);
103	Oid			stacoll2 BKI_LOOKUP_OPT(pg_collation);
104	Oid			stacoll3 BKI_LOOKUP_OPT(pg_collation);
105	Oid			stacoll4 BKI_LOOKUP_OPT(pg_collation);
106	Oid			stacoll5 BKI_LOOKUP_OPT(pg_collation);
107
108#ifdef CATALOG_VARLEN			/* variable-length fields start here */
109	float4		stanumbers1[1];
110	float4		stanumbers2[1];
111	float4		stanumbers3[1];
112	float4		stanumbers4[1];
113	float4		stanumbers5[1];
114
115	/*
116	 * Values in these arrays are values of the column's data type, or of some
117	 * related type such as an array element type.  We presently have to cheat
118	 * quite a bit to allow polymorphic arrays of this kind, but perhaps
119	 * someday it'll be a less bogus facility.
120	 */
121	anyarray	stavalues1;
122	anyarray	stavalues2;
123	anyarray	stavalues3;
124	anyarray	stavalues4;
125	anyarray	stavalues5;
126#endif
127} FormData_pg_statistic;

129END_CATALOG_STRUCT

151/*
152 * Several statistical slot "kinds" are defined by core PostgreSQL, as
153 * documented below.  Also, custom data types can define their own "kind"
154 * codes by mutual agreement between a custom typanalyze routine and the
155 * selectivity estimation functions of the type's operators.
156 *
157 * Code reading the pg_statistic relation should not assume that a particular
158 * data "kind" will appear in any particular slot.  Instead, search the
159 * stakind fields to see if the desired data is available.  (The standard
160 * function get_attstatsslot() may be used for this.)
161 *
162 * Note: The pg_stats view needs to be modified whenever a new slot kind is
163 * added to core.
164 */
165
166/*
167 * The present allocation of "kind" codes is:
168 *
169 *	1-99:		reserved for assignment by the core PostgreSQL project
170 *				(values in this range will be documented in this file)
171 *	100-199:	reserved for assignment by the PostGIS project
172 *				(values to be documented in PostGIS documentation)
173 *	200-299:	reserved for assignment by the ESRI ST_Geometry project
174 *				(values to be documented in ESRI ST_Geometry documentation)
175 *	300-9999:	reserved for future public assignments
176 *
177 * For private use you may choose a "kind" code at random in the range
178 * 10000-30000.  However, for code that is to be widely disseminated it is
179 * better to obtain a publicly defined "kind" code by request from the
180 * PostgreSQL Global Development Group.
181 */
182
183/*
184 * In a "most common values" slot, staop is the OID of the "=" operator
185 * used to decide whether values are the same or not, and stacoll is the
186 * collation used (same as column's collation).  stavalues contains
187 * the K most common non-null values appearing in the column, and stanumbers
188 * contains their frequencies (fractions of total row count).  The values
189 * shall be ordered in decreasing frequency.  Note that since the arrays are
190 * variable-size, K may be chosen at ANALYZE time.  Values should not appear
191 * in MCV unless they have been observed to occur more than once; a unique
192 * column will have no MCV slot.
193 */
194#define STATISTIC_KIND_MCV	1
195
196/*
197 * A "histogram" slot describes the distribution of scalar data.  staop is
198 * the OID of the "<" operator that describes the sort ordering, and stacoll
199 * is the relevant collation.  (In theory more than one histogram could appear,
200 * if a datatype has more than one useful sort operator or we care about more
201 * than one collation.  Currently the collation will always be that of the
202 * underlying column.)  stavalues contains M (>=2) non-null values that
203 * divide the non-null column data values into M-1 bins of approximately equal
204 * population.  The first stavalues item is the MIN and the last is the MAX.
205 * stanumbers is not used and should be NULL.  IMPORTANT POINT: if an MCV
206 * slot is also provided, then the histogram describes the data distribution
207 * *after removing the values listed in MCV* (thus, it's a "compressed
208 * histogram" in the technical parlance).  This allows a more accurate
209 * representation of the distribution of a column with some very-common
210 * values.  In a column with only a few distinct values, it's possible that
211 * the MCV list describes the entire data population; in this case the
212 * histogram reduces to empty and should be omitted.
213 */
214#define STATISTIC_KIND_HISTOGRAM  2
215
216/*
217 * A "correlation" slot describes the correlation between the physical order
218 * of table tuples and the ordering of data values of this column, as seen
219 * by the "<" operator identified by staop with the collation identified by
220 * stacoll.  (As with the histogram, more than one entry could theoretically
221 * appear.)  stavalues is not used and should be NULL.  stanumbers contains
222 * a single entry, the correlation coefficient between the sequence of data
223 * values and the sequence of their actual tuple positions.  The coefficient
224 * ranges from +1 to -1.
225 */
226#define STATISTIC_KIND_CORRELATION	3
227
228/*
229 * A "most common elements" slot is similar to a "most common values" slot,
230 * except that it stores the most common non-null *elements* of the column
231 * values.  This is useful when the column datatype is an array or some other
232 * type with identifiable elements (for instance, tsvector).  staop contains
233 * the equality operator appropriate to the element type, and stacoll
234 * contains the collation to use with it.  stavalues contains
235 * the most common element values, and stanumbers their frequencies.  Unlike
236 * MCV slots, frequencies are measured as the fraction of non-null rows the
237 * element value appears in, not the frequency of all rows.  Also unlike
238 * MCV slots, the values are sorted into the element type's default order
239 * (to support binary search for a particular value).  Since this puts the
240 * minimum and maximum frequencies at unpredictable spots in stanumbers,
241 * there are two extra members of stanumbers, holding copies of the minimum
242 * and maximum frequencies.  Optionally, there can be a third extra member,
243 * which holds the frequency of null elements (expressed in the same terms:
244 * the fraction of non-null rows that contain at least one null element).  If
245 * this member is omitted, the column is presumed to contain no null elements.
246 *
247 * Starting in v19, the first extra member can be smaller than the smallest
248 * frequency of any stored MCE, indicating that it's known that no element
249 * not present in the MCE array has frequency greater than that value.
250 *
251 * Note: in current usage for tsvector columns, the stavalues elements are of
252 * type text, even though their representation within tsvector is not
253 * exactly text.
254 */
255#define STATISTIC_KIND_MCELEM  4
256
257/*
258 * A "distinct elements count histogram" slot describes the distribution of
259 * the number of distinct element values present in each row of an array-type
260 * column.  Only non-null rows are considered, and only non-null elements.
261 * staop contains the equality operator appropriate to the element type,
262 * and stacoll contains the collation to use with it.
263 * stavalues is not used and should be NULL.  The last member of stanumbers is
264 * the average count of distinct element values over all non-null rows.  The
265 * preceding M (>=2) members form a histogram that divides the population of
266 * distinct-elements counts into M-1 bins of approximately equal population.
267 * The first of these is the minimum observed count, and the last the maximum.
268 */
269#define STATISTIC_KIND_DECHIST	5
270
271/*
272 * A "length histogram" slot describes the distribution of range lengths in
273 * rows of a range-type column. stanumbers contains a single entry, the
274 * fraction of empty ranges. stavalues is a histogram of non-empty lengths, in
275 * a format similar to STATISTIC_KIND_HISTOGRAM: it contains M (>=2) range
276 * values that divide the column data values into M-1 bins of approximately
277 * equal population. The lengths are stored as float8s, as measured by the
278 * range type's subdiff function. Only non-null, non-empty rows are
279 * considered.
280 */
281#define STATISTIC_KIND_RANGE_LENGTH_HISTOGRAM  6
282
283/*
284 * A "bounds histogram" slot is similar to STATISTIC_KIND_HISTOGRAM, but for
285 * a range-type column.  stavalues contains M (>=2) range values that divide
286 * the column data values into M-1 bins of approximately equal population.
287 * Unlike a regular scalar histogram, this is actually two histograms combined
288 * into a single array, with the lower bounds of each value forming a
289 * histogram of lower bounds, and the upper bounds a histogram of upper
290 * bounds.  Only non-NULL, non-empty ranges are included.
291 */
292#define STATISTIC_KIND_BOUNDS_HISTOGRAM  7
