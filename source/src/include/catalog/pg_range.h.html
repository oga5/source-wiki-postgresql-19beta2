25 *		pg_range definition.  cpp turns this into
26 *		typedef struct FormData_pg_range
27 * ----------------

29BEGIN_CATALOG_STRUCT
30
31CATALOG(pg_range,3541,RangeRelationId)
32{
33	/* OID of owning range type */
34	Oid			rngtypid BKI_LOOKUP(pg_type);
35
36	/* OID of range's element type (subtype) */
37	Oid			rngsubtype BKI_LOOKUP(pg_type);
38
39	/* OID of the range's multirange type */
40	Oid			rngmultitypid BKI_LOOKUP(pg_type);
41
42	/* collation for this range type, or 0 */
43	Oid			rngcollation BKI_DEFAULT(0) BKI_LOOKUP_OPT(pg_collation);
44
45	/* subtype's btree opclass */
46	Oid			rngsubopc BKI_LOOKUP(pg_opclass);
47
48	/* range constructor functions */
49	regproc		rngconstruct2 BKI_LOOKUP(pg_proc);
50	regproc		rngconstruct3 BKI_LOOKUP(pg_proc);
51
52	/* multirange constructor functions */
53	regproc		rngmltconstruct0 BKI_LOOKUP(pg_proc);
54	regproc		rngmltconstruct1 BKI_LOOKUP(pg_proc);
55	regproc		rngmltconstruct2 BKI_LOOKUP(pg_proc);
56
57	/* canonicalize range, or 0 */
58	regproc		rngcanonical BKI_LOOKUP_OPT(pg_proc);
59
60	/* subtype difference as a float8, or 0 */
61	regproc		rngsubdiff BKI_LOOKUP_OPT(pg_proc);
62} FormData_pg_range;
63
64END_CATALOG_STRUCT
