182 * All plan nodes "derive" from the Plan structure by having the

567/* ----------------
568 *		index scan node
569 *
570 * indexqualorig is an implicitly-ANDed list of index qual expressions, each
571 * in the same form it appeared in the query WHERE condition.  Each should
572 * be of the form (indexkey OP comparisonval) or (comparisonval OP indexkey).
573 * The indexkey is a Var or expression referencing column(s) of the index's
574 * base table.  The comparisonval might be any expression, but it won't use
575 * any columns of the base table.  The expressions are ordered by index
576 * column position (but items referencing the same index column can appear
577 * in any order).  indexqualorig is used at runtime only if we have to recheck
578 * a lossy indexqual.
579 *
580 * indexqual has the same form, but the expressions have been commuted if
581 * necessary to put the indexkeys on the left, and the indexkeys are replaced
582 * by Var nodes identifying the index columns (their varno is INDEX_VAR and
583 * their varattno is the index column number).
584 *
585 * indexorderbyorig is similarly the original form of any ORDER BY expressions
586 * that are being implemented by the index, while indexorderby is modified to
587 * have index column Vars on the left-hand side.  Here, multiple expressions
588 * must appear in exactly the ORDER BY order, and this is not necessarily the
589 * index column order.  Only the expressions are provided, not the auxiliary
590 * sort-order information from the ORDER BY SortGroupClauses; it's assumed
591 * that the sort ordering is fully determinable from the top-level operators.
592 * indexorderbyorig is used at runtime to recheck the ordering, if the index
593 * cannot calculate an accurate ordering.  It is also needed for EXPLAIN.
594 *
595 * indexorderbyops is a list of the OIDs of the operators used to sort the
596 * ORDER BY expressions.  This is used together with indexorderbyorig to
597 * recheck ordering at run time.  (Note that indexorderby, indexorderbyorig,
598 * and indexorderbyops are used for amcanorderbyop cases, not amcanorder.)
599 *
600 * indexorderdir specifies the scan ordering, for indexscans on amcanorder
601 * indexes (for other indexes it should be "don't care").
602 * ----------------
603 */
604typedef struct IndexScan
605{
606	Scan		scan;
607	/* OID of index to scan */
608	Oid			indexid;
609	/* list of index quals (usually OpExprs) */
610	List	   *indexqual;
611	/* the same in original form */
612	List	   *indexqualorig;
613	/* list of index ORDER BY exprs */
614	List	   *indexorderby;
615	/* the same in original form */
616	List	   *indexorderbyorig;
617	/* OIDs of sort ops for ORDER BY exprs */
618	List	   *indexorderbyops;
619	/* forward or backward or don't care */
620	ScanDirection indexorderdir;
621} IndexScan;

623/* ----------------
624 *		index-only scan node
625 *
626 * IndexOnlyScan is very similar to IndexScan, but it specifies an
627 * index-only scan, in which the data comes from the index not the heap.
628 * Because of this, *all* Vars in the plan node's targetlist, qual, and
629 * index expressions reference index columns and have varno = INDEX_VAR.
630 *
631 * We could almost use indexqual directly against the index's output tuple
632 * when rechecking lossy index operators, but that won't work for quals on
633 * index columns that are not retrievable.  Hence, recheckqual is needed
634 * for rechecks: it expresses the same condition as indexqual, but using
635 * only index columns that are retrievable.  (We will not generate an
636 * index-only scan if this is not possible.  An example is that if an
637 * index has table column "x" in a retrievable index column "ind1", plus
638 * an expression f(x) in a non-retrievable column "ind2", an indexable
639 * query on f(x) will use "ind2" in indexqual and f(ind1) in recheckqual.
640 * Without the "ind1" column, an index-only scan would be disallowed.)
641 *
642 * We don't currently need a recheckable equivalent of indexorderby,
643 * because we don't support lossy operators in index ORDER BY.
644 *
645 * To help EXPLAIN interpret the index Vars for display, we provide
646 * indextlist, which represents the contents of the index as a targetlist
647 * with one TLE per index column.  Vars appearing in this list reference
648 * the base table, and this is the only field in the plan node that may
649 * contain such Vars.  Also, for the convenience of setrefs.c, TLEs in
650 * indextlist are marked as resjunk if they correspond to columns that
651 * the index AM cannot reconstruct.
652 * ----------------
653 */
654typedef struct IndexOnlyScan
655{
656	Scan		scan;
657	/* OID of index to scan */
658	Oid			indexid;
659	/* list of index quals (usually OpExprs) */
660	List	   *indexqual;
661	/* index quals in recheckable form */
662	List	   *recheckqual;
663	/* list of index ORDER BY exprs */
664	List	   *indexorderby;
665	/* TargetEntry list describing index's cols */
666	List	   *indextlist;
667	/* forward or backward or don't care */
668	ScanDirection indexorderdir;
669} IndexOnlyScan;

1525/*
1526 * RowMarkType -
1527 *	  enums for types of row-marking operations
1528 *
1529 * The first four of these values represent different lock strengths that
1530 * we can take on tuples according to SELECT FOR [KEY] UPDATE/SHARE requests.
1531 * We support these on regular tables, as well as on foreign tables whose FDWs
1532 * report support for late locking.  For other foreign tables, any locking
1533 * that might be done for such requests must happen during the initial row
1534 * fetch; their FDWs provide no mechanism for going back to lock a row later.
1535 * This means that the semantics will be a bit different than for a local
1536 * table; in particular we are likely to lock more rows than would be locked
1537 * locally, since remote rows will be locked even if they then fail
1538 * locally-checked restriction or join quals.  However, the prospect of
1539 * doing a separate remote query to lock each selected row is usually pretty
1540 * unappealing, so early locking remains a credible design choice for FDWs.
1541 *
1542 * When doing UPDATE/DELETE/MERGE/SELECT FOR UPDATE/SHARE, we have to uniquely
1543 * identify all the source rows, not only those from the target relations, so
1544 * that we can perform EvalPlanQual rechecking at need.  For plain tables we
1545 * can just fetch the TID, much as for a target relation; this case is
1546 * represented by ROW_MARK_REFERENCE.  Otherwise (for example for VALUES or
1547 * FUNCTION scans) we have to copy the whole row value.  ROW_MARK_COPY is
1548 * pretty inefficient, since most of the time we'll never need the data; but
1549 * fortunately the overhead is usually not performance-critical in practice.
1550 * By default we use ROW_MARK_COPY for foreign tables, but if the FDW has
1551 * a concept of rowid it can request to use ROW_MARK_REFERENCE instead.
1552 * (Again, this probably doesn't make sense if a physical remote fetch is
1553 * needed, but for FDWs that map to local storage it might be credible.)
1554 */
1555typedef enum RowMarkType
1556{
1557	ROW_MARK_EXCLUSIVE,			/* obtain exclusive tuple lock */
1558	ROW_MARK_NOKEYEXCLUSIVE,	/* obtain no-key exclusive tuple lock */
1559	ROW_MARK_SHARE,				/* obtain shared tuple lock */
1560	ROW_MARK_KEYSHARE,			/* obtain keyshare tuple lock */
1561	ROW_MARK_REFERENCE,			/* just fetch the TID, don't lock it */
1562	ROW_MARK_COPY,				/* physically copy the row value */
1563} RowMarkType;
1564
1565#define RowMarkRequiresRowShareLock(marktype)  ((marktype) <= ROW_MARK_KEYSHARE)

1567/*
1568 * PlanRowMark -
1569 *	   plan-time representation of FOR [KEY] UPDATE/SHARE clauses
1570 *
1571 * When doing UPDATE/DELETE/MERGE/SELECT FOR UPDATE/SHARE, we create a separate
1572 * PlanRowMark node for each non-target relation in the query.  Relations that
1573 * are not specified as FOR UPDATE/SHARE are marked ROW_MARK_REFERENCE (if
1574 * regular tables or supported foreign tables) or ROW_MARK_COPY (if not).
1575 *
1576 * Initially all PlanRowMarks have rti == prti and isParent == false.
1577 * When the planner discovers that a relation is the root of an inheritance
1578 * tree, it sets isParent true, and adds an additional PlanRowMark to the
1579 * list for each child relation (including the target rel itself in its role
1580 * as a child, if it is not a partitioned table).  Any non-leaf partitioned
1581 * child relations will also have entries with isParent = true.  The child
1582 * entries have rti == child rel's RT index and prti == top parent's RT index,
1583 * and can therefore be recognized as children by the fact that prti != rti.
1584 * The parent's allMarkTypes field gets the OR of (1<<markType) across all
1585 * its children (this definition allows children to use different markTypes).
1586 *
1587 * The planner also adds resjunk output columns to the plan that carry
1588 * information sufficient to identify the locked or fetched rows.  When
1589 * markType != ROW_MARK_COPY, these columns are named
1590 *		tableoid%u			OID of table
1591 *		ctid%u				TID of row
1592 * The tableoid column is only present for an inheritance hierarchy.
1593 * When markType == ROW_MARK_COPY, there is instead a single column named
1594 *		wholerow%u			whole-row value of relation
1595 * (An inheritance hierarchy could have all three resjunk output columns,
1596 * if some children use a different markType than others.)
1597 * In all three cases, %u represents the rowmark ID number (rowmarkId).
1598 * This number is unique within a plan tree, except that child relation
1599 * entries copy their parent's rowmarkId.  (Assigning unique numbers
1600 * means we needn't renumber rowmarkIds when flattening subqueries, which
1601 * would require finding and renaming the resjunk columns as well.)
1602 * Note this means that all tables in an inheritance hierarchy share the
1603 * same resjunk column names.
1604 */
1605typedef struct PlanRowMark
1606{
1607	pg_node_attr(no_equal, no_query_jumble)
1608
1609	NodeTag		type;
1610	/* range table index of markable relation */
1611	Index		rti;
1612	/* range table index of parent relation */
1613	Index		prti;
1614	/* unique identifier for resjunk columns */
1615	Index		rowmarkId;
1616	/* see enum above */
1617	RowMarkType markType;
1618	/* OR of (1<<markType) for all children */
1619	int			allMarkTypes;
1620	/* LockingClause's strength, or LCS_NONE */
1621	LockClauseStrength strength;
1622	/* NOWAIT and SKIP LOCKED options */
1623	LockWaitPolicy waitPolicy;
1624	/* true if this is a "dummy" parent entry */
1625	bool		isParent;
1626} PlanRowMark;
