24/*
25 * Report a subscription error.
26 */
27void
28pgstat_report_subscription_error(Oid subid)
29{
30	PgStat_EntryRef *entry_ref;
31	PgStat_BackendSubEntry *pending;
32	LogicalRepWorkerType wtype = get_logical_worker_type();
33
34	entry_ref = pgstat_prep_pending_entry(PGSTAT_KIND_SUBSCRIPTION,
35										  InvalidOid, subid, NULL);
36	pending = entry_ref->pending;
37
38	switch (wtype)
39	{
40		case WORKERTYPE_APPLY:
41			pending->apply_error_count++;
42			break;
43
44		case WORKERTYPE_SEQUENCESYNC:
45			pending->sync_seq_error_count++;
46			break;
47
48		case WORKERTYPE_TABLESYNC:
49			pending->sync_table_error_count++;
50			break;
51
52		default:
53			/* Should never happen. */
54			Assert(0);
55			break;
56	}
57}

59/*
60 * Report a subscription conflict.
61 */
62void
63pgstat_report_subscription_conflict(Oid subid, ConflictType type)
64{
65	PgStat_EntryRef *entry_ref;
66	PgStat_BackendSubEntry *pending;
67
68	entry_ref = pgstat_prep_pending_entry(PGSTAT_KIND_SUBSCRIPTION,
69										  InvalidOid, subid, NULL);
70	pending = entry_ref->pending;
71	pending->conflict_count[type]++;
72}

74/*
75 * Report creating the subscription.
76 */
77void
78pgstat_create_subscription(Oid subid)
79{
80	/* Ensures that stats are dropped if transaction rolls back */
81	pgstat_create_transactional(PGSTAT_KIND_SUBSCRIPTION,
82								InvalidOid, subid);
83
84	/* Create and initialize the subscription stats entry */
85	pgstat_get_entry_ref(PGSTAT_KIND_SUBSCRIPTION, InvalidOid, subid,
86						 true, NULL);
87	pgstat_reset_entry(PGSTAT_KIND_SUBSCRIPTION, InvalidOid, subid, 0);
88}
89
90/*
91 * Report dropping the subscription.
92 *
93 * Ensures that stats are dropped if transaction commits.
94 */
95void
96pgstat_drop_subscription(Oid subid)
97{
98	pgstat_drop_transactional(PGSTAT_KIND_SUBSCRIPTION,
99							  InvalidOid, subid);
100}

102/*
103 * Support function for the SQL-callable pgstat* functions. Returns
104 * the collected statistics for one subscription or NULL.
105 */
106PgStat_StatSubEntry *
107pgstat_fetch_stat_subscription(Oid subid)
108{
109	return (PgStat_StatSubEntry *)
110		pgstat_fetch_entry(PGSTAT_KIND_SUBSCRIPTION, InvalidOid, subid, NULL);
111}

113/*
114 * Flush out pending stats for the entry
115 *
116 * If nowait is true and the lock could not be immediately acquired, returns
117 * false without flushing the entry.  Otherwise returns true.
118 */
119bool
120pgstat_subscription_flush_cb(PgStat_EntryRef *entry_ref, bool nowait)
121{
122	PgStat_BackendSubEntry *localent;
123	PgStatShared_Subscription *shsubent;
124
125	localent = (PgStat_BackendSubEntry *) entry_ref->pending;
126	shsubent = (PgStatShared_Subscription *) entry_ref->shared_stats;
127
128	/* localent always has non-zero content */
129
130	if (!pgstat_lock_entry(entry_ref, nowait))
131		return false;
132
133#define SUB_ACC(fld) shsubent->stats.fld += localent->fld
134	SUB_ACC(apply_error_count);
135	SUB_ACC(sync_seq_error_count);
136	SUB_ACC(sync_table_error_count);
137	for (int i = 0; i < CONFLICT_NUM_TYPES; i++)
138		SUB_ACC(conflict_count[i]);
139#undef SUB_ACC
140
141	pgstat_unlock_entry(entry_ref);
142	return true;
143}
