22static PgStat_PendingLock PendingLockStats;
23static bool have_lockstats = false;
24
25PgStat_Lock *
26pgstat_fetch_stat_lock(void)
27{
28	pgstat_snapshot_fixed(PGSTAT_KIND_LOCK);
29
30	return &pgStatLocal.snapshot.lock;
31}

33/*
34 * Simpler wrapper of pgstat_lock_flush_cb()
35 */
36void
37pgstat_lock_flush(bool nowait)
38{
39	(void) pgstat_lock_flush_cb(nowait);
40}
41
42/*
43 * Flush out locally pending lock statistics
44 *
45 * If no stats have been recorded, this function returns false.
46 *
47 * If nowait is true, this function returns true if the lock could not be
48 * acquired. Otherwise, return false.
49 */
50bool
51pgstat_lock_flush_cb(bool nowait)
52{
53	LWLock	   *lckstat_lock;
54	PgStatShared_Lock *shstats;
55
56	if (!have_lockstats)
57		return false;
58
59	shstats = &pgStatLocal.shmem->lock;
60	lckstat_lock = &shstats->lock;
61
62	if (!nowait)
63		LWLockAcquire(lckstat_lock, LW_EXCLUSIVE);
64	else if (!LWLockConditionalAcquire(lckstat_lock, LW_EXCLUSIVE))
65		return true;
66
67	for (int i = 0; i <= LOCKTAG_LAST_TYPE; i++)
68	{
69#define LOCKSTAT_ACC(fld) \
70	(shstats->stats.stats[i].fld += PendingLockStats.stats[i].fld)
71		LOCKSTAT_ACC(waits);
72		LOCKSTAT_ACC(wait_time);
73		LOCKSTAT_ACC(fastpath_exceeded);
74#undef LOCKSTAT_ACC
75	}
76
77	LWLockRelease(lckstat_lock);
78
79	memset(&PendingLockStats, 0, sizeof(PendingLockStats));
80	have_lockstats = false;
81
82	return false;
83}

85void
86pgstat_lock_init_shmem_cb(void *stats)
87{
88	PgStatShared_Lock *stat_shmem = (PgStatShared_Lock *) stats;
89
90	LWLockInitialize(&stat_shmem->lock, LWTRANCHE_PGSTATS_DATA);
91}
92
93void
94pgstat_lock_reset_all_cb(TimestampTz ts)
95{
96	LWLock	   *lckstat_lock = &pgStatLocal.shmem->lock.lock;
97
98	LWLockAcquire(lckstat_lock, LW_EXCLUSIVE);
99
100	pgStatLocal.shmem->lock.stats.stat_reset_timestamp = ts;
101
102	memset(pgStatLocal.shmem->lock.stats.stats, 0,
103		   sizeof(pgStatLocal.shmem->lock.stats.stats));
104
105	LWLockRelease(lckstat_lock);
106}
107
108void
109pgstat_lock_snapshot_cb(void)
110{
111	LWLock	   *lckstat_lock = &pgStatLocal.shmem->lock.lock;
112
113	LWLockAcquire(lckstat_lock, LW_SHARED);
114
115	pgStatLocal.snapshot.lock = pgStatLocal.shmem->lock.stats;
116
117	LWLockRelease(lckstat_lock);
118}

120/*
121 * Increment counter for lock not acquired with the fast-path, per lock
122 * type, due to the fast-path slot limit reached.
123 *
124 * Note: This function should not be called in performance-sensitive paths,
125 * like lock acquisitions.
126 */
127void
128pgstat_count_lock_fastpath_exceeded(uint8 locktag_type)
129{
130	Assert(locktag_type <= LOCKTAG_LAST_TYPE);
131	PendingLockStats.stats[locktag_type].fastpath_exceeded++;
132	have_lockstats = true;
133	pgstat_report_fixed = true;
134}
135
136/*
137 * Increment the number of waits and wait time, per lock type.
138 *
139 * Note: This function should not be called in performance-sensitive paths,
140 * like lock acquisitions.
141 */
142void
143pgstat_count_lock_waits(uint8 locktag_type, PgStat_Counter usecs)
144{
145	Assert(locktag_type <= LOCKTAG_LAST_TYPE);
146	PendingLockStats.stats[locktag_type].waits++;
147	PendingLockStats.stats[locktag_type].wait_time += usecs;
148	have_lockstats = true;
149	pgstat_report_fixed = true;
150}
