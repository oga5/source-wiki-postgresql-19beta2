13use strict;
14use warnings FATAL => 'all';
15
16use FindBin;
17use lib "$FindBin::RealBin/../../catalog";
18use Catalog;
19
20die "Usage: $0 INPUT_FILE OUTPUT_FILE\n" unless @ARGV == 2;
21my ($input_fname, $output_fname) = @ARGV;
22
23my $parse = Catalog::ParseData($input_fname);
24
25open my $ofh, '>', $output_fname or die;
26
27print_boilerplate($ofh, $output_fname, 'GUC tables');
28print_table($ofh);
29
30close $ofh;

33# Adds double quotes and escapes as necessary for C strings.
34sub dquote
35{
36	my ($s) = @_;
37
38	return q{"} . $s =~ s/"/\\"/gr . q{"};
39}
40
41sub validate_guc_entry
42{
43	my ($entry) = @_;
44
45	my @required_common =
46	  qw(name type context group short_desc variable boot_val);
47
48	my %required_by_type = (
49		int => [qw(min max)],
50		real => [qw(min max)],
51		enum => [qw(options)],
52		bool => [],      # no extra required fields
53		string => [],    # no extra required fields
54	);
55
56	# All fields recognized by the generator.  "line_number" is injected
57	# by Catalog::ParseData and is not a user-facing field.
58	my %valid_fields = map { $_ => 1 } (
59		@required_common,
60		qw(long_desc flags ifdef min max options
61		  check_hook assign_hook show_hook
62		  line_number));
63
64	for my $f (sort keys %$entry)
65	{
66		unless ($valid_fields{$f})
67		{
68			die sprintf(
69				qq{%s:%d: error: entry "%s" has unrecognized field "%s"\n},
70				$input_fname, $entry->{line_number},
71				$entry->{name} // '<unknown>', $f);
72		}
73	}
74
75	for my $f (@required_common)
76	{
77		unless (defined $entry->{$f})
78		{
79			die sprintf(
80				qq{%s:%d: error: entry "%s" is missing required field "%s"\n},
81				$input_fname, $entry->{line_number},
82				$entry->{name} // '<unknown>', $f);
83		}
84	}
85
86	unless (exists $required_by_type{ $entry->{type} })
87	{
88		die sprintf(
89			qq{%s:%d: error: entry "%s" has unrecognized GUC type "%s"\n},
90			$input_fname, $entry->{line_number},
91			$entry->{name}, $entry->{type} // '<unknown>');
92	}
93
94	for my $f (@{ $required_by_type{ $entry->{type} } })
95	{
96		unless (defined $entry->{$f})
97		{
98			die sprintf(
99				qq{%s:%d: error: entry "%s" of type "%s" is missing required field "%s"\n},
100				$input_fname, $entry->{line_number}, $entry->{name},
101				$entry->{type}, $f);
102		}
103	}
104}

106# Print GUC table.
107sub print_table
108{
109	my ($ofh) = @_;
110	my $prev_name = undef;
111
112	print $ofh "\n\n";
113	print $ofh "struct config_generic ConfigureNames[] =\n";
114	print $ofh "{\n";
115
116	foreach my $entry (@{$parse})
117	{
118		validate_guc_entry($entry);
119
120		if (defined($prev_name) && lc($prev_name) eq lc($entry->{name}))
121		{
122			die sprintf(qq{%s:%d: error: duplicate entry "%s"\n},
123				$input_fname, $entry->{line_number}, $entry->{name});
124		}
125		if (defined($prev_name) && lc($prev_name) gt lc($entry->{name}))
126		{
127			die sprintf(
128				qq{%s:%d: error: entries are not in alphabetical order: "%s", "%s"\n},
129				$input_fname, $entry->{line_number},
130				$prev_name, $entry->{name});
131		}
132
133		print $ofh "#ifdef $entry->{ifdef}\n" if $entry->{ifdef};
134		print $ofh "\t{\n";
135		printf $ofh "\t\t.name = %s,\n", dquote($entry->{name});
136		printf $ofh "\t\t.context = %s,\n", $entry->{context};
137		printf $ofh "\t\t.group = %s,\n", $entry->{group};
138		printf $ofh
139		  "\t\t/* translator: GUC parameter \"%s\" short description */\n",
140		  $entry->{name};
141		printf $ofh "\t\t.short_desc = gettext_noop(%s),\n",
142		  dquote($entry->{short_desc});
143
144		if ($entry->{long_desc})
145		{
146			printf $ofh
147			  "\t\t/* translator: GUC parameter \"%s\" long description */\n",
148			  $entry->{name};
149			printf $ofh "\t\t.long_desc = gettext_noop(%s),\n",
150			  dquote($entry->{long_desc});
151		}
152		printf $ofh "\t\t.flags = %s,\n", $entry->{flags} if $entry->{flags};
153		printf $ofh "\t\t.vartype = %s,\n", ('PGC_' . uc($entry->{type}));
154		printf $ofh "\t\t._%s = {\n", $entry->{type};
155		printf $ofh "\t\t\t.variable = &%s,\n", $entry->{variable};
156		printf $ofh "\t\t\t.boot_val = %s,\n", $entry->{boot_val};
157		printf $ofh "\t\t\t.min = %s,\n", $entry->{min}
158		  if $entry->{type} eq 'int' || $entry->{type} eq 'real';
159		printf $ofh "\t\t\t.max = %s,\n", $entry->{max}
160		  if $entry->{type} eq 'int' || $entry->{type} eq 'real';
161		printf $ofh "\t\t\t.options = %s,\n", $entry->{options}
162		  if $entry->{type} eq 'enum';
163		printf $ofh "\t\t\t.check_hook = %s,\n", $entry->{check_hook}
164		  if $entry->{check_hook};
165		printf $ofh "\t\t\t.assign_hook = %s,\n", $entry->{assign_hook}
166		  if $entry->{assign_hook};
167		printf $ofh "\t\t\t.show_hook = %s,\n", $entry->{show_hook}
168		  if $entry->{show_hook};
169		print $ofh "\t\t},\n";
170		print $ofh "\t},\n";
171		print $ofh "#endif\n" if $entry->{ifdef};
172		print $ofh "\n";
173
174		$prev_name = $entry->{name};
175	}
176
177	print $ofh "\t/* End-of-list marker */\n";
178	print $ofh "\t{0}\n";
179	print $ofh "};\n";
180
181	return;
182}

184sub print_boilerplate
185{
186	my ($fh, $fname, $descr) = @_;
187	printf $fh <<EOM, $fname, $descr;
188/*-------------------------------------------------------------------------
189 *
190 * %s
191 *    %s
192 *
193 * Portions Copyright (c) 1996-2026, PostgreSQL Global Development Group
194 * Portions Copyright (c) 1994, Regents of the University of California
195 *
196 * NOTES
197 *  ******************************
198 *  *** DO NOT EDIT THIS FILE! ***
199 *  ******************************
200 *
201 *  It has been GENERATED by src/backend/utils/misc/gen_guc_tables.pl
202 *
203 *-------------------------------------------------------------------------
204 */
205EOM
206
207	return;
208}
