1/*-------------------------------------------------------------------------
2 * sequencesync.c
3 *	  PostgreSQL logical replication: sequence synchronization
4 *
5 * Copyright (c) 2025-2026, PostgreSQL Global Development Group
6 *
7 * IDENTIFICATION
8 *	  src/backend/replication/logical/sequencesync.c
9 *
10 * NOTES
11 *	  This file contains code for sequence synchronization for
12 *	  logical replication.
13 *
14 * Sequences requiring synchronization are tracked in the pg_subscription_rel
15 * catalog.
16 *
17 * Sequences to be synchronized will be added with state INIT when either of
18 * the following commands is executed:
19 * CREATE SUBSCRIPTION
20 * ALTER SUBSCRIPTION ... REFRESH PUBLICATION
21 *
22 * Executing the following command resets all sequences in the subscription to
23 * state INIT, triggering re-synchronization:
24 * ALTER SUBSCRIPTION ... REFRESH SEQUENCES
25 *
26 * The apply worker periodically scans pg_subscription_rel for sequences in
27 * INIT state. When such sequences are found, it spawns a sequencesync worker
28 * to handle synchronization.
29 *
30 * A single sequencesync worker is responsible for synchronizing all sequences.
31 * It begins by retrieving the list of sequences that are flagged for
32 * synchronization, i.e., those in the INIT state. These sequences are then
33 * processed in batches, allowing multiple entries to be synchronized within a
34 * single transaction. The worker fetches the current sequence values and page
35 * LSNs from the remote publisher, updates the corresponding sequences on the
36 * local subscriber, and finally marks each sequence as READY upon successful
37 * synchronization.
38 *
39 * Sequence state transitions follow this pattern:
40 *   INIT -> READY
41 *
42 * To avoid creating too many transactions, up to MAX_SEQUENCES_SYNC_PER_BATCH
43 * sequences are synchronized per transaction. The locks on the sequence
44 * relation will be periodically released at each transaction commit.
45 *
46 * XXX: We didn't choose launcher process to maintain the launch of sequencesync
47 * worker as it didn't have database connection to access the sequences from the
48 * pg_subscription_rel system catalog that need to be synchronized.
49 *-------------------------------------------------------------------------
50 */

75#define REMOTE_SEQ_COL_COUNT 11

77typedef enum CopySeqResult
78{
79	COPYSEQ_SUCCESS,
80	COPYSEQ_MISMATCH,
81	COPYSEQ_SUBSCRIBER_INSUFFICIENT_PERM,
82	COPYSEQ_PUBLISHER_INSUFFICIENT_PERM,
83	COPYSEQ_SKIPPED
84} CopySeqResult;
85
86static List *seqinfos = NIL;
87
88/*
89 * Apply worker determines if sequence synchronization is needed.
90 *
91 * Start a sequencesync worker if one is not already running. The active
92 * sequencesync worker will handle all pending sequence synchronization. If any
93 * sequences remain unsynchronized after it exits, a new worker can be started
94 * in the next iteration.
95 */
96void
97ProcessSequencesForSync(void)
98{
99	LogicalRepWorker *sequencesync_worker;
100	int			nsyncworkers;
101	bool		has_pending_sequences;
102	bool		started_tx;
103
104	FetchRelationStates(NULL, &has_pending_sequences, &started_tx);
105
106	if (started_tx)
107	{
108		CommitTransactionCommand();
109		pgstat_report_stat(true);
110	}
111
112	if (!has_pending_sequences)
113		return;
114
115	LWLockAcquire(LogicalRepWorkerLock, LW_SHARED);
116
117	/* Check if there is a sequencesync worker already running? */
118	sequencesync_worker = logicalrep_worker_find(WORKERTYPE_SEQUENCESYNC,
119												 MyLogicalRepWorker->subid,
120												 InvalidOid, true);
121	if (sequencesync_worker)
122	{
123		LWLockRelease(LogicalRepWorkerLock);
124		return;
125	}
126
127	/*
128	 * Count running sync workers for this subscription, while we have the
129	 * lock.
130	 */
131	nsyncworkers = logicalrep_sync_worker_count(MyLogicalRepWorker->subid);
132	LWLockRelease(LogicalRepWorkerLock);
133
134	/*
135	 * It is okay to read/update last_seqsync_start_time here in apply worker
136	 * as we have already ensured that sync worker doesn't exist.
137	 */
138	launch_sync_worker(WORKERTYPE_SEQUENCESYNC, nsyncworkers, InvalidOid,
139					   &MyLogicalRepWorker->last_seqsync_start_time);
140}

142/*
143 * get_sequences_string
144 *
145 * Build a comma-separated string of schema-qualified sequence names
146 * for the given list of sequence indexes.
147 */
148static void
149get_sequences_string(List *seqindexes, StringInfo buf)
150{
151	resetStringInfo(buf);
152	foreach_int(seqidx, seqindexes)
153	{
154		LogicalRepSequenceInfo *seqinfo =
155			(LogicalRepSequenceInfo *) list_nth(seqinfos, seqidx);
156
157		if (buf->len > 0)
158			appendStringInfoString(buf, ", ");
159
160		appendStringInfo(buf, "\"%s.%s\"", seqinfo->nspname, seqinfo->seqname);
161	}
162}

164/*
165 * report_sequence_errors
166 *
167 * Report discrepancies found during sequence synchronization between
168 * the publisher and subscriber. Emits warnings for:
169 * a) mismatched definitions or concurrent rename
170 * b) insufficient privileges on the subscriber
171 * c) insufficient privileges on the publisher
172 * d) missing sequences on the publisher
173 * Then raises an ERROR to indicate synchronization failure.
174 */
175static void
176report_sequence_errors(List *mismatched_seqs_idx,
177					   List *sub_insuffperm_seqs_idx,
178					   List *pub_insuffperm_seqs_idx,
179					   List *missing_seqs_idx)
180{
181	StringInfoData seqstr;
182
183	/* Quick exit if there are no errors to report */
184	if (!mismatched_seqs_idx && !sub_insuffperm_seqs_idx &&
185		!pub_insuffperm_seqs_idx && !missing_seqs_idx)
186		return;
187
188	initStringInfo(&seqstr);
189
190	if (mismatched_seqs_idx)
191	{
192		get_sequences_string(mismatched_seqs_idx, &seqstr);
193		ereport(WARNING,
194				errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
195				errmsg_plural("mismatched or renamed sequence on subscriber (%s)",
196							  "mismatched or renamed sequences on subscriber (%s)",
197							  list_length(mismatched_seqs_idx),
198							  seqstr.data));
199	}
200
201	if (sub_insuffperm_seqs_idx)
202	{
203		get_sequences_string(sub_insuffperm_seqs_idx, &seqstr);
204
205		/*
206		 * With run_as_owner enabled, sequence synchronization runs as the
207		 * subscription owner, so a missing UPDATE privilege should be granted
208		 * to that role. Otherwise, the worker switches to the sequence owner
209		 * before checking privileges, so no useful GRANT hint can be
210		 * provided.
211		 */
212		ereport(WARNING,
213				errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
214				errmsg_plural("insufficient privileges on subscriber sequence (%s)",
215							  "insufficient privileges on subscriber sequences (%s)",
216							  list_length(sub_insuffperm_seqs_idx),
217							  seqstr.data),
218				MySubscription->runasowner ?
219				errhint_plural("Grant UPDATE on the sequence to the subscription "
220							   "owner on the subscriber.",
221							   "Grant UPDATE on the sequences to the subscription "
222							   "owner on the subscriber.",
223							   list_length(sub_insuffperm_seqs_idx)) : 0);
224	}
225
226	if (pub_insuffperm_seqs_idx)
227	{
228		get_sequences_string(pub_insuffperm_seqs_idx, &seqstr);
229		ereport(WARNING,
230				errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
231				errmsg_plural("insufficient privileges on publisher sequence (%s)",
232							  "insufficient privileges on publisher sequences (%s)",
233							  list_length(pub_insuffperm_seqs_idx),
234							  seqstr.data),
235				errhint_plural("Grant SELECT on the sequence to the role used for "
236							   "the replication connection on the publisher.",
237							   "Grant SELECT on the sequences to the role used for "
238							   "the replication connection on the publisher.",
239							   list_length(pub_insuffperm_seqs_idx)));
240	}
241
242	if (missing_seqs_idx)
243	{
244		get_sequences_string(missing_seqs_idx, &seqstr);
245		ereport(WARNING,
246				errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
247				errmsg_plural("missing sequence on publisher (%s)",
248							  "missing sequences on publisher (%s)",
249							  list_length(missing_seqs_idx),
250							  seqstr.data));
251	}
252
253	ereport(ERROR,
254			errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
255			errmsg("logical replication sequence synchronization failed for subscription \"%s\"",
256				   MySubscription->name));
257}

259/*
260 * get_and_validate_seq_info
261 *
262 * Extracts remote sequence information from the tuple slot received from the
263 * publisher, and validates it against the corresponding local sequence
264 * definition.
265 */
266static CopySeqResult
267get_and_validate_seq_info(TupleTableSlot *slot, Relation *sequence_rel,
268						  LogicalRepSequenceInfo **seqinfo, int *seqidx)
269{
270	bool		isnull;
271	int			col = 0;
272	Datum		datum;
273	bool		remote_has_select_priv;
274	Oid			remote_typid;
275	int64		remote_start;
276	int64		remote_increment;
277	int64		remote_min;
278	int64		remote_max;
279	bool		remote_cycle;
280	CopySeqResult result = COPYSEQ_SUCCESS;
281	HeapTuple	tup;
282	Form_pg_sequence local_seq;
283	LogicalRepSequenceInfo *seqinfo_local;
284
285	*seqidx = DatumGetInt32(slot_getattr(slot, ++col, &isnull));
286	Assert(!isnull);
287
288	/* Identify the corresponding local sequence for the given index. */
289	*seqinfo = seqinfo_local =
290		(LogicalRepSequenceInfo *) list_nth(seqinfos, *seqidx);
291
292	/*
293	 * The remote sequence state can be NULL if the publisher lacks the
294	 * required privileges or if the sequence was dropped concurrently after
295	 * it was identified in the catalog snapshot (see pg_get_sequence_data()).
296	 */
297	remote_has_select_priv = DatumGetBool(slot_getattr(slot, ++col, &isnull));
298	Assert(!isnull);
299
300	datum = slot_getattr(slot, ++col, &isnull);
301	if (isnull)
302		return remote_has_select_priv ? COPYSEQ_SKIPPED :
303			COPYSEQ_PUBLISHER_INSUFFICIENT_PERM;
304
305	seqinfo_local->last_value = DatumGetInt64(datum);
306
307	seqinfo_local->is_called = DatumGetBool(slot_getattr(slot, ++col, &isnull));
308	Assert(!isnull);
309
310	seqinfo_local->page_lsn = DatumGetLSN(slot_getattr(slot, ++col, &isnull));
311	Assert(!isnull);
312
313	remote_typid = DatumGetObjectId(slot_getattr(slot, ++col, &isnull));
314	Assert(!isnull);
315
316	remote_start = DatumGetInt64(slot_getattr(slot, ++col, &isnull));
317	Assert(!isnull);
318
319	remote_increment = DatumGetInt64(slot_getattr(slot, ++col, &isnull));
320	Assert(!isnull);
321
322	remote_min = DatumGetInt64(slot_getattr(slot, ++col, &isnull));
323	Assert(!isnull);
324
325	remote_max = DatumGetInt64(slot_getattr(slot, ++col, &isnull));
326	Assert(!isnull);
327
328	remote_cycle = DatumGetBool(slot_getattr(slot, ++col, &isnull));
329	Assert(!isnull);
330
331	/* Sanity check */
332	Assert(col == REMOTE_SEQ_COL_COUNT);
333
334	seqinfo_local->found_on_pub = true;
335
336	*sequence_rel = try_table_open(seqinfo_local->localrelid, RowExclusiveLock);
337
338	/* Sequence was concurrently dropped? */
339	if (!*sequence_rel)
340		return COPYSEQ_SKIPPED;
341
342	tup = SearchSysCache1(SEQRELID, ObjectIdGetDatum(seqinfo_local->localrelid));
343
344	/* Sequence was concurrently dropped? */
345	if (!HeapTupleIsValid(tup))
346		elog(ERROR, "cache lookup failed for sequence %u",
347			 seqinfo_local->localrelid);
348
349	local_seq = (Form_pg_sequence) GETSTRUCT(tup);
350
351	/* Sequence parameters for remote/local are the same? */
352	if (local_seq->seqtypid != remote_typid ||
353		local_seq->seqstart != remote_start ||
354		local_seq->seqincrement != remote_increment ||
355		local_seq->seqmin != remote_min ||
356		local_seq->seqmax != remote_max ||
357		local_seq->seqcycle != remote_cycle)
358		result = COPYSEQ_MISMATCH;
359
360	/* Sequence was concurrently renamed? */
361	if (strcmp(seqinfo_local->nspname,
362			   get_namespace_name(RelationGetNamespace(*sequence_rel))) ||
363		strcmp(seqinfo_local->seqname, RelationGetRelationName(*sequence_rel)))
364		result = COPYSEQ_MISMATCH;
365
366	ReleaseSysCache(tup);
367	return result;
368}

370/*
371 * Apply remote sequence state to local sequence and mark it as
372 * synchronized (READY).
373 */
374static CopySeqResult
375copy_sequence(LogicalRepSequenceInfo *seqinfo, Oid seqowner)
376{
377	UserContext ucxt;
378	AclResult	aclresult;
379	bool		run_as_owner = MySubscription->runasowner;
380	Oid			seqoid = seqinfo->localrelid;
381
382	/*
383	 * If the user did not opt to run as the owner of the subscription
384	 * ('run_as_owner'), then copy the sequence as the owner of the sequence.
385	 */
386	if (!run_as_owner)
387		SwitchToUntrustedUser(seqowner, &ucxt);
388
389	aclresult = pg_class_aclcheck(seqoid, GetUserId(), ACL_UPDATE);
390
391	if (aclresult != ACLCHECK_OK)
392	{
393		if (!run_as_owner)
394			RestoreUserContext(&ucxt);
395
396		return COPYSEQ_SUBSCRIBER_INSUFFICIENT_PERM;
397	}
398
399	/*
400	 * The log counter (log_cnt) tracks how many sequence values are still
401	 * unused locally. It is only relevant to the local node and managed
402	 * internally by nextval() when allocating new ranges. Since log_cnt does
403	 * not affect the visible sequence state (like last_value or is_called)
404	 * and is only used for local caching, it need not be copied to the
405	 * subscriber during synchronization.
406	 */
407	SetSequence(seqoid, seqinfo->last_value, seqinfo->is_called);
408
409	if (!run_as_owner)
410		RestoreUserContext(&ucxt);
411
412	/*
413	 * Record the remote sequence's LSN in pg_subscription_rel and mark the
414	 * sequence as READY.
415	 */
416	UpdateSubscriptionRelState(MySubscription->oid, seqoid, SUBREL_STATE_READY,
417							   seqinfo->page_lsn, false);
418
419	return COPYSEQ_SUCCESS;
420}

422/*
423 * Copy existing data of sequences from the publisher.
424 */
425static void
426copy_sequences(WalReceiverConn *conn)
427{
428	int			cur_batch_base_index = 0;
429	int			n_seqinfos = list_length(seqinfos);
430	List	   *mismatched_seqs_idx = NIL;
431	List	   *missing_seqs_idx = NIL;
432	List	   *sub_insuffperm_seqs_idx = NIL;
433	List	   *pub_insuffperm_seqs_idx = NIL;
434	StringInfoData seqstr;
435	StringInfoData cmd;
436	MemoryContext oldctx;
437
438	initStringInfo(&seqstr);
439	initStringInfo(&cmd);
440
441#define MAX_SEQUENCES_SYNC_PER_BATCH 100
442
443	elog(DEBUG1,
444		 "logical replication sequence synchronization for subscription \"%s\" - total unsynchronized: %d",
445		 MySubscription->name, n_seqinfos);
446
447	while (cur_batch_base_index < n_seqinfos)
448	{
449		Oid			seqRow[REMOTE_SEQ_COL_COUNT] = {INT8OID, BOOLOID, INT8OID,
450		BOOLOID, LSNOID, OIDOID, INT8OID, INT8OID, INT8OID, INT8OID, BOOLOID};
451		int			batch_size = 0;
452		int			batch_succeeded_count = 0;
453		int			batch_mismatched_count = 0;
454		int			batch_skipped_count = 0;
455		int			batch_sub_insuffperm_count = 0;
456		int			batch_pub_insuffperm_count = 0;
457		int			batch_missing_count;
458
459		WalRcvExecResult *res;
460		TupleTableSlot *slot;
461
462		StartTransactionCommand();
463
464		for (int idx = cur_batch_base_index; idx < n_seqinfos; idx++)
465		{
466			char	   *nspname_literal;
467			char	   *seqname_literal;
468
469			LogicalRepSequenceInfo *seqinfo =
470				(LogicalRepSequenceInfo *) list_nth(seqinfos, idx);
471
472			if (seqstr.len > 0)
473				appendStringInfoString(&seqstr, ", ");
474
475			nspname_literal = quote_literal_cstr(seqinfo->nspname);
476			seqname_literal = quote_literal_cstr(seqinfo->seqname);
477
478			appendStringInfo(&seqstr, "(%s, %s, %d)",
479							 nspname_literal, seqname_literal, idx);
480
481			if (++batch_size == MAX_SEQUENCES_SYNC_PER_BATCH)
482				break;
483		}
484
485		/*
486		 * We deliberately avoid acquiring a local lock on the sequence before
487		 * querying the publisher to prevent potential distributed deadlocks
488		 * in bi-directional replication setups.
489		 *
490		 * Example scenario:
491		 *
492		 * - On each node, a background worker acquires a lock on a sequence
493		 * as part of a sync operation.
494		 *
495		 * - Concurrently, a user transaction attempts to alter the same
496		 * sequence, waiting on the background worker's lock.
497		 *
498		 * - Meanwhile, a query from the other node tries to access metadata
499		 * that depends on the completion of the alter operation.
500		 *
501		 * - This creates a circular wait across nodes:
502		 *
503		 * Node-1: Query -> waits on Alter -> waits on Sync Worker
504		 *
505		 * Node-2: Query -> waits on Alter -> waits on Sync Worker
506		 *
507		 * Since each node only sees part of the wait graph, the deadlock may
508		 * go undetected, leading to indefinite blocking.
509		 *
510		 * Note: Each entry in VALUES includes an index 'seqidx' that
511		 * represents the sequence's position in the local 'seqinfos' list.
512		 * This index is propagated to the query results and later used to
513		 * directly map the fetched publisher sequence rows back to their
514		 * corresponding local entries without relying on result order or name
515		 * matching.
516		 */
517		appendStringInfo(&cmd,
518						 "SELECT s.seqidx, has_sequence_privilege(c.oid, 'SELECT'),\n"
519						 "       ps.*, seq.seqtypid,\n"
520						 "       seq.seqstart, seq.seqincrement, seq.seqmin,\n"
521						 "       seq.seqmax, seq.seqcycle\n"
522						 "FROM ( VALUES %s ) AS s (schname, seqname, seqidx)\n"
523						 "JOIN pg_namespace n ON n.nspname = s.schname\n"
524						 "JOIN pg_class c ON c.relnamespace = n.oid AND c.relname = s.seqname\n"
525						 "JOIN pg_sequence seq ON seq.seqrelid = c.oid\n"
526						 "JOIN LATERAL pg_get_sequence_data(seq.seqrelid) AS ps ON true\n",
527						 seqstr.data);
528
529		res = walrcv_exec(conn, cmd.data, lengthof(seqRow), seqRow);
530		if (res->status != WALRCV_OK_TUPLES)
531			ereport(ERROR,
532					errcode(ERRCODE_CONNECTION_FAILURE),
533					errmsg("could not fetch sequence information from the publisher: %s",
534						   res->err));
535
536		slot = MakeSingleTupleTableSlot(res->tupledesc, &TTSOpsMinimalTuple);
537		while (tuplestore_gettupleslot(res->tuplestore, true, false, slot))
538		{
539			CopySeqResult sync_status;
540			LogicalRepSequenceInfo *seqinfo;
541			Relation	sequence_rel = NULL;
542			int			seqidx;
543
544			CHECK_FOR_INTERRUPTS();
545
546			if (ConfigReloadPending)
547			{
548				ConfigReloadPending = false;
549				ProcessConfigFile(PGC_SIGHUP);
550			}
551
552			sync_status = get_and_validate_seq_info(slot, &sequence_rel,
553													&seqinfo, &seqidx);
554			if (sync_status == COPYSEQ_SUCCESS)
555				sync_status = copy_sequence(seqinfo,
556											sequence_rel->rd_rel->relowner);
557
558			switch (sync_status)
559			{
560				case COPYSEQ_SUCCESS:
561					elog(DEBUG1,
562						 "logical replication synchronization for subscription \"%s\", sequence \"%s.%s\" has finished",
563						 MySubscription->name, seqinfo->nspname,
564						 seqinfo->seqname);
565					batch_succeeded_count++;
566					break;
567				case COPYSEQ_MISMATCH:
568
569					/*
570					 * Remember mismatched sequences in a long-lived memory
571					 * context since these will be used after the transaction
572					 * is committed.
573					 */
574					oldctx = MemoryContextSwitchTo(ApplyContext);
575					mismatched_seqs_idx = lappend_int(mismatched_seqs_idx,
576													  seqidx);
577					MemoryContextSwitchTo(oldctx);
578					batch_mismatched_count++;
579					break;
580				case COPYSEQ_SUBSCRIBER_INSUFFICIENT_PERM:
581
582					/*
583					 * Remember sequences with insufficient privileges in a
584					 * long-lived memory context since these will be used
585					 * after the transaction is committed.
586					 */
587					oldctx = MemoryContextSwitchTo(ApplyContext);
588					sub_insuffperm_seqs_idx = lappend_int(sub_insuffperm_seqs_idx,
589														  seqidx);
590					MemoryContextSwitchTo(oldctx);
591					batch_sub_insuffperm_count++;
592					break;
593				case COPYSEQ_PUBLISHER_INSUFFICIENT_PERM:
594
595					/*
596					 * Remember sequences for which the publisher lacks the
597					 * privileges required by pg_get_sequence_data().
598					 */
599					oldctx = MemoryContextSwitchTo(ApplyContext);
600					pub_insuffperm_seqs_idx = lappend_int(pub_insuffperm_seqs_idx,
601														  seqidx);
602					MemoryContextSwitchTo(oldctx);
603					batch_pub_insuffperm_count++;
604					break;
605				case COPYSEQ_SKIPPED:
606
607					/*
608					 * Concurrent removal of a sequence on the subscriber is
609					 * treated as success, since the only viable action is to
610					 * skip the corresponding sequence data. Missing sequences
611					 * on the publisher are treated as ERROR.
612					 */
613					if (seqinfo->found_on_pub)
614					{
615						ereport(LOG,
616								errmsg("skip synchronization of sequence \"%s.%s\" because it has been dropped concurrently",
617									   seqinfo->nspname,
618									   seqinfo->seqname));
619						batch_skipped_count++;
620					}
621					break;
622			}
623
624			if (sequence_rel)
625				table_close(sequence_rel, NoLock);
626		}
627
628		ExecDropSingleTupleTableSlot(slot);
629		walrcv_clear_result(res);
630		resetStringInfo(&seqstr);
631		resetStringInfo(&cmd);
632
633		batch_missing_count = batch_size - (batch_succeeded_count +
634											batch_mismatched_count +
635											batch_sub_insuffperm_count +
636											batch_pub_insuffperm_count +
637											batch_skipped_count);
638
639		elog(DEBUG1,
640			 "logical replication sequence synchronization for subscription \"%s\" - batch #%d = %d attempted, %d succeeded, %d mismatched, %d subscriber insufficient permission, %d publisher insufficient permission, %d missing from publisher, %d skipped",
641			 MySubscription->name,
642			 (cur_batch_base_index / MAX_SEQUENCES_SYNC_PER_BATCH) + 1,
643			 batch_size, batch_succeeded_count, batch_mismatched_count,
644			 batch_sub_insuffperm_count, batch_pub_insuffperm_count, batch_missing_count, batch_skipped_count);
645
646		/* Commit this batch, and prepare for next batch */
647		CommitTransactionCommand();
648
649		if (batch_missing_count)
650		{
651			for (int idx = cur_batch_base_index; idx < cur_batch_base_index + batch_size; idx++)
652			{
653				LogicalRepSequenceInfo *seqinfo =
654					(LogicalRepSequenceInfo *) list_nth(seqinfos, idx);
655
656				/* If the sequence was not found on publisher, record it */
657				if (!seqinfo->found_on_pub)
658					missing_seqs_idx = lappend_int(missing_seqs_idx, idx);
659			}
660		}
661
662		/*
663		 * cur_batch_base_index is not incremented sequentially because some
664		 * sequences may be missing, and the number of fetched rows may not
665		 * match the batch size.
666		 */
667		cur_batch_base_index += batch_size;
668	}
669
670	/* Report mismatches, permission issues, or missing sequences */
671	report_sequence_errors(mismatched_seqs_idx, sub_insuffperm_seqs_idx,
672						   pub_insuffperm_seqs_idx, missing_seqs_idx);
673}

675/*
676 * Identifies sequences that require synchronization and initiates the
677 * synchronization process.
678 */
679static void
680LogicalRepSyncSequences(void)
681{
682	char	   *err;
683	bool		must_use_password;
684	Relation	rel;
685	HeapTuple	tup;
686	ScanKeyData skey[2];
687	SysScanDesc scan;
688	Oid			subid = MyLogicalRepWorker->subid;
689	StringInfoData app_name;
690
691	StartTransactionCommand();
692
693	rel = table_open(SubscriptionRelRelationId, AccessShareLock);
694
695	ScanKeyInit(&skey[0],
696				Anum_pg_subscription_rel_srsubid,
697				BTEqualStrategyNumber, F_OIDEQ,
698				ObjectIdGetDatum(subid));
699
700	ScanKeyInit(&skey[1],
701				Anum_pg_subscription_rel_srsubstate,
702				BTEqualStrategyNumber, F_CHAREQ,
703				CharGetDatum(SUBREL_STATE_INIT));
704
705	scan = systable_beginscan(rel, InvalidOid, false,
706							  NULL, 2, skey);
707	while (HeapTupleIsValid(tup = systable_getnext(scan)))
708	{
709		Form_pg_subscription_rel subrel;
710		LogicalRepSequenceInfo *seq;
711		Relation	sequence_rel;
712		MemoryContext oldctx;
713
714		CHECK_FOR_INTERRUPTS();
715
716		subrel = (Form_pg_subscription_rel) GETSTRUCT(tup);
717
718		sequence_rel = try_table_open(subrel->srrelid, RowExclusiveLock);
719
720		/* Skip if sequence was dropped concurrently */
721		if (!sequence_rel)
722			continue;
723
724		/* Skip if the relation is not a sequence */
725		if (sequence_rel->rd_rel->relkind != RELKIND_SEQUENCE)
726		{
727			table_close(sequence_rel, NoLock);
728			continue;
729		}
730
731		/*
732		 * Worker needs to process sequences across transaction boundary, so
733		 * allocate them under long-lived context.
734		 */
735		oldctx = MemoryContextSwitchTo(ApplyContext);
736
737		seq = palloc0_object(LogicalRepSequenceInfo);
738		seq->localrelid = subrel->srrelid;
739		seq->nspname = get_namespace_name(RelationGetNamespace(sequence_rel));
740		seq->seqname = pstrdup(RelationGetRelationName(sequence_rel));
741		seqinfos = lappend(seqinfos, seq);
742
743		MemoryContextSwitchTo(oldctx);
744
745		table_close(sequence_rel, NoLock);
746	}
747
748	/* Cleanup */
749	systable_endscan(scan);
750	table_close(rel, AccessShareLock);
751
752	CommitTransactionCommand();
753
754	/*
755	 * Exit early if no catalog entries found, likely due to concurrent drops.
756	 */
757	if (!seqinfos)
758		return;
759
760	/* Is the use of a password mandatory? */
761	must_use_password = MySubscription->passwordrequired &&
762		!MySubscription->ownersuperuser;
763
764	initStringInfo(&app_name);
765	appendStringInfo(&app_name, "pg_%u_sequence_sync_" UINT64_FORMAT,
766					 MySubscription->oid, GetSystemIdentifier());
767
768	/*
769	 * Establish the connection to the publisher for sequence synchronization.
770	 */
771	LogRepWorkerWalRcvConn =
772		walrcv_connect(MySubscription->conninfo, true, true,
773					   must_use_password,
774					   app_name.data, &err);
775	if (LogRepWorkerWalRcvConn == NULL)
776		ereport(ERROR,
777				errcode(ERRCODE_CONNECTION_FAILURE),
778				errmsg("sequencesync worker for subscription \"%s\" could not connect to the publisher: %s",
779					   MySubscription->name, err));
780
781	pfree(app_name.data);
782
783	copy_sequences(LogRepWorkerWalRcvConn);
784}
785
786/*
787 * Execute the initial sync with error handling. Disable the subscription,
788 * if required.
789 *
790 * Note that we don't handle FATAL errors which are probably because of system
791 * resource error and are not repeatable.
792 */
793static void
794start_sequence_sync(void)
795{
796	Assert(am_sequencesync_worker());
797
798	PG_TRY();
799	{
800		/* Call initial sync. */
801		LogicalRepSyncSequences();
802	}
803	PG_CATCH();
804	{
805		if (MySubscription->disableonerr)
806			DisableSubscriptionAndExit();
807		else
808		{
809			/*
810			 * Report the worker failed during sequence synchronization. Abort
811			 * the current transaction so that the stats message is sent in an
812			 * idle state.
813			 */
814			AbortOutOfAnyTransaction();
815			pgstat_report_subscription_error(MySubscription->oid);
816
817			PG_RE_THROW();
818		}
819	}
820	PG_END_TRY();
821}
822
823/* Logical Replication sequencesync worker entry point */
824void
825SequenceSyncWorkerMain(Datum main_arg)
826{
827	int			worker_slot = DatumGetInt32(main_arg);
828
829	SetupApplyOrSyncWorker(worker_slot);
830
831	start_sequence_sync();
832
833	FinishSyncWorker();
