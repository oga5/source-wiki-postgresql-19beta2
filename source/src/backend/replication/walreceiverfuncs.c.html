46/*
47 * How long to wait for walreceiver to start up after requesting
48 * postmaster to launch it. In seconds.
49 */
50#define WALRCV_STARTUP_TIMEOUT 10
51
52/* Register shared memory space needed by walreceiver */
53static void
54WalRcvShmemRequest(void *arg)
55{
56	ShmemRequestStruct(.name = "Wal Receiver Ctl",
57					   .size = sizeof(WalRcvData),
58					   .ptr = (void **) &WalRcv,
59		);
60}
61
62/* Initialize walreceiver-related shared memory */
63static void
64WalRcvShmemInit(void *arg)
65{
66	MemSet(WalRcv, 0, sizeof(WalRcvData));
67	WalRcv->walRcvState = WALRCV_STOPPED;
68	ConditionVariableInit(&WalRcv->walRcvStoppedCV);
69	SpinLockInit(&WalRcv->mutex);
70	pg_atomic_init_u64(&WalRcv->writtenUpto, 0);
71	WalRcv->procno = INVALID_PROC_NUMBER;
72}

74/* Is walreceiver running (or starting up)? */
75bool
76WalRcvRunning(void)
77{
78	WalRcvData *walrcv = WalRcv;
79	WalRcvState state;
80	pg_time_t	startTime;
81
82	SpinLockAcquire(&walrcv->mutex);
83
84	state = walrcv->walRcvState;
85	startTime = walrcv->startTime;
86
87	SpinLockRelease(&walrcv->mutex);
88
89	/*
90	 * If it has taken too long for walreceiver to start up, give up. Setting
91	 * the state to STOPPED ensures that if walreceiver later does start up
92	 * after all, it will see that it's not supposed to be running and die
93	 * without doing anything.
94	 */
95	if (state == WALRCV_STARTING)
96	{
97		pg_time_t	now = (pg_time_t) time(NULL);
98
99		if ((now - startTime) > WALRCV_STARTUP_TIMEOUT)
100		{
101			bool		stopped = false;
102
103			SpinLockAcquire(&walrcv->mutex);
104			if (walrcv->walRcvState == WALRCV_STARTING)
105			{
106				state = walrcv->walRcvState = WALRCV_STOPPED;
107				stopped = true;
108			}
109			SpinLockRelease(&walrcv->mutex);
110
111			if (stopped)
112				ConditionVariableBroadcast(&walrcv->walRcvStoppedCV);
113		}
114	}
115
116	if (state != WALRCV_STOPPED)
117		return true;
118	else
119		return false;
120}

122/* Return the state of the walreceiver. */
123WalRcvState
124WalRcvGetState(void)
125{
126	WalRcvData *walrcv = WalRcv;
127	WalRcvState state;
128
129	SpinLockAcquire(&walrcv->mutex);
130	state = walrcv->walRcvState;
131	SpinLockRelease(&walrcv->mutex);
132
133	return state;
134}

157	 * after all, it will see that it's not supposed to be running and die

188/*
189 * Stop walreceiver (if running) and wait for it to die.
190 * Executed by the Startup process.
191 */
192void
193ShutdownWalRcv(void)
194{
195	WalRcvData *walrcv = WalRcv;
196	pid_t		walrcvpid = 0;
197	bool		stopped = false;
198
199	/*
200	 * Request walreceiver to stop. Walreceiver will switch to WALRCV_STOPPED
201	 * mode once it's finished, and will also request postmaster to not
202	 * restart itself.
203	 */
204	SpinLockAcquire(&walrcv->mutex);
205	switch (walrcv->walRcvState)
206	{
207		case WALRCV_STOPPED:
208			break;
209		case WALRCV_STARTING:
210			walrcv->walRcvState = WALRCV_STOPPED;
211			stopped = true;
212			break;
213
214		case WALRCV_CONNECTING:
215		case WALRCV_STREAMING:
216		case WALRCV_WAITING:
217		case WALRCV_RESTARTING:
218			walrcv->walRcvState = WALRCV_STOPPING;
219			pg_fallthrough;
220		case WALRCV_STOPPING:
221			walrcvpid = walrcv->pid;
222			break;
223	}
224	SpinLockRelease(&walrcv->mutex);
225
226	/* Unnecessary but consistent. */
227	if (stopped)
228		ConditionVariableBroadcast(&walrcv->walRcvStoppedCV);
229
230	/*
231	 * Signal walreceiver process if it was still running.
232	 */
233	if (walrcvpid != 0)
234		kill(walrcvpid, SIGTERM);
235
236	/*
237	 * Wait for walreceiver to acknowledge its death by setting state to
238	 * WALRCV_STOPPED.
239	 */
240	ConditionVariablePrepareToSleep(&walrcv->walRcvStoppedCV);
241	while (WalRcvRunning())
242		ConditionVariableSleep(&walrcv->walRcvStoppedCV,
243							   WAIT_EVENT_WAL_RECEIVER_EXIT);
244	ConditionVariableCancelSleep();
245}

247/*
248 * Request postmaster to start walreceiver.
249 *
250 * "recptr" indicates the position where streaming should begin.  "conninfo"
251 * is a libpq connection string to use.  "slotname" is, optionally, the name
252 * of a replication slot to acquire.  "create_temp_slot" indicates to create
253 * a temporary slot when no "slotname" is given.
254 *
255 * WAL receivers do not directly load GUC parameters used for the connection
256 * to the primary, and rely on the values passed down by the caller of this
257 * routine instead.  Hence, the addition of any new parameters should happen
258 * through this code path.
259 */
260void
261RequestXLogStreaming(TimeLineID tli, XLogRecPtr recptr, const char *conninfo,
262					 const char *slotname, bool create_temp_slot)
263{
264	WalRcvData *walrcv = WalRcv;
265	bool		launch = false;
266	pg_time_t	now = (pg_time_t) time(NULL);
267	ProcNumber	walrcv_proc;
268
269	/*
270	 * We always start at the beginning of the segment. That prevents a broken
271	 * segment (i.e., with no records in the first half of a segment) from
272	 * being created by XLOG streaming, which might cause trouble later on if
273	 * the segment is e.g archived.
274	 */
275	if (XLogSegmentOffset(recptr, wal_segment_size) != 0)
276		recptr -= XLogSegmentOffset(recptr, wal_segment_size);
277
278	SpinLockAcquire(&walrcv->mutex);
279
280	/* It better be stopped if we try to restart it */
281	Assert(walrcv->walRcvState == WALRCV_STOPPED ||
282		   walrcv->walRcvState == WALRCV_WAITING);
283
284	/*
285	 * Use configured replication slot if present, and ignore the value of
286	 * create_temp_slot as the slot name should be persistent.  Otherwise, use
287	 * create_temp_slot to determine whether this WAL receiver should create a
288	 * temporary slot by itself and use it, or not.
289	 */
290	if (slotname != NULL && slotname[0] != '\0')
291	{
292		strlcpy(walrcv->slotname, slotname, NAMEDATALEN);
293		walrcv->is_temp_slot = false;
294	}
295	else
296	{
297		walrcv->slotname[0] = '\0';
298		walrcv->is_temp_slot = create_temp_slot;
299	}
300
301	/*
302	 * While waiting for instructions, the WAL receiver uses the same
303	 * connection, so do not clobber the user-visible conninfo already saved.
304	 */
305	if (walrcv->walRcvState == WALRCV_STOPPED)
306	{
307		launch = true;
308		walrcv->walRcvState = WALRCV_STARTING;
309
310		if (conninfo != NULL)
311			strlcpy(walrcv->conninfo, conninfo, MAXCONNINFO);
312		else
313			walrcv->conninfo[0] = '\0';
314	}
315	else
316		walrcv->walRcvState = WALRCV_RESTARTING;
317	walrcv->startTime = now;
318
319	/*
320	 * If this is the first startup of walreceiver (on this timeline),
321	 * initialize flushedUpto and latestChunkStart to the starting point.
322	 */
323	if (!XLogRecPtrIsValid(walrcv->receiveStart) || walrcv->receivedTLI != tli)
324	{
325		walrcv->flushedUpto = recptr;
326		walrcv->receivedTLI = tli;
327		walrcv->latestChunkStart = recptr;
328
329		/*
330		 * Pairs with pg_atomic_read_membarrier_u64() in
331		 * GetWalRcvWriteRecPtr().
332		 */
333		pg_atomic_write_membarrier_u64(&walrcv->writtenUpto, recptr);
334	}
335	walrcv->receiveStart = recptr;
336	walrcv->receiveStartTLI = tli;
337
338	walrcv_proc = walrcv->procno;
339
340	SpinLockRelease(&walrcv->mutex);
341
342	if (launch)
343		SendPostmasterSignal(PMSIGNAL_START_WALRECEIVER);
344	else if (walrcv_proc != INVALID_PROC_NUMBER)
345		SetLatch(&GetPGProcByNumber(walrcv_proc)->procLatch);
346}

348/*
349 * Returns the last+1 byte position that walreceiver has flushed.
350 *
351 * Optionally, returns the previous chunk start, that is the first byte
352 * written in the most recent walreceiver flush cycle.  Callers not
353 * interested in that value may pass NULL for latestChunkStart. Same for
354 * receiveTLI.
355 */
356XLogRecPtr
357GetWalRcvFlushRecPtr(XLogRecPtr *latestChunkStart, TimeLineID *receiveTLI)
358{
359	WalRcvData *walrcv = WalRcv;
360	XLogRecPtr	recptr;
361
362	SpinLockAcquire(&walrcv->mutex);
363	recptr = walrcv->flushedUpto;
364	if (latestChunkStart)
365		*latestChunkStart = walrcv->latestChunkStart;
366	if (receiveTLI)
367		*receiveTLI = walrcv->receivedTLI;
368	SpinLockRelease(&walrcv->mutex);
369
370	return recptr;
371}

373/*
374 * Returns the last+1 byte position that walreceiver has written.
375 *
376 * Use pg_atomic_read_membarrier_u64() to ensure that callers see up-to-date
377 * shared memory state, matching the barrier semantics provided by the
378 * spinlock in GetWalRcvFlushRecPtr() and other LSN-position functions.
379 */
380XLogRecPtr
381GetWalRcvWriteRecPtr(void)
382{
383	WalRcvData *walrcv = WalRcv;
384
385	return pg_atomic_read_membarrier_u64(&walrcv->writtenUpto);
386}

388/*
389 * Returns the replication apply delay in ms or -1
390 * if the apply delay info is not available
391 */
392int
393GetReplicationApplyDelay(void)
394{
395	WalRcvData *walrcv = WalRcv;
396	XLogRecPtr	receivePtr;
397	XLogRecPtr	replayPtr;
398	TimestampTz chunkReplayStartTime;
399
400	SpinLockAcquire(&walrcv->mutex);
401	receivePtr = walrcv->flushedUpto;
402	SpinLockRelease(&walrcv->mutex);
403
404	replayPtr = GetXLogReplayRecPtr(NULL);
405
406	if (receivePtr == replayPtr)
407		return 0;
408
409	chunkReplayStartTime = GetCurrentChunkReplayStartTime();
410
411	if (chunkReplayStartTime == 0)
412		return -1;
413
414	return TimestampDifferenceMilliseconds(chunkReplayStartTime,
415										   GetCurrentTimestamp());
416}

418/*
419 * Returns the network latency in ms, note that this includes any
420 * difference in clock settings between the servers, as well as timezone.
421 */
422int
423GetReplicationTransferLatency(void)
424{
425	WalRcvData *walrcv = WalRcv;
426	TimestampTz lastMsgSendTime;
427	TimestampTz lastMsgReceiptTime;
428
429	SpinLockAcquire(&walrcv->mutex);
430	lastMsgSendTime = walrcv->lastMsgSendTime;
431	lastMsgReceiptTime = walrcv->lastMsgReceiptTime;
432	SpinLockRelease(&walrcv->mutex);
433
434	return TimestampDifferenceMilliseconds(lastMsgSendTime,
435										   lastMsgReceiptTime);
