30/*
31 * Create a large object having the given LO identifier.
32 *
33 * We create a new large object by inserting an entry into
34 * pg_largeobject_metadata without any data pages, so that the object
35 * will appear to exist with size 0.
36 */
37Oid
38LargeObjectCreate(Oid loid)
39{
40	Relation	pg_lo_meta;
41	HeapTuple	ntup;
42	Oid			loid_new;
43	Datum		values[Natts_pg_largeobject_metadata];
44	bool		nulls[Natts_pg_largeobject_metadata];
45	Oid			ownerId;
46	Acl		   *lomacl;
47
48	pg_lo_meta = table_open(LargeObjectMetadataRelationId,
49							RowExclusiveLock);
50
51	/*
52	 * Insert metadata of the largeobject
53	 */
54	memset(values, 0, sizeof(values));
55	memset(nulls, false, sizeof(nulls));
56
57	if (OidIsValid(loid))
58		loid_new = loid;
59	else
60		loid_new = GetNewOidWithIndex(pg_lo_meta,
61									  LargeObjectMetadataOidIndexId,
62									  Anum_pg_largeobject_metadata_oid);
63	ownerId = GetUserId();
64	lomacl = get_user_default_acl(OBJECT_LARGEOBJECT, ownerId, InvalidOid);
65
66	values[Anum_pg_largeobject_metadata_oid - 1] = ObjectIdGetDatum(loid_new);
67	values[Anum_pg_largeobject_metadata_lomowner - 1]
68		= ObjectIdGetDatum(ownerId);
69
70	if (lomacl != NULL)
71		values[Anum_pg_largeobject_metadata_lomacl - 1]
72			= PointerGetDatum(lomacl);
73	else
74		nulls[Anum_pg_largeobject_metadata_lomacl - 1] = true;
75
76	ntup = heap_form_tuple(RelationGetDescr(pg_lo_meta),
77						   values, nulls);
78
79	CatalogTupleInsert(pg_lo_meta, ntup);
80
81	heap_freetuple(ntup);
82
83	table_close(pg_lo_meta, RowExclusiveLock);
84
85	/* dependencies on roles mentioned in default ACL */
86	recordDependencyOnNewAcl(LargeObjectRelationId, loid_new, 0,
87							 ownerId, lomacl);
88
89	return loid_new;
90}

92/*
93 * Drop a large object having the given LO identifier.  Both the data pages
94 * and metadata must be dropped.
95 */
96void
97LargeObjectDrop(Oid loid)
98{
99	Relation	pg_lo_meta;
100	Relation	pg_largeobject;
101	ScanKeyData skey[1];
102	SysScanDesc scan;
103	HeapTuple	tuple;
104
105	pg_lo_meta = table_open(LargeObjectMetadataRelationId,
106							RowExclusiveLock);
107
108	pg_largeobject = table_open(LargeObjectRelationId,
109								RowExclusiveLock);
110
111	/*
112	 * Delete an entry from pg_largeobject_metadata
113	 */
114	ScanKeyInit(&skey[0],
115				Anum_pg_largeobject_metadata_oid,
116				BTEqualStrategyNumber, F_OIDEQ,
117				ObjectIdGetDatum(loid));
118
119	scan = systable_beginscan(pg_lo_meta,
120							  LargeObjectMetadataOidIndexId, true,
121							  NULL, 1, skey);
122
123	tuple = systable_getnext(scan);
124	if (!HeapTupleIsValid(tuple))
125		ereport(ERROR,
126				(errcode(ERRCODE_UNDEFINED_OBJECT),
127				 errmsg("large object %u does not exist", loid)));
128
129	CatalogTupleDelete(pg_lo_meta, &tuple->t_self);
130
131	systable_endscan(scan);
132
133	/*
134	 * Delete all the associated entries from pg_largeobject
135	 */
136	ScanKeyInit(&skey[0],
137				Anum_pg_largeobject_loid,
138				BTEqualStrategyNumber, F_OIDEQ,
139				ObjectIdGetDatum(loid));
140
141	scan = systable_beginscan(pg_largeobject,
142							  LargeObjectLOidPNIndexId, true,
143							  NULL, 1, skey);
144	while (HeapTupleIsValid(tuple = systable_getnext(scan)))
145	{
146		CatalogTupleDelete(pg_largeobject, &tuple->t_self);
147	}
148
149	systable_endscan(scan);
150
151	table_close(pg_largeobject, RowExclusiveLock);
152
153	table_close(pg_lo_meta, RowExclusiveLock);
154}

156/*
157 * LargeObjectExists
158 *
159 * We don't use the system cache for large object metadata, for fear of
160 * using too much local memory.
161 *
162 * This function always scans the system catalog using an up-to-date snapshot,
163 * so it should not be used when a large object is opened in read-only mode
164 * (because large objects opened in read only mode are supposed to be viewed
165 * relative to the caller's snapshot, whereas in read-write mode they are
166 * relative to a current snapshot).
167 */
168bool
169LargeObjectExists(Oid loid)
170{
171	return LargeObjectExistsWithSnapshot(loid, NULL);
172}
173
174/*
175 * Same as LargeObjectExists(), except snapshot to read with can be specified.
176 */
177bool
178LargeObjectExistsWithSnapshot(Oid loid, Snapshot snapshot)
179{
180	Relation	pg_lo_meta;
181	ScanKeyData skey[1];
182	SysScanDesc sd;
183	HeapTuple	tuple;
184	bool		retval = false;
185
186	ScanKeyInit(&skey[0],
187				Anum_pg_largeobject_metadata_oid,
188				BTEqualStrategyNumber, F_OIDEQ,
189				ObjectIdGetDatum(loid));
190
191	pg_lo_meta = table_open(LargeObjectMetadataRelationId,
192							AccessShareLock);
193
194	sd = systable_beginscan(pg_lo_meta,
195							LargeObjectMetadataOidIndexId, true,
196							snapshot, 1, skey);
197
198	tuple = systable_getnext(sd);
199	if (HeapTupleIsValid(tuple))
200		retval = true;
201
202	systable_endscan(sd);
203
204	table_close(pg_lo_meta, AccessShareLock);
205
206	return retval;
207}
