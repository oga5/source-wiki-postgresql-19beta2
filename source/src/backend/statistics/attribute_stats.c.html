117/*
118 * Insert or Update Attribute Statistics
119 *
120 * See pg_statistic.h for an explanation of how each statistic kind is
121 * stored. Custom statistics kinds are not supported.
122 *
123 * Depending on the statistics kind, we need to derive information from the
124 * attribute for which we're storing the stats. For instance, the MCVs are
125 * stored as an anyarray, and the representation of the array needs to store
126 * the correct element type, which must be derived from the attribute.
127 *
128 * Major errors, such as the table not existing, the attribute not existing,
129 * or a permissions failure are always reported at ERROR. Other errors, such
130 * as a conversion failure on one statistic kind, are reported as a WARNING
131 * and other statistic kinds may still be updated.
132 */
133static bool
134attribute_statistics_update(FunctionCallInfo fcinfo)
135{
136	char	   *nspname;
137	char	   *relname;
138	Oid			reloid;
139	char	   *attname;
140	AttrNumber	attnum;
141	bool		inherited;
142	Oid			locked_table = InvalidOid;
143
144	stats_check_required_arg(fcinfo, attarginfo, ATTRELSCHEMA_ARG);
145	stats_check_required_arg(fcinfo, attarginfo, ATTRELNAME_ARG);
146
147	nspname = TextDatumGetCString(PG_GETARG_DATUM(ATTRELSCHEMA_ARG));
148	relname = TextDatumGetCString(PG_GETARG_DATUM(ATTRELNAME_ARG));
149
150	if (RecoveryInProgress())
151		ereport(ERROR,
152				(errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
153				 errmsg("recovery is in progress"),
154				 errhint("Statistics cannot be modified during recovery.")));
155
156	/* lock before looking up attribute */
157	reloid = RangeVarGetRelidExtended(makeRangeVar(nspname, relname, -1),
158									  ShareUpdateExclusiveLock, 0,
159									  RangeVarCallbackForStats, &locked_table);
160
161	/* user can specify either attname or attnum, but not both */
162	if (!PG_ARGISNULL(ATTNAME_ARG))
163	{
164		if (!PG_ARGISNULL(ATTNUM_ARG))
165			ereport(ERROR,
166					(errcode(ERRCODE_INVALID_PARAMETER_VALUE),
167					 errmsg("cannot specify both \"%s\" and \"%s\"", "attname", "attnum")));
168		attname = TextDatumGetCString(PG_GETARG_DATUM(ATTNAME_ARG));
169		attnum = get_attnum(reloid, attname);
170		/* note that this test covers attisdropped cases too: */
171		if (attnum == InvalidAttrNumber)
172			ereport(ERROR,
173					(errcode(ERRCODE_UNDEFINED_COLUMN),
174					 errmsg("column \"%s\" of relation \"%s\" does not exist",
175							attname, relname)));
176	}
177	else if (!PG_ARGISNULL(ATTNUM_ARG))
178	{
179		attnum = PG_GETARG_INT16(ATTNUM_ARG);
180		attname = get_attname(reloid, attnum, true);
181		/* annoyingly, get_attname doesn't check attisdropped */
182		if (attname == NULL ||
183			!SearchSysCacheExistsAttName(reloid, attname))
184			ereport(ERROR,
185					(errcode(ERRCODE_UNDEFINED_COLUMN),
186					 errmsg("column %d of relation \"%s\" does not exist",
187							attnum, relname)));
188	}
189	else
190	{
191		ereport(ERROR,
192				(errcode(ERRCODE_INVALID_PARAMETER_VALUE),
193				 errmsg("must specify either \"%s\" or \"%s\"", "attname", "attnum")));
194		attname = NULL;			/* keep compiler quiet */
195		attnum = 0;
196	}
197
198	if (attnum < 0)
199		ereport(ERROR,
200				(errcode(ERRCODE_FEATURE_NOT_SUPPORTED),
201				 errmsg("cannot modify statistics on system column \"%s\"",
202						attname)));
203
204	stats_check_required_arg(fcinfo, attarginfo, INHERITED_ARG);
205	inherited = PG_GETARG_BOOL(INHERITED_ARG);
206
207	return attribute_statistics_update_internal(reloid, attname, attnum,
208												inherited, fcinfo);
209}

211/*
212 * Workhorse function for attribute_statistics_update.
213 */
214static bool
215attribute_statistics_update_internal(Oid reloid,
216									 const char *attname, AttrNumber attnum,
217									 bool inherited, FunctionCallInfo fcinfo)
218{
219	Relation	starel;
220	HeapTuple	statup;
221
222	Oid			atttypid = InvalidOid;
223	int32		atttypmod;
224	char		atttyptype;
225	Oid			atttypcoll = InvalidOid;
226	Oid			eq_opr = InvalidOid;
227	Oid			lt_opr = InvalidOid;
228
229	Oid			elemtypid = InvalidOid;
230	Oid			elem_eq_opr = InvalidOid;
231
232	FmgrInfo	array_in_fn;
233
234	bool		do_mcv = !PG_ARGISNULL(MOST_COMMON_FREQS_ARG) &&
235		!PG_ARGISNULL(MOST_COMMON_VALS_ARG);
236	bool		do_histogram = !PG_ARGISNULL(HISTOGRAM_BOUNDS_ARG);
237	bool		do_correlation = !PG_ARGISNULL(CORRELATION_ARG);
238	bool		do_mcelem = !PG_ARGISNULL(MOST_COMMON_ELEMS_ARG) &&
239		!PG_ARGISNULL(MOST_COMMON_ELEM_FREQS_ARG);
240	bool		do_dechist = !PG_ARGISNULL(ELEM_COUNT_HISTOGRAM_ARG);
241	bool		do_bounds_histogram = !PG_ARGISNULL(RANGE_BOUNDS_HISTOGRAM_ARG);
242	bool		do_range_length_histogram = !PG_ARGISNULL(RANGE_LENGTH_HISTOGRAM_ARG) &&
243		!PG_ARGISNULL(RANGE_EMPTY_FRAC_ARG);
244
245	Datum		values[Natts_pg_statistic] = {0};
246	bool		nulls[Natts_pg_statistic] = {0};
247	bool		replaces[Natts_pg_statistic] = {0};
248
249	bool		result = true;
250
251	/*
252	 * Check argument sanity. If some arguments are unusable, emit a WARNING
253	 * and set the corresponding argument to NULL in fcinfo.
254	 */
255
256	if (!stats_check_arg_array(fcinfo, attarginfo, MOST_COMMON_FREQS_ARG))
257	{
258		do_mcv = false;
259		result = false;
260	}
261
262	if (!stats_check_arg_array(fcinfo, attarginfo, MOST_COMMON_ELEM_FREQS_ARG))
263	{
264		do_mcelem = false;
265		result = false;
266	}
267	if (!stats_check_arg_array(fcinfo, attarginfo, ELEM_COUNT_HISTOGRAM_ARG))
268	{
269		do_dechist = false;
270		result = false;
271	}
272
273	if (!stats_check_arg_pair(fcinfo, attarginfo,
274							  MOST_COMMON_VALS_ARG, MOST_COMMON_FREQS_ARG))
275	{
276		do_mcv = false;
277		result = false;
278	}
279
280	if (!stats_check_arg_pair(fcinfo, attarginfo,
281							  MOST_COMMON_ELEMS_ARG,
282							  MOST_COMMON_ELEM_FREQS_ARG))
283	{
284		do_mcelem = false;
285		result = false;
286	}
287
288	if (!stats_check_arg_pair(fcinfo, attarginfo,
289							  RANGE_LENGTH_HISTOGRAM_ARG,
290							  RANGE_EMPTY_FRAC_ARG))
291	{
292		do_range_length_histogram = false;
293		result = false;
294	}
295
296	/* derive information from attribute */
297	statatt_get_type(reloid, attnum,
298					 &atttypid, &atttypmod,
299					 &atttyptype, &atttypcoll,
300					 &eq_opr, &lt_opr);
301
302	/* if needed, derive element type */
303	if (do_mcelem || do_dechist)
304	{
305		if (!statatt_get_elem_type(atttypid, atttyptype,
306								   &elemtypid, &elem_eq_opr))
307		{
308			ereport(WARNING,
309					(errmsg("could not determine element type of column \"%s\"", attname),
310					 errdetail("Cannot set %s or %s.",
311							   "STATISTIC_KIND_MCELEM", "STATISTIC_KIND_DECHIST")));
312			elemtypid = InvalidOid;
313			elem_eq_opr = InvalidOid;
314
315			do_mcelem = false;
316			do_dechist = false;
317			result = false;
318		}
319	}
320
321	/* histogram and correlation require less-than operator */
322	if ((do_histogram || do_correlation) && !OidIsValid(lt_opr))
323	{
324		ereport(WARNING,
325				(errcode(ERRCODE_INVALID_PARAMETER_VALUE),
326				 errmsg("could not determine less-than operator for column \"%s\"", attname),
327				 errdetail("Cannot set %s or %s.",
328						   "STATISTIC_KIND_HISTOGRAM", "STATISTIC_KIND_CORRELATION")));
329
330		do_histogram = false;
331		do_correlation = false;
332		result = false;
333	}
334
335	/* only range types can have range stats */
336	if ((do_range_length_histogram || do_bounds_histogram) &&
337		!(atttyptype == TYPTYPE_RANGE || atttyptype == TYPTYPE_MULTIRANGE))
338	{
339		ereport(WARNING,
340				(errcode(ERRCODE_INVALID_PARAMETER_VALUE),
341				 errmsg("column \"%s\" is not a range type", attname),
342				 errdetail("Cannot set %s or %s.",
343						   "STATISTIC_KIND_RANGE_LENGTH_HISTOGRAM", "STATISTIC_KIND_BOUNDS_HISTOGRAM")));
344
345		do_bounds_histogram = false;
346		do_range_length_histogram = false;
347		result = false;
348	}
349
350	fmgr_info(F_ARRAY_IN, &array_in_fn);
351
352	starel = table_open(StatisticRelationId, RowExclusiveLock);
353
354	statup = SearchSysCache3(STATRELATTINH, ObjectIdGetDatum(reloid), Int16GetDatum(attnum), BoolGetDatum(inherited));
355
356	/* initialize from existing tuple if exists */
357	if (HeapTupleIsValid(statup))
358		heap_deform_tuple(statup, RelationGetDescr(starel), values, nulls);
359	else
360		statatt_init_empty_tuple(reloid, attnum, inherited, values, nulls,
361								 replaces);
362
363	/* if specified, set to argument values */
364	if (!PG_ARGISNULL(NULL_FRAC_ARG))
365	{
366		values[Anum_pg_statistic_stanullfrac - 1] = PG_GETARG_DATUM(NULL_FRAC_ARG);
367		replaces[Anum_pg_statistic_stanullfrac - 1] = true;
368	}
369	if (!PG_ARGISNULL(AVG_WIDTH_ARG))
370	{
371		values[Anum_pg_statistic_stawidth - 1] = PG_GETARG_DATUM(AVG_WIDTH_ARG);
372		replaces[Anum_pg_statistic_stawidth - 1] = true;
373	}
374	if (!PG_ARGISNULL(N_DISTINCT_ARG))
375	{
376		values[Anum_pg_statistic_stadistinct - 1] = PG_GETARG_DATUM(N_DISTINCT_ARG);
377		replaces[Anum_pg_statistic_stadistinct - 1] = true;
378	}
379
380	/* STATISTIC_KIND_MCV */
381	if (do_mcv)
382	{
383		bool		converted;
384		Datum		stanumbers = PG_GETARG_DATUM(MOST_COMMON_FREQS_ARG);
385		Datum		stavalues = statatt_build_stavalues("most_common_vals",
386														&array_in_fn,
387														PG_GETARG_DATUM(MOST_COMMON_VALS_ARG),
388														atttypid, atttypmod,
389														&converted);
390
391		if (converted)
392		{
393			ArrayType  *vals_arr = DatumGetArrayTypeP(stavalues);
394			ArrayType  *nums_arr = DatumGetArrayTypeP(stanumbers);
395			int			nvals = ARR_DIMS(vals_arr)[0];
396			int			nnums = ARR_DIMS(nums_arr)[0];
397
398			if (nvals != nnums)
399			{
400				ereport(WARNING,
401						(errcode(ERRCODE_INVALID_PARAMETER_VALUE),
402						 errmsg("could not parse \"%s\": incorrect number of elements (same as \"%s\" required)",
403								"most_common_vals",
404								"most_common_freqs")));
405				result = false;
406			}
407			else
408			{
409				statatt_set_slot(values, nulls, replaces,
410								 STATISTIC_KIND_MCV,
411								 eq_opr, atttypcoll,
412								 stanumbers, false, stavalues, false);
413			}
414		}
415		else
416			result = false;
417	}
418
419	/* STATISTIC_KIND_HISTOGRAM */
420	if (do_histogram)
421	{
422		Datum		stavalues;
423		bool		converted = false;
424
425		stavalues = statatt_build_stavalues("histogram_bounds",
426											&array_in_fn,
427											PG_GETARG_DATUM(HISTOGRAM_BOUNDS_ARG),
428											atttypid, atttypmod,
429											&converted);
430
431		if (converted)
432		{
433			statatt_set_slot(values, nulls, replaces,
434							 STATISTIC_KIND_HISTOGRAM,
435							 lt_opr, atttypcoll,
436							 0, true, stavalues, false);
437		}
438		else
439			result = false;
440	}
441
442	/* STATISTIC_KIND_CORRELATION */
443	if (do_correlation)
444	{
445		Datum		elems[] = {PG_GETARG_DATUM(CORRELATION_ARG)};
446		ArrayType  *arry = construct_array_builtin(elems, 1, FLOAT4OID);
447		Datum		stanumbers = PointerGetDatum(arry);
448
449		statatt_set_slot(values, nulls, replaces,
450						 STATISTIC_KIND_CORRELATION,
451						 lt_opr, atttypcoll,
452						 stanumbers, false, 0, true);
453	}
454
455	/* STATISTIC_KIND_MCELEM */
456	if (do_mcelem)
457	{
458		Datum		stanumbers = PG_GETARG_DATUM(MOST_COMMON_ELEM_FREQS_ARG);
459		bool		converted = false;
460		Datum		stavalues;
461
462		stavalues = statatt_build_stavalues("most_common_elems",
463											&array_in_fn,
464											PG_GETARG_DATUM(MOST_COMMON_ELEMS_ARG),
465											elemtypid, atttypmod,
466											&converted);
467
468		if (converted)
469		{
470			statatt_set_slot(values, nulls, replaces,
471							 STATISTIC_KIND_MCELEM,
472							 elem_eq_opr, atttypcoll,
473							 stanumbers, false, stavalues, false);
474		}
475		else
476			result = false;
477	}
478
479	/* STATISTIC_KIND_DECHIST */
480	if (do_dechist)
481	{
482		Datum		stanumbers = PG_GETARG_DATUM(ELEM_COUNT_HISTOGRAM_ARG);
483
484		statatt_set_slot(values, nulls, replaces,
485						 STATISTIC_KIND_DECHIST,
486						 elem_eq_opr, atttypcoll,
487						 stanumbers, false, 0, true);
488	}
489
490	/*
491	 * STATISTIC_KIND_BOUNDS_HISTOGRAM
492	 *
493	 * This stakind appears before STATISTIC_KIND_RANGE_LENGTH_HISTOGRAM even
494	 * though it is numerically greater, and all other stakinds appear in
495	 * numerical order. We duplicate this quirk for consistency.
496	 */
497	if (do_bounds_histogram)
498	{
499		bool		converted = false;
500		Datum		stavalues;
501
502		stavalues = statatt_build_stavalues("range_bounds_histogram",
503											&array_in_fn,
504											PG_GETARG_DATUM(RANGE_BOUNDS_HISTOGRAM_ARG),
505											atttypid, atttypmod,
506											&converted);
507
508		if (converted)
509		{
510			statatt_set_slot(values, nulls, replaces,
511							 STATISTIC_KIND_BOUNDS_HISTOGRAM,
512							 InvalidOid, InvalidOid,
513							 0, true, stavalues, false);
514		}
515		else
516			result = false;
517	}
518
519	/* STATISTIC_KIND_RANGE_LENGTH_HISTOGRAM */
520	if (do_range_length_histogram)
521	{
522		/* The anyarray is always a float8[] for this stakind */
523		Datum		elems[] = {PG_GETARG_DATUM(RANGE_EMPTY_FRAC_ARG)};
524		ArrayType  *arry = construct_array_builtin(elems, 1, FLOAT4OID);
525		Datum		stanumbers = PointerGetDatum(arry);
526
527		bool		converted = false;
528		Datum		stavalues;
529
530		stavalues = statatt_build_stavalues("range_length_histogram",
531											&array_in_fn,
532											PG_GETARG_DATUM(RANGE_LENGTH_HISTOGRAM_ARG),
533											FLOAT8OID, 0, &converted);
534
535		if (converted)
536		{
537			statatt_set_slot(values, nulls, replaces,
538							 STATISTIC_KIND_RANGE_LENGTH_HISTOGRAM,
539							 Float8LessOperator, InvalidOid,
540							 stanumbers, false, stavalues, false);
541		}
542		else
543			result = false;
544	}
545
546	upsert_pg_statistic(starel, statup, values, nulls, replaces);
547
548	if (HeapTupleIsValid(statup))
549		ReleaseSysCache(statup);
550	table_close(starel, RowExclusiveLock);
551
552	return result;
553}

555/*
556 * Upsert the pg_statistic record.
557 */
558static void
559upsert_pg_statistic(Relation starel, HeapTuple oldtup,
560					const Datum *values, const bool *nulls, const bool *replaces)
561{
562	HeapTuple	newtup;
563
564	if (HeapTupleIsValid(oldtup))
565	{
566		newtup = heap_modify_tuple(oldtup, RelationGetDescr(starel),
567								   values, nulls, replaces);
568		CatalogTupleUpdate(starel, &newtup->t_self, newtup);
569	}
570	else
571	{
572		newtup = heap_form_tuple(RelationGetDescr(starel), values, nulls);
573		CatalogTupleInsert(starel, newtup);
574	}
575
576	heap_freetuple(newtup);
577
578	CommandCounterIncrement();
579}

581/*
582 * Delete pg_statistic record.
583 */
584static bool
585delete_pg_statistic(Oid reloid, AttrNumber attnum, bool stainherit)
586{
587	Relation	sd = table_open(StatisticRelationId, RowExclusiveLock);
588	HeapTuple	oldtup;
589	bool		result = false;
590
591	/* Is there already a pg_statistic tuple for this attribute? */
592	oldtup = SearchSysCache3(STATRELATTINH,
593							 ObjectIdGetDatum(reloid),
594							 Int16GetDatum(attnum),
595							 BoolGetDatum(stainherit));
596
597	if (HeapTupleIsValid(oldtup))
598	{
599		CatalogTupleDelete(sd, &oldtup->t_self);
600		ReleaseSysCache(oldtup);
601		result = true;
602	}
603
604	table_close(sd, RowExclusiveLock);
605
606	CommandCounterIncrement();
607
608	return result;
609}

611/*
612 * Delete statistics for the given attribute.
613 */
614Datum
615pg_clear_attribute_stats(PG_FUNCTION_ARGS)
616{
617	char	   *nspname;
618	char	   *relname;
619	Oid			reloid;
620	char	   *attname;
621	AttrNumber	attnum;
622	bool		inherited;
623	Oid			locked_table = InvalidOid;
624
625	stats_check_required_arg(fcinfo, cleararginfo, C_ATTRELSCHEMA_ARG);
626	stats_check_required_arg(fcinfo, cleararginfo, C_ATTRELNAME_ARG);
627	stats_check_required_arg(fcinfo, cleararginfo, C_ATTNAME_ARG);
628	stats_check_required_arg(fcinfo, cleararginfo, C_INHERITED_ARG);
629
630	nspname = TextDatumGetCString(PG_GETARG_DATUM(C_ATTRELSCHEMA_ARG));
631	relname = TextDatumGetCString(PG_GETARG_DATUM(C_ATTRELNAME_ARG));
632
633	if (RecoveryInProgress())
634		ereport(ERROR,
635				(errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
636				 errmsg("recovery is in progress"),
637				 errhint("Statistics cannot be modified during recovery.")));
638
639	reloid = RangeVarGetRelidExtended(makeRangeVar(nspname, relname, -1),
640									  ShareUpdateExclusiveLock, 0,
641									  RangeVarCallbackForStats, &locked_table);
642
643	attname = TextDatumGetCString(PG_GETARG_DATUM(C_ATTNAME_ARG));
644	attnum = get_attnum(reloid, attname);
645
646	if (attnum < 0)
647		ereport(ERROR,
648				(errcode(ERRCODE_FEATURE_NOT_SUPPORTED),
649				 errmsg("cannot clear statistics on system column \"%s\"",
650						attname)));
651
652	if (attnum == InvalidAttrNumber)
653		ereport(ERROR,
654				(errcode(ERRCODE_UNDEFINED_COLUMN),
655				 errmsg("column \"%s\" of relation \"%s\" does not exist",
656						attname, get_rel_name(reloid))));
657
658	inherited = PG_GETARG_BOOL(C_INHERITED_ARG);
659
660	delete_pg_statistic(reloid, attnum, inherited);
661	PG_RETURN_VOID();
662}

664/*
665 * Import statistics for a given relation attribute.
666 *
667 * Inserts or replaces a row in pg_statistic for the given relation and
668 * attribute name or number. It takes input parameters that correspond to
669 * columns in the view pg_stats.
670 *
671 * Parameters are given in a pseudo named-attribute style: they must be
672 * pairs of parameter names (as text) and values (of appropriate types).
673 * We do that, rather than using regular named-parameter notation, so
674 * that we can add or change parameters without fear of breaking
675 * carelessly-written calls.
676 *
677 * Parameters null_frac, avg_width, and n_distinct all correspond to NOT NULL
678 * columns in pg_statistic. The remaining parameters all belong to a specific
679 * stakind. Some stakinds require multiple parameters, which must be specified
680 * together (or neither specified).
681 *
682 * Parameters are only superficially validated. Omitting a parameter or
683 * passing NULL leaves the statistic unchanged.
684 *
685 * Parameters corresponding to ANYARRAY columns are instead passed in as text
686 * values, which is a valid input string for an array of the type or element
687 * type of the attribute. Any error generated by the array_in() function will
688 * in turn fail the function.
689 */
690Datum
691pg_restore_attribute_stats(PG_FUNCTION_ARGS)
692{
693	LOCAL_FCINFO(positional_fcinfo, NUM_ATTRIBUTE_STATS_ARGS);
694	bool		result = true;
695
696	InitFunctionCallInfoData(*positional_fcinfo, NULL, NUM_ATTRIBUTE_STATS_ARGS,
697							 InvalidOid, NULL, NULL);
698
699	if (!stats_fill_fcinfo_from_arg_pairs(fcinfo, positional_fcinfo,
700										  attarginfo))
701		result = false;
702
703	if (!attribute_statistics_update(positional_fcinfo))
704		result = false;
705
706	PG_RETURN_BOOL(result);
707}

709/*
710 * Import attribute statistics from NullableDatum inputs for all statistical
711 * values.
712 *
713 * For now, the 'version' argument is ignored. In the future it can be used
714 * to interpret older statistics properly.
715 */
716bool
717import_attribute_statistics(Relation rel, AttrNumber attnum, bool inherited,
718							const NullableDatum *version,
719							const NullableDatum *null_frac,
720							const NullableDatum *avg_width,
721							const NullableDatum *n_distinct,
722							const NullableDatum *most_common_vals,
723							const NullableDatum *most_common_freqs,
724							const NullableDatum *histogram_bounds,
725							const NullableDatum *correlation,
726							const NullableDatum *most_common_elems,
727							const NullableDatum *most_common_elem_freqs,
728							const NullableDatum *elem_count_histogram,
729							const NullableDatum *range_length_histogram,
730							const NullableDatum *range_empty_frac,
731							const NullableDatum *range_bounds_histogram)
732{
733	LOCAL_FCINFO(newfcinfo, NUM_ATTRIBUTE_STATS_ARGS);
734	Oid			reloid = RelationGetRelid(rel);
735	char	   *relname = RelationGetRelationName(rel);
736	char	   *attname = get_attname(reloid, attnum, true);
737
738	Assert(null_frac);
739	Assert(avg_width);
740	Assert(n_distinct);
741	Assert(most_common_vals);
742	Assert(most_common_freqs);
743	Assert(histogram_bounds);
744	Assert(correlation);
745	Assert(most_common_elems);
746	Assert(most_common_elem_freqs);
747	Assert(elem_count_histogram);
748	Assert(range_length_histogram);
749	Assert(range_empty_frac);
750	Assert(range_bounds_histogram);
751
752	/* annoyingly, get_attname doesn't check attisdropped */
753	if (attname == NULL ||
754		!SearchSysCacheExistsAttName(reloid, attname))
755		ereport(ERROR,
756				(errcode(ERRCODE_UNDEFINED_COLUMN),
757				 errmsg("column %d of relation \"%s\" does not exist",
758						attnum, relname)));
759
760	InitFunctionCallInfoData(*newfcinfo, NULL, NUM_ATTRIBUTE_STATS_ARGS,
761							 InvalidOid, NULL, NULL);
762
763	newfcinfo->args[ATTRELSCHEMA_ARG].value =
764		CStringGetTextDatum(get_namespace_name(RelationGetNamespace(rel)));
765	newfcinfo->args[ATTRELSCHEMA_ARG].isnull = false;
766	newfcinfo->args[ATTRELNAME_ARG].value = CStringGetTextDatum(relname);
767	newfcinfo->args[ATTRELNAME_ARG].isnull = false;
768	newfcinfo->args[ATTNAME_ARG].value = CStringGetTextDatum(attname);
769	newfcinfo->args[ATTNAME_ARG].isnull = false;
770	newfcinfo->args[ATTNUM_ARG].value = Int16GetDatum(attnum);
771	newfcinfo->args[ATTNUM_ARG].isnull = false;
772	newfcinfo->args[INHERITED_ARG].value = BoolGetDatum(inherited);
773	newfcinfo->args[INHERITED_ARG].isnull = false;
774
775	newfcinfo->args[NULL_FRAC_ARG] = *null_frac;
776	newfcinfo->args[AVG_WIDTH_ARG] = *avg_width;
777	newfcinfo->args[N_DISTINCT_ARG] = *n_distinct;
778	newfcinfo->args[MOST_COMMON_VALS_ARG] = *most_common_vals;
779	newfcinfo->args[MOST_COMMON_FREQS_ARG] = *most_common_freqs;
780	newfcinfo->args[HISTOGRAM_BOUNDS_ARG] = *histogram_bounds;
781	newfcinfo->args[CORRELATION_ARG] = *correlation;
782	newfcinfo->args[MOST_COMMON_ELEMS_ARG] = *most_common_elems;
783	newfcinfo->args[MOST_COMMON_ELEM_FREQS_ARG] = *most_common_elem_freqs;
784	newfcinfo->args[ELEM_COUNT_HISTOGRAM_ARG] = *elem_count_histogram;
785	newfcinfo->args[RANGE_LENGTH_HISTOGRAM_ARG] = *range_length_histogram;
786	newfcinfo->args[RANGE_EMPTY_FRAC_ARG] = *range_empty_frac;
787	newfcinfo->args[RANGE_BOUNDS_HISTOGRAM_ARG] = *range_bounds_histogram;
788
789	return attribute_statistics_update_internal(reloid, attname, attnum,
790												inherited, newfcinfo);
791}
