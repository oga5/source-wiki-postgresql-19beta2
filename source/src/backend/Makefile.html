127# Linking to a single .o with -r is a lot faster than building a .a or passing
128# all objects to MKLDEXPORT.
129#
130# It looks alluring to use $(CC) -r instead of ld -r, but that doesn't
131# trivially work with gcc, due to gcc specific static libraries linked in with
132# -r.
133$(POSTGRES_IMP): $(OBJS)
134	ld -b64 -r -o SUBSYS.o $(call expand_subsys,$^)
135	$(MKLDEXPORT) SUBSYS.o . > $@
136	@rm -f SUBSYS.o
137
138endif # aix

143# The following targets are specified in make commands that appear in
144# the make files in our subdirectories. Note that it's important we
145# match the dependencies shown in the subdirectory makefiles!
146# Also, in cases where a subdirectory makefile generates two files in
147# what's really one step, such as bison producing both gram.h and gram.c,
148# we must request making the one that is shown as the secondary (dependent)
149# output, else the timestamp on it might be wrong.  By project convention,
150# the .h file is the dependent one for bison output, so we need only request
151# that; but in other cases, request both for safety.
152
153parser/gram.h: parser/gram.y
154	$(MAKE) -C parser gram.h
155
156storage/lmgr/lwlocknames.h: storage/lmgr/generate-lwlocknames.pl ../include/storage/lwlocklist.h utils/activity/wait_event_names.txt
157	$(MAKE) -C storage/lmgr lwlocknames.h
158
159# run this unconditionally to avoid needing to know its dependencies here:
160submake-catalog-headers:
161	$(MAKE) -C ../include/catalog generated-headers
162

164submake-nodes-headers:
165	$(MAKE) -C nodes generated-header-symlinks
166

168submake-utils-headers:
169	$(MAKE) -C utils generated-header-symlinks

171.PHONY: submake-catalog-headers submake-nodes-headers submake-utils-headers
172
173# Make symlinks for these headers in the include directory. That way
174# we can cut down on the -I options. Also, a symlink is automatically
175# up to date when we update the base file.
176#
177# The point of the prereqdir incantation in some of the rules below is to
178# force the symlink to use an absolute path rather than a relative path.
179# This is needed to support platforms without ln -s.
180
181.PHONY: generated-headers
182
183generated-headers: $(top_builddir)/src/include/storage/lwlocknames.h submake-catalog-headers submake-nodes-headers submake-utils-headers parser/gram.h
184
185$(top_builddir)/src/include/storage/lwlocknames.h: storage/lmgr/lwlocknames.h
186	prereqdir=`cd '$(dir $<)' >/dev/null && pwd` && \
187	  cd '$(dir $@)' && rm -f $(notdir $@) && \
188	  $(LN_S) "$$prereqdir/$(notdir $<)" .
189
190utils/probes.o: utils/probes.d $(SUBDIROBJS)
191	$(DTRACE) $(DTRACEFLAGS) -C -G -s $(call expand_subsys,$^) -o $@
