src/backend/port/aix/mkldexport.sh
1#!/bin/sh
2#
3# mkldexport
4#	create an AIX exports file from an object file
5#
6# src/backend/port/aix/mkldexport.sh
7#
8# Usage:
9#	mkldexport objectfile [location]
10# where
11#	objectfile is the current location of the object file.
12#	location is the eventual (installed) location of the
13#		object file (if different from the current
14#		working directory).
15#
16# On AIX, executables do not automatically expose their symbols to shared
17# modules. Extensions therefore cannot call functions in the main Postgres
18# binary unless those symbols are explicitly exported. Unlike other platforms,
19# AIX executables are not default symbol providers; each shared module must
20# link against an export list that defines which symbols it can use.
21#
22# The mkldexport.sh script fixes AIX's symbol export issue by generating an
23# explicit export list. It uses nm to gather all symbols from the Postgres
24# object files, then writes them into the export file. When invoked with ".",
25# it outputs #! ., which tells AIX the list applies to the main executable.
26# This way, extension modules can link against that list and resolve their
27# undefined symbols directly from the Postgres binary.
28#
29
30# Search for the nm command binary.
31if [ -x /usr/ucb/nm ]
32then NM=/usr/ucb/nm
33elif [ -x /usr/bin/nm ]
34then NM=/usr/bin/nm
35elif [ -x /usr/ccs/bin/nm ]
36then NM=/usr/ccs/bin/nm
37elif [ -x /usr/usg/bin/nm ]
38then NM=/usr/usg/bin/nm
39else echo "Fatal error: cannot find `nm' ... please check your installation."
40     exit 1
41fi
42
43# instruct nm to process 64-bit objects
44export OBJECT_MODE=64
45
46CMDNAME=`basename $0`
47if [ -z "$1" ]; then
48	echo "Usage: $CMDNAME object [location]"
49	exit 1
50fi
51OBJNAME=`basename $1`
52if [ "`basename $OBJNAME`" != "`basename $OBJNAME .o`" ]; then
53	OBJNAME=`basename $OBJNAME .o`.so
54fi
55if [ -z "$2" ]; then
56	echo '#!'
57else
58	if [ "$2" = "." ]; then
59		# for the base executable (AIX 4.2 and up)
60		echo '#! .'
61	else
62		echo '#!' $2
63	fi
64fi
65$NM -BCg $1 | \
66	grep ' [TDB] ' | \
67	sed -e 's/.* //' | \
68	grep -v '\$' | \
69	sed -e 's/^[.]//' | \
70	sort | \
71	uniq
