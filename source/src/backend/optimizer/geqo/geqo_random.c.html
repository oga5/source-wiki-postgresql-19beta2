14#include "postgres.h"
15
16#include "optimizer/geqo_random.h"
17
18void
19geqo_set_seed(PlannerInfo *root, double seed)
20{
21	GeqoPrivateData *private = GetGeqoPrivateData(root);
22
23	pg_prng_fseed(&private->random_state, seed);
24}
25
26double
27geqo_rand(PlannerInfo *root)
28{
29	GeqoPrivateData *private = GetGeqoPrivateData(root);
30
31	return pg_prng_double(&private->random_state);
32}
33
34int
35geqo_randint(PlannerInfo *root, int upper, int lower)
36{
37	GeqoPrivateData *private = GetGeqoPrivateData(root);
38
39	/*
40	 * In current usage, "lower" is never negative so we can just use
41	 * pg_prng_uint64_range directly.
42	 */
43	return (int) pg_prng_uint64_range(&private->random_state, lower, upper);
44}
