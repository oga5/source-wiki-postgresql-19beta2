src/backend/storage/lmgr/generate-lwlocknames.pl
1#!/usr/bin/perl
2#
3# Generate lwlocknames.h from lwlocklist.h
4# Copyright (c) 2000-2026, PostgreSQL Global Development Group
5
6use strict;
7use warnings FATAL => 'all';
8use Getopt::Long;
9
10my $output_path = '.';
11
12my $lastlockidx = -1;
13
14GetOptions('outdir:s' => \$output_path);
15
16open my $lwlocklist, '<', $ARGV[0] or die;
17open my $wait_event_names, '<', $ARGV[1] or die;
18
19# Include PID in suffix in case parallel make runs this multiple times.
20my $htmp = "$output_path/lwlocknames.h.tmp$$";
21open my $h, '>', $htmp or die "Could not open $htmp: $!";
22
23my $autogen =
24  "/* autogenerated from src/include/storage/lwlocklist.h, do not edit */\n";
25print $h $autogen;
26print $h "/* there is deliberately not an #ifndef LWLOCKNAMES_H here */\n\n";
27
28
29#
30# First, record the predefined LWLocks and built-in tranches listed in
31# wait_event_names.txt.  We'll cross-check those with the ones in lwlocklist.h.
32#
33my @wait_event_tranches;
34my @wait_event_lwlocks;
35my $record_lwlocks = 0;
36my $in_tranches = 0;
37
38while (<$wait_event_names>)
39{
40	chomp;
41
42	# Check for end marker.
43	if (/^# END OF PREDEFINED LWLOCKS/)
44	{
45		$in_tranches = 1;
46		next;
47	}
48
49	# Skip comments and empty lines.
50	next if /^#/;
51	next if /^\s*$/;
52
53	# Start recording LWLocks when we find the WaitEventLWLock section.
54	if (/^Section: ClassName - WaitEventLWLock$/)
55	{
56		$record_lwlocks = 1;
57		next;
58	}
59
60	# Go to the next line if we are not yet recording LWLocks.
61	next if not $record_lwlocks;
62
63	# Stop recording if we reach another section.
64	last if /^Section:/;
65
66	# Record the LWLock.
67	(my $waiteventname, my $waitevendocsentence) = split(/\t/, $_);
68
69	if ($in_tranches)
70	{
71		push(@wait_event_tranches, $waiteventname);
72	}
73	else
74	{
75		push(@wait_event_lwlocks, $waiteventname);
76	}
77}
78
79#
80# While gathering the list of predefined LWLocks, cross-check the lists in
81# lwlocklist.h with the wait events we just recorded.
82#
83my $in_comment = 0;
84my $lwlock_count = 0;
85my $tranche_count = 0;
86while (<$lwlocklist>)
87{
88	chomp;
89
90	# Skip single-line C comments and empty lines
91	next if m{^\s*/\*.*\*/$};
92	next if /^\s*$/;
93
94	# skip multiline C comments
95	if ($in_comment == 1)
96	{
97		$in_comment = 0 if m{\*/};
98		next;
99	}
100	elsif (m{^\s*/\*})
101	{
102		$in_comment = 1;
103		next;
104	}
105
106	#
107	# Gather list of predefined LWLocks and cross-check with the wait events.
108	#
109	if (/^PG_LWLOCK\((\d+),\s+(\w+)\)$/)
110	{
111		my ($lockidx, $lockname) = ($1, $2);
112
113		die "lwlocklist.h not in order" if $lockidx < $lastlockidx;
114		die "lwlocklist.h has duplicates" if $lockidx == $lastlockidx;
115
116		die "$lockname defined in lwlocklist.h but missing from "
117		  . "wait_event_names.txt"
118		  if $lwlock_count >= scalar @wait_event_lwlocks;
119		die "lists of predefined LWLocks do not match (first mismatch at "
120		  . "$wait_event_lwlocks[$lwlock_count] in wait_event_names.txt and "
121		  . "$lockname in lwlocklist.h)"
122		  if $wait_event_lwlocks[$lwlock_count] ne $lockname;
123
124		$lwlock_count++;
125
126		while ($lastlockidx < $lockidx - 1)
127		{
128			++$lastlockidx;
129		}
130		$lastlockidx = $lockidx;
131
132		# Add a "Lock" suffix to each lock name, as the C code depends on that.
133		printf $h "#define %-32s (&MainLWLockArray[$lockidx].lock)\n",
134		  $lockname . "Lock";
135
136		next;
137	}
138
139	#
140	# Cross-check the built-in LWLock tranches with the wait events.
141	#
142	if (/^PG_LWLOCKTRANCHE\((\w+),\s+(\w+)\)$/)
143	{
144		my ($tranche_id, $tranche_name) = ($1, $2);
145
146		die "$tranche_name defined in lwlocklist.h but missing from "
147		  . "wait_event_names.txt"
148		  if $tranche_count >= scalar @wait_event_tranches;
149		die
150		  "lists of built-in LWLock tranches do not match (first mismatch at "
151		  . "$wait_event_tranches[$tranche_count] in wait_event_names.txt and "
152		  . "$tranche_name in lwlocklist.h)"
153		  if $wait_event_tranches[$tranche_count] ne $tranche_name;
154
155		$tranche_count++;
156
157		next;
158	}
159
160	die "unable to parse lwlocklist.h line \"$_\"";
161}
162
163die "$wait_event_lwlocks[$lwlock_count] defined in wait_event_names.txt but "
164  . " missing from lwlocklist.h"
165  if $lwlock_count < scalar @wait_event_lwlocks;
166
167die
168  "$wait_event_tranches[$tranche_count] defined in wait_event_names.txt but "
169  . "missing from lwlocklist.h"
170  if $tranche_count < scalar @wait_event_tranches;
171
172print $h "\n";
173printf $h "#define NUM_INDIVIDUAL_LWLOCKS		%s\n", $lastlockidx + 1;
174
175close $h;
176
177rename($htmp, "$output_path/lwlocknames.h")
178  || die "rename: $htmp to $output_path/lwlocknames.h: $!";
179
180close $lwlocklist;
