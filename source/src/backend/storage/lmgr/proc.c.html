101/*
102 * Calculate shared-memory space needed by Fast-Path locks.
103 */
104static Size
105CalculateFastPathLockShmemSize(void)
106{
107	Size		size = 0;
108	Size		fpLockBitsSize,
109				fpRelIdSize;
110
111	/*
112	 * Memory needed for PGPROC fast-path lock arrays. Make sure the sizes are
113	 * nicely aligned in each backend.
114	 */
115	fpLockBitsSize = MAXALIGN(FastPathLockGroupsPerBackend * sizeof(uint64));
116	fpRelIdSize = MAXALIGN(FastPathLockSlotsPerBackend() * sizeof(Oid));
117
118	size = add_size(size, mul_size(TotalProcs, (fpLockBitsSize + fpRelIdSize)));
119
120	Assert(TotalProcs > 0);
121	Assert(size > 0);
122
123	return size;
124}

126/*
127 * Report number of semaphores needed by ProcGlobalShmemInit.
128 */
129int
130ProcGlobalSemas(void)
131{
132	/*
133	 * We need a sema per backend (including autovacuum), plus one for each
134	 * auxiliary process.
135	 */
136	return MaxBackends + NUM_AUXILIARY_PROCS;
137}

139/*
140 * ProcGlobalShmemRequest
141 *	  Register shared memory needs.
142 *
143 * This is called during postmaster or standalone backend startup, and also
144 * during backend startup in EXEC_BACKEND mode.
145 */
146static void
147ProcGlobalShmemRequest(void *arg)
148{
149	Size		size;
150
151	/*
152	 * Reserve all the PGPROC structures we'll need.  There are six separate
153	 * consumers: (1) normal backends, (2) autovacuum workers and special
154	 * workers, (3) background workers, (4) walsenders, (5) auxiliary
155	 * processes, and (6) prepared transactions.  (For largely-historical
156	 * reasons, we combine autovacuum and special workers into one category
157	 * with a single freelist.)  Each PGPROC structure is dedicated to exactly
158	 * one of these purposes, and they do not move between groups.
159	 */
160	TotalProcs =
161		add_size(MaxBackends, add_size(NUM_AUXILIARY_PROCS, max_prepared_xacts));
162
163	size = 0;
164	size = add_size(size, mul_size(TotalProcs, sizeof(PGPROC)));
165	size = add_size(size, mul_size(TotalProcs, sizeof(*ProcGlobal->xids)));
166	size = add_size(size, mul_size(TotalProcs, sizeof(*ProcGlobal->subxidStates)));
167	size = add_size(size, mul_size(TotalProcs, sizeof(*ProcGlobal->statusFlags)));
168	ProcGlobalAllProcsShmemSize = size;
169	ShmemRequestStruct(.name = "PGPROC structures",
170					   .size = ProcGlobalAllProcsShmemSize,
171					   .ptr = &AllProcsShmemPtr,
172		);
173
174	if (!IsUnderPostmaster)
175		size = FastPathLockArrayShmemSize = CalculateFastPathLockShmemSize();
176	else
177		size = SHMEM_ATTACH_UNKNOWN_SIZE;
178	ShmemRequestStruct(.name = "Fast-Path Lock Array",
179					   .size = size,
180					   .ptr = &FastPathLockArrayShmemPtr,
181		);
182
183	/*
184	 * ProcGlobal is registered here in .ptr as usual, but it needs to be
185	 * propagated specially in EXEC_BACKEND mode, because ProcGlobal needs to
186	 * be accessed early at backend startup, before ShmemAttachRequested() has
187	 * been called.
188	 */
189	ShmemRequestStruct(.name = "Proc Header",
190					   .size = sizeof(PROC_HDR),
191					   .ptr = (void **) &ProcGlobal,
192		);
193
194	/* Let the semaphore implementation register its shared memory needs */
195	PGSemaphoreShmemRequest(ProcGlobalSemas());
196}

199/*
200 * ProcGlobalShmemInit -
201 *	  Initialize the global process table during postmaster or standalone
202 *	  backend startup.
203 *
204 *	  We also create all the per-process semaphores we will need to support
205 *	  the requested number of backends.  We used to allocate semaphores
206 *	  only when backends were actually started up, but that is bad because
207 *	  it lets Postgres fail under load --- a lot of Unix systems are
208 *	  (mis)configured with small limits on the number of semaphores, and
209 *	  running out when trying to start another backend is a common failure.
210 *	  So, now we grab enough semaphores to support the desired max number
211 *	  of backends immediately at initialization --- if the sysadmin has set
212 *	  MaxConnections, max_worker_processes, max_wal_senders, or
213 *	  autovacuum_worker_slots higher than his kernel will support, he'll
214 *	  find out sooner rather than later.
215 *
216 *	  Another reason for creating semaphores here is that the semaphore
217 *	  implementation typically requires us to create semaphores in the
218 *	  postmaster, not in backends.
219 */
220static void
221ProcGlobalShmemInit(void *arg)
222{
223	char	   *ptr;
224	size_t		requestSize;
225	PGPROC	   *procs;
226	int			i,
227				j;
228
229	/* Used for setup of per-backend fast-path slots. */
230	char	   *fpPtr,
231			   *fpEndPtr PG_USED_FOR_ASSERTS_ONLY;
232	Size		fpLockBitsSize,
233				fpRelIdSize;
234
235	Assert(ProcGlobal);
236	ProcGlobal->spins_per_delay = DEFAULT_SPINS_PER_DELAY;
237	SpinLockInit(&ProcGlobal->freeProcsLock);
238	dlist_init(&ProcGlobal->freeProcs);
239	dlist_init(&ProcGlobal->autovacFreeProcs);
240	dlist_init(&ProcGlobal->bgworkerFreeProcs);
241	dlist_init(&ProcGlobal->walsenderFreeProcs);
242	ProcGlobal->startupBufferPinWaitBufId = -1;
243	ProcGlobal->walwriterProc = INVALID_PROC_NUMBER;
244	ProcGlobal->checkpointerProc = INVALID_PROC_NUMBER;
245	pg_atomic_init_u32(&ProcGlobal->procArrayGroupFirst, INVALID_PROC_NUMBER);
246	pg_atomic_init_u32(&ProcGlobal->clogGroupFirst, INVALID_PROC_NUMBER);
247
248	ptr = AllProcsShmemPtr;
249	requestSize = ProcGlobalAllProcsShmemSize;
250	MemSet(ptr, 0, requestSize);
251
252	/* Carve out the allProcs array from the shared memory area */
253	procs = (PGPROC *) ptr;
254	ptr = ptr + TotalProcs * sizeof(PGPROC);
255
256	ProcGlobal->allProcs = procs;
257	/* XXX allProcCount isn't really all of them; it excludes prepared xacts */
258	ProcGlobal->allProcCount = MaxBackends + NUM_AUXILIARY_PROCS;
259
260	/*
261	 * Carve out arrays mirroring PGPROC fields in a dense manner. See
262	 * PROC_HDR.
263	 *
264	 * XXX: It might make sense to increase padding for these arrays, given
265	 * how hotly they are accessed.
266	 */
267	ProcGlobal->xids = (TransactionId *) ptr;
268	ptr = ptr + (TotalProcs * sizeof(*ProcGlobal->xids));
269
270	ProcGlobal->subxidStates = (XidCacheStatus *) ptr;
271	ptr = ptr + (TotalProcs * sizeof(*ProcGlobal->subxidStates));
272
273	ProcGlobal->statusFlags = (uint8 *) ptr;
274	ptr = ptr + (TotalProcs * sizeof(*ProcGlobal->statusFlags));
275
276	/* make sure we didn't overflow */
277	Assert((ptr > (char *) procs) && (ptr <= (char *) procs + requestSize));
278
279	/*
280	 * Initialize arrays for fast-path locks. Those are variable-length, so
281	 * can't be included in PGPROC directly. We allocate a separate piece of
282	 * shared memory and then divide that between backends.
283	 */
284	fpLockBitsSize = MAXALIGN(FastPathLockGroupsPerBackend * sizeof(uint64));
285	fpRelIdSize = MAXALIGN(FastPathLockSlotsPerBackend() * sizeof(Oid));
286
287	fpPtr = FastPathLockArrayShmemPtr;
288	requestSize = FastPathLockArrayShmemSize;
289	memset(fpPtr, 0, requestSize);
290
291	/* For asserts checking we did not overflow. */
292	fpEndPtr = fpPtr + requestSize;
293
294	/* Initialize semaphores */
295	PGSemaphoreInit(ProcGlobalSemas());
296
297	for (i = 0; i < TotalProcs; i++)
298	{
299		PGPROC	   *proc = &procs[i];
300
301		/* Common initialization for all PGPROCs, regardless of type. */
302
303		/*
304		 * Set the fast-path lock arrays, and move the pointer. We interleave
305		 * the two arrays, to (hopefully) get some locality for each backend.
306		 */
307		proc->fpLockBits = (uint64 *) fpPtr;
308		fpPtr += fpLockBitsSize;
309
310		proc->fpRelId = (Oid *) fpPtr;
311		fpPtr += fpRelIdSize;
312
313		Assert(fpPtr <= fpEndPtr);
314
315		/*
316		 * Set up per-PGPROC semaphore, latch, and fpInfoLock.  Prepared xact
317		 * dummy PGPROCs don't need these though - they're never associated
318		 * with a real process
319		 */
320		if (i < FIRST_PREPARED_XACT_PROC_NUMBER)
321		{
322			proc->sem = PGSemaphoreCreate();
323			InitSharedLatch(&(proc->procLatch));
324			LWLockInitialize(&(proc->fpInfoLock), LWTRANCHE_LOCK_FASTPATH);
325		}
326
327		/*
328		 * Newly created PGPROCs for normal backends, autovacuum workers,
329		 * special workers, bgworkers, and walsenders must be queued up on the
330		 * appropriate free list.  Because there can only ever be a small,
331		 * fixed number of auxiliary processes, no free list is used in that
332		 * case; InitAuxiliaryProcess() instead uses a linear search.  PGPROCs
333		 * for prepared transactions are added to a free list by
334		 * TwoPhaseShmemInit().
335		 */
336		if (i < MaxConnections)
337		{
338			/* PGPROC for normal backend, add to freeProcs list */
339			dlist_push_tail(&ProcGlobal->freeProcs, &proc->freeProcsLink);
340			proc->procgloballist = &ProcGlobal->freeProcs;
341		}
342		else if (i < MaxConnections + autovacuum_worker_slots + NUM_SPECIAL_WORKER_PROCS)
343		{
344			/* PGPROC for AV or special worker, add to autovacFreeProcs list */
345			dlist_push_tail(&ProcGlobal->autovacFreeProcs, &proc->freeProcsLink);
346			proc->procgloballist = &ProcGlobal->autovacFreeProcs;
347		}
348		else if (i < MaxConnections + autovacuum_worker_slots + NUM_SPECIAL_WORKER_PROCS + max_worker_processes)
349		{
350			/* PGPROC for bgworker, add to bgworkerFreeProcs list */
351			dlist_push_tail(&ProcGlobal->bgworkerFreeProcs, &proc->freeProcsLink);
352			proc->procgloballist = &ProcGlobal->bgworkerFreeProcs;
353		}
354		else if (i < MaxBackends)
355		{
356			/* PGPROC for walsender, add to walsenderFreeProcs list */
357			dlist_push_tail(&ProcGlobal->walsenderFreeProcs, &proc->freeProcsLink);
358			proc->procgloballist = &ProcGlobal->walsenderFreeProcs;
359		}
360
361		/* Initialize myProcLocks[] shared memory queues. */
362		for (j = 0; j < NUM_LOCK_PARTITIONS; j++)
363			dlist_init(&(proc->myProcLocks[j]));
364
365		/* Initialize lockGroupMembers list. */
366		dlist_init(&proc->lockGroupMembers);
367
368		/*
369		 * Initialize the atomic variables, otherwise, it won't be safe to
370		 * access them for backends that aren't currently in use.
371		 */
372		pg_atomic_init_u32(&(proc->procArrayGroupNext), INVALID_PROC_NUMBER);
373		pg_atomic_init_u32(&(proc->clogGroupNext), INVALID_PROC_NUMBER);
374		pg_atomic_init_u64(&(proc->waitStart), 0);
375	}
376
377	/* Should have consumed exactly the expected amount of fast-path memory. */
378	Assert(fpPtr == fpEndPtr);
379
380	/*
381	 * Save pointers to the blocks of PGPROC structures reserved for auxiliary
382	 * processes and prepared transactions.
383	 */
384	AuxiliaryProcs = &procs[MaxBackends];
385	PreparedXactProcs = &procs[FIRST_PREPARED_XACT_PROC_NUMBER];
386}

388/*
389 * InitProcess -- initialize a per-process PGPROC entry for this backend
390 */
391void
392InitProcess(void)
393{
394	dlist_head *procgloballist;
395
396	/*
397	 * ProcGlobal should be set up already (if we are a backend, we inherit
398	 * this by fork() or EXEC_BACKEND mechanism from the postmaster).
399	 */
400	if (ProcGlobal == NULL)
401		elog(PANIC, "proc header uninitialized");
402
403	if (MyProc != NULL)
404		elog(ERROR, "you already exist");
405
406	/*
407	 * Before we start accessing the shared memory in a serious way, mark
408	 * ourselves as an active postmaster child; this is so that the postmaster
409	 * can detect it if we exit without cleaning up.
410	 */
411	if (IsUnderPostmaster)
412		RegisterPostmasterChildActive();
413
414	/*
415	 * Decide which list should supply our PGPROC.  This logic must match the
416	 * way the freelists were constructed in ProcGlobalShmemInit().
417	 */
418	if (AmAutoVacuumWorkerProcess() || AmSpecialWorkerProcess())
419		procgloballist = &ProcGlobal->autovacFreeProcs;
420	else if (AmBackgroundWorkerProcess())
421		procgloballist = &ProcGlobal->bgworkerFreeProcs;
422	else if (AmWalSenderProcess())
423		procgloballist = &ProcGlobal->walsenderFreeProcs;
424	else
425		procgloballist = &ProcGlobal->freeProcs;
426
427	/*
428	 * Try to get a proc struct from the appropriate free list.  If this
429	 * fails, we must be out of PGPROC structures (not to mention semaphores).
430	 *
431	 * While we are holding the spinlock, also copy the current shared
432	 * estimate of spins_per_delay to local storage.
433	 */
434	SpinLockAcquire(&ProcGlobal->freeProcsLock);
435
436	set_spins_per_delay(ProcGlobal->spins_per_delay);
437
438	if (!dlist_is_empty(procgloballist))
439	{
440		MyProc = dlist_container(PGPROC, freeProcsLink, dlist_pop_head_node(procgloballist));
441		SpinLockRelease(&ProcGlobal->freeProcsLock);
442	}
443	else
444	{
445		/*
446		 * If we reach here, all the PGPROCs are in use.  This is one of the
447		 * possible places to detect "too many backends", so give the standard
448		 * error message.  XXX do we need to give a different failure message
449		 * in the autovacuum case?
450		 */
451		SpinLockRelease(&ProcGlobal->freeProcsLock);
452		if (AmWalSenderProcess())
453			ereport(FATAL,
454					(errcode(ERRCODE_TOO_MANY_CONNECTIONS),
455					 errmsg("number of requested standby connections exceeds \"max_wal_senders\" (currently %d)",
456							max_wal_senders)));
457		ereport(FATAL,
458				(errcode(ERRCODE_TOO_MANY_CONNECTIONS),
459				 errmsg("sorry, too many clients already")));
460	}
461	MyProcNumber = GetNumberFromPGProc(MyProc);
462
463	/*
464	 * Cross-check that the PGPROC is of the type we expect; if this were not
465	 * the case, it would get returned to the wrong list.
466	 */
467	Assert(MyProc->procgloballist == procgloballist);
468
469	/*
470	 * Initialize all fields of MyProc, except for those previously
471	 * initialized by ProcGlobalShmemInit.
472	 */
473	dlist_node_init(&MyProc->freeProcsLink);
474	MyProc->waitStatus = PROC_WAIT_STATUS_OK;
475	MyProc->fpVXIDLock = false;
476	MyProc->fpLocalTransactionId = InvalidLocalTransactionId;
477	MyProc->xid = InvalidTransactionId;
478	MyProc->xmin = InvalidTransactionId;
479	MyProc->pid = MyProcPid;
480	MyProc->vxid.procNumber = MyProcNumber;
481	MyProc->vxid.lxid = InvalidLocalTransactionId;
482	/* databaseId and roleId will be filled in later */
483	MyProc->databaseId = InvalidOid;
484	MyProc->roleId = InvalidOid;
485	MyProc->tempNamespaceId = InvalidOid;
486	MyProc->backendType = MyBackendType;
487	MyProc->delayChkptFlags = 0;
488	MyProc->statusFlags = 0;
489	/* NB -- autovac launcher intentionally does not set IS_AUTOVACUUM */
490	if (AmAutoVacuumWorkerProcess())
491		MyProc->statusFlags |= PROC_IS_AUTOVACUUM;
492	MyProc->lwWaiting = LW_WS_NOT_WAITING;
493	MyProc->lwWaitMode = 0;
494	MyProc->waitLock = NULL;
495	dlist_node_init(&MyProc->waitLink);
496	MyProc->waitProcLock = NULL;
497	pg_atomic_write_u64(&MyProc->waitStart, 0);
498#ifdef USE_ASSERT_CHECKING
499	{
500		int			i;
501
502		/* Last process should have released all locks. */
503		for (i = 0; i < NUM_LOCK_PARTITIONS; i++)
504			Assert(dlist_is_empty(&(MyProc->myProcLocks[i])));
505	}
506#endif
507	pg_atomic_write_u32(&MyProc->pendingRecoveryConflicts, 0);
508
509	/* Initialize fields for sync rep */
510	MyProc->waitLSN = InvalidXLogRecPtr;
511	MyProc->syncRepState = SYNC_REP_NOT_WAITING;
512	dlist_node_init(&MyProc->syncRepLinks);
513
514	/* Initialize fields for group XID clearing. */
515	MyProc->procArrayGroupMember = false;
516	MyProc->procArrayGroupMemberXid = InvalidTransactionId;
517	Assert(pg_atomic_read_u32(&MyProc->procArrayGroupNext) == INVALID_PROC_NUMBER);
518
519	/* Check that group locking fields are in a proper initial state. */
520	Assert(MyProc->lockGroupLeader == NULL);
521	Assert(dlist_is_empty(&MyProc->lockGroupMembers));
522
523	/* Initialize wait event information. */
524	MyProc->wait_event_info = 0;
525
526	/* Initialize fields for group transaction status update. */
527	MyProc->clogGroupMember = false;
528	MyProc->clogGroupMemberXid = InvalidTransactionId;
529	MyProc->clogGroupMemberXidStatus = TRANSACTION_STATUS_IN_PROGRESS;
530	MyProc->clogGroupMemberPage = -1;
531	MyProc->clogGroupMemberLsn = InvalidXLogRecPtr;
532	Assert(pg_atomic_read_u32(&MyProc->clogGroupNext) == INVALID_PROC_NUMBER);
533
534	/*
535	 * Acquire ownership of the PGPROC's latch, so that we can use WaitLatch
536	 * on it.  That allows us to repoint the process latch, which so far
537	 * points to process local one, to the shared one.
538	 */
539	OwnLatch(&MyProc->procLatch);
540	SwitchToSharedLatch();
541
542	/* now that we have a proc, report wait events to shared memory */
543	pgstat_set_wait_event_storage(&MyProc->wait_event_info);
544
545	/*
546	 * We might be reusing a semaphore that belonged to a failed process. So
547	 * be careful and reinitialize its value here.  (This is not strictly
548	 * necessary anymore, but seems like a good idea for cleanliness.)
549	 */
550	PGSemaphoreReset(MyProc->sem);
551
552	/*
553	 * Arrange to clean up at backend exit.
554	 */
555	on_shmem_exit(ProcKill, 0);
556
557	/*
558	 * Now that we have a PGPROC, we could try to acquire locks, so initialize
559	 * local state needed for LWLocks, and the deadlock checker.
560	 */
561	InitLWLockAccess();
562	InitDeadLockChecking();
563
564#ifdef EXEC_BACKEND
565
566	/*
567	 * Initialize backend-local pointers to all the shared data structures.
568	 * (We couldn't do this until now because it needs LWLocks.)
569	 */
570	if (IsUnderPostmaster)
571		AttachSharedMemoryStructs();
572#endif
573}

575/*
576 * InitProcessPhase2 -- make MyProc visible in the shared ProcArray.
577 *
578 * This is separate from InitProcess because we can't acquire LWLocks until
579 * we've created a PGPROC, but in the EXEC_BACKEND case ProcArrayAdd won't
580 * work until after we've done AttachSharedMemoryStructs.
581 */
582void
583InitProcessPhase2(void)
584{
585	Assert(MyProc != NULL);
586
587	/*
588	 * Add our PGPROC to the PGPROC array in shared memory.

590	ProcArrayAdd(MyProc);
591
592	/*
593	 * Arrange to clean that up at backend exit.
594	 */
595	on_shmem_exit(RemoveProcFromArray, 0);
596}

598/*
599 * InitAuxiliaryProcess -- create a PGPROC entry for an auxiliary process
600 *
601 * This is called by bgwriter and similar processes so that they will have a
602 * MyProc value that's real enough to let them wait for LWLocks.  The PGPROC
603 * and sema that are assigned are one of the extra ones created during
604 * ProcGlobalShmemInit.
605 *
606 * Auxiliary processes are presently not expected to wait for real (lockmgr)
607 * locks, so we need not set up the deadlock checker.  They are never added
608 * to the ProcArray or the sinval messaging mechanism, either.  They also
609 * don't get a VXID assigned, since this is only useful when we actually
610 * hold lockmgr locks.
611 *
612 * Startup process however uses locks but never waits for them in the
613 * normal backend sense. Startup process also takes part in sinval messaging
614 * as a sendOnly process, so never reads messages from sinval queue. So
615 * Startup process does have a VXID and does show up in pg_locks.
616 */
617void
618InitAuxiliaryProcess(void)
619{
620	PGPROC	   *auxproc;
621	int			proctype;
622
623	/*
624	 * ProcGlobal should be set up already (if we are a backend, we inherit
625	 * this by fork() or EXEC_BACKEND mechanism from the postmaster).
626	 */
627	if (ProcGlobal == NULL || AuxiliaryProcs == NULL)
628		elog(PANIC, "proc header uninitialized");
629
630	if (MyProc != NULL)
631		elog(ERROR, "you already exist");
632
633	if (IsUnderPostmaster)
634		RegisterPostmasterChildActive();
635
636	/*
637	 * We use the freeProcsLock to protect assignment and releasing of
638	 * AuxiliaryProcs entries.
639	 *
640	 * While we are holding the spinlock, also copy the current shared
641	 * estimate of spins_per_delay to local storage.
642	 */
643	SpinLockAcquire(&ProcGlobal->freeProcsLock);
644
645	set_spins_per_delay(ProcGlobal->spins_per_delay);
646
647	/*
648	 * Find a free auxproc ... *big* trouble if there isn't one ...
649	 */
650	for (proctype = 0; proctype < NUM_AUXILIARY_PROCS; proctype++)
651	{
652		auxproc = &AuxiliaryProcs[proctype];
653		if (auxproc->pid == 0)
654			break;
655	}
656	if (proctype >= NUM_AUXILIARY_PROCS)
657	{
658		SpinLockRelease(&ProcGlobal->freeProcsLock);
659		elog(FATAL, "all AuxiliaryProcs are in use");
660	}
661
662	/* Mark auxiliary proc as in use by me */
663	/* use volatile pointer to prevent code rearrangement */
664	((volatile PGPROC *) auxproc)->pid = MyProcPid;
665
666	SpinLockRelease(&ProcGlobal->freeProcsLock);
667
668	MyProc = auxproc;
669	MyProcNumber = GetNumberFromPGProc(MyProc);
670
671	/*
672	 * Initialize all fields of MyProc, except for those previously
673	 * initialized by ProcGlobalShmemInit.
674	 */
675	dlist_node_init(&MyProc->freeProcsLink);
676	MyProc->waitStatus = PROC_WAIT_STATUS_OK;
677	MyProc->fpVXIDLock = false;
678	MyProc->fpLocalTransactionId = InvalidLocalTransactionId;
679	MyProc->xid = InvalidTransactionId;
680	MyProc->xmin = InvalidTransactionId;
681	MyProc->vxid.procNumber = INVALID_PROC_NUMBER;
682	MyProc->vxid.lxid = InvalidLocalTransactionId;
683	MyProc->databaseId = InvalidOid;
684	MyProc->roleId = InvalidOid;
685	MyProc->tempNamespaceId = InvalidOid;
686	MyProc->backendType = MyBackendType;
687	MyProc->delayChkptFlags = 0;
688	MyProc->statusFlags = 0;
689	MyProc->lwWaiting = LW_WS_NOT_WAITING;
690	MyProc->lwWaitMode = 0;
691	MyProc->waitLock = NULL;
692	dlist_node_init(&MyProc->waitLink);
693	MyProc->waitProcLock = NULL;
694	pg_atomic_write_u64(&MyProc->waitStart, 0);
695#ifdef USE_ASSERT_CHECKING
696	{
697		int			i;
698
699		/* Last process should have released all locks. */
700		for (i = 0; i < NUM_LOCK_PARTITIONS; i++)
701			Assert(dlist_is_empty(&(MyProc->myProcLocks[i])));
702	}
703#endif
704	pg_atomic_write_u32(&MyProc->pendingRecoveryConflicts, 0);
705
706	/*
707	 * Acquire ownership of the PGPROC's latch, so that we can use WaitLatch
708	 * on it.  That allows us to repoint the process latch, which so far
709	 * points to process local one, to the shared one.
710	 */
711	OwnLatch(&MyProc->procLatch);
712	SwitchToSharedLatch();
713
714	/* now that we have a proc, report wait events to shared memory */
715	pgstat_set_wait_event_storage(&MyProc->wait_event_info);
716
717	/* Check that group locking fields are in a proper initial state. */
718	Assert(MyProc->lockGroupLeader == NULL);
719	Assert(dlist_is_empty(&MyProc->lockGroupMembers));
720
721	/*
722	 * We might be reusing a semaphore that belonged to a failed process. So
723	 * be careful and reinitialize its value here.  (This is not strictly
724	 * necessary anymore, but seems like a good idea for cleanliness.)
725	 */
726	PGSemaphoreReset(MyProc->sem);
727
728	/*
729	 * Arrange to clean up at process exit.
730	 */
731	on_shmem_exit(AuxiliaryProcKill, Int32GetDatum(proctype));
732
733	/*
734	 * Now that we have a PGPROC, we could try to acquire lightweight locks.
735	 * Initialize local state needed for them.  (Heavyweight locks cannot be
736	 * acquired in aux processes.)
737	 */
738	InitLWLockAccess();
739
740#ifdef EXEC_BACKEND
741
742	/*
743	 * Initialize backend-local pointers to all the shared data structures.
744	 * (We couldn't do this until now because it needs LWLocks.)
745	 */
746	if (IsUnderPostmaster)
747		AttachSharedMemoryStructs();
748#endif
749}

809/*
810 * Cancel any pending wait for lock, when aborting a transaction, and revert
811 * any strong lock count acquisition for a lock being acquired.
812 *
813 * (Normally, this would only happen if we accept a cancel/die
814 * interrupt while waiting; but an ereport(ERROR) before or during the lock
815 * wait is within the realm of possibility, too.)
816 */
817void
818LockErrorCleanup(void)
819{
820	LOCALLOCK  *lockAwaited;
821	LWLock	   *partitionLock;
822	DisableTimeoutParams timeouts[2];
823
824	HOLD_INTERRUPTS();
825
826	AbortStrongLockAcquire();
827
828	/* Nothing to do if we weren't waiting for a lock */
829	lockAwaited = GetAwaitedLock();
830	if (lockAwaited == NULL)
831	{
832		RESUME_INTERRUPTS();
833		return;
834	}
835
836	/*
837	 * Turn off the deadlock and lock timeout timers, if they are still
838	 * running (see ProcSleep).  Note we must preserve the LOCK_TIMEOUT
839	 * indicator flag, since this function is executed before
840	 * ProcessInterrupts when responding to SIGINT; else we'd lose the
841	 * knowledge that the SIGINT came from a lock timeout and not an external
842	 * source.
843	 */
844	timeouts[0].id = DEADLOCK_TIMEOUT;
845	timeouts[0].keep_indicator = false;
846	timeouts[1].id = LOCK_TIMEOUT;
847	timeouts[1].keep_indicator = true;
848	disable_timeouts(timeouts, 2);
849
850	/* Unlink myself from the wait queue, if on it (might not be anymore!) */
851	partitionLock = LockHashPartitionLock(lockAwaited->hashcode);
852	LWLockAcquire(partitionLock, LW_EXCLUSIVE);
853
854	if (!dlist_node_is_detached(&MyProc->waitLink))
855	{
856		/* We could not have been granted the lock yet */
857		RemoveFromWaitQueue(MyProc, lockAwaited->hashcode);
858	}
859	else
860	{
861		/*
862		 * Somebody kicked us off the lock queue already.  Perhaps they
863		 * granted us the lock, or perhaps they detected a deadlock. If they
864		 * did grant us the lock, we'd better remember it in our local lock
865		 * table.
866		 */
867		if (MyProc->waitStatus == PROC_WAIT_STATUS_OK)
868			GrantAwaitedLock();
869	}
870
871	ResetAwaitedLock();
872
873	LWLockRelease(partitionLock);
874
875	RESUME_INTERRUPTS();
876}

879/*
880 * ProcReleaseLocks() -- release locks associated with current transaction
881 *			at main transaction commit or abort
882 *
883 * At main transaction commit, we release standard locks except session locks.
884 * At main transaction abort, we release all locks including session locks.
885 *
886 * Advisory locks are released only if they are transaction-level;
887 * session-level holds remain, whether this is a commit or not.
888 *
889 * At subtransaction commit, we don't release any locks (so this func is not
890 * needed at all); we will defer the releasing to the parent transaction.
891 * At subtransaction abort, we release all locks held by the subtransaction;
892 * this is implemented by retail releasing of the locks under control of
893 * the ResourceOwner mechanism.
894 */
895void
896ProcReleaseLocks(bool isCommit)
897{
898	if (!MyProc)
899		return;
900	/* If waiting, get off wait queue (should only be needed after error) */
901	LockErrorCleanup();
902	/* Release standard locks, including session-level if aborting */
903	LockReleaseAll(DEFAULT_LOCKMETHOD, !isCommit);
904	/* Release transaction-level advisory locks */
905	LockReleaseAll(USER_LOCKMETHOD, false);
906}

919/*
920 * ProcKill() -- Destroy the per-proc data structure for
921 *		this process. Release any of its held LW locks.
922 */
923static void
924ProcKill(int code, Datum arg)
925{
926	PGPROC	   *proc;
927	PGPROC	   *leader;
928	dlist_head *procgloballist;
929	bool		push_leader;
930	bool		push_self;
931
932	Assert(MyProc != NULL);
933
934	/* not safe if forked by system(), etc. */
935	if (MyProc->pid != (int) getpid())
936		elog(PANIC, "ProcKill() called in child process");
937
938	/* Make sure we're out of the sync rep lists */
939	SyncRepCleanupAtProcExit();
940
941#ifdef USE_ASSERT_CHECKING
942	{
943		int			i;
944
945		/* Last process should have released all locks. */
946		for (i = 0; i < NUM_LOCK_PARTITIONS; i++)
947			Assert(dlist_is_empty(&(MyProc->myProcLocks[i])));
948	}
949#endif
950
951	/*
952	 * Release any LW locks I am holding.  There really shouldn't be any, but
953	 * it's cheap to check again before we cut the knees off the LWLock
954	 * facility by releasing our PGPROC ...
955	 */
956	LWLockReleaseAll();
957
958	/*
959	 * Cleanup waiting for LSN if any.
960	 */
961	WaitLSNCleanup();
962
963	/* Cancel any pending condition variable sleep, too */
964	ConditionVariableCancelSleep();
965
966	/*
967	 * Reset MyLatch to the process local one and disown the shared latch, so
968	 * that signal handlers et al can continue using the latch after the
969	 * shared latch isn't ours anymore.
970	 *
971	 * DisownLatch() must happen before our PGPROC can appear on a freelist: a
972	 * newly-forked backend that pops our slot and calls OwnLatch() would
973	 * PANIC on a still-owned latch.
974	 *
975	 * pgstat_reset_wait_event_storage() is intentionally deferred until after
976	 * the lock-group block so that wait_event_info remains visible in our
977	 * PGPROC slot while we may be observed there.  It is safe to defer
978	 * because our slot is not yet on any freelist at this point, and useful
979	 * for testing purposes.
980	 */
981	SwitchBackToLocalLatch();
982	DisownLatch(&MyProc->procLatch);
983
984	proc = MyProc;
985	procgloballist = proc->procgloballist;
986
987	/*
988	 * Detach from any lock group of which we are a member, deciding under
989	 * leader_lwlock whether we (via push_self) and/or the leader (via
990	 * push_leader) need to be pushed onto a freelist.  The actual pushes
991	 * happen after evaluating if any of these are required, under a single
992	 * ProcGlobal->freeProcsLock.
993	 *
994	 * The decision whether any of the freelists needs to be updated is taken
995	 * under a single leader_lwlock.
996	 */
997	push_leader = false;
998	push_self = true;
999	leader = NULL;
1000
1001	if (proc->lockGroupLeader != NULL)
1002	{
1003		LWLock	   *leader_lwlock;
1004
1005		leader = proc->lockGroupLeader;
1006		leader_lwlock = LockHashPartitionLockByProc(leader);
1007
1008		LWLockAcquire(leader_lwlock, LW_EXCLUSIVE);
1009		Assert(!dlist_is_empty(&leader->lockGroupMembers));
1010		dlist_delete(&proc->lockGroupLink);
1011		if (dlist_is_empty(&leader->lockGroupMembers))
1012		{
1013			leader->lockGroupLeader = NULL;
1014			if (leader != proc)
1015			{
1016				/*
1017				 * We are the last follower and the leader exited earlier; its
1018				 * PGPROC is still allocated and must be pushed here.
1019				 */
1020				push_leader = true;
1021				proc->lockGroupLeader = NULL;
1022			}
1023		}
1024		else if (leader != proc)
1025		{
1026			/* Non-last follower; leader still present in the group. */
1027			proc->lockGroupLeader = NULL;
1028		}
1029		else
1030		{
1031			/*
1032			 * We are the leader and followers remain.  Skip our own push; the
1033			 * last follower to exit will push us back to the freelist.
1034			 */
1035			push_self = false;
1036		}
1037		LWLockRelease(leader_lwlock);
1038	}
1039
1040	/* See comment above, close to DisownLatch() */
1041	pgstat_reset_wait_event_storage();
1042
1043	MyProc = NULL;
1044	MyProcNumber = INVALID_PROC_NUMBER;
1045
1046	/* Mark the proc no longer in use */
1047	proc->pid = 0;
1048	proc->vxid.procNumber = INVALID_PROC_NUMBER;
1049	proc->vxid.lxid = InvalidTransactionId;
1050
1051	SpinLockAcquire(&ProcGlobal->freeProcsLock);
1052	if (push_leader)
1053	{
1054		/* Return leader PGPROC (and semaphore) to appropriate freelist */
1055		dlist_push_head(leader->procgloballist, &leader->freeProcsLink);
1056	}
1057	if (push_self)
1058	{
1059		Assert(proc->lockGroupLeader == NULL);
1060		/* Since lockGroupLeader is NULL, lockGroupMembers should be empty. */
1061		Assert(dlist_is_empty(&proc->lockGroupMembers));
1062
1063		/* Return PGPROC structure (and semaphore) to appropriate freelist */
1064		dlist_push_tail(procgloballist, &proc->freeProcsLink);
1065	}
1066
1067	/* Update shared estimate of spins_per_delay */
1068	ProcGlobal->spins_per_delay = update_spins_per_delay(ProcGlobal->spins_per_delay);
1069
1070	SpinLockRelease(&ProcGlobal->freeProcsLock);
1071}

1152/*
1153 * JoinWaitQueue -- join the wait queue on the specified lock
1154 *
1155 * It's not actually guaranteed that we need to wait when this function is
1156 * called, because it could be that when we try to find a position at which
1157 * to insert ourself into the wait queue, we discover that we must be inserted
1158 * ahead of everyone who wants a lock that conflict with ours. In that case,
1159 * we get the lock immediately. Because of this, it's sensible for this function
1160 * to have a dontWait argument, despite the name.
1161 *
1162 * On entry, the caller has already set up LOCK and PROCLOCK entries to
1163 * reflect that we have "requested" the lock.  The caller is responsible for
1164 * cleaning that up, if we end up not joining the queue after all.
1165 *
1166 * The lock table's partition lock must be held at entry, and is still held
1167 * at exit.  The caller must release it before calling ProcSleep().
1168 *
1169 * Result is one of the following:
1170 *
1171 *  PROC_WAIT_STATUS_OK       - lock was immediately granted
1172 *  PROC_WAIT_STATUS_WAITING  - joined the wait queue; call ProcSleep()
1173 *  PROC_WAIT_STATUS_ERROR    - immediate deadlock was detected, or would
1174 *                              need to wait and dontWait == true
1175 *
1176 * NOTES: The process queue is now a priority queue for locking.
1177 */
1178ProcWaitStatus
1179JoinWaitQueue(LOCALLOCK *locallock, LockMethod lockMethodTable, bool dontWait)
1180{
1181	LOCKMODE	lockmode = locallock->tag.mode;
1182	LOCK	   *lock = locallock->lock;
1183	PROCLOCK   *proclock = locallock->proclock;
1184	uint32		hashcode = locallock->hashcode;
1185	LWLock	   *partitionLock PG_USED_FOR_ASSERTS_ONLY = LockHashPartitionLock(hashcode);
1186	dclist_head *waitQueue = &lock->waitProcs;
1187	PGPROC	   *insert_before = NULL;
1188	LOCKMASK	myProcHeldLocks;
1189	LOCKMASK	myHeldLocks;
1190	bool		early_deadlock = false;
1191	PGPROC	   *leader = MyProc->lockGroupLeader;
1192
1193	Assert(LWLockHeldByMeInMode(partitionLock, LW_EXCLUSIVE));
1194
1195	/*
1196	 * Set bitmask of locks this process already holds on this object.
1197	 */
1198	myHeldLocks = MyProc->heldLocks = proclock->holdMask;
1199
1200	/*
1201	 * Determine which locks we're already holding.
1202	 *
1203	 * If group locking is in use, locks held by members of my locking group
1204	 * need to be included in myHeldLocks.  This is not required for relation
1205	 * extension lock which conflict among group members. However, including
1206	 * them in myHeldLocks will give group members the priority to get those
1207	 * locks as compared to other backends which are also trying to acquire
1208	 * those locks.  OTOH, we can avoid giving priority to group members for
1209	 * that kind of locks, but there doesn't appear to be a clear advantage of
1210	 * the same.
1211	 */
1212	myProcHeldLocks = proclock->holdMask;
1213	myHeldLocks = myProcHeldLocks;
1214	if (leader != NULL)
1215	{
1216		dlist_iter	iter;
1217
1218		dlist_foreach(iter, &lock->procLocks)
1219		{
1220			PROCLOCK   *otherproclock;
1221
1222			otherproclock = dlist_container(PROCLOCK, lockLink, iter.cur);
1223
1224			if (otherproclock->groupLeader == leader)
1225				myHeldLocks |= otherproclock->holdMask;
1226		}
1227	}
1228
1229	/*
1230	 * Determine where to add myself in the wait queue.
1231	 *
1232	 * Normally I should go at the end of the queue.  However, if I already
1233	 * hold locks that conflict with the request of any previous waiter, put
1234	 * myself in the queue just in front of the first such waiter. This is not
1235	 * a necessary step, since deadlock detection would move me to before that
1236	 * waiter anyway; but it's relatively cheap to detect such a conflict
1237	 * immediately, and avoid delaying till deadlock timeout.
1238	 *
1239	 * Special case: if I find I should go in front of some waiter, check to
1240	 * see if I conflict with already-held locks or the requests before that
1241	 * waiter.  If not, then just grant myself the requested lock immediately.
1242	 * This is the same as the test for immediate grant in LockAcquire, except
1243	 * we are only considering the part of the wait queue before my insertion
1244	 * point.
1245	 */
1246	if (myHeldLocks != 0 && !dclist_is_empty(waitQueue))
1247	{
1248		LOCKMASK	aheadRequests = 0;
1249		dlist_iter	iter;
1250
1251		dclist_foreach(iter, waitQueue)
1252		{
1253			PGPROC	   *proc = dlist_container(PGPROC, waitLink, iter.cur);
1254
1255			/*
1256			 * If we're part of the same locking group as this waiter, its
1257			 * locks neither conflict with ours nor contribute to
1258			 * aheadRequests.
1259			 */
1260			if (leader != NULL && leader == proc->lockGroupLeader)
1261				continue;
1262
1263			/* Must he wait for me? */
1264			if (lockMethodTable->conflictTab[proc->waitLockMode] & myHeldLocks)
1265			{
1266				/* Must I wait for him ? */
1267				if (lockMethodTable->conflictTab[lockmode] & proc->heldLocks)
1268				{
1269					/*
1270					 * Yes, so we have a deadlock.  Easiest way to clean up
1271					 * correctly is to call RemoveFromWaitQueue(), but we
1272					 * can't do that until we are *on* the wait queue. So, set
1273					 * a flag to check below, and break out of loop.  Also,
1274					 * record deadlock info for later message.
1275					 */
1276					RememberSimpleDeadLock(MyProc, lockmode, lock, proc);
1277					early_deadlock = true;
1278					break;
1279				}
1280				/* I must go before this waiter.  Check special case. */
1281				if ((lockMethodTable->conflictTab[lockmode] & aheadRequests) == 0 &&
1282					!LockCheckConflicts(lockMethodTable, lockmode, lock,
1283										proclock))
1284				{
1285					/* Skip the wait and just grant myself the lock. */
1286					GrantLock(lock, proclock, lockmode);
1287					return PROC_WAIT_STATUS_OK;
1288				}
1289
1290				/* Put myself into wait queue before conflicting process */
1291				insert_before = proc;
1292				break;
1293			}
1294			/* Nope, so advance to next waiter */
1295			aheadRequests |= LOCKBIT_ON(proc->waitLockMode);
1296		}
1297	}
1298
1299	/*
1300	 * If we detected deadlock, give up without waiting.  This must agree with
1301	 * CheckDeadLock's recovery code.
1302	 */
1303	if (early_deadlock)
1304		return PROC_WAIT_STATUS_ERROR;
1305
1306	/*
1307	 * At this point we know that we'd really need to sleep. If we've been
1308	 * commanded not to do that, bail out.
1309	 */
1310	if (dontWait)
1311		return PROC_WAIT_STATUS_ERROR;
1312
1313	/*
1314	 * Insert self into queue, at the position determined above.
1315	 */
1316	if (insert_before)
1317		dclist_insert_before(waitQueue, &insert_before->waitLink, &MyProc->waitLink);
1318	else
1319		dclist_push_tail(waitQueue, &MyProc->waitLink);
1320
1321	lock->waitMask |= LOCKBIT_ON(lockmode);
1322
1323	/* Set up wait information in PGPROC object, too */
1324	MyProc->heldLocks = myProcHeldLocks;
1325	MyProc->waitLock = lock;
1326	MyProc->waitProcLock = proclock;
1327	MyProc->waitLockMode = lockmode;
1328
1329	MyProc->waitStatus = PROC_WAIT_STATUS_WAITING;
1330
1331	return PROC_WAIT_STATUS_WAITING;
1332}

1334/*
1335 * ProcSleep -- put process to sleep waiting on lock
1336 *
1337 * This must be called when JoinWaitQueue() returns PROC_WAIT_STATUS_WAITING.
1338 * Returns after the lock has been granted, or if a deadlock is detected.  Can
1339 * also bail out with ereport(ERROR), if some other error condition, or a
1340 * timeout or cancellation is triggered.
1341 *
1342 * Result is one of the following:
1343 *
1344 *  PROC_WAIT_STATUS_OK      - lock was granted
1345 *  PROC_WAIT_STATUS_ERROR   - a deadlock was detected
1346 */
1347ProcWaitStatus
1348ProcSleep(LOCALLOCK *locallock)
1349{
1350	LOCKMODE	lockmode = locallock->tag.mode;
1351	LOCK	   *lock = locallock->lock;
1352	uint32		hashcode = locallock->hashcode;
1353	LWLock	   *partitionLock = LockHashPartitionLock(hashcode);
1354	TimestampTz standbyWaitStart = 0;
1355	bool		allow_autovacuum_cancel = true;
1356	bool		logged_recovery_conflict = false;
1357	bool		logged_lock_wait = false;
1358	ProcWaitStatus myWaitStatus;
1359	DeadLockState deadlock_state;
1360
1361	/* The caller must've armed the on-error cleanup mechanism */
1362	Assert(GetAwaitedLock() == locallock);
1363	Assert(!LWLockHeldByMe(partitionLock));
1364
1365	/*
1366	 * Now that we will successfully clean up after an ereport, it's safe to
1367	 * check to see if there's a buffer pin deadlock against the Startup
1368	 * process.  Of course, that's only necessary if we're doing Hot Standby
1369	 * and are not the Startup process ourselves.
1370	 */
1371	if (RecoveryInProgress() && !InRecovery)
1372		CheckRecoveryConflictDeadlock();
1373
1374	/* Reset deadlock_state before enabling the timeout handler */
1375	deadlock_state = DS_NOT_YET_CHECKED;
1376	got_deadlock_timeout = false;
1377
1378	/*
1379	 * Set timer so we can wake up after awhile and check for a deadlock. If a
1380	 * deadlock is detected, the handler sets MyProc->waitStatus =
1381	 * PROC_WAIT_STATUS_ERROR, allowing us to know that we must report failure
1382	 * rather than success.
1383	 *
1384	 * By delaying the check until we've waited for a bit, we can avoid
1385	 * running the rather expensive deadlock-check code in most cases.
1386	 *
1387	 * If LockTimeout is set, also enable the timeout for that.  We can save a
1388	 * few cycles by enabling both timeout sources in one call.
1389	 *
1390	 * If InHotStandby we set lock waits slightly later for clarity with other
1391	 * code.
1392	 */
1393	if (!InHotStandby)
1394	{
1395		if (LockTimeout > 0)
1396		{
1397			EnableTimeoutParams timeouts[2];
1398
1399			timeouts[0].id = DEADLOCK_TIMEOUT;
1400			timeouts[0].type = TMPARAM_AFTER;
1401			timeouts[0].delay_ms = DeadlockTimeout;
1402			timeouts[1].id = LOCK_TIMEOUT;
1403			timeouts[1].type = TMPARAM_AFTER;
1404			timeouts[1].delay_ms = LockTimeout;
1405			enable_timeouts(timeouts, 2);
1406		}
1407		else
1408			enable_timeout_after(DEADLOCK_TIMEOUT, DeadlockTimeout);
1409
1410		/*
1411		 * Use the current time obtained for the deadlock timeout timer as
1412		 * waitStart (i.e., the time when this process started waiting for the
1413		 * lock). Since getting the current time newly can cause overhead, we
1414		 * reuse the already-obtained time to avoid that overhead.
1415		 *
1416		 * Note that waitStart is updated without holding the lock table's
1417		 * partition lock, to avoid the overhead by additional lock
1418		 * acquisition. This can cause "waitstart" in pg_locks to become NULL
1419		 * for a very short period of time after the wait started even though
1420		 * "granted" is false. This is OK in practice because we can assume
1421		 * that users are likely to look at "waitstart" when waiting for the
1422		 * lock for a long time.
1423		 */
1424		pg_atomic_write_u64(&MyProc->waitStart,
1425							get_timeout_start_time(DEADLOCK_TIMEOUT));
1426	}
1427	else if (log_recovery_conflict_waits)
1428	{
1429		/*
1430		 * Set the wait start timestamp if logging is enabled and in hot
1431		 * standby.
1432		 */
1433		standbyWaitStart = GetCurrentTimestamp();
1434	}
1435
1436	/*
1437	 * If somebody wakes us between LWLockRelease and WaitLatch, the latch
1438	 * will not wait. But a set latch does not necessarily mean that the lock
1439	 * is free now, as there are many other sources for latch sets than
1440	 * somebody releasing the lock.
1441	 *
1442	 * We process interrupts whenever the latch has been set, so cancel/die
1443	 * interrupts are processed quickly. This means we must not mind losing
1444	 * control to a cancel/die interrupt here.  We don't, because we have no
1445	 * shared-state-change work to do after being granted the lock (the
1446	 * grantor did it all).  We do have to worry about canceling the deadlock
1447	 * timeout and updating the locallock table, but if we lose control to an
1448	 * error, LockErrorCleanup will fix that up.
1449	 */
1450	do
1451	{
1452		if (InHotStandby)
1453		{
1454			bool		maybe_log_conflict =
1455				(standbyWaitStart != 0 && !logged_recovery_conflict);
1456
1457			/* Set a timer and wait for that or for the lock to be granted */
1458			ResolveRecoveryConflictWithLock(locallock->tag.lock,
1459											maybe_log_conflict);
1460
1461			/*
1462			 * Emit the log message if the startup process is waiting longer
1463			 * than deadlock_timeout for recovery conflict on lock.
1464			 */
1465			if (maybe_log_conflict)
1466			{
1467				TimestampTz now = GetCurrentTimestamp();
1468
1469				if (TimestampDifferenceExceeds(standbyWaitStart, now,
1470											   DeadlockTimeout))
1471				{
1472					VirtualTransactionId *vxids;
1473					int			cnt;
1474
1475					vxids = GetLockConflicts(&locallock->tag.lock,
1476											 AccessExclusiveLock, &cnt);
1477
1478					/*
1479					 * Log the recovery conflict and the list of PIDs of
1480					 * backends holding the conflicting lock. Note that we do
1481					 * logging even if there are no such backends right now
1482					 * because the startup process here has already waited
1483					 * longer than deadlock_timeout.
1484					 */
1485					LogRecoveryConflict(RECOVERY_CONFLICT_LOCK,
1486										standbyWaitStart, now,
1487										cnt > 0 ? vxids : NULL, true);
1488					logged_recovery_conflict = true;
1489				}
1490			}
1491		}
1492		else
1493		{
1494			(void) WaitLatch(MyLatch, WL_LATCH_SET | WL_EXIT_ON_PM_DEATH, 0,
1495							 PG_WAIT_LOCK | locallock->tag.lock.locktag_type);
1496			ResetLatch(MyLatch);
1497			/* check for deadlocks first, as that's probably log-worthy */
1498			if (got_deadlock_timeout)
1499			{
1500				deadlock_state = CheckDeadLock();
1501				got_deadlock_timeout = false;
1502			}
1503			CHECK_FOR_INTERRUPTS();
1504		}
1505
1506		/*
1507		 * waitStatus could change from PROC_WAIT_STATUS_WAITING to something
1508		 * else asynchronously.  Read it just once per loop to prevent
1509		 * surprising behavior (such as missing log messages).
1510		 */
1511		myWaitStatus = *((volatile ProcWaitStatus *) &MyProc->waitStatus);
1512
1513		/*
1514		 * If we are not deadlocked, but are waiting on an autovacuum-induced
1515		 * task, send a signal to interrupt it.
1516		 */
1517		if (deadlock_state == DS_BLOCKED_BY_AUTOVACUUM && allow_autovacuum_cancel)
1518		{
1519			PGPROC	   *autovac = GetBlockingAutoVacuumPgproc();
1520			uint8		statusFlags;
1521			uint8		lockmethod_copy;
1522			LOCKTAG		locktag_copy;
1523
1524			/*
1525			 * Grab info we need, then release lock immediately.  Note this
1526			 * coding means that there is a tiny chance that the process
1527			 * terminates its current transaction and starts a different one
1528			 * before we have a change to send the signal; the worst possible
1529			 * consequence is that a for-wraparound vacuum is canceled.  But
1530			 * that could happen in any case unless we were to do kill() with
1531			 * the lock held, which is much more undesirable.
1532			 */
1533			LWLockAcquire(ProcArrayLock, LW_EXCLUSIVE);
1534			statusFlags = ProcGlobal->statusFlags[autovac->pgxactoff];
1535			lockmethod_copy = lock->tag.locktag_lockmethodid;
1536			locktag_copy = lock->tag;
1537			LWLockRelease(ProcArrayLock);
1538
1539			/*
1540			 * Only do it if the worker is not working to protect against Xid
1541			 * wraparound.
1542			 */
1543			if ((statusFlags & PROC_IS_AUTOVACUUM) &&
1544				!(statusFlags & PROC_VACUUM_FOR_WRAPAROUND))
1545			{
1546				int			pid = autovac->pid;
1547
1548				/* report the case, if configured to do so */
1549				if (message_level_is_interesting(DEBUG1))
1550				{
1551					StringInfoData locktagbuf;
1552					StringInfoData logbuf;	/* errdetail for server log */
1553
1554					initStringInfo(&locktagbuf);
1555					initStringInfo(&logbuf);
1556					DescribeLockTag(&locktagbuf, &locktag_copy);
1557					appendStringInfo(&logbuf,
1558									 "Process %d waits for %s on %s.",
1559									 MyProcPid,
1560									 GetLockmodeName(lockmethod_copy, lockmode),
1561									 locktagbuf.data);
1562
1563					ereport(DEBUG1,
1564							(errmsg_internal("sending cancel to blocking autovacuum PID %d",
1565											 pid),
1566							 errdetail_log("%s", logbuf.data)));
1567
1568					pfree(locktagbuf.data);
1569					pfree(logbuf.data);
1570				}
1571
1572				/* send the autovacuum worker Back to Old Kent Road */
1573				if (kill(pid, SIGINT) < 0)
1574				{
1575					/*
1576					 * There's a race condition here: once we release the
1577					 * ProcArrayLock, it's possible for the autovac worker to
1578					 * close up shop and exit before we can do the kill().
1579					 * Therefore, we do not whinge about no-such-process.
1580					 * Other errors such as EPERM could conceivably happen if
1581					 * the kernel recycles the PID fast enough, but such cases
1582					 * seem improbable enough that it's probably best to issue
1583					 * a warning if we see some other errno.
1584					 */
1585					if (errno != ESRCH)
1586						ereport(WARNING,
1587								(errmsg("could not send signal to process %d: %m",
1588										pid)));
1589				}
1590			}
1591
1592			/* prevent signal from being sent again more than once */
1593			allow_autovacuum_cancel = false;
1594		}
1595
1596		/*
1597		 * If awoken after the deadlock check interrupt has run, increment the
1598		 * lock statistics counters and if log_lock_waits is on, then report
1599		 * about the wait.
1600		 */
1601		if (deadlock_state != DS_NOT_YET_CHECKED)
1602		{
1603			long		secs;
1604			int			usecs;
1605			long		msecs;
1606
1607			INJECTION_POINT("deadlock-timeout-fired", NULL);
1608			TimestampDifference(get_timeout_start_time(DEADLOCK_TIMEOUT),
1609								GetCurrentTimestamp(),
1610								&secs, &usecs);
1611			/* Increment the lock statistics counters if done waiting. */
1612			if (myWaitStatus == PROC_WAIT_STATUS_OK)
1613				pgstat_count_lock_waits(locallock->tag.lock.locktag_type,
1614										(PgStat_Counter) secs * 1000000 + usecs);
1615
1616			msecs = secs * 1000 + usecs / 1000;
1617			usecs = usecs % 1000;
1618
1619			if (log_lock_waits)
1620			{
1621				StringInfoData buf,
1622							lock_waiters_sbuf,
1623							lock_holders_sbuf;
1624				const char *modename;
1625				int			lockHoldersNum = 0;
1626
1627				initStringInfo(&buf);
1628				initStringInfo(&lock_waiters_sbuf);
1629				initStringInfo(&lock_holders_sbuf);
1630
1631				DescribeLockTag(&buf, &locallock->tag.lock);
1632				modename = GetLockmodeName(locallock->tag.lock.locktag_lockmethodid,
1633										   lockmode);
1634
1635				/* Gather a list of all lock holders and waiters */
1636				LWLockAcquire(partitionLock, LW_SHARED);
1637				GetLockHoldersAndWaiters(locallock, &lock_holders_sbuf,
1638										 &lock_waiters_sbuf, &lockHoldersNum);
1639				LWLockRelease(partitionLock);
1640
1641				if (deadlock_state == DS_SOFT_DEADLOCK)
1642					ereport(LOG,
1643							(errmsg("process %d avoided deadlock for %s on %s by rearranging queue order after %ld.%03d ms",
1644									MyProcPid, modename, buf.data, msecs, usecs),
1645							 (errdetail_log_plural("Process holding the lock: %s. Wait queue: %s.",
1646												   "Processes holding the lock: %s. Wait queue: %s.",
1647												   lockHoldersNum, lock_holders_sbuf.data, lock_waiters_sbuf.data))));
1648				else if (deadlock_state == DS_HARD_DEADLOCK)
1649				{
1650					/*
1651					 * This message is a bit redundant with the error that
1652					 * will be reported subsequently, but in some cases the
1653					 * error report might not make it to the log (eg, if it's
1654					 * caught by an exception handler), and we want to ensure
1655					 * all long-wait events get logged.
1656					 */
1657					ereport(LOG,
1658							(errmsg("process %d detected deadlock while waiting for %s on %s after %ld.%03d ms",
1659									MyProcPid, modename, buf.data, msecs, usecs),
1660							 (errdetail_log_plural("Process holding the lock: %s. Wait queue: %s.",
1661												   "Processes holding the lock: %s. Wait queue: %s.",
1662												   lockHoldersNum, lock_holders_sbuf.data, lock_waiters_sbuf.data))));
1663				}
1664
1665				if (myWaitStatus == PROC_WAIT_STATUS_WAITING)
1666				{
1667					/*
1668					 * Guard the "still waiting on lock" log message so it is
1669					 * reported at most once while waiting for the lock.
1670					 *
1671					 * Without this guard, the message can be emitted whenever
1672					 * the lock-wait sleep is interrupted (for example by
1673					 * SIGHUP for config reload or by
1674					 * client_connection_check_interval). For example, if
1675					 * client_connection_check_interval is set very low (e.g.,
1676					 * 100 ms), the message could be logged repeatedly,
1677					 * flooding the log and making it difficult to use.
1678					 */
1679					if (!logged_lock_wait)
1680					{
1681						ereport(LOG,
1682								(errmsg("process %d still waiting for %s on %s after %ld.%03d ms",
1683										MyProcPid, modename, buf.data, msecs, usecs),
1684								 (errdetail_log_plural("Process holding the lock: %s. Wait queue: %s.",
1685													   "Processes holding the lock: %s. Wait queue: %s.",
1686													   lockHoldersNum, lock_holders_sbuf.data, lock_waiters_sbuf.data))));
1687						logged_lock_wait = true;
1688					}
1689				}
1690				else if (myWaitStatus == PROC_WAIT_STATUS_OK)
1691					ereport(LOG,
1692							(errmsg("process %d acquired %s on %s after %ld.%03d ms",
1693									MyProcPid, modename, buf.data, msecs, usecs)));
1694				else
1695				{
1696					Assert(myWaitStatus == PROC_WAIT_STATUS_ERROR);
1697
1698					/*
1699					 * Currently, the deadlock checker always kicks its own
1700					 * process, which means that we'll only see
1701					 * PROC_WAIT_STATUS_ERROR when deadlock_state ==
1702					 * DS_HARD_DEADLOCK, and there's no need to print
1703					 * redundant messages.  But for completeness and
1704					 * future-proofing, print a message if it looks like
1705					 * someone else kicked us off the lock.
1706					 */
1707					if (deadlock_state != DS_HARD_DEADLOCK)
1708						ereport(LOG,
1709								(errmsg("process %d failed to acquire %s on %s after %ld.%03d ms",
1710										MyProcPid, modename, buf.data, msecs, usecs),
1711								 (errdetail_log_plural("Process holding the lock: %s. Wait queue: %s.",
1712													   "Processes holding the lock: %s. Wait queue: %s.",
1713													   lockHoldersNum, lock_holders_sbuf.data, lock_waiters_sbuf.data))));
1714				}
1715				pfree(buf.data);
1716				pfree(lock_holders_sbuf.data);
1717				pfree(lock_waiters_sbuf.data);
1718			}
1719
1720			/*
1721			 * At this point we might still need to wait for the lock. Reset
1722			 * state so we don't print the above messages again if
1723			 * log_lock_waits is on.
1724			 */
1725			deadlock_state = DS_NO_DEADLOCK;
1726		}
1727	} while (myWaitStatus == PROC_WAIT_STATUS_WAITING);
1728
1729	/*
1730	 * Disable the timers, if they are still running.  As in LockErrorCleanup,
1731	 * we must preserve the LOCK_TIMEOUT indicator flag: if a lock timeout has
1732	 * already caused QueryCancelPending to become set, we want the cancel to
1733	 * be reported as a lock timeout, not a user cancel.
1734	 */
1735	if (!InHotStandby)
1736	{
1737		if (LockTimeout > 0)
1738		{
1739			DisableTimeoutParams timeouts[2];
1740
1741			timeouts[0].id = DEADLOCK_TIMEOUT;
1742			timeouts[0].keep_indicator = false;
1743			timeouts[1].id = LOCK_TIMEOUT;
1744			timeouts[1].keep_indicator = true;
1745			disable_timeouts(timeouts, 2);
1746		}
1747		else
1748			disable_timeout(DEADLOCK_TIMEOUT, false);
1749	}
1750
1751	/*
1752	 * Emit the log message if recovery conflict on lock was resolved but the
1753	 * startup process waited longer than deadlock_timeout for it.
1754	 */
1755	if (InHotStandby && logged_recovery_conflict)
1756		LogRecoveryConflict(RECOVERY_CONFLICT_LOCK,
1757							standbyWaitStart, GetCurrentTimestamp(),
1758							NULL, false);
1759
1760	/*
1761	 * We don't have to do anything else, because the awaker did all the
1762	 * necessary updates of the lock table and MyProc. (The caller is
1763	 * responsible for updating the local lock table.)
1764	 */
1765	return myWaitStatus;
1766}

1769/*
1770 * ProcWakeup -- wake up a process by setting its latch.
1771 *
1772 *	 Also remove the process from the wait queue and set its waitLink invalid.
1773 *
1774 * The appropriate lock partition lock must be held by caller.
1775 *
1776 * XXX: presently, this code is only used for the "success" case, and only
1777 * works correctly for that case.  To clean up in failure case, would need
1778 * to twiddle the lock's request counts too --- see RemoveFromWaitQueue.
1779 * Hence, in practice the waitStatus parameter must be PROC_WAIT_STATUS_OK.
1780 */
1781void
1782ProcWakeup(PGPROC *proc, ProcWaitStatus waitStatus)
1783{
1784	if (dlist_node_is_detached(&proc->waitLink))
1785		return;
1786
1787	Assert(proc->waitStatus == PROC_WAIT_STATUS_WAITING);
1788
1789	/* Remove process from wait queue */
1790	dclist_delete_from_thoroughly(&proc->waitLock->waitProcs, &proc->waitLink);
1791
1792	/* Clean up process' state and pass it the ok/fail signal */
1793	proc->waitLock = NULL;
1794	proc->waitProcLock = NULL;
1795	proc->waitStatus = waitStatus;
1796	pg_atomic_write_u64(&proc->waitStart, 0);
1797
1798	/* And awaken it */
1799	SetLatch(&proc->procLatch);
1800}

1802/*
1803 * ProcLockWakeup -- routine for waking up processes when a lock is
1804 *		released (or a prior waiter is aborted).  Scan all waiters
1805 *		for lock, waken any that are no longer blocked.
1806 *
1807 * The appropriate lock partition lock must be held by caller.
1808 */
1809void
1810ProcLockWakeup(LockMethod lockMethodTable, LOCK *lock)
1811{
1812	dclist_head *waitQueue = &lock->waitProcs;
1813	LOCKMASK	aheadRequests = 0;
1814	dlist_mutable_iter miter;
1815
1816	if (dclist_is_empty(waitQueue))
1817		return;
1818
1819	dclist_foreach_modify(miter, waitQueue)
1820	{
1821		PGPROC	   *proc = dlist_container(PGPROC, waitLink, miter.cur);
1822		LOCKMODE	lockmode = proc->waitLockMode;
1823
1824		/*
1825		 * Waken if (a) doesn't conflict with requests of earlier waiters, and
1826		 * (b) doesn't conflict with already-held locks.
1827		 */
1828		if ((lockMethodTable->conflictTab[lockmode] & aheadRequests) == 0 &&
1829			!LockCheckConflicts(lockMethodTable, lockmode, lock,
1830								proc->waitProcLock))
1831		{
1832			/* OK to waken */
1833			GrantLock(lock, proc->waitProcLock, lockmode);
1834			/* removes proc from the lock's waiting process queue */
1835			ProcWakeup(proc, PROC_WAIT_STATUS_OK);
1836		}
1837		else
1838		{
1839			/*
1840			 * Lock conflicts: Don't wake, but remember requested mode for
1841			 * later checks.
1842			 */
1843			aheadRequests |= LOCKBIT_ON(lockmode);
1844		}
1845	}
1846}

1848/*
1849 * CheckDeadLock
1850 *
1851 * We only get to this routine, if DEADLOCK_TIMEOUT fired while waiting for a
1852 * lock to be released by some other process.  Check if there's a deadlock; if
1853 * not, just return.  If we have a real deadlock, remove ourselves from the
1854 * lock's wait queue.
1855 */
1856static DeadLockState
1857CheckDeadLock(void)
1858{
1859	int			i;
1860	DeadLockState result;
1861
1862	/*
1863	 * Acquire exclusive lock on the entire shared lock data structures. Must
1864	 * grab LWLocks in partition-number order to avoid LWLock deadlock.
1865	 *
1866	 * Note that the deadlock check interrupt had better not be enabled
1867	 * anywhere that this process itself holds lock partition locks, else this
1868	 * will wait forever.  Also note that LWLockAcquire creates a critical
1869	 * section, so that this routine cannot be interrupted by cancel/die
1870	 * interrupts.
1871	 */
1872	for (i = 0; i < NUM_LOCK_PARTITIONS; i++)
1873		LWLockAcquire(LockHashPartitionLockByIndex(i), LW_EXCLUSIVE);
1874
1875	/*
1876	 * Check to see if we've been awoken by anyone in the interim.
1877	 *
1878	 * If we have, we can return and resume our transaction -- happy day.
1879	 * Before we are awoken the process releasing the lock grants it to us so
1880	 * we know that we don't have to wait anymore.
1881	 *
1882	 * We check by looking to see if we've been unlinked from the wait queue.
1883	 * This is safe because we hold the lock partition lock.
1884	 */
1885	if (dlist_node_is_detached(&MyProc->waitLink))
1886	{
1887		result = DS_NO_DEADLOCK;
1888		goto check_done;
1889	}
1890
1891#ifdef LOCK_DEBUG
1892	if (Debug_deadlocks)
1893		DumpAllLocks();
1894#endif
1895
1896	/* Run the deadlock check */
1897	result = DeadLockCheck(MyProc);
1898
1899	if (result == DS_HARD_DEADLOCK)
1900	{
1901		/*
1902		 * Oops.  We have a deadlock.
1903		 *
1904		 * Get this process out of wait state. (Note: we could do this more
1905		 * efficiently by relying on lockAwaited, but use this coding to
1906		 * preserve the flexibility to kill some other transaction than the
1907		 * one detecting the deadlock.)
1908		 *
1909		 * RemoveFromWaitQueue sets MyProc->waitStatus to
1910		 * PROC_WAIT_STATUS_ERROR, so ProcSleep will report an error after we
1911		 * return.
1912		 */
1913		Assert(MyProc->waitLock != NULL);
1914		RemoveFromWaitQueue(MyProc, LockTagHashCode(&(MyProc->waitLock->tag)));
1915
1916		/*
1917		 * We're done here.  Transaction abort caused by the error that
1918		 * ProcSleep will raise will cause any other locks we hold to be
1919		 * released, thus allowing other processes to wake up; we don't need
1920		 * to do that here.  NOTE: an exception is that releasing locks we
1921		 * hold doesn't consider the possibility of waiters that were blocked
1922		 * behind us on the lock we just failed to get, and might now be
1923		 * wakable because we're not in front of them anymore.  However,
1924		 * RemoveFromWaitQueue took care of waking up any such processes.
1925		 */
1926	}
1927
1928	/*
1929	 * And release locks.  We do this in reverse order for two reasons: (1)
1930	 * Anyone else who needs more than one of the locks will be trying to lock
1931	 * them in increasing order; we don't want to release the other process
1932	 * until it can get all the locks it needs. (2) This avoids O(N^2)
1933	 * behavior inside LWLockRelease.
1934	 */
1935check_done:
1936	for (i = NUM_LOCK_PARTITIONS; --i >= 0;)
1937		LWLockRelease(LockHashPartitionLockByIndex(i));
1938
1939	return result;
1940}

1966/*
1967 * GetLockHoldersAndWaiters - get lock holders and waiters for a lock
1968 *
1969 * Fill lock_holders_sbuf and lock_waiters_sbuf with the PIDs of processes holding
1970 * and waiting for the lock, and set lockHoldersNum to the number of lock holders.
1971 *
1972 * The lock table's partition lock must be held on entry and remains held on exit.
1973 */
1974void
1975GetLockHoldersAndWaiters(LOCALLOCK *locallock, StringInfo lock_holders_sbuf,
1976						 StringInfo lock_waiters_sbuf, int *lockHoldersNum)
1977{
1978	dlist_iter	proc_iter;
1979	PROCLOCK   *curproclock;
1980	LOCK	   *lock = locallock->lock;
1981	bool		first_holder = true,
1982				first_waiter = true;
1983
1984#ifdef USE_ASSERT_CHECKING
1985	{
1986		uint32		hashcode = locallock->hashcode;
1987		LWLock	   *partitionLock = LockHashPartitionLock(hashcode);
1988
1989		Assert(LWLockHeldByMe(partitionLock));
1990	}
1991#endif
1992
1993	*lockHoldersNum = 0;
1994
1995	/*
1996	 * Loop over the lock's procLocks to gather a list of all holders and
1997	 * waiters. Thus we will be able to provide more detailed information for
1998	 * lock debugging purposes.
1999	 *
2000	 * lock->procLocks contains all processes which hold or wait for this
2001	 * lock.
2002	 */
2003	dlist_foreach(proc_iter, &lock->procLocks)
2004	{
2005		curproclock =
2006			dlist_container(PROCLOCK, lockLink, proc_iter.cur);
2007
2008		/*
2009		 * We are a waiter if myProc->waitProcLock == curproclock; we are a
2010		 * holder if it is NULL or something different.
2011		 */
2012		if (curproclock->tag.myProc->waitProcLock == curproclock)
2013		{
2014			if (first_waiter)
2015			{
2016				appendStringInfo(lock_waiters_sbuf, "%d",
2017								 curproclock->tag.myProc->pid);
2018				first_waiter = false;
2019			}
2020			else
2021				appendStringInfo(lock_waiters_sbuf, ", %d",
2022								 curproclock->tag.myProc->pid);
2023		}
2024		else
2025		{
2026			if (first_holder)
2027			{
2028				appendStringInfo(lock_holders_sbuf, "%d",
2029								 curproclock->tag.myProc->pid);
2030				first_holder = false;
2031			}
2032			else
2033				appendStringInfo(lock_holders_sbuf, ", %d",
2034								 curproclock->tag.myProc->pid);
2035
2036			(*lockHoldersNum)++;
2037		}
2038	}
2039}

2069/*
2070 * BecomeLockGroupLeader - designate process as lock group leader
2071 *
2072 * Once this function has returned, other processes can join the lock group
2073 * by calling BecomeLockGroupMember.
2074 */
2075void
2076BecomeLockGroupLeader(void)
2077{
2078	LWLock	   *leader_lwlock;
2079
2080	/* If we already did it, we don't need to do it again. */
2081	if (MyProc->lockGroupLeader == MyProc)
2082		return;
2083
2084	/* We had better not be a follower. */
2085	Assert(MyProc->lockGroupLeader == NULL);
2086
2087	/* Create single-member group, containing only ourselves. */
2088	leader_lwlock = LockHashPartitionLockByProc(MyProc);
2089	LWLockAcquire(leader_lwlock, LW_EXCLUSIVE);
2090	MyProc->lockGroupLeader = MyProc;
2091	dlist_push_head(&MyProc->lockGroupMembers, &MyProc->lockGroupLink);
2092	LWLockRelease(leader_lwlock);
2093}

2095/*
2096 * BecomeLockGroupMember - designate process as lock group member
2097 *
2098 * This is pretty straightforward except for the possibility that the leader
2099 * whose group we're trying to join might exit before we manage to do so;
2100 * and the PGPROC might get recycled for an unrelated process.  To avoid
2101 * that, we require the caller to pass the PID of the intended PGPROC as
2102 * an interlock.  Returns true if we successfully join the intended lock
2103 * group, and false if not.
2104 */
2105bool
2106BecomeLockGroupMember(PGPROC *leader, int pid)
2107{
2108	LWLock	   *leader_lwlock;
2109	bool		ok = false;
2110
2111	/* Group leader can't become member of group */
2112	Assert(MyProc != leader);
2113
2114	/* Can't already be a member of a group */
2115	Assert(MyProc->lockGroupLeader == NULL);
2116
2117	/* PID must be valid. */
2118	Assert(pid != 0);
2119
2120	/*
2121	 * Get lock protecting the group fields.  Note LockHashPartitionLockByProc
2122	 * calculates the proc number based on the PGPROC slot without looking at
2123	 * its contents, so we will acquire the correct lock even if the leader
2124	 * PGPROC is in process of being recycled.
2125	 */
2126	leader_lwlock = LockHashPartitionLockByProc(leader);
2127	LWLockAcquire(leader_lwlock, LW_EXCLUSIVE);
2128
2129	/* Is this the leader we're looking for? */
2130	if (leader->pid == pid && leader->lockGroupLeader == leader)
2131	{
2132		/* OK, join the group */
2133		ok = true;
2134		MyProc->lockGroupLeader = leader;
2135		dlist_push_tail(&leader->lockGroupMembers, &MyProc->lockGroupLink);
2136	}
2137	LWLockRelease(leader_lwlock);
2138
2139	return ok;
2140}
