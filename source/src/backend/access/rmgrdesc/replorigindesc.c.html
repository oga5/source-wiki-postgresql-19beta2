16#include "replication/origin.h"
17
18void
19replorigin_desc(StringInfo buf, XLogReaderState *record)
20{
21	char	   *rec = XLogRecGetData(record);
22	uint8		info = XLogRecGetInfo(record) & ~XLR_INFO_MASK;
23
24	switch (info)
25	{
26		case XLOG_REPLORIGIN_SET:
27			{
28				xl_replorigin_set *xlrec;
29
30				xlrec = (xl_replorigin_set *) rec;
31
32				appendStringInfo(buf, "set %u; lsn %X/%08X; force: %d",
33								 xlrec->node_id,
34								 LSN_FORMAT_ARGS(xlrec->remote_lsn),
35								 xlrec->force);
36				break;
37			}
38		case XLOG_REPLORIGIN_DROP:
39			{
40				xl_replorigin_drop *xlrec;
41
42				xlrec = (xl_replorigin_drop *) rec;
43
44				appendStringInfo(buf, "drop %u", xlrec->node_id);
45				break;
46			}
47	}
48}
49
50const char *
51replorigin_identify(uint8 info)
52{
53	switch (info)
54	{
55		case XLOG_REPLORIGIN_SET:
56			return "SET";
57		case XLOG_REPLORIGIN_DROP:
58			return "DROP";
59		default:
60			return NULL;
61	}
62}
