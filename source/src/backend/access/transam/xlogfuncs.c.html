41/*
42 * Backup-related variables.
43 */
44static BackupState *backup_state = NULL;
45static StringInfo tablespace_map = NULL;

47/* Session-level context for the SQL-callable backup functions */
48static MemoryContext backupcontext = NULL;

51/*
52 * Return a string constant representing the recovery pause state. This is
53 * used in system functions and views, and should *not* be translated.
54 */
55static const char *
56GetRecoveryPauseStateString(RecoveryPauseState pause_state)
57{
58	const char *statestr = NULL;
59
60	switch (pause_state)
61	{
62		case RECOVERY_NOT_PAUSED:
63			statestr = "not paused";
64			break;
65		case RECOVERY_PAUSE_REQUESTED:
66			statestr = "pause requested";
67			break;
68		case RECOVERY_PAUSED:
69			statestr = "paused";
70			break;
71	}
72
73	Assert(statestr != NULL);
74	return statestr;
75}

77/*
78 * pg_backup_start: set up for taking an on-line backup dump
79 *
80 * Essentially what this does is to create the contents required for the
81 * backup_label file and the tablespace map.
82 *
83 * Permission checking for this function is managed through the normal
84 * GRANT system.
85 */
86Datum
87pg_backup_start(PG_FUNCTION_ARGS)
88{
89	text	   *backupid = PG_GETARG_TEXT_PP(0);
90	bool		fast = PG_GETARG_BOOL(1);
91	char	   *backupidstr;
92	SessionBackupState status = get_backup_status();
93	MemoryContext oldcontext;
94
95	backupidstr = text_to_cstring(backupid);
96
97	if (status == SESSION_BACKUP_RUNNING)
98		ereport(ERROR,
99				(errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
100				 errmsg("a backup is already in progress in this session")));
101
102	/*
103	 * backup_state and tablespace_map need to be long-lived as they are used
104	 * in pg_backup_stop().  These are allocated in a dedicated memory context
105	 * child of TopMemoryContext, deleted at the end of pg_backup_stop().  If
106	 * an error happens before ending the backup, memory would be leaked in
107	 * this context until pg_backup_start() is called again.
108	 */
109	if (backupcontext == NULL)
110	{
111		backupcontext = AllocSetContextCreate(TopMemoryContext,
112											  "on-line backup context",
113											  ALLOCSET_START_SMALL_SIZES);
114	}
115	else
116	{
117		backup_state = NULL;
118		tablespace_map = NULL;
119		MemoryContextReset(backupcontext);
120	}
121
122	oldcontext = MemoryContextSwitchTo(backupcontext);
123	backup_state = palloc0_object(BackupState);
124	tablespace_map = makeStringInfo();
125	MemoryContextSwitchTo(oldcontext);
126
127	register_persistent_abort_backup_handler();
128	do_pg_backup_start(backupidstr, fast, NULL, backup_state, tablespace_map);
129
130	PG_RETURN_LSN(backup_state->startpoint);
131}

134/*
135 * pg_backup_stop: finish taking an on-line backup.
136 *
137 * The first parameter (variable 'waitforarchive'), which is optional,
138 * allows the user to choose if they want to wait for the WAL to be archived
139 * or if we should just return as soon as the WAL record is written.
140 *
141 * This function stops an in-progress backup, creates backup_label contents and
142 * it returns the backup stop LSN, backup_label and tablespace_map contents.
143 *
144 * The backup_label contains the user-supplied label string (typically this
145 * would be used to tell where the backup dump will be stored), the starting
146 * time, starting WAL location for the dump and so on.  It is the caller's
147 * responsibility to write the backup_label and tablespace_map files in the
148 * data folder that will be restored from this backup.
149 *
150 * Permission checking for this function is managed through the normal
151 * GRANT system.
152 */
153Datum
154pg_backup_stop(PG_FUNCTION_ARGS)
155{
156#define PG_BACKUP_STOP_V2_COLS 3
157	TupleDesc	tupdesc;
158	Datum		values[PG_BACKUP_STOP_V2_COLS] = {0};
159	bool		nulls[PG_BACKUP_STOP_V2_COLS] = {0};
160	bool		waitforarchive = PG_GETARG_BOOL(0);
161	char	   *backup_label;
162	SessionBackupState status = get_backup_status();
163
164	/* Initialize attributes information in the tuple descriptor */
165	if (get_call_result_type(fcinfo, NULL, &tupdesc) != TYPEFUNC_COMPOSITE)
166		elog(ERROR, "return type must be a row type");
167
168	if (status != SESSION_BACKUP_RUNNING)
169		ereport(ERROR,
170				(errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
171				 errmsg("backup is not in progress"),
172				 errhint("Did you call pg_backup_start()?")));
173
174	Assert(backup_state != NULL);
175	Assert(tablespace_map != NULL);
176
177	/* Stop the backup */
178	do_pg_backup_stop(backup_state, waitforarchive);
179
180	/* Build the contents of backup_label */
181	backup_label = build_backup_content(backup_state, false);
182
183	values[0] = LSNGetDatum(backup_state->stoppoint);
184	values[1] = CStringGetTextDatum(backup_label);
185	values[2] = CStringGetTextDatum(tablespace_map->data);
186
187	/* Deallocate backup-related variables */
188	pfree(backup_label);
189
190	/* Clean up the session-level state and its memory context */
191	backup_state = NULL;
192	tablespace_map = NULL;
193	MemoryContextDelete(backupcontext);
194	backupcontext = NULL;
195
196	/* Returns the record as Datum */
197	PG_RETURN_DATUM(HeapTupleGetDatum(heap_form_tuple(tupdesc, values, nulls)));
198}

399/*
400 * Compute an xlog file name and decimal byte offset given a WAL location,
401 * such as is returned by pg_backup_stop() or pg_switch_wal().
402 */
403Datum
404pg_walfile_name_offset(PG_FUNCTION_ARGS)
405{
406	XLogSegNo	xlogsegno;
407	uint32		xrecoff;
408	XLogRecPtr	locationpoint = PG_GETARG_LSN(0);
409	char		xlogfilename[MAXFNAMELEN];
410	Datum		values[2];
411	bool		isnull[2];
412	TupleDesc	resultTupleDesc;
413	HeapTuple	resultHeapTuple;
414	Datum		result;
415
416	if (RecoveryInProgress())
417		ereport(ERROR,
418				(errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
419				 errmsg("recovery is in progress"),
420				 errhint("%s cannot be executed during recovery.",
421						 "pg_walfile_name_offset()")));
422
423	/*
424	 * Construct a tuple descriptor for the result row.  This must match this
425	 * function's pg_proc entry!
426	 */
427	resultTupleDesc = CreateTemplateTupleDesc(2);
428	TupleDescInitEntry(resultTupleDesc, (AttrNumber) 1, "file_name",
429					   TEXTOID, -1, 0);
430	TupleDescInitEntry(resultTupleDesc, (AttrNumber) 2, "file_offset",
431					   INT4OID, -1, 0);
432
433	TupleDescFinalize(resultTupleDesc);
434	resultTupleDesc = BlessTupleDesc(resultTupleDesc);
435
436	/*
437	 * xlogfilename
438	 */
439	XLByteToSeg(locationpoint, xlogsegno, wal_segment_size);
440	XLogFileName(xlogfilename, GetWALInsertionTimeLine(), xlogsegno,
441				 wal_segment_size);
442
443	values[0] = CStringGetTextDatum(xlogfilename);
444	isnull[0] = false;
445
446	/*
447	 * offset
448	 */
449	xrecoff = XLogSegmentOffset(locationpoint, wal_segment_size);
450
451	values[1] = UInt32GetDatum(xrecoff);
452	isnull[1] = false;
453
454	/*
455	 * Tuple jam: Having first prepared your Datums, then squash together
456	 */
457	resultHeapTuple = heap_form_tuple(resultTupleDesc, values, isnull);
458
459	result = HeapTupleGetDatum(resultHeapTuple);
460
461	PG_RETURN_DATUM(result);
462}

489/*
490 * Extract the sequence number and the timeline ID from given a WAL file
491 * name.
492 */
493Datum
494pg_split_walfile_name(PG_FUNCTION_ARGS)
495{
496#define PG_SPLIT_WALFILE_NAME_COLS 2
497	char	   *fname = text_to_cstring(PG_GETARG_TEXT_PP(0));
498	char	   *fname_upper;
499	char	   *p;
500	TimeLineID	tli;
501	XLogSegNo	segno;
502	Datum		values[PG_SPLIT_WALFILE_NAME_COLS] = {0};
503	bool		isnull[PG_SPLIT_WALFILE_NAME_COLS] = {0};
504	TupleDesc	tupdesc;
505	HeapTuple	tuple;
506	char		buf[256];
507	Datum		result;
508
509	fname_upper = pstrdup(fname);
510
511	/* Capitalize WAL file name. */
512	for (p = fname_upper; *p; p++)
513		*p = pg_ascii_toupper((unsigned char) *p);
514
515	if (!IsXLogFileName(fname_upper))
516		ereport(ERROR,
517				(errcode(ERRCODE_INVALID_PARAMETER_VALUE),
518				 errmsg("invalid WAL file name \"%s\"", fname)));
519
520	XLogFromFileName(fname_upper, &tli, &segno, wal_segment_size);
521
522	if (get_call_result_type(fcinfo, NULL, &tupdesc) != TYPEFUNC_COMPOSITE)
523		elog(ERROR, "return type must be a row type");
524
525	/* Convert to numeric. */
526	snprintf(buf, sizeof buf, UINT64_FORMAT, segno);
527	values[0] = DirectFunctionCall3(numeric_in,
528									CStringGetDatum(buf),
529									ObjectIdGetDatum(0),
530									Int32GetDatum(-1));
531
532	values[1] = Int64GetDatum(tli);
533
534	tuple = heap_form_tuple(tupdesc, values, isnull);
535	result = HeapTupleGetDatum(tuple);
536
537	PG_RETURN_DATUM(result);
538
539#undef PG_SPLIT_WALFILE_NAME_COLS
540}

544 *
545 * Permission checking for this function is managed through the normal

547 */
548Datum
549pg_wal_replay_pause(PG_FUNCTION_ARGS)
550{
551	if (!RecoveryInProgress())

562						 "pg_wal_replay_pause()")));
563
564	SetRecoveryPause(true);
565
566	/* wake up the recovery process so that it can process the pause request */
567	WakeupRecovery();
568
569	PG_RETURN_VOID();
570}

682/*
683 * Promotes a standby server.
684 *
685 * A result of "true" means that promotion has been completed if "wait" is
686 * "true", or initiated if "wait" is false.
687 */
688Datum
689pg_promote(PG_FUNCTION_ARGS)
690{
691	bool		wait = PG_GETARG_BOOL(0);
692	int			wait_seconds = PG_GETARG_INT32(1);
693	FILE	   *promote_file;
694	TimestampTz end_time;
695
696	if (!RecoveryInProgress())
697		ereport(ERROR,
698				(errcode(ERRCODE_OBJECT_NOT_IN_PREREQUISITE_STATE),
699				 errmsg("recovery is not in progress"),
700				 errhint("Recovery control functions can only be executed during recovery.")));
701
702	if (wait_seconds <= 0)
703		ereport(ERROR,
704				(errcode(ERRCODE_NUMERIC_VALUE_OUT_OF_RANGE),
705				 errmsg("\"wait_seconds\" must not be negative or zero")));
706
707	/* create the promote signal file */
708	promote_file = AllocateFile(PROMOTE_SIGNAL_FILE, "w");
709	if (!promote_file)
710		ereport(ERROR,
711				(errcode_for_file_access(),
712				 errmsg("could not create file \"%s\": %m",
713						PROMOTE_SIGNAL_FILE)));
714
715	if (FreeFile(promote_file))
716		ereport(ERROR,
717				(errcode_for_file_access(),
718				 errmsg("could not write file \"%s\": %m",
719						PROMOTE_SIGNAL_FILE)));
720
721	/* signal the postmaster */
722	if (kill(PostmasterPid, SIGUSR1) != 0)
723	{
724		(void) unlink(PROMOTE_SIGNAL_FILE);
725		ereport(ERROR,
726				(errcode(ERRCODE_SYSTEM_ERROR),
727				 errmsg("failed to send signal to postmaster: %m")));
728	}
729
730	/* return immediately if waiting was not requested */
731	if (!wait)
732		PG_RETURN_BOOL(true);
733
734	/* wait for the amount of time wanted until promotion */
735	end_time = TimestampTzPlusSeconds(GetCurrentTimestamp(), wait_seconds);
736	while (GetCurrentTimestamp() < end_time)
737	{
738		int			rc;
739
740		ResetLatch(MyLatch);
741
742		if (!RecoveryInProgress())
743			PG_RETURN_BOOL(true);
744
745		CHECK_FOR_INTERRUPTS();
746
747		rc = WaitLatch(MyLatch,
748					   WL_LATCH_SET | WL_TIMEOUT | WL_POSTMASTER_DEATH,
749					   100L,
750					   WAIT_EVENT_PROMOTE);
751
752		/*
753		 * Emergency bailout if postmaster has died.  This is to avoid the
754		 * necessity for manual cleanup of all postmaster children.
755		 */
756		if (rc & WL_POSTMASTER_DEATH)
757			ereport(FATAL,
758					(errcode(ERRCODE_ADMIN_SHUTDOWN),
759					 errmsg("terminating connection due to unexpected postmaster exit"),
760					 errcontext("while waiting on promotion")));
761	}
762
763	ereport(WARNING,
764			(errmsg_plural("server did not promote within %d second",
765						   "server did not promote within %d seconds",
766						   wait_seconds,
767						   wait_seconds)));
768	PG_RETURN_BOOL(false);
769}

771/*
772 * pg_stat_get_recovery - returns information about WAL recovery state
773 *
774 * Returns NULL when not in recovery or when the caller lacks
775 * pg_read_all_stats privileges; one row otherwise.
776 */
777Datum
778pg_stat_get_recovery(PG_FUNCTION_ARGS)
779{
780	TupleDesc	tupdesc;
781	Datum	   *values;
782	bool	   *nulls;
783
784	/* Local copies of shared state */
785	bool		promote_triggered;
786	XLogRecPtr	last_replayed_read_lsn;
787	XLogRecPtr	last_replayed_end_lsn;
788	TimeLineID	last_replayed_tli;
789	XLogRecPtr	replay_end_lsn;
790	TimeLineID	replay_end_tli;
791	TimestampTz recovery_last_xact_time;
792	TimestampTz current_chunk_start_time;
793	RecoveryPauseState pause_state;
794
795	if (!RecoveryInProgress())
796		PG_RETURN_NULL();
797
798	if (!has_privs_of_role(GetUserId(), ROLE_PG_READ_ALL_STATS))
799		PG_RETURN_NULL();
800
801	/* Take a lock to ensure value consistency */
802	SpinLockAcquire(&XLogRecoveryCtl->info_lck);
803	promote_triggered = XLogRecoveryCtl->SharedPromoteIsTriggered;
804	last_replayed_read_lsn = XLogRecoveryCtl->lastReplayedReadRecPtr;
805	last_replayed_end_lsn = XLogRecoveryCtl->lastReplayedEndRecPtr;
806	last_replayed_tli = XLogRecoveryCtl->lastReplayedTLI;
807	replay_end_lsn = XLogRecoveryCtl->replayEndRecPtr;
808	replay_end_tli = XLogRecoveryCtl->replayEndTLI;
809	recovery_last_xact_time = XLogRecoveryCtl->recoveryLastXTime;
810	current_chunk_start_time = XLogRecoveryCtl->currentChunkStartTime;
811	pause_state = XLogRecoveryCtl->recoveryPauseState;
812	SpinLockRelease(&XLogRecoveryCtl->info_lck);
813
814	if (get_call_result_type(fcinfo, NULL, &tupdesc) != TYPEFUNC_COMPOSITE)
815		elog(ERROR, "return type must be a row type");
816
817	values = palloc0_array(Datum, tupdesc->natts);
818	nulls = palloc0_array(bool, tupdesc->natts);
819
820	values[0] = BoolGetDatum(promote_triggered);
821
822	if (XLogRecPtrIsValid(last_replayed_read_lsn))
823		values[1] = LSNGetDatum(last_replayed_read_lsn);
824	else
825		nulls[1] = true;
826
827	if (XLogRecPtrIsValid(last_replayed_end_lsn))
828		values[2] = LSNGetDatum(last_replayed_end_lsn);
829	else
830		nulls[2] = true;
831
832	if (XLogRecPtrIsValid(last_replayed_end_lsn))
833		values[3] = Int32GetDatum(last_replayed_tli);
834	else
835		nulls[3] = true;
836
837	if (XLogRecPtrIsValid(replay_end_lsn))
838		values[4] = LSNGetDatum(replay_end_lsn);
839	else
840		nulls[4] = true;
841
842	if (XLogRecPtrIsValid(replay_end_lsn))
843		values[5] = Int32GetDatum(replay_end_tli);
844	else
845		nulls[5] = true;
846
847	if (recovery_last_xact_time != 0)
848		values[6] = TimestampTzGetDatum(recovery_last_xact_time);
849	else
850		nulls[6] = true;
851
852	if (current_chunk_start_time != 0)
853		values[7] = TimestampTzGetDatum(current_chunk_start_time);
854	else
855		nulls[7] = true;
856
857	values[8] = CStringGetTextDatum(GetRecoveryPauseStateString(pause_state));
858
859	PG_RETURN_DATUM(HeapTupleGetDatum(heap_form_tuple(tupdesc, values, nulls)));
860}
