src/makefiles/Makefile.aix
1# MAKE_EXPORTS is required for svr4 loaders that want a file of
2# symbol names to tell them what to export/import.
3MAKE_EXPORTS= true
4
5# -blibpath must contain ALL directories where we should look for libraries
6libpath := $(shell echo $(subst -L,:,$(filter -L/%,$(LDFLAGS))) | sed -e's/ //g'):/opt/freeware/lib:/usr/lib:/lib
7
8# when building with gcc, need to make sure that libgcc can be found
9ifeq ($(GCC), yes)
10libpath := $(libpath):$(dir $(shell gcc -print-libgcc-file-name))
11endif
12
13rpath = -Wl,-blibpath:'$(rpathdir)$(libpath)'
14
15LDFLAGS_SL += -Wl,-bnoentry -Wl,-H512 -Wl,-bM:SRE
16
17# gcc needs to know it's building a shared lib, otherwise it'll not emit
18# correct code / link to the right support libraries
19ifeq ($(GCC), yes)
20LDFLAGS_SL += -shared
21endif
22
23# instruct ar to process 64-bit objects
24AROPT := -X64 $(AROPT)
25
26# env var name to use in place of LD_LIBRARY_PATH
27ld_library_path_var = LIBPATH
28
29
30POSTGRES_IMP= postgres.imp
31
32ifdef PGXS
33BE_DLLLIBS= -Wl,-bI:$(pkglibdir)/$(POSTGRES_IMP)
34else
35BE_DLLLIBS= -Wl,-bI:$(top_builddir)/src/backend/$(POSTGRES_IMP)
36endif
37
38MKLDEXPORT_DIR=src/backend/port/aix
39MKLDEXPORT=$(top_srcdir)/$(MKLDEXPORT_DIR)/mkldexport.sh
40
41%$(DLSUFFIX): %.o
42	$(CC) $(CFLAGS) $*.o $(LDFLAGS) $(LDFLAGS_SL) -o $@ $(BE_DLLLIBS)
