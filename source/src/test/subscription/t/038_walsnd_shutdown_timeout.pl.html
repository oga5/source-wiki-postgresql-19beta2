4# Checks that the publisher is able to shut down without
5# waiting for sending of all pending data to the subscriber
6# with wal_sender_shutdown_timeout set
7use strict;
8use warnings FATAL => 'all';
9use PostgreSQL::Test::Cluster;
10use PostgreSQL::Test::Utils;
11use Test::More;
12use Time::HiRes qw(usleep);
13
14# Initialize publisher node
15my $node_publisher = PostgreSQL::Test::Cluster->new('publisher');
16$node_publisher->init(allows_streaming => 'logical');
17$node_publisher->append_conf(
18	'postgresql.conf',
19	qq(wal_sender_timeout = 1h
20	 wal_sender_shutdown_timeout = 10ms));
21$node_publisher->start;
22
23# Initialize subscriber node
24my $node_subscriber = PostgreSQL::Test::Cluster->new('subscriber');
25$node_subscriber->init;
26$node_subscriber->start;
27
28# Create publication for test table
29$node_publisher->safe_psql(
30	'postgres', qq(
31	CREATE TABLE test_tab (id int PRIMARY KEY);
32	CREATE PUBLICATION test_pub FOR TABLE test_tab;
33));
34
35# Create matching table and subscription on subscriber
36my $publisher_connstr = $node_publisher->connstr . ' dbname=postgres';
37$node_subscriber->safe_psql(
38	'postgres', qq(
39	CREATE TABLE test_tab (id int PRIMARY KEY);
40	CREATE SUBSCRIPTION test_sub CONNECTION '$publisher_connstr' PUBLICATION test_pub	WITH (failover = true);
41));
42
43# Wait for initial table sync to finish
44$node_subscriber->wait_for_subscription_sync($node_publisher, 'test_sub');
45
46# Start a background session on the subscriber to run a transaction later
47# that will block the logical apply worker on a lock.
48my $sub_session = $node_subscriber->background_psql('postgres');
49
50# Test that when the logical apply worker is blocked on a lock and replication
51# is stalled, shutting down the publisher causes the logical walsender to exit
52# due to wal_sender_shutdown_timeout, allowing shutdown to complete.
53
54# Cause the logical apply worker to block on a lock by running conflicting
55# transactions on the publisher and subscriber.
56$sub_session->query_safe("BEGIN; INSERT INTO test_tab VALUES (0);");
57$node_publisher->safe_psql('postgres', "INSERT INTO test_tab VALUES (0);");
58
59my $log_offset = -s $node_publisher->logfile;
60
61# Verify that the walsender exits due to wal_sender_shutdown_timeout.
62$node_publisher->stop('fast');
63ok( $node_publisher->log_contains(
64		qr/WARNING: .* terminating walsender process due to replication shutdown timeout/,
65		$log_offset),
66	"walsender exits due to wal_sender_shutdown_timeout");
67
68$sub_session->query_safe("ABORT;");
69$node_publisher->start;
70$node_publisher->wait_for_catchup('test_sub');
71
72# Test that when the logical apply worker is blocked on a lock, replication
73# is stalled, and the logical walsender's output buffer is full, shutting down
74# the publisher causes the walsender to exit due to
75# wal_sender_shutdown_timeout, allowing shutdown to complete.
76#
77# This test differs from the previous one in that the walsender's output
78# buffer is full (because pending data cannot be transferred).
79
80# Run a transaction on the subscriber that blocks the logical apply worker
81# on a lock.
82$sub_session->query_safe("BEGIN; LOCK TABLE test_tab IN EXCLUSIVE MODE;");
83
84# Generate enough data to fill the logical walsender's output buffer.
85$node_publisher->safe_psql('postgres',
86	"INSERT INTO test_tab VALUES (generate_series(1, 20000));");
87
88# Wait for the logical walsender's output buffer to fill. If the WAL send
89# positions do not advance between checks, treat the buffer as full.
90my $last_sent_lsn = $node_publisher->safe_psql('postgres',
91	"SELECT sent_lsn FROM pg_stat_replication WHERE application_name = 'test_sub';"
92);
93
94my $max_attempts = $PostgreSQL::Test::Utils::timeout_default * 10;
95while ($max_attempts-- >= 0)
96{
97	usleep(100_000);
98
99	my $cur_sent_lsn = $node_publisher->safe_psql('postgres',
100		"SELECT sent_lsn FROM pg_stat_replication WHERE application_name = 'test_sub';"
101	);
102
103	my $diff = $node_publisher->safe_psql('postgres',
104		"SELECT pg_wal_lsn_diff('$cur_sent_lsn', '$last_sent_lsn');");
105	last if $diff == 0;
106
107	$last_sent_lsn = $cur_sent_lsn;
108}
109
110$log_offset = -s $node_publisher->logfile;
111
112# Verify that the walsender exits due to wal_sender_shutdown_timeout.
113$node_publisher->stop('fast');
114ok( $node_publisher->log_contains(
115		qr/WARNING: .* terminating walsender process due to replication shutdown timeout/,
116		$log_offset),
117	"walsender with full output buffer exits due to wal_sender_shutdown_timeout"
118);
119
120$sub_session->query_safe("ABORT;");
121
122# The next test depends on Perl's `kill`, which apparently is not
123# portable to Windows.  (It would be nice to use Test::More's `subtest`,
124# but that's not in the ancient version we require.)
125if ($PostgreSQL::Test::Utils::windows_os)
126{
127	$node_subscriber->stop('fast');
128	done_testing();
129	exit;
130}
131
132$node_publisher->start;
133
134# Test that wal_sender_shutdown_timeout works correctly when both physical
135# and logical replication are active, and slot synchronization is running on
136# the standby.
137#
138# In this scenario, the logical apply worker is blocked on a lock and
139# the standby's walreceiver is stopped (via SIGSTOP signal), stalling both
140# replication streams. Verify that shutting down the publisher (primary)
141# causes both physical and logical walsenders to exit due to
142# wal_sender_shutdown_timeout, allowing shutdown to complete.
143#
144# Skip this test on Windows.
145
146# Create the standby with slot synchronization enabled.
147$node_publisher->backup(
148	'publisher_backup',
149	backup_options => [
150		'--create-slot', '--slot',
151		'test_slot', '-d',
152		'dbname=postgres', '--write-recovery-conf'
153	]);
154
155$node_publisher->append_conf('postgresql.conf',
156	"synchronized_standby_slots = 'test_slot'");
157$node_publisher->reload;
158
159my $node_standby = PostgreSQL::Test::Cluster->new('standby');
160$node_standby->init_from_backup($node_publisher, 'publisher_backup');
161$node_standby->append_conf(
162	'postgresql.conf',
163	qq(sync_replication_slots = on
164	hot_standby_feedback = on));
165$node_standby->start;
166
167# Cause the logical apply worker to block on a lock by running conflicting
168# transactions on the publisher and subscriber, stalling logical replication.
169$node_publisher->wait_for_catchup('test_sub');
170$sub_session->query_safe("BEGIN; LOCK TABLE test_tab IN EXCLUSIVE MODE;");
171$node_publisher->safe_psql('postgres', "INSERT INTO test_tab VALUES (-1); ");
172
173# Cause the standby's walreceiver to be blocked with SIGSTOP signal,
174# stalling physical replication.
175$node_standby->poll_query_until('postgres',
176	"SELECT EXISTS(SELECT 1 FROM pg_stat_wal_receiver)");
177my $receiverpid = $node_standby->safe_psql('postgres',
178	"SELECT pid FROM pg_stat_wal_receiver");
179like($receiverpid, qr/^[0-9]+$/, "have walreceiver pid $receiverpid");
180kill 'STOP', $receiverpid;
181
182$log_offset = -s $node_publisher->logfile;
183
184# Verify that the walsender exits due to wal_sender_shutdown_timeout
185# even when both physical and logical replication are stalled.
186# wal_sender_shutdown_timeout.
187$node_publisher->safe_psql('postgres', "INSERT INTO test_tab VALUES (-2);");
188$node_publisher->stop('fast');
189ok( $node_publisher->log_contains(
190		qr/WARNING: .* terminating walsender process due to replication shutdown timeout/,
191		$log_offset),
192	"walsender exits due to wal_sender_shutdown_timeout even when both physical and logical replication are stalled"
193);
194
195kill 'CONT', $receiverpid;
196$sub_session->quit;
197
198$node_subscriber->stop('fast');
199$node_standby->stop('fast');
200
201done_testing();
