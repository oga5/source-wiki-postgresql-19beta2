src/test/isolation/specs/cluster-toast-value-reuse.spec
1# Tests with CLUSTER for toast values
2
3# This test does a relation rewrite, with toast values reused to make the
4# rewrite cheaper (see data_todo = 0 case in toast_save_datum()).
5#
6# A first session updates the table with an attribute not toasted.  CLUSTER
7# is then executed in a second session.  The comparison of the values
8# allocated for the toasted values are done using a CTAS.  The allocated
9# chunk_ids are saved before the rewrite, and compared after the rewrite.
10
11# ---------- global setup ----------
12setup
13{
14  DROP TABLE IF EXISTS cluster_toast_value CASCADE;
15  DROP TABLE IF EXISTS cluster_chunk_id CASCADE;
16
17  CREATE TABLE cluster_toast_value (
18    id    serial PRIMARY KEY,
19    flag  integer,
20    value text);
21
22  -- Make sure 'value' is large enough to be toasted.
23  ALTER TABLE cluster_toast_value ALTER COLUMN value SET STORAGE EXTERNAL;
24
25  -- Clustering index.
26  CLUSTER cluster_toast_value_pkey ON cluster_toast_value;
27
28  -- Seed data: one row with big string to force TOAST tuple and trigger the todo=0 code path.
29  INSERT INTO cluster_toast_value(flag, value)
30    VALUES (0, repeat(encode(sha256('1'), 'hex'), 120) || repeat('x', 8000));
31
32  CLUSTER cluster_toast_value;
33}
34
35teardown
36{
37  DROP TABLE IF EXISTS cluster_toast_value;
38  DROP TABLE IF EXISTS cluster_chunk_id;
39}
40
41session s1
42step s1_begin  { BEGIN; }
43step s1_update { UPDATE cluster_toast_value SET flag = 1 WHERE TRUE; }
44step s1_commit { COMMIT; }
45
46session s2
47# Store the primary key values and their associated chunk IDs.  This makes
48# sure that some data is captured.
49step s2_store_chunk_ids {
50  CREATE TABLE cluster_chunk_id AS
51    SELECT c.id, pg_column_toast_chunk_id(c.value) AS chunk_id
52    FROM cluster_toast_value c;
53  SELECT count(*) FROM cluster_chunk_id;
54}
55step s2_cluster { CLUSTER cluster_toast_value; }
56
57# Verify that toast values allocated are the same, indicating reuse.
58# This query reports the tuples with toast values that do not match.
59step s2_verify_chunk_ids {
60  SELECT o.id AS chunk_ids_preserved
61    FROM cluster_chunk_id o
62    JOIN cluster_toast_value c ON o.id = c.id
63    WHERE o.chunk_id != pg_column_toast_chunk_id(c.value);
64}
65
66# Run UPDATE with its transaction still open, then store the chunk IDs.
67# CLUSTER will wait until the first transaction commit.  Finally, the chunk
68# IDs are compared.
69permutation s1_begin s1_update s2_store_chunk_ids s2_cluster s1_commit s2_verify_chunk_ids
