src/test/isolation/specs/ddl-dependency-locking.spec
1# Test that concurrent DROP and CREATE commands do not leave behind
2# references to non-existent objects.
3
4setup
5{
6	CREATE SCHEMA testschema;
7	CREATE SCHEMA alterschema;
8	CREATE TYPE public.foo as enum ('one', 'two');
9	CREATE TYPE public.footab as enum ('three', 'four');
10	CREATE DOMAIN id AS int;
11	CREATE FUNCTION f() RETURNS int LANGUAGE SQL RETURN 1;
12	CREATE FUNCTION public.falter() RETURNS int LANGUAGE SQL RETURN 1;
13	CREATE FOREIGN DATA WRAPPER fdw_wrapper;
14	CREATE ROLE regress_dependency;
15}
16
17teardown
18{
19	DROP FUNCTION IF EXISTS testschema.foo();
20	DROP FUNCTION IF EXISTS fooargtype(num foo);
21	DROP FUNCTION IF EXISTS footrettype();
22	DROP FUNCTION IF EXISTS foofunc();
23	DROP FUNCTION IF EXISTS public.falter();
24	DROP FUNCTION IF EXISTS alterschema.falter();
25	DROP DOMAIN IF EXISTS idid;
26	DROP SERVER IF EXISTS srv_fdw_wrapper;
27	DROP TABLE IF EXISTS tabtype;
28	DROP SCHEMA IF EXISTS testschema;
29	DROP SCHEMA IF EXISTS alterschema;
30	DROP TYPE IF EXISTS public.foo;
31	DROP TYPE IF EXISTS public.footab;
32	DROP DOMAIN IF EXISTS id;
33	DROP FUNCTION IF EXISTS f();
34	DROP FOREIGN DATA WRAPPER IF EXISTS fdw_wrapper;
35	DROP ROLE regress_dependency;
36}
37
38session "s1"
39
40step "s1_begin" { BEGIN; }
41step "s1_create_function_in_schema" { CREATE FUNCTION testschema.foo() RETURNS int AS 'select 1' LANGUAGE sql; }
42step "s1_create_function_with_argtype" { CREATE FUNCTION fooargtype(num foo) RETURNS int AS 'select 1' LANGUAGE sql; }
43step "s1_create_function_with_rettype" { CREATE FUNCTION footrettype() RETURNS id LANGUAGE sql RETURN 1; }
44step "s1_create_function_with_function" { CREATE FUNCTION foofunc() RETURNS int LANGUAGE SQL RETURN f() + 1; }
45step "s1_alter_function_owner" { ALTER FUNCTION public.falter() OWNER TO regress_dependency; }
46step "s1_alter_function_schema" { ALTER FUNCTION public.falter() SET SCHEMA alterschema; }
47step "s1_create_domain_with_domain" { CREATE DOMAIN idid as id; }
48step "s1_create_table_with_type" { CREATE TABLE tabtype(a footab); }
49step "s1_create_server_with_fdw_wrapper" { CREATE SERVER srv_fdw_wrapper FOREIGN DATA WRAPPER fdw_wrapper; }
50step "s1_commit" { COMMIT; }
51
52session "s2"
53
54step "s2_begin" { BEGIN; }
55step "s2_drop_schema" { DROP SCHEMA testschema; }
56step "s2_drop_alterschema" { DROP SCHEMA alterschema; }
57step "s2_drop_foo_type" { DROP TYPE public.foo; }
58step "s2_drop_foo_rettype" { DROP DOMAIN id; }
59step "s2_drop_footab_type" { DROP TYPE public.footab; }
60step "s2_drop_function_f" { DROP FUNCTION f(); }
61step "s2_drop_domain_id" { DROP DOMAIN id; }
62step "s2_drop_fdw_wrapper" { DROP FOREIGN DATA WRAPPER fdw_wrapper RESTRICT; }
63step "s2_drop_role" { DROP ROLE regress_dependency; }
64step "s2_commit" { COMMIT; }
