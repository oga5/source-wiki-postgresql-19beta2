src/test/isolation/specs/async-notify.spec
1# Tests for LISTEN/NOTIFY
2
3# Most of these tests use only the "notifier" session and hence exercise only
4# self-notifies, which are convenient because they minimize timing concerns.
5# Note we assume that each step is delivered to the backend as a single Query
6# message so it will run as one transaction.
7
8session notifier
9step listenc	{ LISTEN c1; LISTEN c2; }
10step notify1	{ NOTIFY c1; }
11step notify2	{ NOTIFY c2, 'payload'; }
12step notify3	{ NOTIFY c3, 'payload3'; }  # not listening to c3
13step notifyf	{ SELECT pg_notify('c2', NULL); }
14step notifyd1	{ NOTIFY c2, 'payload'; NOTIFY c1; NOTIFY "c2", 'payload'; }
15step notifyd2	{ NOTIFY c1; NOTIFY c1; NOTIFY c1, 'p1'; NOTIFY c1, 'p2'; }
16step notifys1	{
17	BEGIN;
18	NOTIFY c1, 'payload'; NOTIFY "c2", 'payload';
19	NOTIFY c1, 'payload'; NOTIFY "c2", 'payload';
20	SAVEPOINT s1;
21	NOTIFY c1, 'payload'; NOTIFY "c2", 'payload';
22	NOTIFY c1, 'payloads'; NOTIFY "c2", 'payloads';
23	NOTIFY c1, 'payload'; NOTIFY "c2", 'payload';
24	NOTIFY c1, 'payloads'; NOTIFY "c2", 'payloads';
25	RELEASE SAVEPOINT s1;
26	SAVEPOINT s2;
27	NOTIFY c1, 'rpayload'; NOTIFY "c2", 'rpayload';
28	NOTIFY c1, 'rpayloads'; NOTIFY "c2", 'rpayloads';
29	NOTIFY c1, 'rpayload'; NOTIFY "c2", 'rpayload';
30	NOTIFY c1, 'rpayloads'; NOTIFY "c2", 'rpayloads';
31	ROLLBACK TO SAVEPOINT s2;
32	COMMIT;
33}
34step notifys_simple	{
35	BEGIN;
36	SAVEPOINT s1;
37	NOTIFY c1, 'simple1';
38	NOTIFY c2, 'simple2';
39	RELEASE SAVEPOINT s1;
40	COMMIT;
41}
42step notify_many_with_dup	{
43	BEGIN;
44	SELECT pg_notify('c1', 'msg' || s::text) FROM generate_series(1, 17) s;
45	SELECT pg_notify('c1', 'msg1');
46	COMMIT;
47}
48step usage		{ SELECT pg_notification_queue_usage() > 0 AS nonzero; }
49step bignotify	{ SELECT count(pg_notify('c1', s::text)) FROM generate_series(1, 1000) s; }
50teardown		{ UNLISTEN *; }
51
52# The listener session is used for cross-backend notify checks.
53
54session listener
55step llisten	{ LISTEN c1; LISTEN c2; }
56step lcheck		{ SELECT 1 AS x; }
57step lbegin		{ BEGIN; }
58step lbegins	{ BEGIN ISOLATION LEVEL SERIALIZABLE; }
59step lcommit	{ COMMIT; }
60step lunlisten_all	{ BEGIN; LISTEN c1; UNLISTEN *; COMMIT; }
61teardown		{ UNLISTEN *; }
62
63# In some tests we need a second listener, just to block the queue.
64
65session listener2
66step l2listen	{ LISTEN c1; }
67step l2begin	{ BEGIN; }
68step l2commit	{ COMMIT; }
69step l2stop		{ UNLISTEN *; }
70
71# Listener session for cross-session notification test with channel 'ch'.
72
73session listener_ch
74step lch_listen	{ LISTEN ch; }
75step lch_check	{ SELECT 1 AS x; }
76teardown		{ UNLISTEN *; }
77
78# Notifier session for cross-session notification test with channel 'ch'.
79
80session notifier_ch
81step nch_notify	{ NOTIFY ch, 'aa'; }
82
83# Session for testing LISTEN in subtransaction with separate steps.
84
85session listen_subxact
86step lsbegin	{ BEGIN; }
87step lslisten_outer	{ LISTEN c3; }
88step lssavepoint	{ SAVEPOINT s1; }
89step lslisten	{ LISTEN c1; LISTEN c2; }
90step lsrelease	{ RELEASE SAVEPOINT s1; }
91step lsrollback	{ ROLLBACK TO SAVEPOINT s1; }
92step lscommit	{ COMMIT; }
93step lsnotify	{ NOTIFY c1, 'subxact_test'; }
94step lsnotify_check	{ NOTIFY c1, 'should_not_receive'; }
95teardown		{ UNLISTEN *; }
96

98# Trivial cases.
99permutation listenc notify1 notify2 notify3 notifyf
100
101# Check simple and less-simple deduplication.
102permutation listenc notifyd1 notifyd2 notifys1
103
104# Check simple NOTIFY reparenting when parent has no action.
105permutation listenc notifys_simple
106
107# Check LISTEN reparenting in subtransaction.
108permutation lsbegin lssavepoint lslisten lsrelease lscommit lsnotify
109
110# Check LISTEN merge path when both outer and inner transactions have actions.
111permutation lsbegin lslisten_outer lssavepoint lslisten lsrelease lscommit lsnotify
112
113# Check LISTEN abort path (ROLLBACK TO SAVEPOINT discards pending actions).
114permutation lsbegin lssavepoint lslisten lsrollback lscommit lsnotify_check
115
116# Check UNLISTEN * cancels a LISTEN in the same transaction.
117permutation lunlisten_all notify1 lcheck
118
119# Check notification_match function (triggered by hash table duplicate detection).
120permutation listenc notify_many_with_dup
121
122# Cross-backend notification delivery.  We use a "select 1" to force the
123# listener session to check for notifies.  In principle we could just wait
124# for delivery, but that would require extra support in isolationtester
125# and might have portability-of-timing issues.
126permutation llisten notify1 notify2 notify3 notifyf lcheck
127
128# Again, with local delivery too.
129permutation listenc llisten notify1 notify2 notify3 notifyf lcheck
130
131# Check for bug when initial listen is only action in a serializable xact,
132# and notify queue is not empty
133permutation l2listen l2begin notify1 lbegins llisten lcommit l2commit l2stop
134
135# Check that notifications sent from a backend that has not done LISTEN
136# are properly delivered to a listener in another backend.
137permutation lch_listen nch_notify lch_check
138
139# Verify that pg_notification_queue_usage correctly reports a non-zero result,
140# after submitting notifications while another connection is listening for
141# those notifications and waiting inside an active transaction.  We have to
142# fill a page of the notify SLRU to make this happen, which is a good deal
143# of traffic.  To not bloat the expected output, we intentionally don't
144# commit the listener's transaction, so that it never reports these events.
145# Hence, this should be the last test in this script.
146
147permutation llisten lbegin usage bignotify usage
