1# RI Trigger test
2#
3# Test C-based referential integrity enforcement.
4# Under REPEATABLE READ we need some snapshot trickery in C,
5# or we would permit things that violate referential integrity.
6
7setup
8{
9  CREATE TABLE parent (parent_id SERIAL NOT NULL PRIMARY KEY);
10  CREATE TABLE child (
11	child_id SERIAL NOT NULL PRIMARY KEY,
12	parent_id INTEGER REFERENCES parent);
13  INSERT INTO parent VALUES(1);
14}
15
16teardown { DROP TABLE parent, child; }
17
18session s1
19step s1rc	{ BEGIN ISOLATION LEVEL READ COMMITTED; }
20step s1rr	{ BEGIN ISOLATION LEVEL REPEATABLE READ; }
21step s1ser	{ BEGIN ISOLATION LEVEL SERIALIZABLE; }
22step s1del	{ DELETE FROM parent WHERE parent_id = 1; }
23step s1c	{ COMMIT; }
24
25session s2
26step s2rc	{ BEGIN ISOLATION LEVEL READ COMMITTED; }
27step s2rr	{ BEGIN ISOLATION LEVEL REPEATABLE READ; }
28step s2ser	{ BEGIN ISOLATION LEVEL SERIALIZABLE; }
29step s2ins	{ INSERT INTO child VALUES (1, 1); }
30step s2c	{ COMMIT; }
31
32# Violates referential integrity unless we use a crosscheck snapshot,
33# which is up-to-date compared with the transaction's snapshot.
34permutation s1rr s2rr s2ins s1del s2c s1c
35
36# Raises a can't-serialize exception
37# when the INSERT trigger does SELECT FOR KEY SHARE:
38permutation s1rr s2rr s1del s2ins s1c s2c
39
40# Test the same scenarios in READ COMMITTED:
41# A crosscheck snapshot is not required here.
42permutation s1rc s2rc s2ins s1del s2c s1c
43permutation s1rc s2rc s1del s2ins s1c s2c
44
45# Test the same scenarios in SERIALIZABLE:
46# We should report the FK violation:
47permutation s1ser s2ser s2ins s1del s2c s1c
48# We raise a concurrent update error
49# which is good enough:
50permutation s1ser s2ser s1del s2ins s1c s2c
