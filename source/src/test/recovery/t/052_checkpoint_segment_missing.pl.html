3# Verify crash recovery behavior when the WAL segment containing the
4# checkpoint record referenced by pg_controldata is missing.  This
5# checks the code path where there is no backup_label file, where the
6# startup process should fail with FATAL and log a message about the
7# missing checkpoint record.
8
9use strict;
10use warnings FATAL => 'all';
11use PostgreSQL::Test::Cluster;
12use PostgreSQL::Test::Utils;
13use Test::More;
14
15my $node = PostgreSQL::Test::Cluster->new('testnode');
16$node->init;
17$node->append_conf('postgresql.conf', 'log_checkpoints = on');
18$node->start;
19
20# Force a checkpoint so as pg_controldata points to a checkpoint record we
21# can target.
22$node->safe_psql('postgres', 'CHECKPOINT;');
23
24# Retrieve the checkpoint LSN and derive the WAL segment name.
25my $checkpoint_walfile = $node->safe_psql('postgres',
26	"SELECT pg_walfile_name(checkpoint_lsn) FROM pg_control_checkpoint()");
27
28ok($checkpoint_walfile ne '',
29	"derived checkpoint WAL file name: $checkpoint_walfile");
30
31# Stop the node.
32$node->stop('immediate');
33
34# Remove the WAL segment containing the checkpoint record.
35my $walpath = $node->data_dir . "/pg_wal/$checkpoint_walfile";
36ok(-f $walpath, "checkpoint WAL file exists before deletion: $walpath");
37
38unlink $walpath
39  or die "could not remove WAL file $walpath: $!";
40
41ok(!-e $walpath, "checkpoint WAL file removed: $walpath");
42
43# Use run_log instead of node->start because this test expects that
44# the server ends with an error during recovery.
45run_log(
46	[
47		'pg_ctl',
48		'--pgdata' => $node->data_dir,
49		'--log' => $node->logfile,
50		'start',
51	]);
52
53# Confirm that recovery has failed as expected.
54my $logfile = slurp_file($node->logfile());
55ok( $logfile =~
56	  qr/FATAL: .* could not locate a valid checkpoint record at .*/,
57	"FATAL logged for missing checkpoint record (no backup_label path)");
58
59done_testing();
