5# Tests restarts of postgres due to crashes of a subprocess.
6#
7# Two longer-running psql subprocesses are used: One to kill a
8# backend, triggering a crash-restart cycle, one to detect when
9# postmaster noticed the backend died.  The second backend is
10# necessary because it's otherwise hard to determine if postmaster is
11# still accepting new sessions (because it hasn't noticed that the
12# backend died), or because it's already restarted.
13#
14use strict;
15use warnings FATAL => 'all';
16use PostgreSQL::Test::Cluster;
17use PostgreSQL::Test::Utils;
18use Test::More;
19
20my $psql_timeout = IPC::Run::timer($PostgreSQL::Test::Utils::timeout_default);
21
22my $node = PostgreSQL::Test::Cluster->new('primary');
23$node->init(allows_streaming => 1);
24
25# Enable pg_stat_statements to test restart of shared_preload_libraries.
26$node->append_conf(
27	'postgresql.conf',
28	qq{shared_preload_libraries = 'pg_stat_statements'
29pg_stat_statements.max = 50000
30compute_query_id = 'regress'
31});
32
33$node->start();
34
35# by default PostgreSQL::Test::Cluster doesn't restart after a crash
36$node->safe_psql(
37	'postgres', q[
38		ALTER SYSTEM SET restart_after_crash = 1;
39		ALTER SYSTEM SET log_connections = receipt;
40		SELECT pg_reload_conf();
41	]);
42
43# Remember the time that pg_stat_statements was reset. We'll use it later to
44# verify that it gets re-initialized after crash.
45my $stats_reset = $node->safe_psql(
46	'postgres', q[
47		CREATE EXTENSION pg_stat_statements;
48		SELECT stats_reset FROM pg_stat_statements_info;
49	]);
50
51# Run psql, keeping session alive, so we have an alive backend to kill.
52my ($killme_stdin, $killme_stdout, $killme_stderr) = ('', '', '');
53my $killme = IPC::Run::start(
54	[
55		'psql', '--no-psqlrc', '--quiet', '--no-align', '--tuples-only',
56		'--set' => 'ON_ERROR_STOP=1',
57		'--file' => '-',
58		'--dbname' => $node->connstr('postgres')
59	],
60	'<' => \$killme_stdin,
61	'>' => \$killme_stdout,
62	'2>' => \$killme_stderr,
63	$psql_timeout);
64
65# Need a second psql to check if crash-restart happened.
66my ($monitor_stdin, $monitor_stdout, $monitor_stderr) = ('', '', '');
67my $monitor = IPC::Run::start(
68	[
69		'psql', '--no-psqlrc', '--quiet', '--no-align', '--tuples-only',
70		'--set' => 'ON_ERROR_STOP=1',
71		'--file', '-', '--dbname' => $node->connstr('postgres')
72	],
73	'<' => \$monitor_stdin,
74	'>' => \$monitor_stdout,
75	'2>' => \$monitor_stderr,
76	$psql_timeout);
77
78#create table, insert row that should survive
79$killme_stdin .= q[
80CREATE TABLE alive(status text);
81INSERT INTO alive VALUES($$committed-before-sigquit$$);
82SELECT pg_backend_pid();
83];
84ok( pump_until(
85		$killme, $psql_timeout, \$killme_stdout, qr/[[:digit:]]+[\r\n]$/m),
86	'acquired pid for SIGQUIT');
87my $pid = $killme_stdout;
88chomp($pid);
89$killme_stdout = '';
90$killme_stderr = '';
91
92#insert a row that should *not* survive, due to in-progress xact
93$killme_stdin .= q[
94BEGIN;
95INSERT INTO alive VALUES($$in-progress-before-sigquit$$) RETURNING status;
96];
97ok( pump_until(
98		$killme, $psql_timeout,
99		\$killme_stdout, qr/in-progress-before-sigquit/m),
100	'inserted in-progress-before-sigquit');
101$killme_stdout = '';
102$killme_stderr = '';
103
104
105# Start longrunning query in second session; its failure will signal that
106# crash-restart has occurred.  The initial wait for the trivial select is to
107# be sure that psql successfully connected to backend.
108$monitor_stdin .= q[
109SELECT $$psql-connected$$;
110SELECT pg_sleep(3600);
111];
112ok( pump_until(
113		$monitor, $psql_timeout, \$monitor_stdout, qr/psql-connected/m),
114	'monitor connected');
115$monitor_stdout = '';
116$monitor_stderr = '';
117
118# kill once with QUIT - we expect psql to exit, while emitting error message first
119my $ret = PostgreSQL::Test::Utils::system_log('pg_ctl', 'kill', 'QUIT', $pid);
120
121# Exactly process should have been alive to be killed
122is($ret, 0, "killed process with SIGQUIT");
123
124# Check that psql sees the killed backend as having been terminated
125$killme_stdin .= q[
126SELECT 1;
127];
128ok( pump_until(
129		$killme,
130		$psql_timeout,
131		\$killme_stderr,
132		qr/WARNING:  terminating connection because of unexpected SIGQUIT signal|server closed the connection unexpectedly|connection to server was lost|could not send data to server/m
133	),
134	"psql query died successfully after SIGQUIT");
135$killme_stderr = '';
136$killme_stdout = '';
137$killme->finish;
138
139# Wait till server restarts - we should get the WARNING here, but
140# sometimes the server is unable to send that, if interrupted while
141# sending.
142ok( pump_until(
143		$monitor,
144		$psql_timeout,
145		\$monitor_stderr,
146		qr/WARNING:  terminating connection because of crash of another server process|server closed the connection unexpectedly|connection to server was lost|could not send data to server/m
147	),
148	"psql monitor died successfully after SIGQUIT");
149$monitor->finish;
150
151# Wait till server restarts
152is($node->poll_query_until('postgres', undef, ''),
153	"1", "reconnected after SIGQUIT");
154
155
156# restart psql processes, now that the crash cycle finished
157($killme_stdin, $killme_stdout, $killme_stderr) = ('', '', '');
158$killme->run();
159($monitor_stdin, $monitor_stdout, $monitor_stderr) = ('', '', '');
160$monitor->run();
161
162# Verify that pg_stat_statements, loaded via shared_preload_libraries,
163# was re-initialized at the crash.
164my $stats_reset_after = $node->safe_psql('postgres',
165	q[SELECT stats_reset FROM pg_stat_statements_info]);
166cmp_ok($stats_reset, 'ne', $stats_reset_after,
167	"pg_stat_statements was reset by restart");
168
169
170# Acquire pid of new backend
171$killme_stdin .= q[
172SELECT pg_backend_pid();
173];
174ok( pump_until(
175		$killme, $psql_timeout, \$killme_stdout, qr/[[:digit:]]+[\r\n]$/m),
176	"acquired pid for SIGKILL");
177$pid = $killme_stdout;
178chomp($pid);
179$killme_stdout = '';
180$killme_stderr = '';
181
182# Insert test rows
183$killme_stdin .= q[
184INSERT INTO alive VALUES($$committed-before-sigkill$$) RETURNING status;
185BEGIN;
186INSERT INTO alive VALUES($$in-progress-before-sigkill$$) RETURNING status;
187];
188ok( pump_until(
189		$killme, $psql_timeout,
190		\$killme_stdout, qr/in-progress-before-sigkill/m),
191	'inserted in-progress-before-sigkill');
192$killme_stdout = '';
193$killme_stderr = '';
194
195# Re-start longrunning query in second session; its failure will signal that
196# crash-restart has occurred.  The initial wait for the trivial select is to
197# be sure that psql successfully connected to backend.
198$monitor_stdin .= q[
199SELECT $$psql-connected$$;
200SELECT pg_sleep(3600);
201];
202ok( pump_until(
203		$monitor, $psql_timeout, \$monitor_stdout, qr/psql-connected/m),
204	'monitor connected');
205$monitor_stdout = '';
206$monitor_stderr = '';
207
208
209# kill with SIGKILL this time - we expect the backend to exit, without
210# being able to emit an error message
211$ret = PostgreSQL::Test::Utils::system_log('pg_ctl', 'kill', 'KILL', $pid);
212is($ret, 0, "killed process with KILL");
213
214# Check that psql sees the server as being terminated. No WARNING,
215# because signal handlers aren't being run on SIGKILL.
216$killme_stdin .= q[
217SELECT 1;
218];
219ok( pump_until(
220		$killme,
221		$psql_timeout,
222		\$killme_stderr,
223		qr/server closed the connection unexpectedly|connection to server was lost|could not send data to server/m
224	),
225	"psql query died successfully after SIGKILL");
226$killme->finish;
227
228# Wait till server restarts - we should get the WARNING here, but
229# sometimes the server is unable to send that, if interrupted while
230# sending.
231ok( pump_until(
232		$monitor,
233		$psql_timeout,
234		\$monitor_stderr,
235		qr/WARNING:  terminating connection because of crash of another server process|server closed the connection unexpectedly|connection to server was lost|could not send data to server/m
236	),
237	"psql monitor died successfully after SIGKILL");
238$monitor->finish;
239
240# Wait till server restarts
241is($node->poll_query_until('postgres', undef, ''),
242	"1", "reconnected after SIGKILL");
243
244# Make sure the committed rows survived, in-progress ones not
245is( $node->safe_psql('postgres', 'SELECT * FROM alive'),
246	"committed-before-sigquit\ncommitted-before-sigkill",
247	'data survived');
248
249is( $node->safe_psql(
250		'postgres',
251		'INSERT INTO alive VALUES($$before-orderly-restart$$) RETURNING status'
252	),
253	'before-orderly-restart',
254	'can still write after crash restart');
255
256# Confirm that the logical replication launcher, a background worker
257# without the never-restart flag, has also restarted successfully.
258is( $node->poll_query_until(
259		'postgres',
260		"SELECT count(*) = 1 FROM pg_stat_activity WHERE backend_type = 'logical replication launcher'"
261	),
262	'1',
263	'logical replication launcher restarted after crash');
264
265# Just to be sure, check that an orderly restart now still works
266$node->restart();
267
268is( $node->safe_psql('postgres', 'SELECT * FROM alive'),
269	"committed-before-sigquit\ncommitted-before-sigkill\nbefore-orderly-restart",
270	'data survived');
271
272is( $node->safe_psql(
273		'postgres',
274		'INSERT INTO alive VALUES($$after-orderly-restart$$) RETURNING status'
275	),
276	'after-orderly-restart',
277	'can still write after orderly restart');
278
279$node->stop();
280
281done_testing();
