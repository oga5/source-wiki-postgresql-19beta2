11# Helper functions to control walreceiver for testing wait conditions.
12# These allow us to stop WAL streaming so waiters block, then resume it.
13my $saved_primary_conninfo;
14
15# Stop the walreceiver on $node by clearing primary_conninfo and waiting
16# until pg_stat_wal_receiver becomes empty.  Used to freeze the
17# walreceiver-tracked positions (writtenUpto, flushedUpto) so a fencepost
18# test can rely on them not advancing.  The previous value is saved for
19# resume_walreceiver().
20sub stop_walreceiver
21{
22	my ($node) = @_;
23	$saved_primary_conninfo = $node->safe_psql(
24		'postgres', qq[
25		SELECT pg_catalog.quote_literal(setting)
26		FROM pg_settings
27		WHERE name = 'primary_conninfo';
28	]);
29	$node->safe_psql(
30		'postgres', qq[
31		ALTER SYSTEM SET primary_conninfo = '';
32		SELECT pg_reload_conf();
33	]);
34
35	$node->poll_query_until('postgres',
36		"SELECT NOT EXISTS (SELECT * FROM pg_stat_wal_receiver);");
37}
38
39# Restart the walreceiver on $node by restoring primary_conninfo to the
40# value captured by stop_walreceiver() and waiting until walreceiver
41# reconnects.  Must be paired with a prior stop_walreceiver() call.
42sub resume_walreceiver
43{
44	my ($node) = @_;
45	$node->safe_psql(
46		'postgres', qq[
47		ALTER SYSTEM SET primary_conninfo = $saved_primary_conninfo;
48		SELECT pg_reload_conf();
49	]);
50
51	$node->poll_query_until('postgres',
52		"SELECT EXISTS (SELECT * FROM pg_stat_wal_receiver);");
53}

55# Verify the wait predicate "target <= currentLSN" at the boundary.
56# Given $current_lsn (the frozen position for $mode), check that:
57#   target == current        -> success (predicate is <=)
58#   target == current - 1    -> success
59#   target == current + 1    -> timeout
60# The caller must ensure that the relevant LSN position on $node is
61# actually frozen (e.g. walreceiver stopped and replay paused), otherwise
62# the "+1" case may racily succeed.  Returns ($lsn_minus, $lsn_plus) so
63# the caller can reuse them, e.g. to drive an async wakeup test.
64sub check_wait_for_lsn_fencepost
65{
66	my ($node, $mode, $current_lsn, $label) = @_;
67
68	my $lsn_minus = $node->safe_psql('postgres',
69		"SELECT ('$current_lsn'::pg_lsn - 1)::text");
70	my $lsn_plus = $node->safe_psql('postgres',
71		"SELECT ('$current_lsn'::pg_lsn + 1)::text");
72
73	foreach my $case (
74		[ $current_lsn, 'success', 'target == current succeeds', '5s' ],
75		[ $lsn_minus, 'success', 'target == current - 1 succeeds', '5s' ],
76		[ $lsn_plus, 'timeout', 'target == current + 1 times out', '500ms' ])
77	{
78		my ($target_lsn, $expected, $desc, $timeout) = @$case;
79		my $output = $node->safe_psql(
80			'postgres', qq[
81			WAIT FOR LSN '${target_lsn}'
82				WITH (MODE '$mode', timeout '$timeout', no_throw);]);
83
84		is($output, $expected, "$label: $desc");
85	}
86
87	return ($lsn_minus, $lsn_plus);
88}

216# 4a. Check that aborting a subtransaction during WAIT FOR LSN cleans up the
217# shared wait-state.  Poll pg_stat_activity before canceling the first WAIT
218# FOR to ensure that the backend has registered itself in the waiters heap.
219# After rolling back to the savepoint, a second WAIT FOR in the same backend
220# must be able to register itself again.
221my $subxact_lsn = $node_primary->safe_psql('postgres',
222	"SELECT pg_current_wal_insert_lsn() + 10000000000");
223my $subxact_appname = 'wait_for_lsn_subxact_cleanup';
224my $subxact_session =
225  $node_primary->background_psql('postgres', on_error_stop => 0);
226$subxact_session->query_until(
227	qr/start/, qq[
228	SET application_name = '$subxact_appname';
229	BEGIN;
230	SAVEPOINT wait_cleanup;
231	\\echo start
232	WAIT FOR LSN '${subxact_lsn}' WITH (MODE 'primary_flush');
233	ROLLBACK TO wait_cleanup;
234	WAIT FOR LSN '${subxact_lsn}'
235		WITH (MODE 'primary_flush', timeout '10ms', no_throw);
236	COMMIT;
237]);
238$node_primary->poll_query_until(
239	'postgres',
240	"SELECT count(*) = 1 FROM pg_stat_activity
241	 WHERE application_name = '$subxact_appname'
242	   AND wait_event = 'WaitForWalFlush'"
243) or die "WAIT FOR LSN did not enter the primary_flush wait path";
244my $subxact_cancelled = $node_primary->safe_psql(
245	'postgres',
246	"SELECT pg_cancel_backend(pid) FROM pg_stat_activity
247	 WHERE application_name = '$subxact_appname'
248	   AND wait_event = 'WaitForWalFlush'"
249);
250is($subxact_cancelled, 't', "canceled WAIT FOR LSN in subtransaction");
251$subxact_session->quit;
252chomp($subxact_session->{stdout});
253like(
254	$subxact_session->{stderr},
255	qr/canceling statement due to user request/,
256	"query cancel interrupted WAIT FOR LSN in subtransaction");
257is($subxact_session->{stdout},
258	"timeout", "second WAIT FOR LSN timed out after savepoint rollback");
259unlike(
260	$subxact_session->{stderr},
261	qr/server closed the connection unexpectedly/,
262	"WAIT FOR LSN after savepoint rollback did not disconnect");
