32# Cover walsender error shutdown code
33my ($result, $stdout, $stderr) = $node_primary->psql(
34	'template1',
35	qq[START_REPLICATION SLOT test_slot LOGICAL 0/0],
36	replication => 'database');
37like(
38	$stderr,
39	qr/replication slot "test_slot" was not created in this database/,
40	"Logical decoding correctly fails to start");

42($result, $stdout, $stderr) = $node_primary->psql(
43	'template1',
44	qq[READ_REPLICATION_SLOT test_slot;],
45	replication => 'database');
46like(
47	$stderr,
48	qr/cannot use READ_REPLICATION_SLOT with a logical replication slot/,
49	'READ_REPLICATION_SLOT not supported for logical slots');

51# Check case of walsender not using a database connection.  Logical
52# decoding should not be allowed.
53($result, $stdout, $stderr) = $node_primary->psql(
54	'template1',
55	qq[START_REPLICATION SLOT s1 LOGICAL 0/1],
56	replication => 'true');
57like(
58	$stderr,
59	qr/ERROR:  logical decoding requires a database connection/,
60	"Logical decoding fails on non-database connection");

62$node_primary->safe_psql('postgres',
63	qq[INSERT INTO decoding_test(x,y) SELECT s, s::text FROM generate_series(1,10) s;]
64);

66# Basic decoding works
67$result = $node_primary->safe_psql('postgres',
68	qq[SELECT pg_logical_slot_get_changes('test_slot', NULL, NULL);]);
69is(scalar(my @foobar = split /^/m, $result),
70	12, 'Decoding produced 12 rows inc BEGIN/COMMIT');
71
72# If we immediately crash the server we might lose the progress we just made
73# and replay the same changes again. But a clean shutdown should never repeat
74# the same changes when we use the SQL decoding interface.
75$node_primary->restart;
76
77# There are no new writes, so the result should be empty.
78$result = $node_primary->safe_psql('postgres',
79	qq[SELECT pg_logical_slot_get_changes('test_slot', NULL, NULL);]);
80chomp($result);
81is($result, '', 'Decoding after fast restart repeats no rows');
82
83# Insert some rows and verify that we get the same results from pg_recvlogical
84# and the SQL interface.
85$node_primary->safe_psql('postgres',
86	qq[INSERT INTO decoding_test(x,y) SELECT s, s::text FROM generate_series(1,4) s;]
87);
88
89my $expected = q{BEGIN
90table public.decoding_test: INSERT: x[integer]:1 y[text]:'1'
91table public.decoding_test: INSERT: x[integer]:2 y[text]:'2'
92table public.decoding_test: INSERT: x[integer]:3 y[text]:'3'
93table public.decoding_test: INSERT: x[integer]:4 y[text]:'4'
94COMMIT};

96my $stdout_sql = $node_primary->safe_psql('postgres',
97	qq[SELECT data FROM pg_logical_slot_peek_changes('test_slot', NULL, NULL, 'include-xids', '0', 'skip-empty-xacts', '1');]
98);
99is($stdout_sql, $expected, 'got expected output from SQL decoding session');
100
101my $endpos = $node_primary->safe_psql('postgres',
102	"SELECT lsn FROM pg_logical_slot_peek_changes('test_slot', NULL, NULL) ORDER BY lsn DESC LIMIT 1;"
103);
104print "waiting to replay $endpos\n";
105
106# Insert some rows after $endpos, which we won't read.
107$node_primary->safe_psql('postgres',
108	qq[INSERT INTO decoding_test(x,y) SELECT s, s::text FROM generate_series(5,50) s;]
109);
110
111my $stdout_recv = $node_primary->pg_recvlogical_upto(
112	'postgres', 'test_slot', $endpos,
113	$PostgreSQL::Test::Utils::timeout_default,
114	'include-xids' => '0',
115	'skip-empty-xacts' => '1');
116chomp($stdout_recv);
117is($stdout_recv, $expected,
118	'got same expected output from pg_recvlogical decoding session');

120$node_primary->poll_query_until('postgres',
121	"SELECT EXISTS (SELECT 1 FROM pg_replication_slots WHERE slot_name = 'test_slot' AND active_pid IS NULL)"
122) or die "slot never became inactive";
123
124$stdout_recv = $node_primary->pg_recvlogical_upto(
125	'postgres', 'test_slot', $endpos,
126	$PostgreSQL::Test::Utils::timeout_default,
127	'include-xids' => '0',
128	'skip-empty-xacts' => '1');
129chomp($stdout_recv);
130is($stdout_recv, '', 'pg_recvlogical acknowledged changes');

179# Test logical slot advancing and its durability.
180# Passing failover=true (last arg) should not have any impact on advancing.
181my $logical_slot = 'logical_slot';
182$node_primary->safe_psql('postgres',
183	"SELECT pg_create_logical_replication_slot('$logical_slot', 'test_decoding', false, false, true);"
184);
185$node_primary->psql(
186	'postgres', "
187	CREATE TABLE tab_logical_slot (a int);
188	INSERT INTO tab_logical_slot VALUES (generate_series(1,10));");
189my $current_lsn =
190  $node_primary->safe_psql('postgres', "SELECT pg_current_wal_lsn();");
191chomp($current_lsn);
192my $psql_rc = $node_primary->psql('postgres',
193	"SELECT pg_replication_slot_advance('$logical_slot', '$current_lsn'::pg_lsn);"
194);
195is($psql_rc, '0', 'slot advancing with logical slot');
196my $logical_restart_lsn_pre = $node_primary->safe_psql('postgres',
197	"SELECT restart_lsn from pg_replication_slots WHERE slot_name = '$logical_slot';"
198);
199chomp($logical_restart_lsn_pre);
200# Slot advance should persist across clean restarts.
201$node_primary->restart;
202my $logical_restart_lsn_post = $node_primary->safe_psql('postgres',
203	"SELECT restart_lsn from pg_replication_slots WHERE slot_name = '$logical_slot';"
204);
205chomp($logical_restart_lsn_post);
206is($logical_restart_lsn_pre, $logical_restart_lsn_post,
207	"logical slot advance persists across restarts");

209my $stats_test_slot1 = 'test_slot';
210my $stats_test_slot2 = 'logical_slot';

212# Test that reset works for pg_stat_replication_slots
213
214# Stats exist for stats test slot 1
215is( $node_primary->safe_psql(
216		'postgres',
217		qq(SELECT total_bytes > 0, stats_reset IS NULL FROM pg_stat_replication_slots WHERE slot_name = '$stats_test_slot1')
218	),
219	qq(t|t),
220	qq(Total bytes is > 0 and stats_reset is NULL for slot '$stats_test_slot1'.)
221);
222
223# Do reset of stats for stats test slot 1
224$node_primary->safe_psql('postgres',
225	qq(SELECT pg_stat_reset_replication_slot('$stats_test_slot1')));
226
227# Get reset value after reset
228my $reset1 = $node_primary->safe_psql('postgres',
229	qq(SELECT stats_reset FROM pg_stat_replication_slots WHERE slot_name = '$stats_test_slot1')
230);
231
232# Do reset again
233$node_primary->safe_psql('postgres',
234	qq(SELECT pg_stat_reset_replication_slot('$stats_test_slot1')));
235
236is( $node_primary->safe_psql(
237		'postgres',
238		qq(SELECT stats_reset > '$reset1'::timestamptz, total_bytes = 0 FROM pg_stat_replication_slots WHERE slot_name = '$stats_test_slot1')
239	),
240	qq(t|t),
241	qq(Check that reset timestamp is later after the second reset of stats for slot '$stats_test_slot1' and confirm total_bytes was set to 0.)
242);
243
244# Check that test slot 2 has NULL in reset timestamp
245is( $node_primary->safe_psql(
246		'postgres',
247		qq(SELECT stats_reset IS NULL FROM pg_stat_replication_slots WHERE slot_name = '$stats_test_slot2')
248	),
249	qq(t),
250	qq(Stats_reset is NULL for slot '$stats_test_slot2' before reset.));
251
252# Get reset value again for test slot 1
253$reset1 = $node_primary->safe_psql('postgres',
254	qq(SELECT stats_reset FROM pg_stat_replication_slots WHERE slot_name = '$stats_test_slot1')
255);
256
257# Reset stats for all replication slots
258$node_primary->safe_psql('postgres',
259	qq(SELECT pg_stat_reset_replication_slot(NULL)));
260
261# Check that test slot 2 reset timestamp is no longer NULL after reset
262is( $node_primary->safe_psql(
263		'postgres',
264		qq(SELECT stats_reset IS NOT NULL FROM pg_stat_replication_slots WHERE slot_name = '$stats_test_slot2')
265	),
266	qq(t),
267	qq(Stats_reset is not NULL for slot '$stats_test_slot2' after reset all.)
268);
269
270is( $node_primary->safe_psql(
271		'postgres',
272		qq(SELECT stats_reset > '$reset1'::timestamptz FROM pg_stat_replication_slots WHERE slot_name = '$stats_test_slot1')
273	),
274	qq(t),
275	qq(Check that reset timestamp is later after resetting stats for slot '$stats_test_slot1' again.)
276);
