3# Test that effective_wal_level changes upon logical replication slot creation
4# and deletion.

6use strict;
7use warnings FATAL => 'all';
8use PostgreSQL::Test::Cluster;
9use PostgreSQL::Test::Utils;
10use Test::More;

12# Check both wal_level and effective_wal_level values on the given node
13# are expected.
14sub test_wal_level
15{
16	my ($node, $expected, $msg) = @_;
17
18	is( $node->safe_psql(
19			'postgres',
20			qq[select current_setting('wal_level'), current_setting('effective_wal_level');]
21		),
22		"$expected",
23		"$msg");
24}
25
26# Wait for the checkpointer to decrease effective_wal_level to 'replica'.
27sub wait_for_logical_decoding_disabled
28{
29	my ($node) = @_;
30
31	$node->poll_query_until('postgres',
32		qq[select current_setting('effective_wal_level') = 'replica';]);
33}
34
35# Initialize the primary server with wal_level = 'replica'.
36my $primary = PostgreSQL::Test::Cluster->new('primary');
37$primary->init(allows_streaming => 1);
38$primary->append_conf('postgresql.conf', "log_min_messages = debug1");
39$primary->start();
40
41# Check both initial wal_level and effective_wal_level values.
42test_wal_level($primary, "replica|replica",
43	"wal_level and effective_wal_level start with the same value 'replica'");
44
45# Create a physical slot and verify that it doesn't affect effective_wal_level.
46$primary->safe_psql('postgres',
47	qq[select pg_create_physical_replication_slot('test_phy_slot', false, false)]
48);
49test_wal_level($primary, "replica|replica",
50	"effective_wal_level doesn't change with a new physical slot");
51$primary->safe_psql('postgres',
52	qq[select pg_drop_replication_slot('test_phy_slot')]);
53
54# Create a temporary logical slot but exit without releasing it explicitly.
55# This enables logical decoding but skips disabling it and delegates to the
56# checkpointer.
57$primary->safe_psql('postgres',
58	qq[select pg_create_logical_replication_slot('test_tmp_slot', 'test_decoding', true)]
59);
60ok( $primary->log_contains(
61		"logical decoding is enabled upon creating a new logical replication slot"
62	),
63	"logical decoding has been enabled upon creating a temp slot");
64
65# Wait for the checkpointer to disable logical decoding.
66wait_for_logical_decoding_disabled($primary);
67
68# Test that logical decoding is disabled after repack
69$primary->safe_psql('postgres', qq[create table foo(a int primary key)]);
70$primary->safe_psql('postgres', qq[repack (concurrently) foo;]);
71ok( $primary->log_contains(
72		"logical decoding is enabled upon creating a new logical replication slot"
73	),
74	"logical decoding enabled by repack");
75
76# Wait for the checkpointer to disable logical decoding.
77wait_for_logical_decoding_disabled($primary);
78test_wal_level($primary, "replica|replica",
79	"logical decoding disabled after repack");
80
81# Create a new logical slot and check that effective_wal_level must be increased
82# to 'logical'.
83$primary->safe_psql('postgres',
84	qq[select pg_create_logical_replication_slot('test_slot', 'pgoutput')]);
85test_wal_level($primary, "replica|logical",
86	"effective_wal_level increased to 'logical' upon a logical slot creation"
87);
88
89# Restart the server and check again.
90$primary->restart();
91test_wal_level($primary, "replica|logical",
92	"effective_wal_level remains 'logical' even after a server restart");
93
94# Create and drop another logical slot, then verify that effective_wal_level remains
95# 'logical'.
96$primary->safe_psql('postgres',
97	qq[select pg_create_logical_replication_slot('test_slot2', 'pgoutput')]);
98$primary->safe_psql('postgres',
99	qq[select pg_drop_replication_slot('test_slot2')]);
100test_wal_level($primary, "replica|logical",
101	"effective_wal_level stays 'logical' as one slot remains");
102
103# Verify that the server cannot start with wal_level='minimal' when there is
104# at least one replication slot.
105$primary->adjust_conf('postgresql.conf', 'wal_level', 'minimal');
106$primary->adjust_conf('postgresql.conf', 'max_wal_senders', '0');
107$primary->stop;
108
109command_fails(
110	[
111		'pg_ctl',
112		'--pgdata' => $primary->data_dir,
113		'--log' => $primary->logfile,
114		'start',
115	],
116	"cannot start server with wal_level='minimal' as there is in-use logical slot"
117);
118
119my $logfile = slurp_file($primary->logfile());
120like(
121	$logfile,
122	qr/logical replication slot "test_slot" exists, but "wal_level" < "replica"/,
123	'logical slots requires logical decoding enabled at server startup');
124
125# Revert the modified settings.
126$primary->adjust_conf('postgresql.conf', 'wal_level', 'replica');
127$primary->adjust_conf('postgresql.conf', 'max_wal_senders', '10');
128
129# Add other settings to test if we disable logical decoding when invalidating the last
130# logical slot.
131$primary->append_conf(
132	'postgresql.conf',
133	qq[
134min_wal_size = 32MB
135max_wal_size = 32MB
136max_slot_wal_keep_size = 16MB
137]);
138$primary->start;
139
140# Advance WAL and verify that the slot gets invalidated.
141$primary->advance_wal(2);
142$primary->safe_psql('postgres', qq[CHECKPOINT]);
143is( $primary->safe_psql(
144		'postgres',
145		qq[
146select invalidation_reason = 'wal_removed' from pg_replication_slots where slot_name = 'test_slot';
147			     ]),
148	't',
149	'test_slot gets invalidated due to wal_removed');
150
151# Verify that logical decoding is disabled after invalidating the last logical slot.
152wait_for_logical_decoding_disabled($primary);
153test_wal_level($primary, "replica|replica",
154	"effective_wal_level got decreased to 'replica' after invalidating the last logical slot"
155);
156
157# Revert the modified settings, and restart the server.
158$primary->adjust_conf('postgresql.conf', 'max_slot_wal_keep_size', undef);
159$primary->adjust_conf('postgresql.conf', 'min_wal_size', undef);
160$primary->adjust_conf('postgresql.conf', 'max_wal_size', undef);
161$primary->restart;
162
163# Recreate the logical slot to enable logical decoding again.
164$primary->safe_psql('postgres',
165	qq[select pg_drop_replication_slot('test_slot')]);
166$primary->safe_psql('postgres',
167	qq[select pg_create_logical_replication_slot('test_slot', 'pgoutput')]);
168
169# Take backup during the effective_wal_level being 'logical'. But note that
170# replication slots are not included in the backup.
171$primary->backup('my_backup');
172
173# Initialize standby1 node.
174my $standby1 = PostgreSQL::Test::Cluster->new('standby1');
175$standby1->init_from_backup($primary, 'my_backup', has_streaming => 1);
176$standby1->start;
177
178# Creating a logical slot on standby should succeed as the primary enables
179# it.
180$primary->wait_for_replay_catchup($standby1);
181$standby1->create_logical_slot_on_standby($primary, 'standby1_slot',
182	'postgres');
183
184# Promote the standby1 node that has one logical slot. So effective_wal_level
185# remains 'logical' even after the promotion.
186$standby1->promote;
187test_wal_level($standby1, "replica|logical",
188	"effective_wal_level remains 'logical' even after the promotion");
189
190# Confirm if we can create a logical slot after the promotion.
191$standby1->safe_psql('postgres',
192	qq[select pg_create_logical_replication_slot('standby1_slot2', 'pgoutput')]
193);
194$standby1->stop;
195
196# Initialize standby2 node and start it with wal_level = 'logical'.
197my $standby2 = PostgreSQL::Test::Cluster->new('standby2');
198$standby2->init_from_backup($primary, 'my_backup', has_streaming => 1);
199$standby2->append_conf('postgresql.conf', qq[wal_level = 'logical']);
200$standby2->start();
201$standby2->backup('my_backup3');
202
203# Initialize cascade standby and start with wal_level = 'replica'.
204my $cascade = PostgreSQL::Test::Cluster->new('cascade');
205$cascade->init_from_backup($standby2, 'my_backup3', has_streaming => 1);
206$cascade->adjust_conf('postgresql.conf', 'wal_level', 'replica');
207$cascade->start();
208
209# Regardless of their wal_level values, effective_wal_level values on the
210# standby and the cascaded standby depend on the primary's value, 'logical'.
211test_wal_level($standby2, "logical|logical",
212	"check wal_level and effective_wal_level on standby");
213test_wal_level($cascade, "replica|logical",
214	"check wal_level and effective_wal_level on cascaded standby");
215
216# Drop the primary's last logical slot, decreasing effective_wal_level to
217# 'replica' on all nodes.
218$primary->safe_psql('postgres',
219	qq[select pg_drop_replication_slot('test_slot')]);
220wait_for_logical_decoding_disabled($primary);
221
222$primary->wait_for_replay_catchup($standby2);
223$standby2->wait_for_replay_catchup($cascade, $primary);
224
225test_wal_level($primary, "replica|replica",
226	"effective_wal_level got decreased to 'replica' on primary");
227test_wal_level($standby2, "logical|replica",
228	"effective_wal_level got decreased to 'replica' on standby");
229test_wal_level($cascade, "replica|replica",
230	"effective_wal_level got decreased to 'replica' on cascaded standby");
231
232# Promote standby2, increasing effective_wal_level to 'logical' as its wal_level
233# is set to 'logical'.
234$standby2->promote;
235
236# Verify that effective_wal_level is increased to 'logical' on the cascaded standby.
237$standby2->wait_for_replay_catchup($cascade);
238test_wal_level($cascade, "replica|logical",
239	"effective_wal_level got increased to 'logical' on standby as the new primary has wal_level='logical'"
240);
241
242$standby2->stop;
243$cascade->stop;
244
245# Initialize standby3 node and start it.
246my $standby3 = PostgreSQL::Test::Cluster->new('standby3');
247$standby3->init_from_backup($primary, 'my_backup', has_streaming => 1);
248$standby3->start;
249
250# Create logical slots on both nodes.
251$primary->safe_psql('postgres',
252	qq[select pg_create_logical_replication_slot('test_slot', 'pgoutput')]);
253$primary->wait_for_replay_catchup($standby3);
254$standby3->create_logical_slot_on_standby($primary, 'standby3_slot',
255	'postgres');
256
257# Drop the logical slot from the primary, decreasing effective_wal_level to
258# 'replica' on the primary, which leads to invalidating the logical slot on the
259# standby due to 'wal_level_insufficient'.
260$primary->safe_psql('postgres',
261	qq[select pg_drop_replication_slot('test_slot')]);
262wait_for_logical_decoding_disabled($primary);
263test_wal_level($primary, "replica|replica",
264	"effective_wal_level got decreased to 'replica' on the primary to invalidate standby's slots"
265);
266$standby3->poll_query_until(
267	'postgres', qq[
268select invalidation_reason = 'wal_level_insufficient' from pg_replication_slots where slot_name = 'standby3_slot'
269			    ]);
270
271# Restart the server to verify that the slot is successfully restored during
272# startup.
273$standby3->restart;
274
275# Check that the logical decoding is not enabled on the standby3. Note that it still has
276# the invalidated logical slot.
277test_wal_level($standby3, "replica|replica",
278	"effective_wal_level got decreased to 'replica' on standby");
279
280my ($result, $stdout, $stderr) = $standby3->psql('postgres',
281	qq[select pg_logical_slot_get_changes('standby3_slot', null, null)]);
282like(
283	$stderr,
284	qr/ERROR:  logical decoding on standby requires "effective_wal_level" >= "logical" on the primary/,
285	"cannot use logical decoding on standby as it is disabled on primary");
286
287# Restart the primary with setting wal_level = 'logical' and create a new logical
288# slot.
289$primary->append_conf('postgresql.conf', qq[wal_level = 'logical']);
290$primary->restart;
291$primary->safe_psql('postgres',
292	qq[select pg_create_logical_replication_slot('test_slot', 'pgoutput')]);
293
294# effective_wal_level should be 'logical' on both nodes.
295$primary->wait_for_replay_catchup($standby3);
296test_wal_level($primary, "logical|logical",
297	"check WAL levels on the primary node");
298test_wal_level($standby3, "replica|logical",
299	"effective_wal_level got increased to 'logical' again on standby");
300
301# Set wal_level to 'replica' and restart the primary. Since one logical slot
302# is still present on the primary, effective_wal_level remains 'logical' even
303# if wal_level got decreased to 'replica'.
304$primary->adjust_conf('postgresql.conf', 'wal_level', 'replica');
305$primary->restart;
306$primary->wait_for_replay_catchup($standby3);
307
308# Verify that the effective_wal_level remains 'logical' on both nodes
309test_wal_level($primary, "replica|logical",
310	"effective_wal_level remains 'logical' on primary even after setting wal_level to 'replica'"
311);
312test_wal_level($standby3, "replica|logical",
313	"effective_wal_level remains 'logical' on standby even after setting wal_level to 'replica' on primary"
314);
315
316# Promote the standby3 and verify that effective_wal_level got decreased to
317# 'replica' after the promotion since there is no valid logical slot.
318$standby3->promote;
319test_wal_level($standby3, "replica|replica",
320	"effective_wal_level got decreased to 'replica' as there is no valid logical slot"
321);
322
323# Cleanup the invalidated slot.
324$standby3->safe_psql('postgres',
325	qq[select pg_drop_replication_slot('standby3_slot')]);
326
327$standby3->stop;
328
329# Test the race condition at end of the recovery between the startup and logical
330# decoding status change. This test requires injection points enabled.
331if (   $ENV{enable_injection_points} eq 'yes'
332	&& $primary->check_extension('injection_points'))
333{
334	# Initialize standby4 and start it.
335	my $standby4 = PostgreSQL::Test::Cluster->new('standby4');
336	$standby4->init_from_backup($primary, 'my_backup', has_streaming => 1);
337	$standby4->start;
338
339	# Both servers have one logical slot.
340	$primary->wait_for_replay_catchup($standby4);
341	$standby4->create_logical_slot_on_standby($primary, 'standby4_slot',
342		'postgres');
343
344	# Enable and attach the injection point on the standby4.
345	$primary->safe_psql('postgres', 'create extension injection_points');
346	$primary->wait_for_replay_catchup($standby4);
347	$standby4->safe_psql('postgres',
348		qq[select injection_points_attach('startup-logical-decoding-status-change-end-of-recovery', 'wait');]
349	);
350
351	# Trigger promotion with no wait, and wait for the startup process to reach
352	# the injection point.
353	$standby4->safe_psql('postgres', qq[select pg_promote(false)]);
354	note('promote the standby and waiting for injection_point');
355	$standby4->wait_for_event('startup',
356		'startup-logical-decoding-status-change-end-of-recovery');
357	note(
358		"injection_point 'startup-logical-decoding-status-change-end-of-recovery' is reached"
359	);
360
361	# Drop the logical slot, requesting to disable logical decoding to the checkpointer.
362	$standby4->safe_psql('postgres',
363		qq[select pg_drop_replication_slot('standby4_slot');]);
364
365	# Resume the startup process to complete the recovery.
366	$standby4->safe_psql('postgres',
367		qq[select injection_points_wakeup('startup-logical-decoding-status-change-end-of-recovery')]
368	);
369
370	# Verify that logical decoding got disabled after the recovery.
371	wait_for_logical_decoding_disabled($standby4);
372	test_wal_level($standby4, "replica|replica",
373		"effective_wal_level properly got decreased to 'replica'");
374	$standby4->stop;
375
376	# Test the abort process of logical decoding activation. We drop the primary's
377	# slot to decrease its effective_wal_level to 'replica'.
378	$primary->safe_psql('postgres',
379		qq[select pg_drop_replication_slot('test_slot')]);
380	wait_for_logical_decoding_disabled($primary);
381	test_wal_level($primary, "replica|replica",
382		"effective_wal_level got decreased to 'replica' on primary");
383
384	# Start a psql session to test the case where the activation process is
385	# interrupted.
386	my $psql_create_slot = $primary->background_psql('postgres');
387
388	# Start the logical decoding activation process upon creating the logical
389	# slot, but it will wait due to the injection point.
390	$psql_create_slot->query_until(
391		qr/create_slot_canceled/,
392		q(\echo create_slot_canceled
393select injection_points_set_local();
394select injection_points_attach('logical-decoding-activation', 'wait');
395select pg_create_logical_replication_slot('slot_canceled', 'pgoutput');
396\q
397));
398
399	$primary->wait_for_event('client backend', 'logical-decoding-activation');
400	note("injection_point 'logical-decoding-activation' is reached");
401
402	# Cancel the backend initiated by $psql_create_slot, aborting its activation
403	# process.
404	$primary->safe_psql(
405		'postgres',
406		qq[
407select pg_cancel_backend(pid) from pg_stat_activity where query ~ 'slot_canceled' and pid <> pg_backend_pid()
408]);
409
410	# Verify that the backend aborted the activation process.
411	$primary->wait_for_log("aborting logical decoding activation process");
412	wait_for_logical_decoding_disabled($primary);
413	pass("the activation process aborted");
414
415	# Test concurrent activation processes run and one is interrupted.
416	$psql_create_slot = $primary->background_psql('postgres');
417
418	# Start a psql session and stops in the middle of the activation
419	# process.
420	$psql_create_slot->query_until(
421		qr/create_slot_canceled/,
422		q(\echo create_slot_canceled
423select injection_points_set_local();
424select injection_points_attach('logical-decoding-activation', 'wait');
425select pg_create_logical_replication_slot('slot_canceled2', 'pgoutput');
426\q
427));
428	$primary->wait_for_event('client backend', 'logical-decoding-activation');
429	note("injection_point 'logical-decoding-activation' is reached");
430
431	# Another backend concurrently enables logical decoding.
432	$primary->safe_psql('postgres',
433		qq[select pg_create_logical_replication_slot('test_slot2', 'pgoutput')]
434	);
435
436	# Concurrent slot creation should not be blocked. So wait until
437	# test_slot2 is created and logical decoding is enabled.
438	test_wal_level($primary, "replica|logical",
439		"the concurrent activation has done properly");
440
441	# Cancel the backend initiated by $psql_create_slot, aborting its activation
442	# process.
443	my $log_offset = -s $primary->logfile;
444	$primary->safe_psql(
445		'postgres',
446		qq[
447select pg_cancel_backend(pid) from pg_stat_activity where query ~ 'slot_canceled2' and pid <> pg_backend_pid()
448]);
449	# Canceling the backend should not affect the concurrent slot creation.
450	$primary->wait_for_log("canceling statement due to user request",
451		$log_offset);
452	test_wal_level($primary, "replica|logical",
453		"effective_wal_level remains 'logical' even after the concurrent activation is interrupted"
454	);
455}
456
457$primary->stop;
458
459done_testing();
