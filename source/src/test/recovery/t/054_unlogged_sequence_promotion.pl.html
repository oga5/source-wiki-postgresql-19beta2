3# Test that unlogged sequences created on a primary can be read after
4# promotion of a standby that replayed their init fork.
5
6use strict;
7use warnings FATAL => 'all';
8use PostgreSQL::Test::Cluster;
9use PostgreSQL::Test::Utils;
10use Test::More;
11
12my $node_primary = PostgreSQL::Test::Cluster->new('primary');
13$node_primary->init(allows_streaming => 1);
14$node_primary->start;
15
16my $backup_name = 'my_backup';
17$node_primary->backup($backup_name);
18
19my $node_standby = PostgreSQL::Test::Cluster->new('standby');
20$node_standby->init_from_backup($node_primary, $backup_name,
21	has_streaming => 1);
22$node_standby->start;
23
24# Create the unlogged sequence after the standby has started, so its init fork
25# is generated by WAL replay on the standby.
26$node_primary->safe_psql('postgres', "CREATE UNLOGGED SEQUENCE ulseq");
27$node_primary->wait_for_replay_catchup($node_standby);
28
29$node_standby->promote;
30
31is($node_standby->safe_psql('postgres', "SELECT nextval('ulseq')"),
32	1, 'unlogged sequence can be read after standby promotion');
33
34done_testing();
