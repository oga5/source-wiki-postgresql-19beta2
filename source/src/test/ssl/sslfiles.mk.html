93#
94# Special-case certificates
95#
96
97# Root CA is self-signed.
98ssl/root_ca.crt: ssl/root_ca.key conf/root_ca.config
99	$(OPENSSL) req -new -x509 -config conf/root_ca.config -days 10000 -key $< -out $@
100
101# Certificate using RSA-PSS algorithm. Also self-signed.
102ssl/server-rsapss.crt: ssl/server-rsapss.key conf/server-rsapss.config
103	$(OPENSSL) req -new -x509 -config conf/server-rsapss.config -key $< -out $@
104
105#
106# Special-case keys
107#
108# All targets here are contained in $(SPECIAL_KEYS).
109#
110
111# Password-protected version of server-cn-only.key
112ssl/server-password.key: ssl/server-cn-only.key
113	$(OPENSSL) pkey -aes256 -in $< -out $@ -passout 'pass:secret1'
114
115# Key that uses the RSA-PSS algorithm
116ssl/server-rsapss.key:
117	$(OPENSSL) genpkey -algorithm rsa-pss -out $@
118
119# DER-encoded version of client.key
120ssl/client-der.key: ssl/client.key
121	$(OPENSSL) rsa -in $< -outform DER -out $@
122
123# Convert client.key to encrypted PEM (X.509 text) and DER (X.509 ASN.1)
124# formats to test libpq's support for the sslpassword= option.
125ssl/client-encrypted-pem.key: ssl/client.key
126	$(OPENSSL) pkey -in $< -outform PEM -aes128 -passout 'pass:dUmmyP^#+' -out $@
127# TODO Explicitly choosing -aes128 generates a key unusable to PostgreSQL with
128# OpenSSL 3.0.0, so fall back on the default for now.
129ssl/client-encrypted-der.key: ssl/client.key
130	$(OPENSSL) rsa -in $< -outform DER -passout 'pass:dUmmyP^#+' -out $@

132#
133# Combined files
134#
135# All targets in $(COMBINATIONS) share a single recipe; just declare the
136# necessary dependencies and they'll be smashed together.
137#
138
139# Root certificate file that contains both CA certificates, for testing
140# that multiple certificates can be used.
141ssl/both-cas-1.crt: ssl/root_ca.crt ssl/client_ca.crt ssl/server_ca.crt
142
143# The same, but the certs are in different order
144ssl/both-cas-2.crt: ssl/root_ca.crt ssl/server_ca.crt ssl/client_ca.crt
145
146# A root certificate file for the client, to validate server certs.
147ssl/root+server_ca.crt: ssl/root_ca.crt ssl/server_ca.crt
148
149# and for the server, to validate client certs
150ssl/root+client_ca.crt: ssl/root_ca.crt ssl/client_ca.crt
151
152# and for the client, to present to the server
153ssl/client+client_ca.crt: ssl/client.crt ssl/client_ca.crt
154
155# for the server, to present to a client that only knows the root
156ssl/server-cn-only+server_ca.crt: ssl/server-cn-only.crt ssl/server_ca.crt
157
158# If a CRL is used, OpenSSL requires a CRL file for *all* the CAs in the
159# chain, even if some of them are empty.
160ssl/root+server.crl: ssl/root.crl ssl/server.crl
161ssl/root+client.crl: ssl/root.crl ssl/client.crl

175# Standard certificates

178CA_CERTS     := ssl/server_ca.crt ssl/client_ca.crt
179SERVER_CERTS := $(SERVERS:%=ssl/%.crt)
180CLIENT_CERTS := $(CLIENTS:%=ssl/%.crt)
181
182# See the "CA State" section below.
183root_ca_state_files := ssl/root_ca-certindex ssl/root_ca-certindex.attr ssl/root_ca.srl
184server_ca_state_files := ssl/server_ca-certindex ssl/server_ca-certindex.attr ssl/server_ca.srl
185client_ca_state_files := ssl/client_ca-certindex ssl/client_ca-certindex.attr ssl/client_ca.srl
186
187# These are the workhorse recipes. `openssl ca` can't be safely run from
188# parallel processes, so we must mark the entire Makefile .NOTPARALLEL.
189.NOTPARALLEL:
190$(CA_CERTS): ssl/%.crt: ssl/%.csr conf/%.config conf/cas.config ssl/root_ca.crt | ssl/new_certs_dir $(root_ca_state_files)
191	$(OPENSSL) ca -batch -config conf/cas.config -name root_ca   -notext -in $< -out $@
192
193$(SERVER_CERTS): ssl/%.crt: ssl/%.csr conf/%.config conf/cas.config ssl/server_ca.crt | ssl/new_certs_dir $(server_ca_state_files)
194	$(OPENSSL) ca -batch -config conf/cas.config -name server_ca -notext -in $< -out $@
195
196$(CLIENT_CERTS): ssl/%.crt: ssl/%.csr conf/%.config conf/cas.config ssl/client_ca.crt | ssl/new_certs_dir $(client_ca_state_files)
197	$(OPENSSL) ca -batch -config conf/cas.config -name client_ca -notext -in $< -out $@
198
199# The CSRs don't need to persist after a build.
200.INTERMEDIATE: $(CERTIFICATES:%=ssl/%.csr)
201ssl/%.csr: ssl/%.key conf/%.config
202	$(OPENSSL) req -new -utf8 -key $< -out $@ -config conf/$*.config
203
204#
205# CA State
206#
207# All of these are intended to be order-only dependencies; additionally, the
208# pattern recipes are marked as explicit intermediates. The goal is for Make to
209# create the state files once for each CA, allow them to accumulate whatever
210# state is needed, and then automatically remove them at the end of the run.
211#
212
213.INTERMEDIATE: $(root_ca_state_files) $(server_ca_state_files) $(client_ca_state_files)
214
215# OpenSSL requires a directory to put all generated certificates in. We don't
216# use this for anything, but we need a location.
217ssl/new_certs_dir:
218	mkdir $@
219
220ssl/%-certindex:
221	touch $@
222
223ssl/%-certindex.attr:
224	echo "unique_subject=no" > $@
225
226# The first serial number for each CA is based on the current timestamp, to
227# avoid collisions across Make runs.
228ssl/%.srl:
229	date +%Y%m%d%H%M%S00 > $@
230
231#
232# CRLs
233#
234
235ssl/root.crl: ssl/root_ca.crt | $(root_ca_state_files)
236	$(OPENSSL) ca -config conf/cas.config -name root_ca   -gencrl -out $@
237
238ssl/server.crl: ssl/server-revoked.crt ssl/server_ca.crt | $(server_ca_state_files)
239	$(OPENSSL) ca -config conf/cas.config -name server_ca -revoke $<
240	$(OPENSSL) ca -config conf/cas.config -name server_ca -gencrl -out $@
241
242ssl/client.crl: ssl/client-revoked.crt ssl/client-revoked-utf8.crt ssl/client_ca.crt | $(client_ca_state_files)
243	$(OPENSSL) ca -config conf/cas.config -name client_ca -revoke ssl/client-revoked.crt
244	$(OPENSSL) ca -config conf/cas.config -name client_ca -revoke ssl/client-revoked-utf8.crt
245	$(OPENSSL) ca -config conf/cas.config -name client_ca -gencrl -out $@

247#
248# CRL hash directories
249#
250
251ssl/root+server-crldir: ssl/server.crl ssl/root.crl
252ssl/root+client-crldir: ssl/client.crl ssl/root.crl
253ssl/server-crldir: ssl/server.crl
254ssl/client-crldir: ssl/client.crl
255
256crlhashfile = $(shell $(OPENSSL) crl -hash -noout -in $(1)).r0
257
258ssl/%-crldir:
259	mkdir -p $@
260	rm -f $@/*.r0
261	$(foreach crl,$^,cp $(crl) $@/$(call crlhashfile,$(crl)) &&) true
262	touch $@

264.PHONY: sslfiles-clean
265sslfiles-clean:
266	rm -f $(SSLFILES) ssl/*.old ssl/*.csr ssl/*.srl ssl/*-certindex*
267	rm -rf $(SSLDIRS) ssl/new_certs_dir
268
269# The difference between the below clean targets and sslfiles-clean is that the
270# clean targets will be run during a "standard" recursive clean run from the
271# main build tree. The sslfiles-clean target must be run explicitly from this
272# directory.
273.PHONY: clean distclean
274clean distclean:
275	rm -rf ssl/*.old ssl/new_certs_dir ssl/client*_tmp.key
