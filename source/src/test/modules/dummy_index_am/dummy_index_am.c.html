24/* parse table for fillRelOptions */
25static relopt_parse_elt di_relopt_tab[8];
26
27/* Kind of relation options for dummy index */
28static relopt_kind di_relopt_kind;
29
30typedef enum DummyAmEnum
31{
32	DUMMY_AM_ENUM_ONE,
33	DUMMY_AM_ENUM_TWO,
34}			DummyAmEnum;
35
36/* Dummy index options */
37typedef struct DummyIndexOptions
38{
39	int32		vl_len_;		/* varlena header (do not touch directly!) */
40	int			option_int;
41	double		option_real;
42	bool		option_bool;
43	pg_ternary	option_ternary_1;
44	DummyAmEnum option_enum;
45	int			option_string_val_offset;
46	int			option_string_null_offset;
47}			DummyIndexOptions;
48
49static relopt_enum_elt_def dummyAmEnumValues[] =
50{
51	{"one", DUMMY_AM_ENUM_ONE},
52	{"two", DUMMY_AM_ENUM_TWO},
53	{(const char *) NULL}		/* list terminator */
54};

59/*
60 * Validation function for string relation options.
61 */
62static void
63validate_string_option(const char *value)
64{
65	ereport(NOTICE,
66			(errmsg("new option value for string parameter %s",
67					value ? value : "NULL")));
68}
69
70/*
71 * This function creates a full set of relation option types,
72 * with various patterns.
73 */
74static void
75create_reloptions_table(void)
76{
77	int			i = 0;
78
79	di_relopt_kind = add_reloption_kind();
80
81	add_int_reloption(di_relopt_kind, "option_int",
82					  "Integer option for dummy_index_am",
83					  10, -10, 100, AccessExclusiveLock);
84	di_relopt_tab[i].optname = "option_int";
85	di_relopt_tab[i].opttype = RELOPT_TYPE_INT;
86	di_relopt_tab[i].offset = offsetof(DummyIndexOptions, option_int);
87	i++;
88
89	add_real_reloption(di_relopt_kind, "option_real",
90					   "Real option for dummy_index_am",
91					   3.1415, -10, 100, AccessExclusiveLock);
92	di_relopt_tab[i].optname = "option_real";
93	di_relopt_tab[i].opttype = RELOPT_TYPE_REAL;
94	di_relopt_tab[i].offset = offsetof(DummyIndexOptions, option_real);
95	i++;
96
97	add_bool_reloption(di_relopt_kind, "option_bool",
98					   "Boolean option for dummy_index_am",
99					   true, AccessExclusiveLock);
100	di_relopt_tab[i].optname = "option_bool";
101	di_relopt_tab[i].opttype = RELOPT_TYPE_BOOL;
102	di_relopt_tab[i].offset = offsetof(DummyIndexOptions, option_bool);
103	i++;
104
105	add_ternary_reloption(di_relopt_kind, "option_ternary_1",
106						  "One ternary option for dummy_index_am",
107						  AccessExclusiveLock);
108	di_relopt_tab[i].optname = "option_ternary_1";
109	di_relopt_tab[i].opttype = RELOPT_TYPE_TERNARY;
110	di_relopt_tab[i].offset = offsetof(DummyIndexOptions, option_ternary_1);
111	i++;
112
113	add_enum_reloption(di_relopt_kind, "option_enum",
114					   "Enum option for dummy_index_am",
115					   dummyAmEnumValues,
116					   DUMMY_AM_ENUM_ONE,
117					   "Valid values are \"one\" and \"two\".",
118					   AccessExclusiveLock);
119	di_relopt_tab[i].optname = "option_enum";
120	di_relopt_tab[i].opttype = RELOPT_TYPE_ENUM;
121	di_relopt_tab[i].offset = offsetof(DummyIndexOptions, option_enum);
122	i++;
123
124	add_string_reloption(di_relopt_kind, "option_string_val",
125						 "String option for dummy_index_am with non-NULL default",
126						 "DefaultValue", &validate_string_option,
127						 AccessExclusiveLock);
128	di_relopt_tab[i].optname = "option_string_val";
129	di_relopt_tab[i].opttype = RELOPT_TYPE_STRING;
130	di_relopt_tab[i].offset = offsetof(DummyIndexOptions,
131									   option_string_val_offset);
132	i++;
133
134	/*
135	 * String option for dummy_index_am with NULL default, and without
136	 * description.
137	 */
138	add_string_reloption(di_relopt_kind, "option_string_null",
139						 NULL,	/* description */
140						 NULL, &validate_string_option,
141						 AccessExclusiveLock);
142	di_relopt_tab[i].optname = "option_string_null";
143	di_relopt_tab[i].opttype = RELOPT_TYPE_STRING;
144	di_relopt_tab[i].offset = offsetof(DummyIndexOptions,
145									   option_string_null_offset);
146	i++;
147}

150/*
151 * Build a new index.
152 */
153static IndexBuildResult *
154dibuild(Relation heap, Relation index, IndexInfo *indexInfo)
155{
156	IndexBuildResult *result;
157
158	result = palloc_object(IndexBuildResult);
159
160	/* let's pretend that no tuples were scanned */
161	result->heap_tuples = 0;
162	/* and no index tuples were created (that is true) */
163	result->index_tuples = 0;
164
165	return result;
166}

234/*
235 * Parse relation options for index AM, returning a DummyIndexOptions
236 * structure filled with option values.
237 */
238static bytea *
239dioptions(Datum reloptions, bool validate)
240{
241	return (bytea *) build_reloptions(reloptions, validate,
242									  di_relopt_kind,
243									  sizeof(DummyIndexOptions),
244									  di_relopt_tab, lengthof(di_relopt_tab));
245}
