src/test/modules/nbtree/sql/nbtree_incomplete_splits.sql
1--
2-- Test incomplete splits in B-tree indexes.
3--
4-- We use a test table with integers from 1 to :next_i.  Each integer
5-- occurs exactly once, no gaps or duplicates, although the index does
6-- contain some duplicates because some of the inserting transactions
7-- are rolled back during the test.  The exact contents of the table
8-- depend on the physical layout of the index, which in turn depends
9-- at least on the block size, so instead of checking the exact
10-- contents, we check those invariants.  :next_i psql variable is
11-- maintained at all times to hold the last inserted integer + 1.
12--
13
14-- This uses injection points to cause errors that leave some page
15-- splits in "incomplete" state
16set client_min_messages TO 'warning';
17create extension if not exists injection_points;
18create extension if not exists amcheck;
19reset client_min_messages;
20
21-- Make all injection points local to this process, for concurrency.
22SELECT injection_points_set_local();
23
24-- Use the index for all the queries
25set enable_seqscan=off;
26
27-- Print a NOTICE whenever an incomplete split gets fixed
28SELECT injection_points_attach('nbtree-finish-incomplete-split', 'notice');
29
30--
31-- First create the test table and some helper functions
32--
33create table nbtree_incomplete_splits(i int4) with (autovacuum_enabled = off);
34
35create index nbtree_incomplete_splits_i_idx on nbtree_incomplete_splits using btree (i);
36
37-- Inserts 'n' rows to the test table. Pass :next_i as the first
38-- argument, returns the new value for :next_i.
39create function insert_n(first_i int, n int) returns int language plpgsql as $$
40begin
41  insert into nbtree_incomplete_splits select g from generate_series(first_i, first_i + n - 1) as g;
42  return first_i + n;
43end;
44$$;
45
46-- Inserts to the table until an insert fails. Like insert_n(), returns the
47-- new value for :next_i.
48create function insert_until_fail(next_i int, step int default 1) returns int language plpgsql as $$
49declare
50  i integer;
51begin
52  -- Insert rows in batches of 'step' rows each, until an error occurs.
53  i := 0;
54  loop
55    begin
56      select insert_n(next_i, step) into next_i;
57    exception when others then
58      raise notice 'failed with: %', sqlerrm;
59      exit;
60    end;
61
62    -- The caller is expected to set an injection point that eventually
63    -- causes an error. But bail out if still no error after 10000
64    -- attempts, so that we don't get stuck in an infinite loop.
65    i := i + 1;
66    if i = 10000 then
67      raise 'no error on inserts after % iterations', i;
68    end if;
69  end loop;
70
71  return next_i;
72end;
73$$;
74
75-- Check the invariants.
76create function verify(next_i int) returns bool language plpgsql as $$
77declare
78  c integer;
79begin
80  -- Perform a scan over the trailing part of the index, where the
81  -- possible incomplete splits are. (We don't check the whole table,
82  -- because that'd be pretty slow.)
83  --
84  -- Find all rows that overlap with the last 200 inserted integers. Or
85  -- the next 100, which shouldn't exist.
86  select count(*) into c from nbtree_incomplete_splits where i between next_i - 200 and next_i + 100;
87  if c <> 200 then
88    raise 'unexpected count % ', c;
89  end if;
90
91  -- Also check the index with amcheck. Both to test that the index is
92  -- valid, but also to test that amcheck doesn't wrongly complain
93  -- about incomplete splits.
94  perform bt_index_parent_check('nbtree_incomplete_splits_i_idx'::regclass, true, true);
95
96  return true;
97end;
98$$;
99
100-- Insert one array to get started.
101select insert_n(1, 1000) as next_i
102\gset
103select verify(:next_i);
104
105
106--
107-- Test incomplete leaf split
108--
109SELECT injection_points_attach('nbtree-leave-leaf-split-incomplete', 'error');
110select insert_until_fail(:next_i) as next_i
111\gset
112SELECT injection_points_detach('nbtree-leave-leaf-split-incomplete');
113
114-- Verify that a scan works even though there's an incomplete split
115select verify(:next_i);
116
117-- Insert some more rows, finishing the split
118select insert_n(:next_i, 10) as next_i
119\gset
120-- Verify that a scan still works
121select verify(:next_i);
122
123
124--
125-- Test incomplete internal page split
126--
127SELECT injection_points_attach('nbtree-leave-internal-split-incomplete', 'error');
128select insert_until_fail(:next_i, 100) as next_i
129\gset
130SELECT injection_points_detach('nbtree-leave-internal-split-incomplete');
131
132 -- Verify that a scan works even though there's an incomplete split
133select verify(:next_i);
134
135-- Insert some more rows, finishing the split
136select insert_n(:next_i, 10) as next_i
137\gset
138-- Verify that a scan still works
139select verify(:next_i);
140
141SELECT injection_points_detach('nbtree-finish-incomplete-split');
142
143drop extension amcheck;
144drop extension injection_points;
