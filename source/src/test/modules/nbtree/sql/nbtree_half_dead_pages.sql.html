src/test/modules/nbtree/sql/nbtree_half_dead_pages.sql
1--
2-- Test half-dead pages in B-tree indexes.
3--
4-- Half-dead pages is an intermediate state while vacuum is deleting a
5-- page. You can encounter them if you query concurrently with vacuum,
6-- or if vacuum is interrupted while it's deleting a page. A B-tree
7-- with half-dead pages is a valid state, but they rarely observed by
8-- other backends in practice because, so it's good to have some
9-- targeted tests to exercise them.
10--
11
12-- This uses injection points to interrupt some page deletions
13set client_min_messages TO 'warning';
14create extension if not exists injection_points;
15create extension if not exists amcheck;
16reset client_min_messages;
17
18-- Wait until all recently-dead tuples on a table become fully dead
19-- and removable by vacuum. (We don't run any concurrent transactions
20-- in the test itself, but auto-analyze can kick in at any time and
21-- hold a transaction open, holding back the vacuum horizon.)
22CREATE PROCEDURE wait_prunable() LANGUAGE plpgsql AS $$
23	DECLARE
24		barrier xid8;
25		cutoff xid8;
26	BEGIN
27		barrier := pg_current_xact_id();
28		-- Pass a shared catalog rather than the table we'll
29		-- prune, to prevent the cutoff from moving
30		-- backwards. See comments at removable_cutoff()
31		LOOP
32			ROLLBACK;  -- release MyProc->xmin, which could be the oldest
33			cutoff := removable_cutoff('pg_database');
34			EXIT WHEN cutoff >= barrier;
35			RAISE LOG 'removable cutoff %; waiting for %', cutoff, barrier;
36			PERFORM pg_sleep(.1);
37		END LOOP;
38	END
39$$;
40
41-- Make all injection points local to this process, for concurrency.
42SELECT injection_points_set_local();
43
44-- Use the index for all the queries
45set enable_seqscan=off;
46
47-- Print a NOTICE whenever a half-dead page is deleted
48SELECT injection_points_attach('nbtree-finish-half-dead-page-vacuum', 'notice');
49
50create table nbtree_half_dead_pages(id bigint) with (autovacuum_enabled = off);
51
52insert into nbtree_half_dead_pages SELECT g from generate_series(1, 150000) g;
53
54create index nbtree_half_dead_pages_id_idx on nbtree_half_dead_pages using btree (id);
55
56delete from nbtree_half_dead_pages where id > 100000 and id < 120000;
57
58-- Run VACUUM and interrupt it so that it leaves behind a half-dead page
59call wait_prunable();
60SELECT injection_points_attach('nbtree-leave-page-half-dead', 'error');
61vacuum nbtree_half_dead_pages;
62SELECT injection_points_detach('nbtree-leave-page-half-dead');
63
64select * from nbtree_half_dead_pages where id > 99998 and id < 120002;
65
66-- Also check the index with amcheck
67select bt_index_parent_check('nbtree_half_dead_pages_id_idx'::regclass, true, true);
68
69-- Finish the deletion and re-check
70vacuum nbtree_half_dead_pages;
71select * from nbtree_half_dead_pages where id > 99998 and id < 120002;
72select bt_index_parent_check('nbtree_half_dead_pages_id_idx'::regclass, true, true);
73
74drop extension amcheck;
75drop extension injection_points;
