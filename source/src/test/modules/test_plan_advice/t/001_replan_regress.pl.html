3# Run the core regression tests under pg_plan_advice to check for problems.
4use strict;
5use warnings FATAL => 'all';
6
7use Cwd            qw(abs_path);
8use File::Basename qw(dirname);
9
10use PostgreSQL::Test::Cluster;
11use PostgreSQL::Test::Utils;
12use Test::More;
13
14# Initialize the primary node
15my $node = PostgreSQL::Test::Cluster->new('main');
16$node->init();
17
18# Set up our desired configuration.
19$node->append_conf('postgresql.conf', <<EOM);
20shared_preload_libraries='test_plan_advice'
21wal_level=replica
22pg_plan_advice.always_explain_supplied_advice=false
23pg_plan_advice.feedback_warnings=true
24EOM
25$node->start;
26
27my $srcdir = abs_path("../../../..");
28
29# --dlpath is needed to be able to find the location of regress.so
30# and any libraries the regression tests require.
31my $dlpath = dirname($ENV{REGRESS_SHLIB});
32
33# --outputdir points to the path where to place the output files.
34my $outputdir = $PostgreSQL::Test::Utils::tmp_check;
35
36# --inputdir points to the path of the input files.
37my $inputdir = "$srcdir/src/test/regress";
38
39# Run the tests.
40my $rc =
41  system($ENV{PG_REGRESS} . " "
42	  . "--bindir= "
43	  . "--dlpath=\"$dlpath\" "
44	  . "--host="
45	  . $node->host . " "
46	  . "--port="
47	  . $node->port . " "
48	  . "--schedule=$srcdir/src/test/regress/parallel_schedule "
49	  . "--max-concurrent-tests=20 "
50	  . "--inputdir=\"$inputdir\" "
51	  . "--outputdir=\"$outputdir\"");
52
53# Dump out the regression diffs file, if there is one
54if ($rc != 0)
55{
56	my $diffs = "$outputdir/regression.diffs";
57	if (-e $diffs)
58	{
59		print "=== dumping $diffs ===\n";
60		print slurp_file($diffs);
61		print "=== EOF ===\n";
62	}
63}
64
65# Report results
66is($rc, 0, 'regression tests pass');
67
68done_testing();
