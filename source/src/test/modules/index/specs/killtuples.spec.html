src/test/modules/index/specs/killtuples.spec
1# Basic testing of killtuples / kill_prior_tuples / all_dead testing
2# for various index AMs
3#
4# This tests just enough to ensure that the kill* routines are actually
5# executed and does something approximately reasonable. It's *not* sufficient
6# testing for adding killitems support to a new AM!
7#
8# This doesn't really need to be an isolation test, it could be written as a
9# regular regression test. However, writing it as an isolation test ends up a
10# *lot* less verbose.
11
12setup
13{
14    CREATE TABLE counter(heap_accesses int);
15    INSERT INTO counter(heap_accesses) VALUES (0);
16}
17
18teardown
19{
20    DROP TABLE counter;
21}
22
23session s1
24# to ensure GUCs are reset
25setup { RESET ALL; }
26
27step disable_seq { SET enable_seqscan = false; }
28
29step disable_bitmap { SET enable_bitmapscan = false; }
30
31# use a temporary table to make sure no other session can interfere with
32# visibility determinations
33step create_table { CREATE TEMPORARY TABLE kill_prior_tuple(key int not null, cat text not null); }
34
35step fill_10 { INSERT INTO kill_prior_tuple(key, cat) SELECT g.i, 'a' FROM generate_series(1, 10) g(i); }
36
37step fill_500 { INSERT INTO kill_prior_tuple(key, cat) SELECT g.i, 'a' FROM generate_series(1, 500) g(i); }
38
39step fill_same { INSERT INTO kill_prior_tuple(key, cat) SELECT 1, 'a' FROM generate_series(1, 408) g(i); }
40
41# column-less select to make output easier to read
42step flush { SELECT FROM pg_stat_force_next_flush(); }
43
44step measure { UPDATE counter SET heap_accesses = (SELECT heap_blks_read + heap_blks_hit FROM pg_statio_all_tables WHERE relname = 'kill_prior_tuple'); }
45
46step result { SELECT ((heap_blks_read + heap_blks_hit - counter.heap_accesses) > 0) AS has_new_heap_accesses FROM counter, pg_statio_all_tables WHERE relname = 'kill_prior_tuple'; }
47
48step access { EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY OFF, BUFFERS OFF) SELECT * FROM kill_prior_tuple WHERE key = 1; }
49
50step delete { DELETE FROM kill_prior_tuple; }
51
52step drop_table { DROP TABLE IF EXISTS kill_prior_tuple; }
53
54### steps for testing btree indexes ###
55step create_btree { CREATE INDEX kill_prior_tuple_btree ON kill_prior_tuple USING btree (key); }
56
57### steps for testing gist indexes ###
58# Creating the extensions takes time, so we don't want to do so when testing
59# other AMs
60step create_ext_btree_gist { CREATE EXTENSION btree_gist; }
61step drop_ext_btree_gist { DROP EXTENSION btree_gist; }
62step create_gist { CREATE INDEX kill_prior_tuple_gist ON kill_prior_tuple USING gist (key); }
63
64### steps for testing gin indexes ###
65# See create_ext_btree_gist
66step create_ext_btree_gin { CREATE EXTENSION btree_gin; }
67step drop_ext_btree_gin { DROP EXTENSION btree_gin; }
68step create_gin { CREATE INDEX kill_prior_tuple_gin ON kill_prior_tuple USING gin (key); }
69
70### steps for testing hash indexes ###
71step create_hash { CREATE INDEX kill_prior_tuple_hash ON kill_prior_tuple USING hash (key); }
72
73

75permutation
76  create_table fill_500 create_btree flush
77  disable_seq disable_bitmap
78  # show each access to non-deleted tuple increments heap_blks_*
79  measure access flush result
80  measure access flush result
81  delete flush
82  # first access after accessing deleted tuple still needs to access heap
83  measure access flush result
84  # but after kill_prior_tuple did its thing, we shouldn't access heap anymore
85  measure access flush result
86  drop_table
