3use strict;
4use warnings FATAL => 'all';
5
6use PostgreSQL::Test::Cluster;
7use PostgreSQL::Test::Utils;
8use Test::More;
9
10###
11# Test allocating memory after startup, i.e. when the library is not
12# in shared_preload_libraries
13###
14my $node = PostgreSQL::Test::Cluster->new('main');
15$node->init;
16$node->start;
17
18
19$node->safe_psql("postgres", "CREATE EXTENSION test_shmem;");
20
21# Check that the attach counter is incremented on a new connection
22my $attach_count1 =
23  $node->safe_psql("postgres", "SELECT get_test_shmem_attach_count();");
24my $attach_count2 =
25  $node->safe_psql("postgres", "SELECT get_test_shmem_attach_count();");
26cmp_ok($attach_count2, '>', $attach_count1,
27	"attach callback is called in each backend");
28$node->stop;
29
30###
31# Test that loading via shared_preload_libraries also works
32###
33$node->append_conf('postgresql.conf',
34	"shared_preload_libraries = 'test_shmem'");
35$node->start;
36
37# When loaded via shared_preload_libraries, the attach callback is
38# called or not, depending on whether this is an EXEC_BACKEND build.
39my $exec_backend =
40  $node->safe_psql("postgres", "SHOW debug_exec_backend;") eq 'on';
41$attach_count1 =
42  $node->safe_psql("postgres", "SELECT get_test_shmem_attach_count();");
43$attach_count2 =
44  $node->safe_psql("postgres", "SELECT get_test_shmem_attach_count();");
45
46if ($exec_backend)
47{
48	cmp_ok($attach_count2, '>', $attach_count1,
49		"attach callback is called in each backend when loaded via shared_preload_libraries"
50	);
51}
52else
53{
54	ok( $attach_count1 == 0 && $attach_count2 == 0,
55		"attach callback is not called when loaded via shared_preload_libraries"
56	);
57}
58
59$node->stop;
60done_testing();
