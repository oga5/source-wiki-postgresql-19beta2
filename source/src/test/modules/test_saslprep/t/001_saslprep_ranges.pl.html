3# Test all ranges of valid UTF-8 codepoints under SASLprep.
4#
5# This test is expensive and enabled with PG_TEST_EXTRA, which is
6# why it exists as a TAP test.
7
8use strict;
9use warnings FATAL => 'all';
10use PostgreSQL::Test::Cluster;
11use Test::More;
12
13if (!$ENV{PG_TEST_EXTRA} || $ENV{PG_TEST_EXTRA} !~ /\bsaslprep\b/)
14{
15	plan skip_all => "test saslprep not enabled in PG_TEST_EXTRA";
16}
17
18# Initialize node
19my $node = PostgreSQL::Test::Cluster->new('main');
20
21$node->init;
22$node->start;
23$node->safe_psql('postgres', 'CREATE EXTENSION test_saslprep;');
24
25# Among all the valid UTF-8 codepoint ranges, our implementation of
26# SASLprep should never return an empty password if the operation is
27# considered a success.
28my $result = $node->safe_psql(
29	'postgres', qq[SELECT * FROM test_saslprep_ranges()
30  WHERE status = 'SUCCESS' AND res IN (NULL, '')
31]);
32
33is($result, '', "valid codepoints returning an empty password");
34
35$node->stop;
36done_testing();
