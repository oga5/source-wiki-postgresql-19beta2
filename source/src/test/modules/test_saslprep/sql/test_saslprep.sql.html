1-- Tests for SASLprep
2
3CREATE EXTENSION test_saslprep;
4
5-- Incomplete UTF-8 sequence.
6SELECT test_saslprep('\xef');
7
8-- Range of ASCII characters.
9SELECT
10    CASE
11      WHEN a = 0   THEN '<NUL>'
12      WHEN a < 32  THEN '<CTL_' || a::text || '>'
13      WHEN a = 127 THEN '<DEL>'
14      ELSE chr(a) END AS dat,
15    set_byte('\x00'::bytea, 0, a) AS byt,
16    test_saslprep(set_byte('\x00'::bytea, 0, a)) AS saslprep
17  FROM generate_series(0,127) AS a;
18
19DROP EXTENSION test_saslprep;
