4use strict;
5use warnings FATAL => 'all';
6
7use File::Copy;
8
9use PostgreSQL::Test::Utils;
10use Test::More;
11use PostgreSQL::Test::Cluster;
12
13unless (($ENV{with_ssl} || "") eq 'openssl')
14{
15	plan skip_all => 'OpenSSL not supported by this build';
16}
17
18my $libressl = not check_pg_config("#define HAVE_SSL_CTX_SET_CERT_CB 1");
19
20my $rot13pass = "SbbOnE1";
21
22# see the Makefile for how the certificate and key have been generated
23
24my $node = PostgreSQL::Test::Cluster->new('main');
25$node->init;
26$node->append_conf('postgresql.conf',
27	"ssl_passphrase.passphrase = '$rot13pass'");
28$node->append_conf('postgresql.conf',
29	"shared_preload_libraries = 'ssl_passphrase_func'");
30$node->append_conf('postgresql.conf', "ssl = 'on'");
31
32my $ddir = $node->data_dir;
33
34# install certificate and protected key
35copy("server.crt", $ddir);
36copy("server.key", $ddir);
37chmod 0600, "$ddir/server.key" or die $!;
38
39$node->start;
40
41# if the server is running we must have successfully transformed the passphrase
42ok(-e "$ddir/postmaster.pid", "postgres started");
43
44$node->stop('fast');
45
46# should get a warning if ssl_passphrase_command is set
47my $log = $node->rotate_logfile();
48
49$node->append_conf('postgresql.conf',
50	"ssl_passphrase_command = 'echo spl0tz'");
51
52$node->start;
53
54$node->stop('fast');
55
56my $log_contents = slurp_file($log);
57
58like(
59	$log_contents,
60	qr/WARNING.*"ssl_passphrase_command" setting ignored by ssl_passphrase_func module/,
61	"ssl_passphrase_command set warning");
62
63# set the wrong passphrase
64$node->append_conf('postgresql.conf', "ssl_passphrase.passphrase = 'blurfl'");
65
66# try to start the server again
67my $ret = PostgreSQL::Test::Utils::system_log(
68	'pg_ctl',
69	'--pgdata' => $node->data_dir,
70	'--log' => $node->logfile,
71	'start');
72
73
74# with a bad passphrase the server should not start
75ok($ret, "pg_ctl fails with bad passphrase");
76ok(!-e "$ddir/postmaster.pid", "postgres not started with bad passphrase");
77
78# just in case
79$node->stop('fast');
80
81# Make sure the hook is bypassed when SNI is enabled.
82SKIP:
83{
84	skip 'SNI not supported with LibreSSL', 2 if ($libressl);
85
86	$node->append_conf(
87		'postgresql.conf', qq{
88ssl_passphrase_command = 'echo FooBaR1'
89ssl_sni = on
90});
91	$node->append_conf(
92		'pg_hosts.conf', qq{
93example.org "$ddir/server.crt" "$ddir/server.key" "" "echo FooBaR1" on
94example.com "$ddir/server.crt" "$ddir/server.key" "" "echo FooBaR1" on
95});
96
97	# If the servers starts and runs, the bad ssl_passphrase.passphrase was
98	# correctly ignored.
99	$node->start;
100	ok(-e "$ddir/postmaster.pid", "postgres started after SNI");
101
102	$node->stop('fast');
103	$log_contents = slurp_file($log);
104	like(
105		$log_contents,
106		qr/WARNING.*SNI is enabled; installed TLS init hook will be ignored/,
107		"server warns that init hook and SNI are incompatible");
108	# Ensure that the warning was printed once and not once per host line
109	my $count = () =
110	  $log_contents =~ m/installed TLS init hook will be ignored/;
111	is($count, 1, 'Only one WARNING');
112}
113
114done_testing();
