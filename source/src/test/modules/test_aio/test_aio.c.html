121						 "test_aio",

154PG_FUNCTION_INFO_V1(errno_from_string);
155Datum
156errno_from_string(PG_FUNCTION_ARGS)
157{
158	const char *sym = text_to_cstring(PG_GETARG_TEXT_PP(0));
159
160	if (strcmp(sym, "EIO") == 0)
161		PG_RETURN_INT32(EIO);
162	else if (strcmp(sym, "EAGAIN") == 0)
163		PG_RETURN_INT32(EAGAIN);
164	else if (strcmp(sym, "EINTR") == 0)
165		PG_RETURN_INT32(EINTR);
166	else if (strcmp(sym, "ENOSPC") == 0)
167		PG_RETURN_INT32(ENOSPC);
168	else if (strcmp(sym, "EROFS") == 0)
169		PG_RETURN_INT32(EROFS);
170
171	ereport(ERROR,
172			errcode(ERRCODE_INVALID_PARAMETER_VALUE),
173			errmsg_internal("%s is not a supported errno value", sym));
174	PG_RETURN_INT32(0);
175}

177PG_FUNCTION_INFO_V1(grow_rel);
178Datum
179grow_rel(PG_FUNCTION_ARGS)
180{
181	Oid			relid = PG_GETARG_OID(0);
182	uint32		nblocks = PG_GETARG_UINT32(1);
183	Relation	rel;
184#define MAX_BUFFERS_TO_EXTEND_BY 64
185	Buffer		victim_buffers[MAX_BUFFERS_TO_EXTEND_BY];
186
187	rel = relation_open(relid, AccessExclusiveLock);
188
189	while (nblocks > 0)
190	{
191		uint32		extend_by_pages;
192
193		extend_by_pages = Min(nblocks, MAX_BUFFERS_TO_EXTEND_BY);
194
195		ExtendBufferedRelBy(BMR_REL(rel),
196							MAIN_FORKNUM,
197							NULL,
198							0,
199							extend_by_pages,
200							victim_buffers,
201							&extend_by_pages);
202
203		nblocks -= extend_by_pages;
204
205		for (uint32 i = 0; i < extend_by_pages; i++)
206		{
207			ReleaseBuffer(victim_buffers[i]);
208		}
209	}
210
211	relation_close(rel, NoLock);
212
213	PG_RETURN_VOID();
214}

216PG_FUNCTION_INFO_V1(modify_rel_block);
217Datum
218modify_rel_block(PG_FUNCTION_ARGS)
219{
220	Oid			relid = PG_GETARG_OID(0);
221	BlockNumber blkno = PG_GETARG_UINT32(1);
222	bool		zero = PG_GETARG_BOOL(2);
223	bool		corrupt_header = PG_GETARG_BOOL(3);
224	bool		corrupt_checksum = PG_GETARG_BOOL(4);
225	Page		page = palloc_aligned(BLCKSZ, PG_IO_ALIGN_SIZE, 0);
226	bool		flushed;
227	Relation	rel;
228	Buffer		buf;
229	PageHeader	ph;
230
231	rel = relation_open(relid, AccessExclusiveLock);
232
233	buf = ReadBufferExtended(rel, MAIN_FORKNUM, blkno,
234							 RBM_ZERO_ON_ERROR, NULL);
235
236	LockBuffer(buf, BUFFER_LOCK_EXCLUSIVE);
237
238	/*
239	 * copy the page to local memory, seems nicer than to directly modify in
240	 * the buffer pool.
241	 */
242	memcpy(page, BufferGetPage(buf), BLCKSZ);
243
244	UnlockReleaseBuffer(buf);
245
246	/*
247	 * Don't want to have a buffer in-memory that's marked valid where the
248	 * on-disk contents are invalid. Particularly not if the in-memory buffer
249	 * could be dirty...
250	 *
251	 * While we hold an AEL on the relation nobody else should be able to read
252	 * the buffer in.
253	 *
254	 * NB: This is probably racy, better don't copy this to non-test code.
255	 */
256	if (BufferIsLocal(buf))
257		InvalidateLocalBuffer(GetLocalBufferDescriptor(-buf - 1), true);
258	else
259		EvictUnpinnedBuffer(buf, &flushed);
260
261	/*
262	 * Now modify the page as asked for by the caller.
263	 */
264	if (zero)
265		memset(page, 0, BufferGetPageSize(buf));
266
267	if (PageIsEmpty(page) && (corrupt_header || corrupt_checksum))
268		PageInit(page, BufferGetPageSize(buf), 0);
269
270	ph = (PageHeader) page;
271
272	if (corrupt_header)
273		ph->pd_special = BLCKSZ + 1;
274
275	if (corrupt_checksum)
276	{
277		bool		successfully_corrupted = 0;
278
279		/*
280		 * Any single modification of the checksum could just end up being
281		 * valid again, due to e.g. corrupt_header changing the data in a way
282		 * that'd result in the "corrupted" checksum, or the checksum already
283		 * being invalid. Retry in that, unlikely, case.
284		 */
285		for (int i = 0; i < 100; i++)
286		{
287			uint16		verify_checksum;
288			uint16		old_checksum;
289
290			old_checksum = ph->pd_checksum;
291			ph->pd_checksum = old_checksum + 1;
292
293			elog(LOG, "corrupting checksum of blk %u from %u to %u",
294				 blkno, old_checksum, ph->pd_checksum);
295
296			verify_checksum = pg_checksum_page(page, blkno);
297			if (verify_checksum != ph->pd_checksum)
298			{
299				successfully_corrupted = true;
300				break;
301			}
302		}
303
304		if (!successfully_corrupted)
305			elog(ERROR, "could not corrupt checksum, what's going on?");
306	}
307	else
308	{
309		PageSetChecksum(page, blkno);
310	}
311
312	smgrwrite(RelationGetSmgr(rel),
313			  MAIN_FORKNUM, blkno, page, true);
314
315	relation_close(rel, NoLock);
316
317	PG_RETURN_VOID();
318}

320/*
321 * Ensures a buffer for rel & blkno is in shared buffers, without actually
322 * caring about the buffer contents. Used to set up test scenarios.
323 */
324static Buffer
325create_toy_buffer(Relation rel, BlockNumber blkno)
326{
327	Buffer		buf;
328	BufferDesc *buf_hdr;
329	uint64		buf_state;
330	bool		was_pinned = false;
331	uint64		unset_bits = 0;
332
333	/* place buffer in shared buffers without erroring out */
334	buf = ReadBufferExtended(rel, MAIN_FORKNUM, blkno, RBM_ZERO_AND_LOCK, NULL);
335	LockBuffer(buf, BUFFER_LOCK_UNLOCK);
336
337	if (RelationUsesLocalBuffers(rel))
338	{
339		buf_hdr = GetLocalBufferDescriptor(-buf - 1);
340		buf_state = pg_atomic_read_u64(&buf_hdr->state);
341	}
342	else
343	{
344		buf_hdr = GetBufferDescriptor(buf - 1);
345		buf_state = LockBufHdr(buf_hdr);
346	}
347
348	/*
349	 * We should be the only backend accessing this buffer. This is just a
350	 * small bit of belt-and-suspenders defense, none of this code should ever
351	 * run in a cluster with real data.
352	 */
353	if (BUF_STATE_GET_REFCOUNT(buf_state) > 1)
354		was_pinned = true;
355	else
356		unset_bits |= BM_VALID | BM_DIRTY;
357
358	if (RelationUsesLocalBuffers(rel))
359	{
360		buf_state &= ~unset_bits;
361		pg_atomic_unlocked_write_u64(&buf_hdr->state, buf_state);
362	}
363	else
364	{
365		UnlockBufHdrExt(buf_hdr, buf_state, 0, unset_bits, 0);
366	}
367
368	if (was_pinned)
369		elog(ERROR, "toy buffer %d was already pinned",
370			 buf);
371
372	return buf;
373}

375/*
376 * A "low level" read. This does similar things to what
377 * StartReadBuffers()/WaitReadBuffers() do, but provides more control (and
378 * less sanity).
379 */
380PG_FUNCTION_INFO_V1(read_rel_block_ll);
381Datum
382read_rel_block_ll(PG_FUNCTION_ARGS)
383{
384	Oid			relid = PG_GETARG_OID(0);
385	BlockNumber blkno = PG_GETARG_UINT32(1);
386	int			nblocks = PG_GETARG_INT32(2);
387	bool		wait_complete = PG_GETARG_BOOL(3);
388	bool		batchmode_enter = PG_GETARG_BOOL(4);
389	bool		call_smgrreleaseall = PG_GETARG_BOOL(5);
390	bool		batchmode_exit = PG_GETARG_BOOL(6);
391	bool		zero_on_error = PG_GETARG_BOOL(7);
392	Relation	rel;
393	Buffer		bufs[PG_IOV_MAX];
394	BufferDesc *buf_hdrs[PG_IOV_MAX];
395	Page		pages[PG_IOV_MAX];
396	uint8		srb_flags = 0;
397	PgAioReturn ior;
398	PgAioHandle *ioh;
399	PgAioWaitRef iow;
400	SMgrRelation smgr;
401
402	if (nblocks <= 0 || nblocks > PG_IOV_MAX)
403		elog(ERROR, "nblocks is out of range");
404
405	rel = relation_open(relid, AccessShareLock);
406
407	for (int i = 0; i < nblocks; i++)
408	{
409		bufs[i] = create_toy_buffer(rel, blkno + i);
410		pages[i] = BufferGetBlock(bufs[i]);
411		buf_hdrs[i] = BufferIsLocal(bufs[i]) ?
412			GetLocalBufferDescriptor(-bufs[i] - 1) :
413			GetBufferDescriptor(bufs[i] - 1);
414	}
415
416	smgr = RelationGetSmgr(rel);
417
418	pgstat_prepare_report_checksum_failure(smgr->smgr_rlocator.locator.dbOid);
419
420	ioh = pgaio_io_acquire(CurrentResourceOwner, &ior);
421	pgaio_io_get_wref(ioh, &iow);
422
423	if (RelationUsesLocalBuffers(rel))
424	{
425		for (int i = 0; i < nblocks; i++)
426			StartLocalBufferIO(buf_hdrs[i], true, true, NULL);
427		pgaio_io_set_flag(ioh, PGAIO_HF_REFERENCES_LOCAL);
428	}
429	else
430	{
431		for (int i = 0; i < nblocks; i++)
432			StartSharedBufferIO(buf_hdrs[i], true, true, NULL);
433	}
434
435	pgaio_io_set_handle_data_32(ioh, (uint32 *) bufs, nblocks);
436
437	if (zero_on_error | zero_damaged_pages)
438		srb_flags |= READ_BUFFERS_ZERO_ON_ERROR;
439	if (ignore_checksum_failure)
440		srb_flags |= READ_BUFFERS_IGNORE_CHECKSUM_FAILURES;
441
442	pgaio_io_register_callbacks(ioh,
443								RelationUsesLocalBuffers(rel) ?
444								PGAIO_HCB_LOCAL_BUFFER_READV :
445								PGAIO_HCB_SHARED_BUFFER_READV,
446								srb_flags);
447
448	if (batchmode_enter)
449		pgaio_enter_batchmode();
450
451	smgrstartreadv(ioh, smgr, MAIN_FORKNUM, blkno,
452				   (void *) pages, nblocks);
453
454	if (call_smgrreleaseall)
455		smgrreleaseall();
456
457	if (batchmode_exit)
458		pgaio_exit_batchmode();
459
460	for (int i = 0; i < nblocks; i++)
461		ReleaseBuffer(bufs[i]);
462
463	if (wait_complete)
464	{
465		pgaio_wref_wait(&iow);
466
467		if (ior.result.status != PGAIO_RS_OK)
468			pgaio_result_report(ior.result,
469								&ior.target_data,
470								ior.result.status == PGAIO_RS_ERROR ?
471								ERROR : WARNING);
472	}
473
474	relation_close(rel, NoLock);
475
476	PG_RETURN_VOID();
477}

479/* helper for invalidate_rel_block() and evict_rel() */
480static void
481invalidate_one_block(Relation rel, ForkNumber forknum, BlockNumber blkno)
482{
483	PrefetchBufferResult pr;
484	Buffer		buf;
485
486	/*
487	 * This is a gross hack, but there's no other API exposed that allows to
488	 * get a buffer ID without actually reading the block in.
489	 */
490	pr = PrefetchBuffer(rel, forknum, blkno);
491	buf = pr.recent_buffer;
492
493	if (BufferIsValid(buf))
494	{
495		/* if the buffer contents aren't valid, this'll return false */
496		if (ReadRecentBuffer(rel->rd_locator, forknum, blkno, buf))
497		{
498			BufferDesc *buf_hdr = BufferIsLocal(buf) ?
499				GetLocalBufferDescriptor(-buf - 1)
500				: GetBufferDescriptor(buf - 1);
501			bool		flushed;
502
503			LockBuffer(buf, BUFFER_LOCK_EXCLUSIVE);
504
505			if (pg_atomic_read_u64(&buf_hdr->state) & BM_DIRTY)
506			{
507				if (BufferIsLocal(buf))
508					FlushLocalBuffer(buf_hdr, NULL);
509				else
510					FlushOneBuffer(buf);
511			}
512			UnlockReleaseBuffer(buf);
513
514			if (BufferIsLocal(buf))
515				InvalidateLocalBuffer(GetLocalBufferDescriptor(-buf - 1), true);
516			else if (!EvictUnpinnedBuffer(buf, &flushed))
517				elog(ERROR, "couldn't evict");
518		}
519	}
520
521}

540PG_FUNCTION_INFO_V1(evict_rel);
541Datum
542evict_rel(PG_FUNCTION_ARGS)
543{
544	Oid			relid = PG_GETARG_OID(0);
545	Relation	rel;
546
547	rel = relation_open(relid, AccessExclusiveLock);
548
549	/*
550	 * EvictRelUnpinnedBuffers() doesn't support temp tables, so for temp
551	 * tables we have to do it the expensive way and evict every possible
552	 * buffer.
553	 */
554	if (RelationUsesLocalBuffers(rel))
555	{
556		SMgrRelation smgr = RelationGetSmgr(rel);
557
558		for (int forknum = MAIN_FORKNUM; forknum <= MAX_FORKNUM; forknum++)
559		{
560			BlockNumber nblocks;
561
562			if (!smgrexists(smgr, forknum))
563				continue;
564
565			nblocks = smgrnblocks(smgr, forknum);
566
567			for (int blkno = 0; blkno < nblocks; blkno++)
568			{
569				invalidate_one_block(rel, forknum, blkno);
570			}
571		}
572	}
573	else
574	{
575		int32		buffers_evicted,
576					buffers_flushed,
577					buffers_skipped;
578
579		EvictRelUnpinnedBuffers(rel, &buffers_evicted, &buffers_flushed,
580								&buffers_skipped);
581	}
582
583	relation_close(rel, AccessExclusiveLock);
584
585
586	PG_RETURN_VOID();
587}

608PG_FUNCTION_INFO_V1(buffer_call_start_io);
609Datum
610buffer_call_start_io(PG_FUNCTION_ARGS)
611{
612	Buffer		buf = PG_GETARG_INT32(0);
613	bool		for_input = PG_GETARG_BOOL(1);
614	bool		wait = PG_GETARG_BOOL(2);
615	StartBufferIOResult result;
616	bool		can_start;
617
618	if (BufferIsLocal(buf))
619		result = StartLocalBufferIO(GetLocalBufferDescriptor(-buf - 1),
620									for_input, wait, NULL);
621	else
622		result = StartSharedBufferIO(GetBufferDescriptor(buf - 1),
623									 for_input, wait, NULL);
624
625	can_start = result == BUFFER_IO_READY_FOR_IO;
626
627	/*
628	 * For tests we don't want the resowner release preventing us from
629	 * orchestrating odd scenarios.
630	 */
631	if (can_start && !BufferIsLocal(buf))
632		ResourceOwnerForgetBufferIO(CurrentResourceOwner,
633									buf);
634
635	ereport(LOG,
636			errmsg("buffer %d after StartBufferIO: %s",
637				   buf, DebugPrintBufferRefcount(buf)),
638			errhidestmt(true), errhidecontext(true));
639
640	PG_RETURN_BOOL(can_start);
641}

643PG_FUNCTION_INFO_V1(buffer_call_terminate_io);
644Datum
645buffer_call_terminate_io(PG_FUNCTION_ARGS)
646{
647	Buffer		buf = PG_GETARG_INT32(0);
648	bool		for_input = PG_GETARG_BOOL(1);
649	bool		succeed = PG_GETARG_BOOL(2);
650	bool		io_error = PG_GETARG_BOOL(3);
651	bool		release_aio = PG_GETARG_BOOL(4);
652	bool		clear_dirty = false;
653	uint64		set_flag_bits = 0;
654
655	if (io_error)
656		set_flag_bits |= BM_IO_ERROR;
657
658	if (for_input)
659	{
660		clear_dirty = false;
661
662		if (succeed)
663			set_flag_bits |= BM_VALID;
664	}
665	else
666	{
667		if (succeed)
668			clear_dirty = true;
669	}
670
671	ereport(LOG,
672			errmsg("buffer %d before Terminate[Local]BufferIO: %s",
673				   buf, DebugPrintBufferRefcount(buf)),
674			errhidestmt(true), errhidecontext(true));
675
676	if (BufferIsLocal(buf))
677		TerminateLocalBufferIO(GetLocalBufferDescriptor(-buf - 1),
678							   clear_dirty, set_flag_bits, release_aio);
679	else
680		TerminateBufferIO(GetBufferDescriptor(buf - 1),
681						  clear_dirty, set_flag_bits, false, release_aio);
682
683	ereport(LOG,
684			errmsg("buffer %d after Terminate[Local]BufferIO: %s",
685				   buf, DebugPrintBufferRefcount(buf)),
686			errhidestmt(true), errhidecontext(true));
687
688	PG_RETURN_VOID();
689}

692/*
693 * Infrastructure to test StartReadBuffers()
694 */
695Datum
696read_buffers(PG_FUNCTION_ARGS)
697{
698	Oid			relid = PG_GETARG_OID(0);
699	BlockNumber startblock = PG_GETARG_UINT32(1);
700	int32		nblocks = PG_GETARG_INT32(2);
701	ReturnSetInfo *rsinfo = (ReturnSetInfo *) fcinfo->resultinfo;
702	Relation	rel;

704	int			nblocks_done = 0;
705	int			nblocks_disp = 0;
706	int			nios = 0;
707	ReadBuffersOperation *operations;
708	Buffer	   *buffers;
709	Datum	   *buffers_datum;
710	bool	   *io_reqds;
711	int		   *nblocks_per_io;
712
713	Assert(nblocks > 0);
714
715	InitMaterializedSRF(fcinfo, 0);
716
717	/* at worst each block gets its own IO */
718	operations = palloc0(sizeof(ReadBuffersOperation) * nblocks);
719	buffers = palloc0(sizeof(Buffer) * nblocks);
720	buffers_datum = palloc0(sizeof(Datum) * nblocks);
721	io_reqds = palloc0(sizeof(bool) * nblocks);
722	nblocks_per_io = palloc0(sizeof(int) * nblocks);
723
724	rel = relation_open(relid, AccessShareLock);
725	smgr = RelationGetSmgr(rel);
726
727	/*
728	 * Do StartReadBuffers() until IO for all the required blocks has been
729	 * started (if required).
730	 */
731	while (nblocks_done < nblocks)
732	{
733		ReadBuffersOperation *operation = &operations[nios];
734		int			nblocks_this_io =
735			Min(nblocks - nblocks_done, io_combine_limit);
736
737		operation->rel = rel;
738		operation->smgr = smgr;
739		operation->persistence = rel->rd_rel->relpersistence;
740		operation->strategy = NULL;
741		operation->forknum = MAIN_FORKNUM;
742
743		io_reqds[nios] = StartReadBuffers(operation,
744										  &buffers[nblocks_done],
745										  startblock + nblocks_done,
746										  &nblocks_this_io,
747										  0);
748		nblocks_per_io[nios] = nblocks_this_io;
749		nios++;
750		nblocks_done += nblocks_this_io;
751	}
752
753	/*
754	 * Now wait for all operations that required IO. This is done at the end,
755	 * as otherwise waiting for IO in progress in other backends could
756	 * influence the result for subsequent buffers / blocks.
757	 */
758	for (int nio = 0; nio < nios; nio++)
759	{
760		ReadBuffersOperation *operation = &operations[nio];
761
762		if (io_reqds[nio])
763			WaitReadBuffers(operation);
764	}
765
766	/*
767	 * Convert what has been done into SQL SRF return value.
768	 */
769	for (int nio = 0; nio < nios; nio++)
770	{
771		ReadBuffersOperation *operation = &operations[nio];
772		int			nblocks_this_io = nblocks_per_io[nio];
773		Datum		values[6] = {0};
774		bool		nulls[6] = {0};
775		ArrayType  *buffers_arr;
776
777		/* convert buffer array to datum array */
778		for (int i = 0; i < nblocks_this_io; i++)
779		{
780			Buffer		buf = buffers[nblocks_disp + i];
781
782			Assert(BufferGetBlockNumber(buf) == startblock + nblocks_disp + i);
783
784			buffers_datum[nblocks_disp + i] = Int32GetDatum(buf);
785		}
786
787		buffers_arr = construct_array_builtin(&buffers_datum[nblocks_disp],
788											  nblocks_this_io,
789											  INT4OID);
790
791		/* blockoff */
792		values[0] = Int32GetDatum(nblocks_disp);
793		nulls[0] = false;
794
795		/* blocknum */
796		values[1] = UInt32GetDatum(startblock + nblocks_disp);
797		nulls[1] = false;
798
799		/* io_reqd */
800		values[2] = BoolGetDatum(io_reqds[nio]);
801		nulls[2] = false;
802
803		/* foreign IO - only valid when IO was required */
804		values[3] = BoolGetDatum(io_reqds[nio] ? operation->foreign_io : false);
805		nulls[3] = false;
806
807		/* nblocks */
808		values[4] = Int32GetDatum(nblocks_this_io);
809		nulls[4] = false;
810
811		/* array of buffers */
812		values[5] = PointerGetDatum(buffers_arr);
813		nulls[5] = false;
814
815		tuplestore_putvalues(rsinfo->setResult, rsinfo->setDesc, values, nulls);
816
817		nblocks_disp += nblocks_this_io;
818	}
819
820	/* release pins on all the buffers */
821	for (int i = 0; i < nblocks_done; i++)
822		ReleaseBuffer(buffers[i]);
823
824	/*
825	 * Free explicitly, to have a chance to detect potential issues with too
826	 * long lived references to the operation.
827	 */
828	pfree(operations);
829	pfree(buffers);
830	pfree(buffers_datum);
831	pfree(io_reqds);
832	pfree(nblocks_per_io);
833
834	relation_close(rel, NoLock);
835
836	return (Datum) 0;
837}

840static BlockNumber
841read_stream_for_blocks_cb(ReadStream *stream,
842						  void *callback_private_data,
843						  void *per_buffer_data)
844{
845	BlocksReadStreamData *stream_data = callback_private_data;
846
847	if (stream_data->curblock >= stream_data->nblocks)
848		return InvalidBlockNumber;
849	return stream_data->blocks[stream_data->curblock++];
850}
851
852PG_FUNCTION_INFO_V1(read_stream_for_blocks);
853Datum
854read_stream_for_blocks(PG_FUNCTION_ARGS)
855{
856	Oid			relid = PG_GETARG_OID(0);
857	ArrayType  *blocksarray = PG_GETARG_ARRAYTYPE_P(1);
858	ReturnSetInfo *rsinfo = (ReturnSetInfo *) fcinfo->resultinfo;
859	Relation	rel;
860	BlocksReadStreamData stream_data;
861	ReadStream *stream;
862
863	InitMaterializedSRF(fcinfo, 0);
864
865	/*
866	 * We expect the input to be an N-element int4 array; verify that. We
867	 * don't need to use deconstruct_array() since the array data is just
868	 * going to look like a C array of N int4 values.
869	 */
870	if (ARR_NDIM(blocksarray) != 1 ||
871		ARR_HASNULL(blocksarray) ||
872		ARR_ELEMTYPE(blocksarray) != INT4OID)
873		elog(ERROR, "expected 1 dimensional int4 array");
874
875	stream_data.curblock = 0;
876	stream_data.nblocks = ARR_DIMS(blocksarray)[0];
877	stream_data.blocks = (uint32 *) ARR_DATA_PTR(blocksarray);
878
879	rel = relation_open(relid, AccessShareLock);
880
881	stream = read_stream_begin_relation(READ_STREAM_FULL,
882										NULL,
883										rel,
884										MAIN_FORKNUM,
885										read_stream_for_blocks_cb,
886										&stream_data,
887										0);
888
889	for (int i = 0; i < stream_data.nblocks; i++)
890	{
891		Buffer		buf = read_stream_next_buffer(stream, NULL);
892		Datum		values[3] = {0};
893		bool		nulls[3] = {0};
894
895		if (!BufferIsValid(buf))
896			elog(ERROR, "read_stream_next_buffer() call %d is unexpectedly invalid", i);
897
898		values[0] = Int32GetDatum(i);
899		values[1] = UInt32GetDatum(stream_data.blocks[i]);
900		values[2] = UInt32GetDatum(buf);
901
902		tuplestore_putvalues(rsinfo->setResult, rsinfo->setDesc, values, nulls);
903
904		ReleaseBuffer(buf);
905	}
906
907	if (read_stream_next_buffer(stream, NULL) != InvalidBuffer)
908		elog(ERROR, "read_stream_next_buffer() call %d is unexpectedly valid",
909			 stream_data.nblocks);
910
911	read_stream_end(stream);
912
913	relation_close(rel, NoLock);
914
915	return (Datum) 0;
916}

996static bool
997inj_io_short_read_matches(PgAioHandle *ioh)
998{
999	PGPROC	   *io_proc;
1000	int32		io_pid;
1001	int32		inj_pid;
1002	PgAioTargetData *td;
1003
1004	if (!inj_io_error_state->enabled_short_read)
1005		return false;
1006
1007	if (!inj_io_error_state->short_read_result_set)
1008		return false;
1009
1010	io_proc = GetPGProcByNumber(pgaio_io_get_owner(ioh));
1011	io_pid = io_proc->pid;
1012	inj_pid = inj_io_error_state->short_read_pid;
1013
1014	if (inj_pid != InvalidPid && inj_pid != io_pid)
1015		return false;
1016
1017	td = pgaio_io_get_target_data(ioh);
1018
1019	if (inj_io_error_state->short_read_relfilenode != InvalidOid &&
1020		td->smgr.rlocator.relNumber != inj_io_error_state->short_read_relfilenode)
1021		return false;
1022
1023	/*
1024	 * Only shorten reads that are actually longer than the target size,
1025	 * otherwise we can trigger over-reads.
1026	 */
1027	if (inj_io_error_state->short_read_result >= ioh->result)
1028		return false;
1029
1030	return true;
1031}

1033static bool
1034inj_io_completion_wait_matches(PgAioHandle *ioh)
1035{
1036	PGPROC	   *io_proc;
1037	int32		io_pid;
1038	PgAioTargetData *td;
1039	int32		inj_pid;
1040	BlockNumber io_blockno;
1041	BlockNumber inj_blockno;
1042	Oid			inj_relfilenode;
1043
1044	if (!inj_io_error_state->enabled_completion_wait)
1045		return false;
1046
1047	io_proc = GetPGProcByNumber(pgaio_io_get_owner(ioh));
1048	io_pid = io_proc->pid;
1049	inj_pid = inj_io_error_state->completion_wait_pid;
1050
1051	if (inj_pid != InvalidPid && inj_pid != io_pid)
1052		return false;
1053
1054	td = pgaio_io_get_target_data(ioh);
1055
1056	inj_relfilenode = inj_io_error_state->completion_wait_relfilenode;
1057	if (inj_relfilenode != InvalidOid &&
1058		td->smgr.rlocator.relNumber != inj_relfilenode)
1059		return false;
1060
1061	inj_blockno = inj_io_error_state->completion_wait_blockno;
1062	io_blockno = td->smgr.blockNum;
1063	if (inj_blockno != InvalidBlockNumber &&
1064		!(inj_blockno >= io_blockno && inj_blockno < (io_blockno + td->smgr.nblocks)))
1065		return false;
1066
1067	return true;
1068}

1070static void
1071inj_io_completion_wait_hook(const char *name, const void *private_data, void *arg)
1072{
1073	PgAioHandle *ioh = (PgAioHandle *) arg;
1074
1075	if (!inj_io_completion_wait_matches(ioh))
1076		return;
1077
1078	ConditionVariablePrepareToSleep(&inj_io_error_state->cv);
1079
1080	while (true)
1081	{
1082		if (!inj_io_completion_wait_matches(ioh))
1083			break;
1084
1085		ConditionVariableSleep(&inj_io_error_state->cv,
1086							   inj_io_error_state->completion_wait_event);
1087	}
1088
1089	ConditionVariableCancelSleep();
1090}

1092static void
1093inj_io_short_read_hook(const char *name, const void *private_data, void *arg)
1094{
1095	PgAioHandle *ioh = (PgAioHandle *) arg;
1096
1097	ereport(LOG,
1098			errmsg("short read injection point called, is enabled: %d",
1099				   inj_io_error_state->enabled_short_read),
1100			errhidestmt(true), errhidecontext(true));
1101
1102	if (inj_io_short_read_matches(ioh))
1103	{
1104		struct iovec *iov = &pgaio_ctl->iovecs[ioh->iovec_off];
1105		int32		old_result = ioh->result;
1106		int32		new_result = inj_io_error_state->short_read_result;
1107		int32		processed = 0;
1108
1109		ereport(LOG,
1110				errmsg("short read inject point, changing result from %d to %d",
1111					   old_result, new_result),
1112				errhidestmt(true), errhidecontext(true));
1113
1114		/*
1115		 * The underlying IO actually completed OK, and thus the "invalid"
1116		 * portion of the IOV actually contains valid data. That can hide a
1117		 * lot of problems, e.g. if we were to wrongly mark a buffer, that
1118		 * wasn't read according to the shortened-read, IO as valid, the
1119		 * contents would look valid and we might miss a bug.
1120		 *
1121		 * To avoid that, iterate through the IOV and zero out the "failed"
1122		 * portion of the IO.
1123		 */
1124		for (int i = 0; i < ioh->op_data.read.iov_length; i++)
1125		{
1126			if (processed + iov[i].iov_len <= new_result)
1127				processed += iov[i].iov_len;
1128			else if (processed <= new_result)
1129			{
1130				uint32		ok_part = new_result - processed;
1131
1132				memset((char *) iov[i].iov_base + ok_part, 0, iov[i].iov_len - ok_part);
1133				processed += iov[i].iov_len;
1134			}
1135			else
1136			{
1137				memset((char *) iov[i].iov_base, 0, iov[i].iov_len);
1138			}
1139		}
1140
1141		ioh->result = new_result;
1142	}
1143}

1145void
1146inj_io_completion_hook(const char *name, const void *private_data, void *arg)
1147{
1148	inj_io_completion_wait_hook(name, private_data, arg);
1149	inj_io_short_read_hook(name, private_data, arg);
1150}
