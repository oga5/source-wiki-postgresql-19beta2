1CREATE EXTENSION test_lwlock_tranches;
2
3-- Test lock tranches created with RequestNamedLWLockTranche()
4SELECT test_startup_lwlocks();
5
6-- Test creating new lock tranches with LWLockNewTrancheId()
7CREATE TEMP TABLE test_tranches(tranche_name text, tranche_id int);
8INSERT INTO test_tranches VALUES
9    ('test_tranche_a', test_lwlock_tranche_create('test_tranche_a')),
10    ('test_tranche_b', test_lwlock_tranche_create('test_tranche_b'));
11
12-- You can create multiple tranches with the same name. It's confusing, so
13-- not recommended, but test it.
14INSERT INTO test_tranches VALUES
15    ('test_tranche_duplicate', test_lwlock_tranche_create('test_tranche_duplicate')),
16    ('test_tranche_duplicate', test_lwlock_tranche_create('test_tranche_duplicate'));
17
18-- Check that all tranches were assigned different tranche IDs, including
19-- the ones with duplicate names
20select count(distinct tranche_id) from  test_tranches;
21
22-- Test GetLWLockIdentifier() on tranches created with LWLockNewTrancheId()
23select tranche_name, test_lwlock_get_lwlock_identifier(tranche_id) as lookup_result
24from test_tranches;
25
26-- negative tests
27SELECT test_lwlock_tranche_create(NULL);
28SELECT test_lwlock_tranche_create(repeat('a', 64));
29SELECT test_lwlock_tranche_lookup('bogus');
30SELECT test_lwlock_tranche_lookup('test_tranche_a');
31SELECT test_lwlock_initialize(65535);
32
33-- Test what happens when you use up all the slots.
34--
35-- MAX_USER_DEFINED_TRANCHES is 256. Two locks were created with
36-- RequestNamedLWLockTranche() when the library was loaded, and we created
37-- four more.
38insert into test_tranches
39    select 'test_tranche_consume_all', test_lwlock_tranche_create('test_tranche_consume_all')
40    from generate_series(1, 256 - 2 - 4);
41
42select test_lwlock_tranche_create('out-of-tranches');
