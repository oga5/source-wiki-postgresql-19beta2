35static TidStore *tidstore = NULL;
36static size_t tidstore_empty_size;

38/* array for verification of some tests */
39typedef struct ItemArray
40{
41	ItemPointerData *insert_tids;
42	ItemPointerData *lookup_tids;
43	ItemPointerData *iter_tids;
44	int			max_tids;
45	int			num_tids;
46} ItemArray;
47
48static ItemArray items;
49
50/* comparator routine for ItemPointer */
51static int
52itemptr_cmp(const void *left, const void *right)
53{
54	BlockNumber lblk,
55				rblk;
56	OffsetNumber loff,
57				roff;
58
59	lblk = ItemPointerGetBlockNumber((const ItemPointerData *) left);
60	rblk = ItemPointerGetBlockNumber((const ItemPointerData *) right);
61
62	if (lblk < rblk)
63		return -1;
64	if (lblk > rblk)
65		return 1;
66
67	loff = ItemPointerGetOffsetNumber((const ItemPointerData *) left);
68	roff = ItemPointerGetOffsetNumber((const ItemPointerData *) right);
69
70	if (loff < roff)
71		return -1;
72	if (loff > roff)
73		return 1;
74
75	return 0;
76}
77
78static int
79offsetnumber_cmp(const void *a, const void *b)
80{
81	OffsetNumber l = *(const OffsetNumber *) a;
82	OffsetNumber r = *(const OffsetNumber *) b;
83
84	if (l < r)
85		return -1;
86	else if (l > r)
87		return 1;
88	return 0;
89}

91/*
92 * Create a TidStore. If shared is false, the tidstore is created
93 * on TopMemoryContext, otherwise on DSA. Although the tidstore
94 * is created on DSA, only the same process can subsequently use
95 * the tidstore. The tidstore handle is not shared anywhere.
96 */
97Datum
98test_create(PG_FUNCTION_ARGS)
99{
100	bool		shared = PG_GETARG_BOOL(0);
101	MemoryContext old_ctx;
102
103	/* doesn't really matter, since it's just a hint */
104	size_t		tidstore_max_size = 2 * 1024 * 1024;
105	size_t		array_init_size = 1024;
106
107	Assert(tidstore == NULL);
108
109	/*
110	 * Create the TidStore on TopMemoryContext so that the same process use it
111	 * for subsequent tests.
112	 */
113	old_ctx = MemoryContextSwitchTo(TopMemoryContext);
114
115	if (shared)
116	{
117		int			tranche_id;
118
119		tranche_id = LWLockNewTrancheId("test_tidstore");
120
121		tidstore = TidStoreCreateShared(tidstore_max_size, tranche_id);
122
123		/*
124		 * Remain attached until end of backend or explicitly detached so that
125		 * the same process use the tidstore for subsequent tests.
126		 */
127		dsa_pin_mapping(TidStoreGetDSA(tidstore));
128	}
129	else
130		/* VACUUM uses insert only, so we test the other option. */
131		tidstore = TidStoreCreateLocal(tidstore_max_size, false);
132
133	tidstore_empty_size = TidStoreMemoryUsage(tidstore);
134
135	items.num_tids = 0;
136	items.max_tids = array_init_size / sizeof(ItemPointerData);
137	items.insert_tids = (ItemPointerData *) palloc0(array_init_size);
138	items.lookup_tids = (ItemPointerData *) palloc0(array_init_size);
139	items.iter_tids = (ItemPointerData *) palloc0(array_init_size);
140
141	MemoryContextSwitchTo(old_ctx);
142
143	PG_RETURN_VOID();
144}

146static void
147sanity_check_array(ArrayType *ta)
148{
149	if (ARR_HASNULL(ta) && array_contains_nulls(ta))
150		ereport(ERROR,
151				(errcode(ERRCODE_NULL_VALUE_NOT_ALLOWED),
152				 errmsg("array must not contain nulls")));
153
154	if (ARR_NDIM(ta) > 1)
155		ereport(ERROR,
156				(errcode(ERRCODE_DATA_EXCEPTION),
157				 errmsg("argument must be empty or one-dimensional array")));
158}

160static void
161check_tidstore_available(void)
162{
163	if (tidstore == NULL)
164		elog(ERROR, "tidstore is not created");
165}

167static void
168purge_from_verification_array(BlockNumber blkno)
169{
170	int			dst = 0;
171
172	for (int src = 0; src < items.num_tids; src++)
173		if (ItemPointerGetBlockNumber(&items.insert_tids[src]) != blkno)
174			items.insert_tids[dst++] = items.insert_tids[src];
175	items.num_tids = dst;
176}

179/* Set the given block and offsets pairs */
180Datum
181do_set_block_offsets(PG_FUNCTION_ARGS)
182{
183	BlockNumber blkno = PG_GETARG_INT64(0);
184	ArrayType  *ta = PG_GETARG_ARRAYTYPE_P_COPY(1);
185	OffsetNumber *offs;
186	int			noffs;
187
188	check_tidstore_available();
189	sanity_check_array(ta);
190
191	noffs = ArrayGetNItems(ARR_NDIM(ta), ARR_DIMS(ta));
192	offs = ((OffsetNumber *) ARR_DATA_PTR(ta));
193
194	/* TidStoreSetBlockOffsets() requires offsets to be strictly ascending. */
195	qsort(offs, noffs, sizeof(OffsetNumber), offsetnumber_cmp);
196
197	/* Set TIDs in the store */
198	TidStoreLockExclusive(tidstore);
199	TidStoreSetBlockOffsets(tidstore, blkno, offs, noffs);
200	TidStoreUnlock(tidstore);
201
202	/* Remove the existing items of blkno from the verification array */
203	purge_from_verification_array(blkno);
204
205	/* Set TIDs in verification array */
206	for (int i = 0; i < noffs; i++)
207	{
208		ItemPointer tid;
209		int			idx = items.num_tids + i;
210
211		/* Enlarge the TID arrays if necessary */
212		if (idx >= items.max_tids)
213		{
214			items.max_tids *= 2;
215			items.insert_tids = repalloc(items.insert_tids, sizeof(ItemPointerData) * items.max_tids);
216			items.lookup_tids = repalloc(items.lookup_tids, sizeof(ItemPointerData) * items.max_tids);
217			items.iter_tids = repalloc(items.iter_tids, sizeof(ItemPointerData) * items.max_tids);
218		}
219
220		tid = &(items.insert_tids[idx]);
221		ItemPointerSet(tid, blkno, offs[i]);
222	}
223
224	/* Update statistics */
225	items.num_tids += noffs;
226
227	PG_RETURN_INT64(blkno);
228}

230/*
231 * Verify TIDs in store against the array.
232 */
233Datum
234check_set_block_offsets(PG_FUNCTION_ARGS)
235{
236	TidStoreIter *iter;
237	TidStoreIterResult *iter_result;
238	int			num_iter_tids = 0;
239	int			num_lookup_tids = 0;
240	BlockNumber prevblkno = 0;
241
242	check_tidstore_available();
243
244	/* lookup each member in the verification array */
245	for (int i = 0; i < items.num_tids; i++)
246		if (!TidStoreIsMember(tidstore, &items.insert_tids[i]))
247			elog(ERROR, "missing TID with block %u, offset %u",
248				 ItemPointerGetBlockNumber(&items.insert_tids[i]),
249				 ItemPointerGetOffsetNumber(&items.insert_tids[i]));
250
251	/*
252	 * Lookup all possible TIDs for each distinct block in the verification
253	 * array and save successful lookups in the lookup array.
254	 */
255
256	for (int i = 0; i < items.num_tids; i++)
257	{
258		BlockNumber blkno = ItemPointerGetBlockNumber(&items.insert_tids[i]);
259
260		if (i > 0 && blkno == prevblkno)
261			continue;
262
263		for (OffsetNumber offset = FirstOffsetNumber; offset < MaxOffsetNumber; offset++)
264		{
265			ItemPointerData tid;
266
267			ItemPointerSet(&tid, blkno, offset);
268
269			TidStoreLockShare(tidstore);
270			if (TidStoreIsMember(tidstore, &tid))
271				ItemPointerSet(&items.lookup_tids[num_lookup_tids++], blkno, offset);
272			TidStoreUnlock(tidstore);
273		}
274
275		prevblkno = blkno;
276	}
277
278	/* Collect TIDs stored in the tidstore, in order */
279
280	TidStoreLockShare(tidstore);
281	iter = TidStoreBeginIterate(tidstore);
282	while ((iter_result = TidStoreIterateNext(iter)) != NULL)
283	{
284		OffsetNumber offsets[MaxOffsetNumber];
285		int			num_offsets;
286
287		num_offsets = TidStoreGetBlockOffsets(iter_result, offsets, lengthof(offsets));
288		Assert(num_offsets <= lengthof(offsets));
289		for (int i = 0; i < num_offsets; i++)
290			ItemPointerSet(&(items.iter_tids[num_iter_tids++]), iter_result->blkno,
291						   offsets[i]);
292	}
293	TidStoreEndIterate(iter);
294	TidStoreUnlock(tidstore);
295
296	/*
297	 * Sort verification and lookup arrays and test that all arrays are the
298	 * same.
299	 */
300
301	if (num_lookup_tids != items.num_tids)
302		elog(ERROR, "should have %d TIDs, have %d", items.num_tids, num_lookup_tids);
303	if (num_iter_tids != items.num_tids)
304		elog(ERROR, "should have %d TIDs, have %d", items.num_tids, num_iter_tids);
305
306	qsort(items.insert_tids, items.num_tids, sizeof(ItemPointerData), itemptr_cmp);
307	qsort(items.lookup_tids, items.num_tids, sizeof(ItemPointerData), itemptr_cmp);
308	for (int i = 0; i < items.num_tids; i++)
309	{
310		if (itemptr_cmp(&items.insert_tids[i], &items.iter_tids[i]) != 0)
311			elog(ERROR, "TID iter array doesn't match verification array, got (%u,%u) expected (%u,%u)",
312				 ItemPointerGetBlockNumber(&items.iter_tids[i]),
313				 ItemPointerGetOffsetNumber(&items.iter_tids[i]),
314				 ItemPointerGetBlockNumber(&items.insert_tids[i]),
315				 ItemPointerGetOffsetNumber(&items.insert_tids[i]));
316		if (itemptr_cmp(&items.insert_tids[i], &items.lookup_tids[i]) != 0)
317			elog(ERROR, "TID lookup array doesn't match verification array, got (%u,%u) expected (%u,%u)",
318				 ItemPointerGetBlockNumber(&items.lookup_tids[i]),
319				 ItemPointerGetOffsetNumber(&items.lookup_tids[i]),
320				 ItemPointerGetBlockNumber(&items.insert_tids[i]),
321				 ItemPointerGetOffsetNumber(&items.insert_tids[i]));
322	}
323
324	PG_RETURN_VOID();
325}

327/*
328 * In real world use, we care if the memory usage is greater than
329 * some configured limit. Here we just want to verify that
330 * TidStoreMemoryUsage is not broken.
331 */
332Datum
333test_is_full(PG_FUNCTION_ARGS)
334{
335	bool		is_full;
336
337	check_tidstore_available();
338
339	is_full = (TidStoreMemoryUsage(tidstore) > tidstore_empty_size);
340
341	PG_RETURN_BOOL(is_full);
342}
343
344/* Free the tidstore */
345Datum
346test_destroy(PG_FUNCTION_ARGS)
347{
348	check_tidstore_available();
349
350	TidStoreDestroy(tidstore);
351	tidstore = NULL;
352	items.num_tids = 0;
353	pfree(items.insert_tids);
354	pfree(items.lookup_tids);
355	pfree(items.iter_tids);
356
357	PG_RETURN_VOID();
