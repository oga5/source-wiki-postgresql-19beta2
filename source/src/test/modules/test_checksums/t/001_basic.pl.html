4# Test suite for testing enabling data checksums in an online cluster
5use strict;
6use warnings FATAL => 'all';
7
8use PostgreSQL::Test::Cluster;
9use PostgreSQL::Test::Utils;
10use Test::More;
11
12use FindBin;
13use lib $FindBin::RealBin;
14
15use DataChecksums::Utils;
16
17# Initialize node with checksums disabled.
18my $node = PostgreSQL::Test::Cluster->new('basic_node');
19$node->init(no_data_checksums => 1);
20$node->start;
21
22# Create some content to have un-checksummed data in the cluster
23$node->safe_psql('postgres',
24	"CREATE TABLE t AS SELECT generate_series(1,10000) AS a;");
25
26# Ensure that checksums are turned off
27test_checksum_state($node, 'off');
28
29# Enable data checksums and wait for the state transition to 'on'
30enable_data_checksums($node, wait => 'on');
31
32# Run a dummy query just to make sure we can read back data
33my $result =
34  $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1 ");
35is($result, '9999', 'ensure checksummed pages can be read back');
36
37# Enable data checksums again which should be a no-op so we explicitly don't
38# wait for any state transition as none should happen here.
39enable_data_checksums($node);
40test_checksum_state($node, 'on');
41# ..and make sure we can still read/write data
42$node->safe_psql('postgres', "UPDATE t SET a = a + 1;");
43$result = $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
44is($result, '10000', 'ensure checksummed pages can be read back');
45
46# Disable checksums again and wait for the state transition
47disable_data_checksums($node, wait => 1);
48
49# Test reading data again
50$result = $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
51is($result, '10000', 'ensure previously checksummed pages can be read back');
52
53# Re-enable checksums and make sure that the underlying data has changed to
54# ensure that checksums will be different.
55$node->safe_psql('postgres', "UPDATE t SET a = a + 1;");
56enable_data_checksums($node, wait => 'on');
57
58# Run a dummy query just to make sure we can read back the data
59$result = $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
60is($result, '10000', 'ensure checksummed pages can be read back');
61
62$node->stop;
63done_testing();
