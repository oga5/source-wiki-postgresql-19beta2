4# Test suite for testing enabling data checksums in an online cluster with

6use strict;
7use warnings FATAL => 'all';
8use PostgreSQL::Test::Cluster;
9use PostgreSQL::Test::Utils;
10use Test::More;
11
12use FindBin;
13use lib $FindBin::RealBin;
14
15use DataChecksums::Utils;
16
17# Initialize primary node
18my $node_primary = PostgreSQL::Test::Cluster->new('standby_restarts_primary');
19$node_primary->init(allows_streaming => 1, no_data_checksums => 1);
20$node_primary->start;
21
22my $slotname = 'physical_slot';
23$node_primary->safe_psql('postgres',
24	"SELECT pg_create_physical_replication_slot('$slotname')");
25
26# Take backup
27my $backup_name = 'my_backup';
28$node_primary->backup($backup_name);
29
30# Create streaming standby linking to primary
31my $node_standby = PostgreSQL::Test::Cluster->new('standby_restarts_standby');
32$node_standby->init_from_backup($node_primary, $backup_name,
33	has_streaming => 1);
34$node_standby->append_conf(
35	'postgresql.conf', qq[
36primary_slot_name = '$slotname'
37]);
38$node_standby->start;
39
40# Create some content on the primary to have un-checksummed data in the cluster
41$node_primary->safe_psql('postgres',
42	"CREATE TABLE t AS SELECT generate_series(1,10000) AS a;");
43
44# Wait for standby to catch up
45$node_primary->wait_for_catchup($node_standby, 'replay',
46	$node_primary->lsn('insert'));
47
48# Check that checksums are turned off on all nodes
49test_checksum_state($node_primary, 'off');
50test_checksum_state($node_standby, 'off');
51
52# ---------------------------------------------------------------------------
53# Enable checksums for the cluster, and make sure that both the primary and
54# standby change state.
55#
56
57# Initiate enabling of checksums and ensure that the primary switches to
58# either "inprogress-on" or "on"
59enable_data_checksums($node_primary);
60my $result = $node_primary->poll_query_until(
61	'postgres',
62	"SELECT setting = 'off' FROM pg_catalog.pg_settings WHERE name = 'data_checksums';",
63	'f');
64is($result, 1, 'ensure primary has transitioned from off');
65# Wait for checksum enable to be replayed
66$node_primary->wait_for_catchup($node_standby, 'replay');
67
68# Ensure that the standby has switched to "inprogress-on" or "on".  Normally it
69# would be "inprogress-on", but it is theoretically possible for the primary to
70# complete the checksum enabling *and* have the standby replay that record
71# before we reach the check below.
72$result = $node_standby->poll_query_until(
73	'postgres',
74	"SELECT setting = 'off' FROM pg_catalog.pg_settings WHERE name = 'data_checksums';",
75	'f');
76is($result, 1, 'ensure standby has absorbed the inprogress-on barrier');
77$result = $node_standby->safe_psql('postgres',
78	"SELECT setting FROM pg_catalog.pg_settings WHERE name = 'data_checksums';"
79);
80
81is(($result eq 'inprogress-on' || $result eq 'on'),
82	1, 'ensure checksums are on, or in progress, on standby_1');
83
84# Insert some more data which should be checksummed on INSERT
85$node_primary->safe_psql('postgres',
86	"INSERT INTO t VALUES (generate_series(1, 10000));");
87
88# Wait for checksums enabled on the primary and standby
89wait_for_checksum_state($node_primary, 'on');
90wait_for_checksum_state($node_standby, 'on');
91
92$result =
93  $node_primary->safe_psql('postgres', "SELECT count(a) FROM t WHERE a > 1");
94is($result, '19998', 'ensure we can safely read all data with checksums');
95
96$result = $node_primary->poll_query_until(
97	'postgres',
98	"SELECT count(*) FROM pg_stat_activity WHERE backend_type LIKE 'datachecksums%';",
99	'0');
100is($result, 1, 'await datachecksums worker/launcher termination');
101
102#
103# Disable checksums and ensure it's propagated to standby and that we can
104# still read all data
105#
106
107# Disable checksums and wait for the operation to be replayed
108disable_data_checksums($node_primary);
109$node_primary->wait_for_catchup($node_standby, 'replay');
110# Ensure that the primary and standby has switched to off
111wait_for_checksum_state($node_primary, 'off');
112wait_for_checksum_state($node_standby, 'off');
113# Double-check reading data without errors
114$result =
115  $node_primary->safe_psql('postgres', "SELECT count(a) FROM t WHERE a > 1");
116is($result, "19998", 'ensure we can safely read all data without checksums');
117
118# ---------------------------------------------------------------------------
119# Test that enabling checksums does not emit WAL for unlogged relations.
120# Unlogged relations are wiped on recovery, so FPIs for them would be
121# pointless and waste WAL traffic / standby I/O.
122#
123# Additionally, exercise standby promotion to ensure the init fork of an
124# unlogged relation is still WAL-logged during checksum enable -- otherwise
125# the standby keeps a stale init fork and the post-promotion main fork
126# fails verification on every read (see ResetUnloggedRelations()).  Both
127# tables must exist BEFORE enable_data_checksums() so that their init
128# forks get re-checksummed during the enable sweep.
129#
130
131$node_primary->safe_psql('postgres',
132	'CREATE UNLOGGED TABLE unlogged_tbl AS SELECT generate_series(1,1000) AS a;'
133);
134# Use a btree index so the init fork is non-trivial (one metapage).
135$node_primary->safe_psql(
136	'postgres', q[
137	CREATE UNLOGGED TABLE unlogged_promo (id int PRIMARY KEY,
138	                                      payload text);
139	INSERT INTO unlogged_promo
140	  SELECT g, repeat('x', 100) FROM generate_series(1, 1000) g;
141	CREATE INDEX unlogged_promo_payload_idx ON unlogged_promo (payload);
142]);
143$node_primary->wait_for_catchup($node_standby, 'replay',
144	$node_primary->lsn('insert'));
145
146# Get the relfilenode and database OID so we can inspect the filesystem
147my $unlogged_rfn = $node_primary->safe_psql('postgres',
148	"SELECT relfilenode FROM pg_class WHERE relname = 'unlogged_tbl';");
149my $db_oid = $node_primary->safe_psql('postgres',
150	"SELECT oid FROM pg_database WHERE datname = 'postgres';");
151
152# Verify the standby only has the init fork (no main fork)
153my $standby_datadir = $node_standby->data_dir;
154ok( !-f "$standby_datadir/base/$db_oid/$unlogged_rfn",
155	'standby has no main fork for unlogged table before enable');
156
157# Re-enable data checksums
158enable_data_checksums($node_primary, wait => 'on');
159wait_for_checksum_state($node_standby, 'on');
160
161# After standby replays, the unlogged main file must still not exist.
162# If the bug were present, FPI replay would materialize the full table.
163$node_primary->wait_for_catchup($node_standby, 'replay',
164	$node_primary->lsn('insert'));
165ok( !-f "$standby_datadir/base/$db_oid/$unlogged_rfn",
166	'standby has no main fork for unlogged table after enable');
167
168# Verify unlogged relation size is 0 on the standby (main fork missing)
169my $standby_size = $node_standby->safe_psql('postgres',
170	"SELECT pg_relation_size('unlogged_tbl', 'main');");
171is($standby_size, '0',
172	'unlogged table has zero size on standby after checksum enable');
173
174# Unlogged table should still be readable on primary
175$result =
176  $node_primary->safe_psql('postgres', 'SELECT count(*) FROM unlogged_tbl;');
177is($result, '1000',
178	'unlogged table readable on primary after checksum enable');
179
180# Alter persistence to logged, and make sure we can read it on both the primary
181# and standby without any page verification errors in the logfiles.
182$node_primary->safe_psql('postgres', 'ALTER TABLE unlogged_tbl SET logged;');
183$node_primary->wait_for_catchup($node_standby, 'replay',
184	$node_primary->lsn('insert'));
185
186$result =
187  $node_primary->safe_psql('postgres', 'SELECT sum(a) FROM unlogged_tbl;');
188is($result, '500500', 'previously unlogged table can be read on primary');
189$result =
190  $node_standby->safe_psql('postgres', 'SELECT sum(a) FROM unlogged_tbl;');
191is($result, '500500', 'previously unlogged table can be read on standby');
192
193# ---------------------------------------------------------------------------
194# Promote the standby and verify the unlogged_promo relation (created above
195# before the enable sweep) is still usable.  Without the init-fork WAL fix,
196# every read of the index would fail with
197# "page verification failed, calculated checksum X but expected 0".
198#
199$node_primary->stop;
200$node_standby->promote;
201
202$result =
203  $node_standby->safe_psql('postgres',
204	'SELECT count(*) FROM unlogged_promo;');
205is($result, '0',
206	'unlogged table readable on promoted standby (truncated as expected)');
207
208$node_standby->safe_psql('postgres',
209	"INSERT INTO unlogged_promo SELECT g, repeat('y',100) FROM generate_series(1,100) g;"
210);
211$result = $node_standby->safe_psql('postgres',
212	'SET enable_seqscan = off; SELECT id FROM unlogged_promo WHERE id = 50;');
213is($result, '50', 'indexed lookup on promoted standby returns expected row');
214
215$node_standby->stop;
216
217# Perform one final pass over the logs and hunt for unexpected errors
218my $log = PostgreSQL::Test::Utils::slurp_file($node_primary->logfile, 0);
219unlike(
220	$log,
221	qr/page verification failed,.+\d$/m,
222	"no checksum validation errors in primary log");
223$log = PostgreSQL::Test::Utils::slurp_file($node_standby->logfile, 0);
224unlike(
225	$log,
226	qr/page verification failed,.+\d$/m,
227	"no checksum validation errors in standby log");
228
229# ---------------------------------------------------------------------------
230# Test that enforced state transitions during promotion (via StartupXLOG) are
231# performed as expected.  When the primary crashes during inprogress-on the
232# standby should revert to off at promotion.  In order to check the transition
233# the test will keep an open psql session with the standby during promotion.
234
235# The cluster is currently broken down from the previous test.  Start up the
236# primary as primary, disable checksums and create a new standby from that
237# state.
238$node_standby->clean_node();
239$node_primary->start();
240disable_data_checksums($node_primary, wait => 'off');
241
242# Re-create a new streaming standby linking to primary.  The replication slot
243# name is reused from earlier but a fresh backup is taken
244$backup_name = 'my_new_backup';
245$node_primary->backup($backup_name);
246$node_standby = PostgreSQL::Test::Cluster->new('standby_restarts_standby');
247$node_standby->init_from_backup($node_primary, $backup_name,
248	has_streaming => 1);
249$node_standby->append_conf(
250	'postgresql.conf', qq[
251primary_slot_name = '$slotname'
252]);
253$node_standby->start;
254$node_primary->wait_for_catchup($node_standby, 'replay');
255
256# Open a background psql connection on the primary and inject a barrier to
257# block progress on to keep the state from advancing past inprogress-on
258my $node_primary_bpsql = $node_primary->background_psql('postgres');
259$node_primary_bpsql->query_safe('CREATE TEMPORARY TABLE tt (a integer);');
260# Also open a background psql connection to the standby to make sure we have
261# an active backend during promotion.
262my $node_standby_bpsql = $node_standby->background_psql('postgres');
263
264# Start to enable checksums and wait until both primary and standby have moved
265# to the inprogress-on state.  Processing will block here as the temporary rel
266# barrier will block the primary from finishing.
267enable_data_checksums($node_primary, wait => 'inprogress-on');
268$node_primary->wait_for_catchup($node_standby, 'replay');
269test_checksum_state($node_standby, 'inprogress-on');
270
271# Crash the primary before checksums are enabled and promote the standby.  The
272# new primary node will now revert the state of 'off' since checksums weren't
273# fully enabled during the crash.
274$node_primary->teardown_node();
275$node_standby->promote;
276wait_for_checksum_state($node_standby, 'off');
277
278# Ensure that the any backend which was active before, and during, promotion
279# sees the new state.
280$result = $node_standby_bpsql->query_safe("SHOW data_checksums;");
281is($result, 'off',
282	'ensure checksums are set to off after promotion during inprogress-on');
283
284$node_standby_bpsql->quit;
285$node_standby->stop;
286
287done_testing();
