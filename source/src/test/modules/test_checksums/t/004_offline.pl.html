4# Test suite for testing enabling data checksums offline from various states
5# of checksum processing
6use strict;
7use warnings FATAL => 'all';
8
9use PostgreSQL::Test::Cluster;
10use PostgreSQL::Test::Utils;
11use Test::More;
12
13use FindBin;
14use lib $FindBin::RealBin;
15
16use DataChecksums::Utils;
17
18# Initialize node with checksums disabled.
19my $node = PostgreSQL::Test::Cluster->new('offline_node');
20$node->init(no_data_checksums => 1);
21$node->start;
22
23# Create some content to have un-checksummed data in the cluster
24$node->safe_psql('postgres',
25	"CREATE TABLE t AS SELECT generate_series(1,10000) AS a;");
26
27# Ensure that checksums are disabled
28test_checksum_state($node, 'off');
29
30# Enable checksums offline using pg_checksums
31$node->stop;
32$node->checksum_enable_offline;
33$node->start;
34
35# Ensure that checksums are enabled
36test_checksum_state($node, 'on');
37
38# Run a dummy query just to make sure we can read back some data
39my $result =
40  $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
41is($result, '9999', 'ensure checksummed pages can be read back');
42
43# Disable checksums offline again using pg_checksums
44$node->stop;
45$node->checksum_disable_offline;
46$node->start;
47
48# Ensure that checksums are disabled
49test_checksum_state($node, 'off');
50
51# Create a barrier for checksum enablement to block on, in this case a pre-
52# existing temporary table which is kept open while processing is started. We
53# can accomplish this by setting up an interactive psql process which keeps the
54# temporary table created as we enable checksums in another psql process.
55
56my $bsession = $node->background_psql('postgres');
57$bsession->query_safe('CREATE TEMPORARY TABLE tt (a integer);');
58
59# In another session, make sure we can see the blocking temp table but start
60# processing anyways and check that we are blocked with a proper wait event.
61$result = $node->safe_psql('postgres',
62	"SELECT relpersistence FROM pg_catalog.pg_class WHERE relname = 'tt';");
63is($result, 't', 'ensure we can see the temporary table');
64
65# Enable, but stop waiting at inprogress-on since it will sit there until the
66# above temporary table is removed.
67enable_data_checksums($node, wait => 'inprogress-on');
68
69# Turn the cluster off and enable checksums offline, then start back up.
70# Stop the cluster before exiting the background session since otherwise
71# checksums might have time to get enabled before shutting down the cluster.
72$node->stop('fast');
73$bsession->quit;
74$node->checksum_enable_offline;
75$node->start;
76
77# Ensure that checksums are now enabled even though processing wasn't
78# restarted
79test_checksum_state($node, 'on');
80
81# Run a dummy query just to make sure we can read back some data
82$result = $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
83is($result, '9999', 'ensure checksummed pages can be read back');
84
85$node->stop;
86done_testing();
