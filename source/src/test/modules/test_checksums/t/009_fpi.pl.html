4use strict;
5use warnings FATAL => 'all';
6
7use PostgreSQL::Test::Cluster;
8use PostgreSQL::Test::Utils;
9use Test::More;
10
11use FindBin;
12use lib $FindBin::RealBin;
13
14use DataChecksums::Utils;
15
16# Create and start a cluster with one node
17my $node = PostgreSQL::Test::Cluster->new('fpi_node');
18$node->init(allows_streaming => 1, no_data_checksums => 1);
19# max_connections need to be bumped in order to accommodate for pgbench clients
20# and log_statement is dialled down since it otherwise will generate enormous
21# amounts of logging. Page verification failures are still logged.
22$node->append_conf(
23	'postgresql.conf',
24	qq[
25max_connections = 100
26log_statement = none
27]);
28$node->start;
29$node->safe_psql('postgres', 'CREATE EXTENSION test_checksums;');
30# Create some content to have un-checksummed data in the cluster
31$node->safe_psql('postgres',
32	"CREATE TABLE t AS SELECT generate_series(1, 1000000) AS a;");
33
34# Enable data checksums and wait for the state transition to 'on'
35enable_data_checksums($node, wait => 'on');
36
37$node->safe_psql('postgres', 'UPDATE t SET a = a + 1;');
38
39disable_data_checksums($node, wait => 1);
40
41$node->append_conf('postgresql.conf', 'full_page_writes = off');
42$node->restart;
43test_checksum_state($node, 'off');
44
45$node->safe_psql('postgres', 'UPDATE t SET a = a + 1;');
46$node->safe_psql('postgres', 'DELETE FROM t WHERE a < 10000;');
47
48$node->adjust_conf('postgresql.conf', 'full_page_writes', 'on');
49$node->restart;
50test_checksum_state($node, 'off');
51
52enable_data_checksums($node, wait => 'on');
53
54my $result = $node->safe_psql('postgres', 'SELECT count(*) FROM t;');
55is($result, '990003', 'Reading back all data from table t');
56
57$node->stop;
58my $log = PostgreSQL::Test::Utils::slurp_file($node->logfile, 0);
59unlike(
60	$log,
61	qr/page verification failed,.+\d$/m,
62	"no checksum validation errors in server log");
63
64done_testing();
