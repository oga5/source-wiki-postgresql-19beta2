20# This test suite is expensive, or very expensive, to execute.  There are two
21# PG_TEST_EXTRA options for running it, "checksum" for a pared-down test suite
22# an "checksum_extended" for the full suite.  The full suite can run for hours
23# on slow or constrained systems.
24my $extended = undef;
25if ($ENV{PG_TEST_EXTRA})
26{
27	$extended = 1 if ($ENV{PG_TEST_EXTRA} =~ /\bchecksum_extended\b/);
28	plan skip_all => 'Expensive data checksums test disabled'
29	  unless ($ENV{PG_TEST_EXTRA} =~ /\bchecksum(_extended)?\b/);
30}
31else
32{
33	plan skip_all => 'Expensive data checksums test disabled';
34}
35
36if ($ENV{enable_injection_points} ne 'yes')
37{
38	plan skip_all => 'Injection points not supported by this build';
39}

41my $node_primary_slot = 'physical_slot';
42my $node_primary_backup = 'primary_backup';
43my $node_primary;
44my $node_primary_loglocation = 0;
45my $node_standby;
46my $node_standby_loglocation = 0;
47
48# The number of full test iterations which will be performed. The exact number
49# of tests performed and the wall time taken is non-deterministic as the test
50# performs a lot of randomized actions, but 5 iterations will be a long test
51# run regardless.
52my $TEST_ITERATIONS = 1;
53$TEST_ITERATIONS = 5 if ($extended);
54
55# Variables which record the current state of the cluster
56my $data_checksum_state = 'off';
57
58my $pgbench_primary = undef;
59my $pgbench_standby = undef;
60
61# Start a pgbench run in the background against the server specified via the
62# port passed as parameter
63sub background_pgbench
64{
65	my ($port, $standby) = @_;
66	my $pgbench = ($standby ? \$pgbench_standby : \$pgbench_primary);
67
68	# Terminate any currently running pgbench process before continuing
69	$$pgbench->finish if $$pgbench;
70
71	my $clients = 1;
72	my $runtime = 5;
73
74	if ($extended)
75	{
76		# Randomize the number of pgbench clients a bit (range 1-16)
77		$clients = 1 + int(rand(15));
78		$runtime = 600;
79	}
80
81	my @cmd = ('pgbench', '-p', $port, '-T', $runtime, '-c', $clients);
82	# Randomize whether we spawn connections or not
83	push(@cmd, '-C') if ($extended && cointoss());
84	# If we run on a standby it needs to be a read-only benchmark
85	push(@cmd, '-S') if ($standby);
86	push(@cmd, '-n') if ($standby);
87	# Finally add the database name to use
88	push(@cmd, 'postgres');
89
90	$$pgbench = IPC::Run::start(
91		\@cmd,
92		'<' => '/dev/null',
93		'>' => '/dev/null',
94		'2>' => '/dev/null',
95		IPC::Run::timer($PostgreSQL::Test::Utils::timeout_default));
96}

98# Invert the state of data checksums in the cluster, if data checksums are on
99# then disable them and vice versa. Also performs proper validation of the
100# before and after state.
101sub flip_data_checksums
102{
103	my $temptablewait = 0;
104
105	# First, make sure the cluster is in the state we expect it to be
106	test_checksum_state($node_primary, $data_checksum_state);
107	test_checksum_state($node_standby, $data_checksum_state);
108
109	if ($data_checksum_state eq 'off')
110	{
111		# Coin-toss to see if we are injecting a retry due to a temptable
112		if (cointoss())
113		{
114			$node_primary->safe_psql('postgres',
115				"SELECT injection_points_attach('datachecksumsworker-fake-temptable-wait', 'notice');"
116			);
117			$temptablewait = 1;
118		}
119
120		# log LSN right before we start changing checksums
121		my $result =
122		  $node_primary->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
123		note("LSN before enabling: " . $result . "\n");
124
125		# Ensure that the primary switches to "inprogress-on"
126		enable_data_checksums($node_primary, wait => 'inprogress-on');
127
128		random_sleep() if ($extended);
129
130		# Wait for checksum enable to be replayed
131		$node_primary->wait_for_catchup($node_standby, 'replay');
132
133		# Ensure that the standby has switched to "inprogress-on" or "on".
134		# Normally it would be "inprogress-on", but it is theoretically
135		# possible for the primary to complete the checksum enabling *and* have
136		# the standby replay that record before we reach the check below.
137		$result = $node_standby->poll_query_until(
138			'postgres',
139			"SELECT setting = 'off' "
140			  . "FROM pg_catalog.pg_settings "
141			  . "WHERE name = 'data_checksums';",
142			'f');
143		is($result, 1,
144			'ensure standby has absorbed the inprogress-on barrier');
145		$result = $node_standby->safe_psql('postgres',
146				"SELECT setting "
147			  . "FROM pg_catalog.pg_settings "
148			  . "WHERE name = 'data_checksums';");
149
150		is( ($result eq 'inprogress-on' || $result eq 'on'),
151			1,
152			'ensure checksums are on, or in progress, on standby_1, got: '
153			  . $result);
154
155		# Wait for checksums enabled on the primary and standby
156		wait_for_checksum_state($node_primary, 'on');
157
158		# log LSN right after the primary flips checksums to "on"
159		$result =
160		  $node_primary->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
161		note("LSN after enabling: " . $result . "\n");
162
163		random_sleep() if ($extended);
164		wait_for_checksum_state($node_standby, 'on');
165
166		$node_primary->safe_psql('postgres',
167			"SELECT injection_points_detach('datachecksumsworker-fake-temptable-wait');"
168		) if ($temptablewait);
169		$data_checksum_state = 'on';
170	}
171	elsif ($data_checksum_state eq 'on')
172	{
173		random_sleep() if ($extended);
174
175		# log LSN right before we start changing checksums
176		my $result =
177		  $node_primary->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
178		note("LSN before disabling: " . $result . "\n");
179
180		disable_data_checksums($node_primary);
181		$node_primary->wait_for_catchup($node_standby, 'replay');
182
183		# Wait for checksums disabled on the primary and standby
184		random_sleep() if ($extended);
185		wait_for_checksum_state($node_primary, 'off');
186		wait_for_checksum_state($node_standby, 'off');
187
188		# log LSN right after the primary flips checksums to "off"
189		$result =
190		  $node_primary->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
191		note("LSN after disabling: " . $result . "\n");
192
193		$data_checksum_state = 'off';
194	}
195	else
196	{
197		# This should only happen due to programmer error when hacking on the
198		# test code, but since that might pass subtly we error out.
199		BAIL_OUT('data_checksum_state variable has invalid state:'
200			  . $data_checksum_state);
201	}
202}

204# Create and start a cluster with one primary and one standby node, and ensure
205# they are caught up and in sync.
206$node_primary = PostgreSQL::Test::Cluster->new('pgbench_standby_main');
207$node_primary->init(allows_streaming => 1, no_data_checksums => 1);
208# max_connections need to be bumped in order to accommodate for pgbench clients
209# and log_statement is dialled down since it otherwise will generate enormous
210# amounts of logging. Page verification failures are still logged.
211$node_primary->append_conf(
212	'postgresql.conf',
213	qq[
214max_connections = 30
215log_statement = none
216hot_standby_feedback = on
217]);
218$node_primary->start;
219$node_primary->safe_psql('postgres', 'CREATE EXTENSION test_checksums;');
220$node_primary->safe_psql('postgres', 'CREATE EXTENSION injection_points;');
221# Create some content to have un-checksummed data in the cluster
222$node_primary->safe_psql('postgres',
223	"CREATE TABLE t AS SELECT generate_series(1, 100000) AS a;");
224$node_primary->safe_psql('postgres',
225	"SELECT pg_create_physical_replication_slot('$node_primary_slot');");
226$node_primary->backup($node_primary_backup);
227
228$node_standby = PostgreSQL::Test::Cluster->new('pgbench_standby_standby');
229$node_standby->init_from_backup($node_primary, $node_primary_backup,
230	has_streaming => 1);
231$node_standby->append_conf(
232	'postgresql.conf', qq[
233primary_slot_name = '$node_primary_slot'
234]);
235$node_standby->start;
236
237# Initialize pgbench and wait for the objects to be created on the standby
238my $scalefactor = ($extended ? 10 : 1);
239$node_primary->command_ok(
240	[
241		'pgbench', '-p', $node_primary->port, '-i', '-s', $scalefactor, '-q',
242		'postgres'
243	]);
244$node_primary->wait_for_catchup($node_standby, 'replay');
245
246# Start the test suite with pgbench running on all nodes
247background_pgbench($node_standby->port, 1);
248background_pgbench($node_primary->port, 0);
249
250# Main test suite. This loop will start a pgbench run on the cluster and while
251# that's running flip the state of data checksums concurrently. It will then
252# randomly restart the cluster and then check for
253# the desired state.  The idea behind doing things randomly is to stress out
254# any timing related issues by subjecting the cluster for varied workloads.
255# A TODO is to generate a trace such that any test failure can be traced to
256# its order of operations for debugging.
257for (my $i = 0; $i < $TEST_ITERATIONS; $i++)
258{
259	note("iteration ", ($i + 1), " of ", $TEST_ITERATIONS);
260
261	if (!$node_primary->is_alive)
262	{
263		# start, to do recovery, and stop
264		$node_primary->start;
265		$node_primary->stop('fast');
266
267		# Since the log isn't being written to now, parse the log and check
268		# for instances of checksum verification failures.
269		my $log = PostgreSQL::Test::Utils::slurp_file($node_primary->logfile,
270			$node_primary_loglocation);
271		unlike(
272			$log,
273			qr/page verification failed,.+\d$/m,
274			"no checksum validation errors in primary log (during WAL recovery)"
275		);
276		$node_primary_loglocation = -s $node_primary->logfile;
277
278		# randomize the WAL size, to trigger checkpoints less/more often
279		my $sb = 32 + int(rand(960));
280		$node_primary->append_conf('postgresql.conf', qq[max_wal_size = $sb]);
281
282		note("changing primary max_wal_size to " . $sb);
283
284		$node_primary->start;
285
286		# Start a pgbench in the background against the primary
287		background_pgbench($node_primary->port, 0);
288	}
289
290	if (!$node_standby->is_alive)
291	{
292		$node_standby->start;
293		$node_standby->stop('fast');
294
295		# Since the log isn't being written to now, parse the log and check
296		# for instances of checksum verification failures.
297		my $log =
298		  PostgreSQL::Test::Utils::slurp_file($node_standby->logfile,
299			$node_standby_loglocation);
300		unlike(
301			$log,
302			qr/page verification failed,.+\d$/m,
303			"no checksum validation errors in standby_1 log (during WAL recovery)"
304		);
305		$node_standby_loglocation = -s $node_standby->logfile;
306
307		# randomize the WAL size, to trigger checkpoints less/more often
308		my $sb = 32 + int(rand(960));
309		$node_standby->append_conf('postgresql.conf', qq[max_wal_size = $sb]);
310
311		note("changing standby max_wal_size to " . $sb);
312
313		$node_standby->start;
314
315		# Start a read-only pgbench in the background on the standby
316		background_pgbench($node_standby->port, 1);
317	}
318
319	$node_primary->safe_psql('postgres', "UPDATE t SET a = a + 1;");
320	$node_primary->wait_for_catchup($node_standby, 'write');
321
322	flip_data_checksums();
323	random_sleep() if ($extended);
324	my $result = $node_primary->safe_psql('postgres',
325		"SELECT count(*) FROM t WHERE a > 1");
326	is($result, '100000', 'ensure data pages can be read back on primary');
327	random_sleep();
328
329	# Potentially powercycle the cluster (the nodes independently). A TODO is
330	# to randomly stop the nodes in the opposite order too.
331	if ($extended && cointoss())
332	{
333		$node_primary->stop(stopmode());
334
335		# print the contents of the control file on the primary
336		PostgreSQL::Test::Utils::system_log("pg_controldata",
337			$node_primary->data_dir);
338
339		# slurp the file after shutdown, so that it doesn't interfere with the recovery
340		my $log = PostgreSQL::Test::Utils::slurp_file($node_primary->logfile,
341			$node_primary_loglocation);
342		unlike(
343			$log,
344			qr/page verification failed,.+\d$/m,
345			"no checksum validation errors in primary log (outside WAL recovery)"
346		);
347		$node_primary_loglocation = -s $node_primary->logfile;
348	}
349
350	random_sleep() if ($extended);
351
352	if ($extended && cointoss())
353	{
354		$node_standby->stop(stopmode());
355
356		# print the contents of the control file on the standby
357		PostgreSQL::Test::Utils::system_log("pg_controldata",
358			$node_standby->data_dir);
359
360		# slurp the file after shutdown, so that it doesn't interfere with the recovery
361		my $log =
362		  PostgreSQL::Test::Utils::slurp_file($node_standby->logfile,
363			$node_standby_loglocation);
364		unlike(
365			$log,
366			qr/page verification failed,.+\d$/m,
367			"no checksum validation errors in standby_1 log (outside WAL recovery)"
368		);
369		$node_standby_loglocation = -s $node_standby->logfile;
370	}
371}
372
373# make sure the nodes are running
374if (!$node_primary->is_alive)
375{
376	$node_primary->start;
377}
378
379if (!$node_standby->is_alive)
380{
381	$node_standby->start;
382}
383
384# Testrun is over, ensure that data reads back as expected and perform a final
385# verification of the data checksum state.
386my $result =
387  $node_primary->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
388is($result, '100000', 'ensure data pages can be read back on primary');
389test_checksum_state($node_primary, $data_checksum_state);
390test_checksum_state($node_standby, $data_checksum_state);
391
392# Perform one final pass over the logs and hunt for unexpected errors
393my $log = PostgreSQL::Test::Utils::slurp_file($node_primary->logfile,
394	$node_primary_loglocation);
395unlike(
396	$log,
397	qr/page verification failed,.+\d$/m,
398	"no checksum validation errors in primary log");
399$node_primary_loglocation = -s $node_primary->logfile;
400$log = PostgreSQL::Test::Utils::slurp_file($node_standby->logfile,
401	$node_standby_loglocation);
402unlike(
403	$log,
404	qr/page verification failed,.+\d$/m,
405	"no checksum validation errors in standby_1 log");
406$node_standby_loglocation = -s $node_standby->logfile;
407
408$node_standby->teardown_node;
409$node_primary->teardown_node;
410
411done_testing();
