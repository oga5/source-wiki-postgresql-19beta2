4# Test suite for testing enabling data checksums in an online cluster with a
5# restart which breaks processing.
6use strict;
7use warnings FATAL => 'all';
8
9use PostgreSQL::Test::Cluster;
10use PostgreSQL::Test::Utils;
11use Test::More;
12
13use FindBin;
14use lib $FindBin::RealBin;
15
16use DataChecksums::Utils;
17
18# Initialize node with checksums disabled.
19my $node = PostgreSQL::Test::Cluster->new('restarts_node');
20$node->init(no_data_checksums => 1);
21$node->start;
22
23# Initialize result storage for queries
24my $result;
25
26# Create some content to have un-checksummed data in the cluster
27$node->safe_psql('postgres',
28	"CREATE TABLE t AS SELECT generate_series(1,10000) AS a;");
29
30# Ensure that checksums are disabled
31test_checksum_state($node, 'off');
32
33SKIP:
34{
35	skip 'Data checksum delay tests not enabled in PG_TEST_EXTRA', 6
36	  if (!$ENV{PG_TEST_EXTRA}
37		|| $ENV{PG_TEST_EXTRA} !~ /\bchecksum_extended\b/);
38
39	# Create a barrier for checksum enablement to block on, in this case a pre-
40	# existing temporary table which is kept open while processing is started.
41	# We can accomplish this by setting up an interactive psql process which
42	# keeps the temporary table created as we enable checksums in another psql
43	# process.
44	#
45	# This is a similar test to the synthetic variant in 005_injection.pl
46	# which fakes this scenario.
47	my $bsession = $node->background_psql('postgres');
48	$bsession->query_safe('CREATE TEMPORARY TABLE tt (a integer);');
49
50	# In another session, make sure we can see the blocking temp table but
51	# start processing anyways and check that we are blocked with a proper
52	# wait event.
53	$result = $node->safe_psql('postgres',
54		"SELECT relpersistence FROM pg_catalog.pg_class WHERE relname = 'tt';"
55	);
56	is($result, 't', 'ensure we can see the temporary table');
57
58	# Enabling data checksums shouldn't work as the process is blocked on the
59	# temporary table held open by $bsession. Ensure that we reach inprogress-
60	# on before we do more tests.
61	enable_data_checksums($node, wait => 'inprogress-on');
62
63	# Wait for processing to finish and the worker waiting for leftover temp
64	# relations to be able to actually finish
65	$result = $node->poll_query_until(
66		'postgres',
67		"SELECT wait_event FROM pg_catalog.pg_stat_activity "
68		  . "WHERE backend_type = 'datachecksums worker';",
69		'ChecksumEnableTemptableWait');
70
71	# The datachecksumsworker waits for temporary tables to disappear for 3
72	# seconds before retrying, so sleep for 4 seconds to be guaranteed to see
73	# a retry cycle
74	sleep(4);
75
76	# Re-check the wait event to ensure we are blocked on the right thing.
77	$result = $node->safe_psql('postgres',
78			"SELECT wait_event FROM pg_catalog.pg_stat_activity "
79		  . "WHERE backend_type = 'datachecksums worker';");
80	is($result, 'ChecksumEnableTemptableWait',
81		'ensure the correct wait condition is set');
82	test_checksum_state($node, 'inprogress-on');
83
84	# Stop the cluster while bsession is still attached.  We can't close the
85	# session first since the brief period between closing and stopping might
86	# be enough for checksums to get enabled.
87	$node->stop;
88	$bsession->quit;
89	$node->start;
90
91	# Ensure the checksums aren't enabled across the restart.  This leaves the
92	# cluster in the same state as before we entered the SKIP block.
93	test_checksum_state($node, 'off');
94}
95
96enable_data_checksums($node, wait => 'on');
97
98$result = $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
99is($result, '9999', 'ensure checksummed pages can be read back');
100
101$result = $node->poll_query_until(
102	'postgres',
103	"SELECT count(*) FROM pg_stat_activity WHERE backend_type LIKE 'datachecksums%';",
104	'0');
105is($result, 1, 'await datachecksums worker/launcher termination');
106
107disable_data_checksums($node, wait => 1);
108
109$node->stop;
110done_testing();
