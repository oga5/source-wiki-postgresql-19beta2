19# This test suite is expensive, or very expensive, to execute.  There are two
20# PG_TEST_EXTRA options for running it, "checksum" for a pared-down test suite
21# an "checksum_extended" for the full suite.  The full suite can run for hours
22# on slow or constrained systems.
23my $extended = undef;
24if ($ENV{PG_TEST_EXTRA})
25{
26	$extended = 1 if ($ENV{PG_TEST_EXTRA} =~ /\bchecksum_extended\b/);
27	plan skip_all => 'Expensive data checksums test disabled'
28	  unless ($ENV{PG_TEST_EXTRA} =~ /\bchecksum(_extended)?\b/);
29}
30else
31{
32	plan skip_all => 'Expensive data checksums test disabled';
33}
34
35if ($ENV{enable_injection_points} ne 'yes')
36{
37	plan skip_all => 'Injection points not supported by this build';
38}
39
40my $node;
41my $node_loglocation = 0;
42
43# The number of full test iterations which will be performed. The exact number
44# of tests performed and the wall time taken is non-deterministic as the test
45# performs a lot of randomized actions, but 10 iterations will be a long test
46# run regardless.
47my $TEST_ITERATIONS = 1;
48$TEST_ITERATIONS = 10 if ($extended);
49
50# Variables which record the current state of the cluster
51my $data_checksum_state = 'off';
52my $pgbench = undef;
53
54# Start a pgbench run in the background against the server specified via the
55# port passed as parameter.
56sub background_rw_pgbench
57{
58	my $port = shift;
59
60	# If a previous pgbench is still running, start by shutting it down.
61	$pgbench->finish if $pgbench;
62
63	my $clients = 1;
64	my $runtime = 2;
65
66	if ($extended)
67	{
68		# Randomize the number of pgbench clients a bit (range 1-16)
69		$clients = 1 + int(rand(15));
70		$runtime = 600;
71	}
72	my @cmd = ('pgbench', '-p', $port, '-T', $runtime, '-c', $clients);
73
74	# Randomize whether we spawn connections or not
75	push(@cmd, '-C') if ($extended && cointoss);
76	# Finally add the database name to use
77	push(@cmd, 'postgres');
78
79	$pgbench = IPC::Run::start(
80		\@cmd,
81		'<' => '/dev/null',
82		'>' => '/dev/null',
83		'2>' => '/dev/null',
84		IPC::Run::timer($PostgreSQL::Test::Utils::timeout_default));
85}
86
87# Invert the state of data checksums in the cluster, if data checksums are on
88# then disable them and vice versa. Also performs proper validation of the
89# before and after state.
90sub flip_data_checksums
91{
92	my $temptablewait = 0;
93
94	# First, make sure the cluster is in the state we expect it to be
95	test_checksum_state($node, $data_checksum_state);
96
97	if ($data_checksum_state eq 'off')
98	{
99		# Coin-toss to see if we are injecting a retry due to a temptable
100		if (cointoss())
101		{
102			$node->safe_psql('postgres',
103				"SELECT injection_points_attach('datachecksumsworker-fake-temptable-wait', 'notice');"
104			);
105			$temptablewait = 1;
106		}
107
108		# log LSN right before we start changing checksums
109		my $result =
110		  $node->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
111		note("LSN before enabling: " . $result . "\n");
112
113		# Ensure that the primary switches to "inprogress-on"
114		enable_data_checksums($node, wait => 'inprogress-on');
115
116		random_sleep() if ($extended);
117
118		# Wait for checksums enabled on the primary
119		wait_for_checksum_state($node, 'on');
120
121		# log LSN right after the primary flips checksums to "on"
122		$result = $node->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
123		note("LSN after enabling: " . $result . "\n");
124
125		random_sleep() if ($extended);
126
127		$node->safe_psql('postgres',
128			"SELECT injection_points_detach('datachecksumsworker-fake-temptable-wait');"
129		) if ($temptablewait);
130		$data_checksum_state = 'on';
131	}
132	elsif ($data_checksum_state eq 'on')
133	{
134		random_sleep() if ($extended);
135
136		# log LSN right before we start changing checksums
137		my $result =
138		  $node->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
139		note("LSN before disabling: " . $result . "\n");
140
141		disable_data_checksums($node);
142
143		# Wait for checksums disabled on the primary
144		wait_for_checksum_state($node, 'off');
145
146		# log LSN right after the primary flips checksums to "off"
147		$result = $node->safe_psql('postgres', "SELECT pg_current_wal_lsn()");
148		note("LSN after disabling: " . $result . "\n");
149
150		random_sleep() if ($extended);
151
152		$data_checksum_state = 'off';
153	}
154	else
155	{
156		# This should only happen due to programmer error when hacking on the
157		# test code, but since that might pass subtly we error out.
158		BAIL_OUT('data_checksum_state variable has invalid state:'
159			  . $data_checksum_state);
160	}
161}

163# Create and start a cluster with one node
164$node = PostgreSQL::Test::Cluster->new('pgbench_single_main');
165$node->init(allows_streaming => 1, no_data_checksums => 1);
166# max_connections need to be bumped in order to accommodate for pgbench clients
167# and log_statement is dialled down since it otherwise will generate enormous
168# amounts of logging. Page verification failures are still logged.
169$node->append_conf(
170	'postgresql.conf',
171	qq[
172max_connections = 100
173log_statement = none
174]);
175$node->start;
176$node->safe_psql('postgres', 'CREATE EXTENSION test_checksums;');
177$node->safe_psql('postgres', 'CREATE EXTENSION injection_points;');
178# Create some content to have un-checksummed data in the cluster
179$node->safe_psql('postgres',
180	"CREATE TABLE t AS SELECT generate_series(1, 100000) AS a;");
181# Initialize pgbench
182my $scalefactor = ($extended ? 10 : 1);
183$node->command_ok(
184	[
185		'pgbench', '-p', $node->port, '-i',
186		'-s', $scalefactor, '-q', 'postgres'
187	]);
188# Start the test suite with pgbench running.
189background_rw_pgbench($node->port);
190
191# Main test suite. This loop will start a pgbench run on the cluster and while
192# that's running flip the state of data checksums concurrently. It will then
193# randomly restart the cluster and then check for
194# the desired state.  The idea behind doing things randomly is to stress out
195# any timing related issues by subjecting the cluster for varied workloads.
196# A TODO is to generate a trace such that any test failure can be traced to
197# its order of operations for debugging.
198for (my $i = 0; $i < $TEST_ITERATIONS; $i++)
199{
200	note("iteration ", ($i + 1), " of ", $TEST_ITERATIONS);
201
202	if (!$node->is_alive)
203	{
204		# Start, to do recovery, and stop
205		$node->start;
206		$node->stop('fast');
207
208		# Since the log isn't being written to now, parse the log and check
209		# for instances of checksum verification failures.
210		my $log = PostgreSQL::Test::Utils::slurp_file($node->logfile,
211			$node_loglocation);
212		unlike(
213			$log,
214			qr/page verification failed,.+\d$/m,
215			"no checksum validation errors in primary log (during WAL recovery)"
216		);
217		$node_loglocation = -s $node->logfile;
218
219		# Randomize the WAL size, to trigger checkpoints less/more often
220		my $sb = 64 + int(rand(1024));
221		$node->append_conf('postgresql.conf', qq[max_wal_size = $sb]);
222		note("changing max_wal_size to " . $sb);
223
224		$node->start;
225
226		# Start a pgbench in the background against the primary
227		background_rw_pgbench($node->port);
228	}
229
230	$node->safe_psql('postgres', "UPDATE t SET a = a + 1;");
231
232	flip_data_checksums();
233	random_sleep() if ($extended);
234	my $result =
235	  $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
236	is($result, '100000', 'ensure data pages can be read back on primary');
237
238	random_sleep() if ($extended);
239
240	# Potentially powercycle the node
241	if (cointoss())
242	{
243		$node->stop(stopmode());
244
245		PostgreSQL::Test::Utils::system_log("pg_controldata",
246			$node->data_dir);
247
248		my $log = PostgreSQL::Test::Utils::slurp_file($node->logfile,
249			$node_loglocation);
250		unlike(
251			$log,
252			qr/page verification failed,.+\d$/m,
253			"no checksum validation errors in primary log (outside WAL recovery)"
254		);
255		$node_loglocation = -s $node->logfile;
256	}
257
258	random_sleep() if ($extended);
259}
260
261# Make sure the node is running
262if (!$node->is_alive)
263{
264	$node->start;
265}
266
267# Testrun is over, ensure that data reads back as expected and perform a final
268# verification of the data checksum state.
269my $result =
270  $node->safe_psql('postgres', "SELECT count(*) FROM t WHERE a > 1");
271is($result, '100000', 'ensure data pages can be read back on primary');
272test_checksum_state($node, $data_checksum_state);
273
274# Perform one final pass over the logs and hunt for unexpected errors
275my $log =
276  PostgreSQL::Test::Utils::slurp_file($node->logfile, $node_loglocation);
277unlike(
278	$log,
279	qr/page verification failed,.+\d$/m,
280	"no checksum validation errors in primary log");
281$node_loglocation = -s $node->logfile;
282
283$node->teardown_node;
284
285done_testing();
