4# Test parallel autovacuum behavior

6use strict;
7use warnings FATAL => 'all';
8use PostgreSQL::Test::Cluster;
9use PostgreSQL::Test::Utils;
10use Test::More;
11
12if ($ENV{enable_injection_points} ne 'yes')
13{
14	plan skip_all => 'Injection points not supported by this build';
15}
16
17# Before each test we should disable autovacuum for 'test_autovac' table and
18# generate some dead tuples in it.
19sub prepare_for_next_test
20{
21	my ($node, $test_number) = @_;
22
23	$node->safe_psql(
24		'postgres', qq{
25		ALTER TABLE test_autovac SET (autovacuum_enabled = false);
26		UPDATE test_autovac SET col_1 = $test_number;
27	});
28}
29
30my $node = PostgreSQL::Test::Cluster->new('main');
31$node->init;
32
33# Limit to one autovacuum worker and disable autovacuum logging globally
34# (enabled only on the test table) so that log checks below match only
35# activity on the expected table.
36$node->append_conf(
37	'postgresql.conf', qq{
38autovacuum_max_workers = 1
39autovacuum_worker_slots = 1
40autovacuum_max_parallel_workers = 2
41max_worker_processes = 10
42max_parallel_workers = 10
43log_min_messages = debug2
44autovacuum_naptime = '1s'
45min_parallel_index_scan_size = 0
46log_autovacuum_min_duration = -1
47});
48$node->start;
49
50# Check if the extension injection_points is available, as it may be
51# possible that this script is run with installcheck, where the module
52# would not be installed by default.
53if (!$node->check_extension('injection_points'))
54{
55	plan skip_all => 'Extension injection_points not installed';
56}
57
58# Create all functions needed for testing
59$node->safe_psql(
60	'postgres', qq{
61	CREATE EXTENSION injection_points;
62});
63
64my $indexes_num = 3;
65my $initial_rows_num = 10_000;
66my $autovacuum_parallel_workers = 2;
67
68# Create table and fill it with some data
69$node->safe_psql(
70	'postgres', qq{
71	CREATE TABLE test_autovac (
72		id SERIAL PRIMARY KEY,
73		col_1 INTEGER,  col_2 INTEGER,  col_3 INTEGER,  col_4 INTEGER
74	) WITH (autovacuum_parallel_workers = $autovacuum_parallel_workers,
75			log_autovacuum_min_duration = 0);
76
77	INSERT INTO test_autovac
78	SELECT
79		g AS col1,
80		g + 1 AS col2,
81		g + 2 AS col3,
82		g + 3 AS col4
83	FROM generate_series(1, $initial_rows_num) AS g;
84});
85
86# Create specified number of b-tree indexes on the table
87$node->safe_psql(
88	'postgres', qq{
89	DO \$\$
90	DECLARE
91		i INTEGER;
92	BEGIN
93		FOR i IN 1..$indexes_num LOOP
94			EXECUTE format('CREATE INDEX idx_col_\%s ON test_autovac (col_\%s);', i, i);
95		END LOOP;
96	END \$\$;
97});
98
99# Test 1 :
100# Our table has enough indexes and appropriate reloptions, so autovacuum must
101# be able to process it in parallel mode. Just check if it can do it.
102
103prepare_for_next_test($node, 1);
104my $log_offset = -s $node->logfile;
105
106$node->safe_psql(
107	'postgres', qq{
108	ALTER TABLE test_autovac SET (autovacuum_enabled = true);
109});
110
111# Wait for parallel autovacuum to complete; check worker count matches reloptions.
112$node->wait_for_log(
113	qr/parallel workers: index vacuum: 2 planned, 2 launched in total/,
114	$log_offset);
115ok(1, "parallel autovacuum on test_autovac table");
116
117# Test 2:
118# Check whether parallel autovacuum leader can propagate cost-based parameters
119# to the parallel workers.
120
121prepare_for_next_test($node, 2);
122$log_offset = -s $node->logfile;
123
124$node->safe_psql(
125	'postgres', qq{
126	SELECT injection_points_attach('autovacuum-start-parallel-vacuum', 'wait');
127
128	ALTER TABLE test_autovac SET (autovacuum_parallel_workers = 1, autovacuum_enabled = true);
129});
130
131# Wait until parallel autovacuum is inited
132$node->wait_for_event('autovacuum worker',
133	'autovacuum-start-parallel-vacuum');
134
135# Update the shared cost-based delay parameters.
136$node->safe_psql(
137	'postgres', qq{
138	ALTER SYSTEM SET autovacuum_vacuum_cost_limit = 500;
139	ALTER SYSTEM SET autovacuum_vacuum_cost_delay = 5;
140	ALTER SYSTEM SET vacuum_cost_page_miss = 10;
141	ALTER SYSTEM SET vacuum_cost_page_dirty = 10;
142	ALTER SYSTEM SET vacuum_cost_page_hit = 10;
143	SELECT pg_reload_conf();
144});
145
146# Resume the leader process to update the shared parameters during heap scan (i.e.
147# vacuum_delay_point() is called) and launch a parallel vacuum worker, but it stops
148# before vacuuming indexes due to the injection point.
149$node->safe_psql(
150	'postgres', qq{
151	SELECT injection_points_wakeup('autovacuum-start-parallel-vacuum');
152});
153
154# Check whether parallel worker successfully updated all parameters during
155# index processing.
156$node->wait_for_log(
157	qr/parallel autovacuum worker updated cost params: cost_limit=500, cost_delay=5, cost_page_miss=10, cost_page_dirty=10, cost_page_hit=10/,
158	$log_offset);
159
160# Cleanup
161$node->safe_psql(
162	'postgres', qq{
163	SELECT injection_points_detach('autovacuum-start-parallel-vacuum');
164});
165
166ok(1,
167	"vacuum delay parameter changes are propagated to parallel vacuum workers"
168);
169
170$node->stop;
171done_testing();
