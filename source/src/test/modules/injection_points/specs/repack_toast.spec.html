1# REPACK (CONCURRENTLY);
2#
3# Test handling of TOAST. At the same time, no tuplesort.
4setup
5{
6
7	CREATE EXTENSION IF NOT EXISTS injection_points;
8
9	-- Generate text consisting of repeated strings so that it can be
10	-- compressed easily.
11	CREATE FUNCTION gen_compressible(seed int)
12	RETURNS text
13	LANGUAGE sql IMMUTABLE as $$
14		SELECT repeat(md5((seed * 1000)::text), 50);
15	$$;
16
17	-- Like above, but too big even after compression.
18	CREATE FUNCTION gen_compressible_external(seed int)
19	RETURNS text
20	LANGUAGE sql IMMUTABLE as $$
21		SELECT repeat(md5((seed * 1000)::text), 10000);
22	$$;
23
24	-- Generate a string of random characters that is not likely to be
25	-- compressed, but is big enough to be stored externally.
26	CREATE FUNCTION gen_external()
27	RETURNS text
28	LANGUAGE sql as $$
29		SELECT string_agg(chr(65 + trunc(25 * random())::int), '')
30		FROM generate_series(1, 2048) s(x);
31	$$;
32
33	-- Not compressible like above, but small enough to stay in-line.
34	CREATE FUNCTION gen_inline()
35	RETURNS text
36	LANGUAGE sql as $$
37		SELECT string_agg(chr(65 + trunc(25 * random())::int), '')
38		FROM generate_series(1, 1024) s(x);
39	$$;
40
41	-- A varlena short enough to have a one-byte header.
42	CREATE FUNCTION gen_short()
43	RETURNS text
44	LANGUAGE sql as $$
45		SELECT string_agg(chr(65 + trunc(25 * random())::int), '')
46		FROM generate_series(1, 120) s(x);
47	$$;
48
49	CREATE TABLE repack_toast(drop1 int, i int PRIMARY KEY, drop2 int,
50		j text COMPRESSION pglz, k text COMPRESSION pglz);
51	INSERT INTO repack_toast(drop1, i, drop2, j, k)
52	SELECT 42, gs, 42, gen_external(), gen_compressible(gs) FROM generate_series(1, 10) gs;
53	ALTER TABLE repack_toast DROP COLUMN drop1, DROP COLUMN drop2;
54	ALTER TABLE repack_toast ALTER COLUMN k SET COMPRESSION default;
55	INSERT INTO repack_toast(i, j, k)
56	SELECT gs, gen_external(), gen_compressible(142857) FROM generate_series(11, 20) gs;
57
58	ALTER TABLE repack_toast SET (toast_tuple_target = 128);
59
60	CREATE TABLE relfilenodes(node oid);
61
62	CREATE TABLE data_s1 (i int, j text, j_toast oid, k text, k_toast oid);
63	CREATE TABLE data_s2 (LIKE data_s1);
64}
65
66
67teardown
68{
69	DROP TABLE repack_toast;
70	DROP EXTENSION injection_points;
71	DROP FUNCTION gen_compressible(int);
72	DROP FUNCTION gen_compressible_external(int);
73	DROP FUNCTION gen_external();
74	DROP FUNCTION gen_inline();
75	DROP FUNCTION gen_short();
76
77	DROP TABLE relfilenodes;
78	DROP TABLE data_s1;
79	DROP TABLE data_s2;
80}

83setup
84{
85	SELECT injection_points_set_local();
86	SELECT injection_points_attach('repack-concurrently-before-lock', 'wait');

89# Perform the initial load and wait for s2 to do some data changes.
90step s1_wait_before_lock
91{
92	REPACK (CONCURRENTLY) repack_toast;
93}
94
95# Check the table, after REPACK has completed.  s2 must have saved the data
96# as it was visible to it.  We check that the relfilenode changed in addition
97# to verifying that the actual data matches.
98step s1_check
99{
100	INSERT INTO relfilenodes(node)
101	SELECT c2.relfilenode
102	FROM pg_class c1 JOIN pg_class c2 ON c2.oid = c1.oid OR c2.oid = c1.reltoastrelid
103	WHERE c1.relname='repack_toast';
104
105	SELECT count(DISTINCT node) FROM relfilenodes;
106
107	INSERT INTO data_s1(i, j, j_toast, k, k_toast)
108	SELECT i,
109	j, COALESCE(pg_column_toast_chunk_id(j), 0) AS j_toast,
110	k, COALESCE(pg_column_toast_chunk_id(k), 0) AS k_toast
111	FROM repack_toast;
112
113	-- this should be empty
114	SELECT d1.i, substring(d1.j FOR 12) AS d1_j, substring(d1.k FOR 12) AS d1_k,
115		d2.i, substring(d2.j FOR 12) AS d2_j, substring(d2.k FOR 12) AS d2_k,
116		d1.j_toast as d1_j_tst, d2.j_toast as d2_j_tst,
117		d1.k_toast as d1_k_tst, d2.k_toast AS d2_k_tst
118	FROM data_s1 d1 FULL JOIN data_s2 d2 USING (i, j, k)
119	WHERE d1.i ISNULL OR d2.i ISNULL;
120}
121teardown
122{
123    SELECT injection_points_detach('repack-concurrently-before-lock');
124}

126session s2
127
128# Test different kinds of toast data changes.
129step s2_updates
130{
131	DELETE FROM repack_toast WHERE i=1;
132	INSERT INTO repack_toast(i, j, k) VALUES (1, gen_external(), gen_compressible(1));
133
134	-- existing toast data unchanged.  (This covers the case where we
135	-- adjust the toast pointer.)
136	UPDATE repack_toast SET i=i+300 where i % 10 = 2 RETURNING OLD.i, NEW.i;
137
138	-- "j" is here an external indirect, written to the file separately.
139	UPDATE repack_toast SET j=gen_external() where i % 10 = 3 RETURNING OLD.i, NEW.i;
140
141	-- the updated value of "j" is compressed.
142	UPDATE repack_toast SET j=gen_compressible(1), k=k||'' where i % 10 = 4 RETURNING i;
143
144	-- the updated value of "j" is compressed externally.
145	UPDATE repack_toast SET j=gen_compressible_external(2) where i % 10 = 5 RETURNING i;
146
147	-- the updated value of "j" stays inline.
148	UPDATE repack_toast SET j=gen_inline(), k=repeat(k,5) where i % 10 = 6 RETURNING i;
149
150	-- updated value of "j" is a short varlena; "k" is written separately.
151	UPDATE repack_toast SET j=gen_short(), k=gen_external() where i % 10 = 7 RETURNING i;
152}
153
154# Check the table from the perspective of s2.  This saves data so that it can
155# be verified later.
156step s2_check
157{
158	INSERT INTO relfilenodes(node)
159	SELECT c2.relfilenode
160	FROM pg_class c1 JOIN pg_class c2 ON c2.oid = c1.oid OR c2.oid = c1.reltoastrelid
161	WHERE c1.relname='repack_toast';
162
163	INSERT INTO data_s2(i, j, j_toast, k, k_toast)
164	SELECT i, j, COALESCE(pg_column_toast_chunk_id(j), 0) AS j_toast,
165	k, COALESCE(pg_column_toast_chunk_id(k), 0) AS k_toast
166	FROM repack_toast;
167}
168step s2_wakeup_before_lock
169{
170	SELECT injection_points_wakeup('repack-concurrently-before-lock');
171}

174# CONCURRENTLY find their way into the table.
175permutation

177	s2_updates
178	s2_check
179	s2_wakeup_before_lock
