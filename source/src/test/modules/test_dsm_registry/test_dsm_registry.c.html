18#include "utils/builtins.h"

20PG_MODULE_MAGIC;
21
22typedef struct TestDSMRegistryStruct
23{
24	int			val;
25	LWLock		lck;
26} TestDSMRegistryStruct;
27
28typedef struct TestDSMRegistryHashEntry
29{
30	char		key[64];
31	dsa_pointer val;
32} TestDSMRegistryHashEntry;
33
34static TestDSMRegistryStruct *tdr_dsm;
35static dsa_area *tdr_dsa;
36static dshash_table *tdr_hash;
37
38static const dshash_parameters dsh_params = {
39	offsetof(TestDSMRegistryHashEntry, val),
40	sizeof(TestDSMRegistryHashEntry),
41	dshash_strcmp,
42	dshash_strhash,
43	dshash_strcpy
44};
45
46static void
47init_tdr_dsm(void *ptr, void *arg)
48{
49	TestDSMRegistryStruct *dsm = (TestDSMRegistryStruct *) ptr;
50
51	if ((int) (intptr_t) arg != 5432)
52		elog(ERROR, "unexpected arg value %d", (int) (intptr_t) arg);
53
54	LWLockInitialize(&dsm->lck, LWLockNewTrancheId("test_dsm_registry"));
55	dsm->val = 0;
56}
57
58static void
59tdr_attach_shmem(void)
60{
61	bool		found;
62
63	tdr_dsm = GetNamedDSMSegment("test_dsm_registry_dsm",
64								 sizeof(TestDSMRegistryStruct),
65								 init_tdr_dsm,
66								 &found, (void *) (intptr_t) 5432);
67
68	if (tdr_dsa == NULL)
69		tdr_dsa = GetNamedDSA("test_dsm_registry_dsa", &found);
70
71	if (tdr_hash == NULL)
72		tdr_hash = GetNamedDSHash("test_dsm_registry_hash", &dsh_params, &found);
73}
74
75PG_FUNCTION_INFO_V1(set_val_in_shmem);
76Datum
77set_val_in_shmem(PG_FUNCTION_ARGS)
78{
79	tdr_attach_shmem();
80
81	LWLockAcquire(&tdr_dsm->lck, LW_EXCLUSIVE);
82	tdr_dsm->val = PG_GETARG_INT32(0);
83	LWLockRelease(&tdr_dsm->lck);
84
85	PG_RETURN_VOID();
86}
87
88PG_FUNCTION_INFO_V1(get_val_in_shmem);
89Datum
90get_val_in_shmem(PG_FUNCTION_ARGS)
91{
92	int			ret;
93
94	tdr_attach_shmem();
95
96	LWLockAcquire(&tdr_dsm->lck, LW_SHARED);
97	ret = tdr_dsm->val;
98	LWLockRelease(&tdr_dsm->lck);
99
100	PG_RETURN_INT32(ret);
101}
102
103PG_FUNCTION_INFO_V1(set_val_in_hash);
104Datum
105set_val_in_hash(PG_FUNCTION_ARGS)
106{
107	TestDSMRegistryHashEntry *entry;
108	char	   *key = TextDatumGetCString(PG_GETARG_DATUM(0));
109	char	   *val = TextDatumGetCString(PG_GETARG_DATUM(1));
110	bool		found;
111
112	if (strlen(key) >= offsetof(TestDSMRegistryHashEntry, val))
113		ereport(ERROR,
114				(errmsg("key too long")));
115
116	tdr_attach_shmem();
117
118	entry = dshash_find_or_insert(tdr_hash, key, &found);
119	if (found)
120		dsa_free(tdr_dsa, entry->val);
121
122	entry->val = dsa_allocate(tdr_dsa, strlen(val) + 1);
123	strcpy(dsa_get_address(tdr_dsa, entry->val), val);
124
125	dshash_release_lock(tdr_hash, entry);
126
127	PG_RETURN_VOID();
128}
129
130PG_FUNCTION_INFO_V1(get_val_in_hash);
131Datum
132get_val_in_hash(PG_FUNCTION_ARGS)
133{
134	TestDSMRegistryHashEntry *entry;
135	char	   *key = TextDatumGetCString(PG_GETARG_DATUM(0));
136	text	   *val = NULL;
137
138	tdr_attach_shmem();
139
140	entry = dshash_find(tdr_hash, key, false);
141	if (entry == NULL)
142		PG_RETURN_NULL();
143
144	val = cstring_to_text(dsa_get_address(tdr_dsa, entry->val));
145
146	dshash_release_lock(tdr_hash, entry);
147
148	PG_RETURN_TEXT_P(val);
149}
