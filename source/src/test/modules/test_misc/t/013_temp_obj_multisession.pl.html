3# Tests that one session cannot read or modify data in another session's
4# temporary table.  Each session keeps its temp data in its own local
5# buffer pool, and a different backend has no visibility into those
6# buffers, so any command that needs to look at the data must be
7# rejected.
8#
9# DROP TABLE is intentionally allowed: it does not touch the table's
10# contents, and autovacuum relies on this to clean up orphaned temp
11# relations left behind by a crashed backend.
12#
13# A regression caught here typically means a new buffer-access entry
14# point bypasses the RELATION_IS_OTHER_TEMP() check.  See
15# ReadBuffer_common(), StartReadBuffersImpl(), and read_stream_begin_impl()
16# for the existing checks.  When adding a new command or buffer-access
17# path, also add a corresponding case below.
18
19use strict;
20use warnings;
21use PostgreSQL::Test::Cluster;
22use PostgreSQL::Test::Utils;
23use PostgreSQL::Test::BackgroundPsql;
24use Test::More;
25
26my $node = PostgreSQL::Test::Cluster->new('temp_lock');
27$node->init;
28$node->start;
29
30# Owner session.  Created via background_psql so it stays alive while
31# the second session probes its temp objects.
32my $psql1 = $node->background_psql('postgres');
33
34# Initially create the table without an index, so read paths go straight
35# through the read-stream / buffer-manager entry points without being
36# masked by an index scan that would hit ReadBuffer_common from nbtree.
37$psql1->query_safe(q(CREATE TEMP TABLE foo AS SELECT 42 AS val;));
38
39# Also create an empty table, so read path go straight through the
40# extend-relation entry point.
41$psql1->query_safe(q(CREATE TEMP TABLE empty_foo (val INT);));
42
43# Resolve the owner's temp schema so the probing session can refer to
44# the table by a fully-qualified name.
45my $tempschema = $node->safe_psql(
46	'postgres',
47	q{
48      SELECT n.nspname
49      FROM pg_class c
50      JOIN pg_namespace n ON n.oid = c.relnamespace
51      WHERE relname = 'foo' AND relpersistence = 't';
52    }
53);
54chomp $tempschema;
55ok($tempschema =~ /^pg_temp_\d+$/, "got temp schema: $tempschema");
56
57my ($stdout, $stderr);
58
59# DML and SELECT have to read the table's data and therefore go through
60# the buffer manager.  With no index on the table, the planner cannot
61# use index access, so SELECT/UPDATE/DELETE/MERGE/COPY all run through
62# the read-stream path and are caught by read_stream_begin_impl().
63
64$node->psql(
65	'postgres',
66	"SELECT val FROM $tempschema.foo;",
67	stderr => \$stderr);
68like(
69	$stderr,
70	qr/cannot access temporary tables of other sessions/,
71	'SELECT (seqscan via read_stream)');
72
73# INSERT into empty table goes through hio.c which calls RelationAddBlocks() to
74# extend the table; that hits the check before new pages are created for the
75# table.
76$node->psql(
77	'postgres',
78	"INSERT INTO $tempschema.empty_foo VALUES (42);",
79	stderr => \$stderr);
80like(
81	$stderr,
82	qr/cannot access temporary tables of other sessions/,
83	'INSERT (caught via hio.c)');
84
85# INSERT goes through hio.c which calls ReadBufferExtended() to find a
86# page with free space; that hits the existing check before any data
87# is written.
88$node->psql(
89	'postgres',
90	"INSERT INTO $tempschema.foo VALUES (73);",
91	stderr => \$stderr);
92like(
93	$stderr,
94	qr/cannot access temporary tables of other sessions/,
95	'INSERT (caught via hio.c)');
96
97$node->psql(
98	'postgres',
99	"UPDATE $tempschema.foo SET val = NULL;",
100	stderr => \$stderr);
101like($stderr, qr/cannot access temporary tables of other sessions/, 'UPDATE');
102
103$node->psql('postgres', "DELETE FROM $tempschema.foo;", stderr => \$stderr);
104like($stderr, qr/cannot access temporary tables of other sessions/, 'DELETE');
105
106$node->psql(
107	'postgres',
108	"MERGE INTO $tempschema.foo USING (VALUES (42)) AS s(val) "
109	  . "ON foo.val = s.val WHEN MATCHED THEN DELETE;",
110	stderr => \$stderr);
111like($stderr, qr/cannot access temporary tables of other sessions/, 'MERGE');
112
113$node->psql('postgres', "COPY $tempschema.foo TO STDOUT;",
114	stderr => \$stderr);
115like($stderr, qr/cannot access temporary tables of other sessions/, 'COPY');
116
117# DDL and maintenance commands have their own command-specific checks
118# (older than the buffer-manager check above), so they fail with
119# command-specific error messages.  Verifying them here documents the
120# expected behaviour and guards against accidental removal of those
121# checks.
122
123$node->psql('postgres', "TRUNCATE TABLE $tempschema.foo;",
124	stderr => \$stderr);
125like($stderr, qr/cannot truncate temporary tables of other sessions/,
126	'TRUNCATE');
127
128$node->psql(
129	'postgres',
130	"ALTER TABLE $tempschema.foo ALTER COLUMN val TYPE bigint;",
131	stderr => \$stderr);
132like($stderr, qr/cannot alter temporary tables of other sessions/,
133	'ALTER TABLE');
134
135# VACUUM silently skips other sessions' temp tables (vacuum_rel() returns
136# without warning to avoid noise during database-wide VACUUM).  Verify
137# that no error is reported, and that no buffer-access path is hit.
138$node->psql('postgres', "VACUUM $tempschema.foo;", stderr => \$stderr);
139is($stderr, '', 'VACUUM is silently skipped');
140
141$node->psql('postgres', "CLUSTER $tempschema.foo;", stderr => \$stderr);
142like($stderr,
143	qr/cannot execute CLUSTER on temporary tables of other sessions/,
144	'CLUSTER');
145
146# Now create an index to exercise the index-scan path.  nbtree calls
147# ReadBuffer (which is ReadBufferExtended -> ReadBuffer_common), so
148# this exercises a different chain of buffer-manager entry points.
149$psql1->query_safe(q(CREATE INDEX ON foo(val);));
150
151$node->psql(
152	'postgres',
153	"SET enable_seqscan = off; SELECT val FROM $tempschema.foo WHERE val = 42;",
154	stderr => \$stderr);
155like(
156	$stderr,
157	qr/cannot access temporary tables of other sessions/,
158	'index scan (ReadBuffer_common via nbtree)');
159
160# ALTER INDEX goes through the same CheckAlterTableIsSafe() path as
161# ALTER TABLE, so it produces the same error.
162$node->psql(
163	'postgres',
164	"ALTER INDEX $tempschema.foo_val_idx SET (fillfactor = 50);",
165	stderr => \$stderr);
166like($stderr, qr/cannot alter temporary tables of other sessions/,
167	'ALTER INDEX');
168
169# A function created by the owner in its own pg_temp using its own
170# row type can be observed via the catalog by a separate session.
171# ALTER FUNCTION and DROP FUNCTION on it must work as catalog
172# operations -- they don't read the underlying table -- which
173# documents the boundary between catalog and data access for temp
174# objects.
175$psql1->query_safe(
176		q[CREATE FUNCTION pg_temp.foo_id(r foo) RETURNS int LANGUAGE SQL ]
177	  . q[AS 'SELECT r.val';]);
178
179$node->psql(
180	'postgres',
181	"ALTER FUNCTION $tempschema.foo_id($tempschema.foo) "
182	  . "SET search_path = pg_catalog;",
183	stderr => \$stderr);
184is($stderr, '', 'ALTER FUNCTION on function over other session\'s row type');
185
186$node->psql(
187	'postgres',
188	"DROP FUNCTION $tempschema.foo_id($tempschema.foo);",
189	stderr => \$stderr);
190is($stderr, '', 'DROP FUNCTION on function over other session\'s row type');
191
192# DROP TABLE on another session's temp table is intentionally permitted.
193# DROP doesn't touch the table's contents, and autovacuum relies on this
194# to remove temp relations orphaned by a crashed backend.  Verify that
195# the bare DROP succeeds without error.
196$node->psql('postgres', "DROP TABLE $tempschema.foo;", stderr => \$stderr);
197is($stderr, '', 'DROP TABLE is allowed');
198
199# Cross-session CREATE FUNCTION scenario.  The owner creates a fresh
200# temp table foo2 in its pg_temp namespace, and a separate session
201# then creates a function whose argument type is that row type.
202# PostgreSQL allows this and emits a NOTICE: the function is moved
203# into the creator's pg_temp namespace with an auto-dependency on
204# the borrowed type, so it disappears together with the session that
205# created it.
206$psql1->query_safe(q(CREATE TEMP TABLE foo2 AS SELECT 42 AS val;));
207
208$node->psql(
209	'postgres',
210	"CREATE FUNCTION public.cross_session_func(r $tempschema.foo2) "
211	  . "RETURNS int LANGUAGE SQL AS 'SELECT 1';",
212	stderr => \$stderr);
213like(
214	$stderr,
215	qr/function "cross_session_func" will be effectively temporary/,
216	'CREATE FUNCTION using other session\'s row type is effectively temporary'
217);
218
219# A bare DROP TABLE on foo2 now fails because cross_session_func
220# depends on its row type.  This is normal SQL dependency behaviour
221# and documents that DROP itself is not blocked by buffer-manager
222# checks -- we get a catalog-level error instead.
223$node->psql('postgres', "DROP TABLE $tempschema.foo2;", stderr => \$stderr);
224like(
225	$stderr,
226	qr/cannot drop table .*\.foo2 because other objects depend on it/,
227	'DROP TABLE blocked by cross-session dependency');
228
229my $foo2_oid = $node->safe_psql('postgres',
230	"SELECT oid FROM pg_class WHERE relname='foo2';");
231
232# Cross-session LOCK TABLE scenario.  Ensure that LockRelationOid is working
233# properly for other temp tables since this mechanism is also used by
234# autovacuum during orphaned tables cleanup.
235my $psql2 = $node->background_psql('postgres');
236$psql2->query_safe(
237	qq{
238	BEGIN;
239	LOCK TABLE $tempschema.foo2 IN ACCESS SHARE MODE;
240});
241
242# When the owner session ends, its temp objects are dropped via the
243# normal session-exit cleanup, which cascades through
244# DEPENDENCY_NORMAL and also removes the cross-session function that
245# depended on the temp row type.  This is the same mechanism
246# autovacuum relies on to clean up temp relations left behind by a
247# crashed backend.
248# Access share lock on the foo2 will block session-exit cleanup, because an
249# owner will try to acquire deletion lock all its temp objects via
250# findDependentObjects.
251my $log_offset = -s $node->logfile;
252$psql1->quit;
253
254# Check whether session-exit cleanup is blocked.
255$node->wait_for_log(qr/waiting for AccessExclusiveLock on relation $foo2_oid/,
256	$log_offset);
257
258# Release lock on foo2 and allow session-exit cleanup to finish.
259$psql2->query_safe(q(COMMIT;));
260$psql2->quit;
261
262# After releasing the lock, the owner can finally acquire
263# AccessExclusiveLock on foo2 and finish session-exit cleanup.  Verify
264# directly that both foo2 (the locked temp table) and cross_session_func
265# (which depended on its row type) have been dropped.  Both being gone
266# confirms the owner's cleanup got past the blocked findDependentObjects()
267# call and completed normally.
268$node->poll_query_until('postgres',
269	"SELECT NOT EXISTS (SELECT 1 FROM pg_class WHERE oid = $foo2_oid)")
270  or die "foo2 was not cleaned up after owner session exit";
271
272is( $node->safe_psql(
273		'postgres',
274		"SELECT count(*) FROM pg_proc WHERE proname = 'cross_session_func'"),
275	'0',
276	'cross_session_func cleaned up by cascade from foo2');
277
278done_testing();
