4# Tests to cross-check the consistency of GUC parameters with
5# postgresql.conf.sample.
6
7use strict;
8use warnings FATAL => 'all';
9use PostgreSQL::Test::Cluster;
10use PostgreSQL::Test::Utils;
11use Test::More;
12
13my $node = PostgreSQL::Test::Cluster->new('main');
14$node->init;
15$node->start;
16
17# Grab the names of all the parameters that can be listed in the
18# configuration sample file.  config_file is an exception, it is not
19# in postgresql.conf.sample but is part of the lists from guc_tables.c.
20# Custom GUCs loaded by extensions are excluded.
21my $all_params = $node->safe_psql(
22	'postgres',
23	"SELECT name
24     FROM pg_settings
25   WHERE NOT 'NOT_IN_SAMPLE' = ANY (pg_settings_get_flags(name)) AND
26       name <> 'config_file' AND category <> 'Customized Options'
27     ORDER BY 1");
28# Note the lower-case conversion, for consistency.
29my @all_params_array = split("\n", lc($all_params));
30
31# Grab the names of all parameters marked as NOT_IN_SAMPLE.
32my $not_in_sample = $node->safe_psql(
33	'postgres',
34	"SELECT name
35     FROM pg_settings
36   WHERE 'NOT_IN_SAMPLE' = ANY (pg_settings_get_flags(name))
37     ORDER BY 1");
38my @not_in_sample_array = split("\n", lc($not_in_sample));
39
40# use the sample file from the temp install
41my $share_dir = $node->config_data('--sharedir');
42my $sample_file = "$share_dir/postgresql.conf.sample";
43
44# List of all the GUCs found in the sample file.
45my @gucs_in_file;
46
47# List of all lines with tabs in the sample file.
48my @lines_with_tabs;
49
50# Read the sample file line-by-line, checking its contents to build a list
51# of everything known as a GUC.
52my $num_tests = 0;
53my $line_num = 0;
54open(my $contents, '<', $sample_file)
55  || die "Could not open $sample_file: $!";
56while (my $line = <$contents>)
57{
58	$line_num++;
59	push @lines_with_tabs, $line_num
60	  if $line =~ /\t/;
61
62	# Check if this line matches a GUC parameter:
63	# - Each parameter is preceded by "#", but not "# " in the sample
64	# file.
65	# - Valid configuration options are followed immediately by " = ",
66	# with one space before and after the equal sign.
67	if ($line =~ m/^#([_[:alnum:]]+) = .*/)
68	{
69		# Lower-case conversion matters for some of the GUCs.
70		my $param_name = lc($1);
71
72		# Ignore some exceptions.
73		next if $param_name eq "include";
74		next if $param_name eq "include_dir";
75		next if $param_name eq "include_if_exists";
76
77		# Update the list of GUCs found in the sample file, for the
78		# follow-up tests.
79		push @gucs_in_file, $param_name;
80
81		next;
82	}
83	# Make sure each line starts with either a # or whitespace
84	fail("$line missing initial # in postgresql.conf.sample")
85	  if $line =~ /^\s*[^#\s]/;
86}
87
88close $contents;
89
90# Cross-check that all the GUCs found in the sample file match the ones
91# fetched above.  This maps the arrays to a hash, making the creation of
92# each exclude and intersection list easier.
93my %gucs_in_file_hash = map { $_ => 1 } @gucs_in_file;
94my %all_params_hash = map { $_ => 1 } @all_params_array;
95my %not_in_sample_hash = map { $_ => 1 } @not_in_sample_array;
96
97my @missing_from_file = grep(!$gucs_in_file_hash{$_}, @all_params_array);
98is(scalar(@missing_from_file),
99	0, "no parameters missing from postgresql.conf.sample");
100
101my @missing_from_list = grep(!$all_params_hash{$_}, @gucs_in_file);
102is(scalar(@missing_from_list), 0, "no parameters missing from guc_tables.c");
103
104my @sample_intersect = grep($not_in_sample_hash{$_}, @gucs_in_file);
105is(scalar(@sample_intersect),
106	0, "no parameters marked as NOT_IN_SAMPLE in postgresql.conf.sample");
107
108is(scalar(@lines_with_tabs),
109	0, "no lines with tabs in postgresql.conf.sample");
110
111# These would log some information only on errors.
112foreach my $param (@missing_from_file)
113{
114	print(
115		"found GUC $param in guc_tables.c, missing from postgresql.conf.sample\n"
116	);
117}
118foreach my $param (@missing_from_list)
119{
120	print(
121		"found GUC $param in postgresql.conf.sample, with incorrect info in guc_tables.c\n"
122	);
123}
124foreach my $param (@sample_intersect)
125{
126	print(
127		"found GUC $param in postgresql.conf.sample, marked as NOT_IN_SAMPLE\n"
128	);
129}
130foreach my $param (@lines_with_tabs)
131{
132	print("found tab in line $param in postgresql.conf.sample\n");
133}
134
135done_testing();
