7
8PostgreSQL::Test::Cluster - class representing PostgreSQL server instance
9
10=head1 SYNOPSIS
11
12  use PostgreSQL::Test::Cluster;
13
14  my $node = PostgreSQL::Test::Cluster->new('mynode');
15
16  # Create a data directory with initdb
17  $node->init();
18
19  # Start the PostgreSQL server
20  $node->start();
21
22  # Add a setting and restart
23  $node->append_conf('postgresql.conf', 'hot_standby = on');
24  $node->restart();
25
26  # Modify or delete an existing setting
27  $node->adjust_conf('postgresql.conf', 'max_wal_senders', '10');
28
29  # get pg_config settings
30  # all the settings in one string
31  $pgconfig = $node->config_data;
32  # all the settings as a map
33  %config_map = ($node->config_data);
34  # specified settings
35  ($incdir, $sharedir) = $node->config_data(qw(--includedir --sharedir));
36
37  # run a query with psql, like:
38  #   echo 'SELECT 1' | psql -qAXt postgres -v ON_ERROR_STOP=1
39  $psql_stdout = $node->safe_psql('postgres', 'SELECT 1');
40
41  # Run psql with a timeout, capturing stdout and stderr
42  # as well as the psql exit code. Pass some extra psql
43  # options. If there's an error from psql raise an exception.
44  my ($stdout, $stderr, $timed_out);
45  my $cmdret = $node->psql('postgres', 'SELECT pg_sleep(600)',
46	  stdout => \$stdout, stderr => \$stderr,
47	  timeout => $PostgreSQL::Test::Utils::timeout_default,
48	  timed_out => \$timed_out,
49	  extra_params => ['--single-transaction'],
50	  on_error_die => 1)
51  print "Sleep timed out" if $timed_out;
52
53  # Similar thing, more convenient in common cases
54  my ($cmdret, $stdout, $stderr) =
55      $node->psql('postgres', 'SELECT 1');
56
57  # run query every second until it returns 't'
58  # or times out
59  $node->poll_query_until('postgres', q|SELECT random() < 0.1;|')
60    or croak "timed out";
61
62  # Do an online pg_basebackup
63  my $ret = $node->backup('testbackup1');
64
65  # Take a backup of a stopped server
66  $node->stop;
67  my $ret = $node->backup_fs_cold('testbackup3')
68
69  # Restore it to create a new independent node (not a replica)
70  my $other_node = PostgreSQL::Test::Cluster->new('mycopy');
71  $other_node->init_from_backup($node, 'testbackup');
72  $other_node->start;
73
74  # Stop the server
75  $node->stop('fast');
76
77  # Find a free, unprivileged TCP port to bind some other service to
78  my $port = PostgreSQL::Test::Cluster::get_free_port();
79

81
82PostgreSQL::Test::Cluster contains a set of routines able to work on a PostgreSQL node,
83allowing to start, stop, backup and initialize it with various options.
84The set of nodes managed by a given test is also managed by this module.
85
86In addition to node management, PostgreSQL::Test::Cluster instances have some wrappers
87around Test::More functions to run commands with an environment set up to

89

320=item $node->raw_connect()
321
322Open a raw TCP or Unix domain socket connection to the server. This is
323used by low-level protocol and connection limit tests.

327sub raw_connect
328{
329	my ($self) = @_;
330	my $pgport = $self->port;
331	my $pghost = $self->host;
332
333	my $socket;
334	if ($PostgreSQL::Test::Utils::use_unix_sockets)
335	{
336		require IO::Socket::UNIX;
337		my $path = "$pghost/.s.PGSQL.$pgport";
338
339		$socket = IO::Socket::UNIX->new(
340			Type => SOCK_STREAM(),
341			Peer => $path,
342		) or croak "Cannot create socket - $IO::Socket::errstr\n";
343	}
344	else
345	{
346		$socket = IO::Socket::INET->new(
347			PeerHost => $pghost,
348			PeerPort => $pgport,
349			Proto => 'tcp'
350		) or croak "Cannot create socket - $IO::Socket::errstr\n";
351	}
352	return $socket;
353}

358
359Check if raw_connect() function works on this platform. This should
360be called to SKIP any tests that require raw_connect().
361
362This tries to connect to the server, to test whether it works or not,,
363so the server is up and running. Otherwise this can return 0 even if

372sub raw_connect_works
373{
374	my ($self) = @_;
375
376	# If we're using Unix domain sockets, we need a working
377	# IO::Socket::UNIX implementation.
378	if ($PostgreSQL::Test::Utils::use_unix_sockets)
379	{
380		eval {
381			my $sock = $self->raw_connect();
382			$sock->close();
383		};
384		if ($@ =~ /not implemented/)
385		{
386			diag "IO::Socket::UNIX does not work: $@";
387			return 0;
388		}
389	}
390	return 1;

487=pod
488
489=item $node->config_data( option ...)
490
491Return configuration data from pg_config, using options (if supplied).
492The options will be things like '--sharedir'.
493
494If no options are supplied, return a string in scalar context or a map in
495array context.
496
497If options are supplied, return the list of values.
498
499=cut
500
501sub config_data
502{
503	my ($self, @options) = @_;
504	local %ENV = $self->_get_env();
505
506	my ($stdout, $stderr);
507	my $result =
508	  IPC::Run::run [ $self->installed_command('pg_config'), @options ],
509	  '>', \$stdout, '2>', \$stderr
510	  or croak "could not execute pg_config";
511	# standardize line endings
512	$stdout =~ s/\r(?=\n)//g;
513	# no options, scalar context: just hand back the output
514	return $stdout unless (wantarray || @options);
515	chomp($stdout);
516	# exactly one option: hand back the output (minus LF)
517	return $stdout if (@options == 1);
518	my @lines = split(/\n/, $stdout);
519	# more than one option: hand back the list of values;
520	return @lines if (@options);
521	# no options, array context: return a map
522	my @map;
523	foreach my $line (@lines)
524	{
525		my ($k, $v) = split(/ = /, $line, 2);
526		push(@map, $k, $v);
527	}
528	return @map;
529}

598=pod
599
600=item $node->init(...)
601
602Initialize a new cluster for testing.
603
604Authentication is set up so that only the current OS user can access the
605cluster. On Unix, we use Unix domain socket connections, with the socket in
606a directory that's only accessible to the current user to ensure that.
607On Windows, we use SSPI authentication to ensure the same (by pg_regress
608--config-auth).
609
610WAL archiving can be enabled on this node by passing the keyword parameter
611has_archiving => 1. This is disabled by default.
612
613Data checksums can be forced off by passing no_data_checksums => 1.
614
615postgresql.conf can be set up for replication by passing the keyword
616parameter allows_streaming => 'logical' or 'physical' (passing 1 will also
617suffice for physical replication) depending on type of replication that
618should be enabled. This is disabled by default.
619
620force_initdb => 1 will force the initialization of the cluster with a new
621initdb rather than copying the data folder from a template.
622
623The new node is set up in a fast but unsafe configuration where fsync is
624disabled.
625
626=cut
627
628sub init
629{
630	my ($self, %params) = @_;
631	my $port = $self->port;
632	my $pgdata = $self->data_dir;
633	my $host = $self->host;
634
635	local %ENV = $self->_get_env();
636
637	$params{allows_streaming} = 0 unless defined $params{allows_streaming};
638	$params{force_initdb} = 0 unless defined $params{force_initdb};
639	$params{has_archiving} = 0 unless defined $params{has_archiving};
640
641	my $initdb_extra_opts_env = $ENV{PG_TEST_INITDB_EXTRA_OPTS};
642	if (defined $initdb_extra_opts_env)
643	{
644		push @{ $params{extra} }, shellwords($initdb_extra_opts_env);
645	}
646
647	# This should override user-supplied initdb options.
648	if ($params{no_data_checksums})
649	{
650		push @{ $params{extra} }, '--no-data-checksums';
651	}
652
653	mkdir $self->backup_dir;
654	mkdir $self->archive_dir;
655
656	# If available, if there aren't any parameters and if force_initdb is
657	# disabled, use a previously initdb'd cluster as a template by copying it.
658	# For a lot of tests, that's substantially cheaper. It does not seem
659	# worth figuring out whether extra parameters affect compatibility, so
660	# initdb is forced if any are defined.
661	#
662	# There's very similar code in pg_regress.c, but we can't easily
663	# deduplicate it until we require perl at build time.
664	if (   $params{force_initdb}
665		or defined $params{extra}
666		or !defined $ENV{INITDB_TEMPLATE})
667	{
668		note("initializing database system by running initdb");
669		PostgreSQL::Test::Utils::system_or_bail(
670			'initdb', '--no-sync',
671			'--pgdata' => $pgdata,
672			'--auth' => 'trust',
673			@{ $params{extra} });
674	}
675	else
676	{
677		my @copycmd;
678		my $expected_exitcode;
679
680		note("initializing database system by copying initdb template");
681
682		if ($PostgreSQL::Test::Utils::windows_os)
683		{
684			@copycmd = qw(robocopy /E /NJS /NJH /NFL /NDL /NP);
685			$expected_exitcode = 1;    # 1 denotes files were copied
686		}
687		else
688		{
689			@copycmd = qw(cp -RPp);
690			$expected_exitcode = 0;
691		}
692
693		@copycmd = (@copycmd, $ENV{INITDB_TEMPLATE}, $pgdata);
694
695		my $ret = PostgreSQL::Test::Utils::system_log(@copycmd);
696
697		# See http://perldoc.perl.org/perlvar.html#%24CHILD_ERROR
698		if ($ret & 127 or $ret >> 8 != $expected_exitcode)
699		{
700			BAIL_OUT(
701				sprintf("failed to execute command \"%s\": $ret",
702					join(" ", @copycmd)));
703		}
704	}
705
706	PostgreSQL::Test::Utils::system_or_bail($ENV{PG_REGRESS},
707		'--config-auth', $pgdata, @{ $params{auth_extra} });
708
709	open my $conf, '>>', "$pgdata/postgresql.conf" or croak $!;
710	print $conf "\n# Added by PostgreSQL::Test::Cluster.pm\n";
711	print $conf "fsync = off\n";
712	print $conf "restart_after_crash = off\n";
713	print $conf "log_line_prefix = '%m %b[%p] %q%a '\n";
714	print $conf "log_statement = all\n";
715	print $conf "log_replication_commands = on\n";
716	print $conf "wal_retrieve_retry_interval = '500ms'\n";
717
718	# If a setting tends to affect whether tests pass or fail, print it after
719	# TEMP_CONFIG.  Otherwise, print it before TEMP_CONFIG, thereby permitting
720	# overrides.  Settings that merely improve performance or ease debugging
721	# belong before TEMP_CONFIG.
722	print $conf PostgreSQL::Test::Utils::slurp_file($ENV{TEMP_CONFIG})
723	  if defined $ENV{TEMP_CONFIG};
724
725	if ($params{allows_streaming})
726	{
727		if ($params{allows_streaming} eq "logical")
728		{
729			print $conf "wal_level = logical\n";
730		}
731		else
732		{
733			print $conf "wal_level = replica\n";
734		}
735		print $conf "max_wal_senders = 10\n";
736		print $conf "max_replication_slots = 10\n";
737		print $conf "autovacuum_worker_slots = 3\n";
738		print $conf "wal_log_hints = on\n";
739		print $conf "hot_standby = on\n";
740		# conservative settings to ensure we can run multiple postmasters:
741		print $conf "shared_buffers = 1MB\n";
742		print $conf "max_connections = 10\n";
743		# limit disk space consumption, too:
744		print $conf "max_wal_size = 128MB\n";
745	}
746	else
747	{
748		print $conf "wal_level = minimal\n";
749		print $conf "max_wal_senders = 0\n";
750	}
751
752	print $conf "port = $port\n";
753	if ($use_tcp)
754	{
755		print $conf "unix_socket_directories = ''\n";
756		print $conf "listen_addresses = '$host'\n";
757	}
758	else
759	{
760		print $conf "unix_socket_directories = '$host'\n";
761		print $conf "listen_addresses = ''\n";
762	}
763	close $conf;
764
765	chmod($self->group_access ? 0640 : 0600, "$pgdata/postgresql.conf")
766	  or croak("unable to set permissions for $pgdata/postgresql.conf");
767

769	$self->enable_archiving if $params{has_archiving};

906
907=item $node->init_from_backup(root_node, backup_name, %params)
908
909Initialize a node from a backup, which may come from this node or a different
910node. root_node must be a PostgreSQL::Test::Cluster reference, backup_name the string name
911of a backup previously created on that node with $node->backup.
912
913Does not start the node after initializing it.
914

916a tar-format backup, pass the name of the tar program to use in the
917keyword parameter tar_program.
918

920a keyword parameter whose values is a hash. When combine_with_prior is used,

922they are tablespace OIDs.  In either case, the values are the tablespace
923pathnames that should be used for the target cluster.

926as a reference to an array of prior backup names with which this backup
927is to be combined using pg_combinebackup.

930parameter has_streaming => 1. This is disabled by default.
931

933by passing the keyword parameter has_restoring => 1. This is disabled by
934default.

937recovery mode instead, pass the keyword parameter standby => 0.
938

940unconditionally set to enable replication connections.

944sub init_from_backup
945{
946	my ($self, $root_node, $backup_name, %params) = @_;
947	my $backup_path = $root_node->backup_dir . '/' . $backup_name;
948	my $host = $self->host;
949	my $port = $self->port;
950	my $node_name = $self->name;
951	my $root_name = $root_node->name;
952
953	$params{has_streaming} = 0 unless defined $params{has_streaming};
954	$params{has_restoring} = 0 unless defined $params{has_restoring};
955	$params{standby} = 1 unless defined $params{standby};
956
957	print
958	  "# Initializing node \"$node_name\" from backup \"$backup_name\" of node \"$root_name\"\n";
959	croak "Backup \"$backup_name\" does not exist at $backup_path"
960	  unless -d $backup_path;
961
962	mkdir $self->backup_dir;
963	mkdir $self->archive_dir;
964
965	my $data_path = $self->data_dir;
966	if (defined $params{combine_with_prior})
967	{
968		my @prior_backups = @{ $params{combine_with_prior} };
969		my @prior_backup_path;
970
971		for my $prior_backup_name (@prior_backups)
972		{
973			push @prior_backup_path,
974			  $root_node->backup_dir . '/' . $prior_backup_name;
975		}
976
977		local %ENV = $self->_get_env();
978		my @combineargs = ('pg_combinebackup', '--debug');
979		if (exists $params{tablespace_map})
980		{
981			while (my ($olddir, $newdir) = each %{ $params{tablespace_map} })
982			{
983				push @combineargs, "-T$olddir=$newdir";
984			}
985		}
986		# use the combine mode (clone/copy-file-range) if specified
987		if (defined $params{combine_mode})
988		{
989			push @combineargs, $params{combine_mode};
990		}
991		push @combineargs, @prior_backup_path, $backup_path,
992		  '--output' => $data_path;
993		PostgreSQL::Test::Utils::system_or_bail(@combineargs);
994	}
995	elsif (defined $params{tar_program})
996	{
997		mkdir($data_path) || croak "mkdir $data_path: $!";
998		PostgreSQL::Test::Utils::system_or_bail(
999			$params{tar_program},
1000			'xf' => $backup_path . '/base.tar',
1001			'-C' => $data_path);
1002		PostgreSQL::Test::Utils::system_or_bail(
1003			$params{tar_program},
1004			'xf' => $backup_path . '/pg_wal.tar',
1005			'-C' => $data_path . '/pg_wal');
1006
1007		# We need to generate a tablespace_map file.
1008		open(my $tsmap, ">", "$data_path/tablespace_map")
1009		  || croak "$data_path/tablespace_map: $!";
1010
1011		# Extract tarfiles and add tablespace_map entries
1012		my @tstars = grep { /^\d+.tar/ }
1013		  PostgreSQL::Test::Utils::slurp_dir($backup_path);
1014		for my $tstar (@tstars)
1015		{
1016			my $tsoid = $tstar;
1017			$tsoid =~ s/\.tar$//;
1018
1019			croak "no tablespace mapping for $tstar"
1020			  if !exists $params{tablespace_map}
1021			  || !exists $params{tablespace_map}{$tsoid};
1022			my $newdir = $params{tablespace_map}{$tsoid};
1023
1024			mkdir($newdir) || croak "mkdir $newdir: $!";
1025			PostgreSQL::Test::Utils::system_or_bail(
1026				$params{tar_program},
1027				'xf' => $backup_path . '/' . $tstar,
1028				'-C' => $newdir);
1029
1030			my $escaped_newdir = $newdir;
1031			$escaped_newdir =~ s/\\/\\\\/g;
1032			print $tsmap "$tsoid $escaped_newdir\n";
1033		}
1034
1035		# Close tablespace_map.
1036		close($tsmap);
1037	}
1038	else
1039	{
1040		my @tsoids;
1041		rmdir($data_path);
1042
1043		# Copy the main backup. If we see a tablespace directory for which we
1044		# have a tablespace mapping, skip it, but remember that we saw it.
1045		PostgreSQL::Test::RecursiveCopy::copypath(
1046			$backup_path,
1047			$data_path,
1048			filterfn => sub {
1049				my ($path) = @_;
1050				if ($path =~ /^pg_tblspc\/(\d+)$/
1051					&& exists $params{tablespace_map}{$1})
1052				{
1053					push @tsoids, $1;
1054					return 0;
1055				}
1056				return 1;
1057			});
1058
1059		if (@tsoids > 0)
1060		{
1061			# We need to generate a tablespace_map file.
1062			open(my $tsmap, ">", "$data_path/tablespace_map")
1063			  || croak "$data_path/tablespace_map: $!";
1064
1065			# Now use the list of tablespace links to copy each tablespace.
1066			for my $tsoid (@tsoids)
1067			{
1068				croak "no tablespace mapping for $tsoid"
1069				  if !exists $params{tablespace_map}
1070				  || !exists $params{tablespace_map}{$tsoid};
1071
1072				my $olddir = $backup_path . '/pg_tblspc/' . $tsoid;
1073				my $newdir = $params{tablespace_map}{$tsoid};
1074				PostgreSQL::Test::RecursiveCopy::copypath($olddir, $newdir);
1075
1076				my $escaped_newdir = $newdir;
1077				$escaped_newdir =~ s/\\/\\\\/g;
1078				print $tsmap "$tsoid $escaped_newdir\n";
1079			}
1080
1081			# Close tablespace_map.
1082			close($tsmap);
1083		}
1084	}
1085	chmod(0700, $data_path) or croak $!;
1086
1087	# Base configuration for this node
1088	$self->append_conf(
1089		'postgresql.conf',
1090		qq(
1091port = $port
1092));
1093	if ($use_tcp)
1094	{
1095		$self->append_conf('postgresql.conf', "listen_addresses = '$host'");
1096	}
1097	else
1098	{
1099		$self->append_conf('postgresql.conf',
1100			"unix_socket_directories = '$host'");
1101	}
1102	$self->enable_streaming($root_node) if $params{has_streaming};
1103	$self->enable_restoring($root_node, $params{standby})
1104	  if $params{has_restoring};
1105	return;
1106}

1758# Private routine to return a copy of the environment with the PATH and
1759# (DY)LD_LIBRARY_PATH correctly set when there is an install path set for
1760# the node.
1761#
1762# Routines that call Postgres binaries need to call this routine like this:
1763#
1764#    local %ENV = $self->_get_env([%extra_settings]);
1765#
1766# A copy of the environment is taken and node's host and port settings are
1767# added as PGHOST and PGPORT, then the extra settings (if any) are applied.
1768# Any setting in %extra_settings with a value that is undefined is deleted;
1769# the remainder are set. Then the PATH and (DY)LD_LIBRARY_PATH are adjusted
1770# if the node's install path is set, and the copy environment is returned.
1771#
1772# The install path set in new() needs to be a directory containing
1773# bin and lib subdirectories as in a standard PostgreSQL installation, so this
1774# can't be used with installations where the bin and lib directories don't have
1775# a common parent directory.
1776sub _get_env
1777{
1778	my $self = shift;
1779	my %inst_env = (%ENV, PGHOST => $self->{_host}, PGPORT => $self->{_port});
1780	# the remaining arguments are modifications to make to the environment
1781	my %mods = (@_);
1782	while (my ($k, $v) = each %mods)
1783	{
1784		if (defined $v)
1785		{
1786			$inst_env{$k} = "$v";
1787		}
1788		else
1789		{
1790			delete $inst_env{$k};
1791		}
1792	}
1793	# now fix up the new environment for the install path
1794	my $inst = $self->{_install_path};
1795	if ($inst)
1796	{
1797		if ($PostgreSQL::Test::Utils::windows_os)
1798		{
1799			# Windows picks up DLLs from the PATH rather than *LD_LIBRARY_PATH
1800			# choose the right path separator
1801			if ($Config{osname} eq 'MSWin32')
1802			{
1803				$inst_env{PATH} = "$inst/bin;$inst/lib;$ENV{PATH}";
1804			}
1805			else
1806			{
1807				$inst_env{PATH} = "$inst/bin:$inst/lib:$ENV{PATH}";
1808			}
1809		}
1810		else
1811		{
1812			my $dylib_name =
1813			  $Config{osname} eq 'darwin'
1814			  ? "DYLD_LIBRARY_PATH"
1815			  : "LD_LIBRARY_PATH";
1816			$inst_env{PATH} = "$inst/bin:$ENV{PATH}";
1817			if (exists $ENV{$dylib_name})
1818			{
1819				$inst_env{$dylib_name} = "$inst/lib:$ENV{$dylib_name}";
1820			}
1821			else
1822			{
1823				$inst_env{$dylib_name} = "$inst/lib";
1824			}
1825		}
1826	}
1827	return (%inst_env);
1828}

1860
1861=item get_free_port()
1862
1863Locate an unprivileged (high) TCP port that's not currently bound to
1864anything.  This is used by C<new()>, and also by some test cases that need to
1865start other, non-Postgres servers.
1866
1867Ports assigned to existing PostgreSQL::Test::Cluster objects are automatically
1868excluded, even if those servers are not currently running.
1869
1870The port number is reserved so that other concurrent test programs will not
1871try to use the same port.
1872
1873Note: this is not an instance method. As it's not exported it should be
1874called from outside the module as C<PostgreSQL::Test::Cluster::get_free_port()>.
1875
1876=cut
1877
1878sub get_free_port
1879{
1880	my $found = 0;
1881	my $port = $last_port_assigned;
1882
1883	while ($found == 0)
1884	{
1885
1886		# advance $port, wrapping correctly around range end
1887		$port = $port_lower_bound if ++$port > $port_upper_bound;
1888		print "# Checking port $port\n";
1889
1890		# Check first that candidate port number is not included in
1891		# the list of already-registered nodes.
1892		$found = 1;
1893		foreach my $node (@all_nodes)
1894		{
1895			$found = 0 if ($node->port == $port);
1896		}
1897
1898		# Check to see if anything else is listening on this TCP port.
1899		# Seek a port available for all possible listen_addresses values,
1900		# so callers can harness this port for the widest range of purposes.
1901		# The 0.0.0.0 test achieves that for MSYS, which automatically sets
1902		# SO_EXCLUSIVEADDRUSE.  Testing 0.0.0.0 is insufficient for Windows
1903		# native Perl (https://stackoverflow.com/a/14388707), so we also
1904		# have to test individual addresses.  Doing that for 127.0.0/24
1905		# addresses other than 127.0.0.1 might fail with EADDRNOTAVAIL on
1906		# non-Linux, non-Windows kernels.
1907		#
1908		# Thus, 0.0.0.0 and individual 127.0.0/24 addresses are tested
1909		# only on Windows and only when TCP usage is requested.
1910		if ($found == 1)
1911		{
1912			foreach my $addr (qw(127.0.0.1),
1913				($use_tcp && $PostgreSQL::Test::Utils::windows_os)
1914				  ? qw(127.0.0.2 127.0.0.3 0.0.0.0)
1915				  : ())
1916			{
1917				if (!can_bind($addr, $port))
1918				{
1919					$found = 0;
1920					last;
1921				}
1922			}
1923			$found = _reserve_port($port) if $found;
1924		}
1925	}
1926
1927	print "# Found port $port\n";
1928
1929	# Update port for next time
1930	$last_port_assigned = $port;
1931
1932	return $port;
1933}

1935# Internal routine to check whether a host:port is available to bind
1936sub can_bind
1937{
1938	my ($host, $port) = @_;
1939	my $iaddr = inet_aton($host);
1940	my $paddr = sockaddr_in($port, $iaddr);
1941
1942	socket(SOCK, PF_INET, SOCK_STREAM, 0)
1943	  or croak "socket failed: $!";
1944
1945	# As in postmaster, don't use SO_REUSEADDR on Windows
1946	setsockopt(SOCK, SOL_SOCKET, SO_REUSEADDR, pack("l", 1))
1947	  unless $PostgreSQL::Test::Utils::windows_os;
1948	my $ret = bind(SOCK, $paddr) && listen(SOCK, SOMAXCONN);
1949	close(SOCK);

1953# Internal routine to reserve a port number
1954# Returns 1 if successful, 0 if port is already reserved.
1955sub _reserve_port
1956{
1957	my $port = shift;
1958	# open in rw mode so we don't have to reopen it and lose the lock
1959	my $filename = "$portdir/$port.rsv";
1960	sysopen(my $portfile, $filename, O_RDWR | O_CREAT)
1961	  || croak "opening port file $filename: $!";
1962	# take an exclusive lock to avoid concurrent access
1963	flock($portfile, LOCK_EX) || croak "locking port file $filename: $!";
1964	# see if someone else has or had a reservation of this port
1965	my $pid = <$portfile> || "0";
1966	chomp $pid;
1967	if ($pid + 0 > 0)
1968	{
1969		if (kill 0, $pid)
1970		{
1971			# process exists and is owned by us, so we can't reserve this port
1972			flock($portfile, LOCK_UN) || croak $!;
1973			close($portfile);
1974			return 0;
1975		}
1976	}
1977	# All good, go ahead and reserve the port
1978	seek($portfile, 0, SEEK_SET) || croak $!;
1979	# print the pid with a fixed width so we don't leave any trailing junk
1980	print $portfile sprintf("%10d\n", $$);
1981	flock($portfile, LOCK_UN) || croak $!;
1982	close($portfile);
1983	push(@port_reservation_files, $filename);
1984	return 1;
1985}
