9-- check the number of estimated/actual rows in the top node
10create function check_estimated_rows(text) returns table (estimated int, actual int)
11language plpgsql as
12$$
13declare
14    ln text;
15    tmp text[];
16    first_row bool := true;
17begin
18    for ln in
19        execute format('explain analyze %s', $1)
20    loop
21        if first_row then
22            first_row := false;
23            tmp := regexp_match(ln, 'rows=(\d*) .* rows=(\d*)');
24            return query select tmp[1]::int, tmp[2]::int;
25        end if;
26    end loop;
27end;
28$$;
