1--
2--Test RLS with GRAPH_TABLE
3--
4--This test verifies that Row Level Security (RLS) policies are correctly
5--enforced when querying tables underlying property graphs using GRAPH_TABLE.
6--graph_table.sql has extensive tests covering interaction of GRAPH_TABLE with
7--other query constructs. rowsecurity.sql has extensive coverage of interaction
8--of RLS and other features of PostgreSQL. This test along with those two tests
9--is sufficient to make sure that all combinations of RLS and GRAPH_TABLE will
10--work as expected.

93-- Basic RLS tests
94--
95ALTER TABLE document_people ENABLE ROW LEVEL SECURITY;
96-- user's security level must be higher than or equal to document's
97CREATE POLICY p1 ON document_people AS PERMISSIVE
98    USING (dlevel <= (SELECT seclv FROM users WHERE pguser = current_user));
99-- but Dave isn't allowed to see document titled 'Scientists'
100CREATE POLICY p2 ON document_people AS RESTRICTIVE TO regress_graph_rls_dave
101    USING (dtitle <> 'Scientists');
102CREATE POLICY p3 ON document_people AS RESTRICTIVE TO regress_graph_rls_group1
103    USING (dlevel < 3);
104CREATE POLICY p4 ON document_people AS PERMISSIVE TO regress_graph_rls_group2
105    USING (dlevel < 3);

109-- Use the same query in all the test cases below. Prepare it once and
110-- use multiple times. Apart from making the test file shorter and avoiding
111-- duplication, it also tests that a prepared statement correctly reflect changes
112-- to RLS policies, session user or RLS settings.
113PREPARE graph_rls_query AS
114SELECT * FROM GRAPH_TABLE (cabinet
115                           MATCH (u IS users)-[a IS accessed]->(d IS document)
116                           WHERE f_leak(d.dtitle)
117                           COLUMNS (u.pguser, a.aid, d.dtitle, d.dlevel))
118         ORDER BY 1, 2, 3, 4;

121SET SESSION AUTHORIZATION regress_graph_rls_bob;
122EXECUTE graph_rls_query;

126EXECUTE graph_rls_query;

130EXECUTE graph_rls_query;
131
132-- RLS policy does not apply to table owner when RLS enabled
133SET SESSION AUTHORIZATION regress_graph_rls_alice;
134SET row_security TO ON;
135EXECUTE graph_rls_query;
136
137SET row_security TO OFF;
138-- database superuser does bypass RLS policy when disabled
139RESET SESSION AUTHORIZATION;
140EXECUTE graph_rls_query;
141
142-- database non-superuser with bypass privilege can bypass RLS policy when disabled

152EXECUTE graph_rls_query; -- error

312-- Create a policy on document that references document itself via GRAPH_TABLE
313CREATE POLICY pr ON document TO regress_graph_rls_dave
314    USING (EXISTS (SELECT 1 FROM GRAPH_TABLE (cabinet
315                           MATCH (u IS users)-[a IS accessed]->(d IS document)
316                           WHERE u.pguser = current_user
317                           COLUMNS (a.aid))));

319EXECUTE graph_rls_query; -- error
320SET SESSION AUTHORIZATION regress_graph_rls_alice;
