185-- For partitioned temp tables, ON COMMIT actions ignore storage-less
186-- partitioned tables.
187begin;
188create temp table temp_parted_oncommit (a int)
189  partition by list (a) on commit delete rows;
190create temp table temp_parted_oncommit_1
191  partition of temp_parted_oncommit
192  for values in (1) on commit delete rows;
193insert into temp_parted_oncommit values (1);
194commit;
195-- partitions are emptied by the previous commit
196select * from temp_parted_oncommit;
197drop table temp_parted_oncommit;

199-- Check dependencies between ON COMMIT actions with a partitioned
200-- table and its partitions.  Using ON COMMIT DROP on a parent removes
201-- the whole set.
202begin;
203create temp table temp_parted_oncommit_test (a int)
204  partition by list (a) on commit drop;
205create temp table temp_parted_oncommit_test1
206  partition of temp_parted_oncommit_test
207  for values in (1) on commit delete rows;
208create temp table temp_parted_oncommit_test2
209  partition of temp_parted_oncommit_test
210  for values in (2) on commit drop;
211insert into temp_parted_oncommit_test values (1), (2);
212commit;
213-- no relations remain in this case.
214select relname from pg_class where relname ~ '^temp_parted_oncommit_test';
215-- Using ON COMMIT DELETE on a partitioned table does not remove
216-- all rows if partitions preserve their data.
217begin;
218create temp table temp_parted_oncommit_test (a int)
219  partition by list (a) on commit delete rows;
220create temp table temp_parted_oncommit_test1
221  partition of temp_parted_oncommit_test
222  for values in (1) on commit preserve rows;
223create temp table temp_parted_oncommit_test2
224  partition of temp_parted_oncommit_test
225  for values in (2) on commit drop;
226insert into temp_parted_oncommit_test values (1), (2);
227commit;
228-- Data from the remaining partition is still here as its rows are
229-- preserved.
230select * from temp_parted_oncommit_test;
231-- two relations remain in this case.
232select relname from pg_class where relname ~ '^temp_parted_oncommit_test'
233  order by relname;
234drop table temp_parted_oncommit_test;

236-- Check dependencies between ON COMMIT actions with inheritance trees.
237-- Using ON COMMIT DROP on a parent removes the whole set.
238begin;
239create temp table temp_inh_oncommit_test (a int) on commit drop;
240create temp table temp_inh_oncommit_test1 ()
241  inherits(temp_inh_oncommit_test) on commit delete rows;

243commit;
244-- no relations remain in this case
245select relname from pg_class where relname ~ '^temp_inh_oncommit_test';
246-- Data on the parent is removed, and the child goes away.
247begin;
248create temp table temp_inh_oncommit_test (a int) on commit delete rows;
249create temp table temp_inh_oncommit_test1 ()
250  inherits(temp_inh_oncommit_test) on commit drop;

331/* helper function to create cursors for each page in [p_start, p_end] */
332CREATE FUNCTION test_temp_pin(p_start int, p_end int)
333RETURNS void
334LANGUAGE plpgsql
335AS $f$
336  DECLARE
337      cursorname text;
338      query text;
339  BEGIN
340    FOR i IN p_start..p_end LOOP
341       cursorname = 'c_'||i;
342       query = format($q$DECLARE %I CURSOR FOR SELECT ctid FROM test_temp WHERE ctid >= '( %s, 1)'::tid $q$, cursorname, i);
343       EXECUTE query;
344       EXECUTE 'FETCH NEXT FROM '||cursorname;
345       -- for test development
346       -- RAISE NOTICE '%: %', cursorname, query;
347    END LOOP;
348  END;
349$f$;
