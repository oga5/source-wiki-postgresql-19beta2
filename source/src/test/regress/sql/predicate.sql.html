312--
313-- Test optimization of IS [NOT] DISTINCT FROM
314--
315
316CREATE TYPE dist_row_t AS (a int, b int);
317CREATE TABLE dist_tab (id int, val_nn int NOT NULL, val_null int, row_nn dist_row_t NOT NULL);
318
319INSERT INTO dist_tab VALUES (1, 10, 10, ROW(1, 1));
320INSERT INTO dist_tab VALUES (2, 20, NULL, ROW(2, 2));
321INSERT INTO dist_tab VALUES (3, 30, 30, ROW(1, NULL));
322
323CREATE INDEX dist_tab_nn_idx ON dist_tab (val_nn);
324
325ANALYZE dist_tab;
326
327-- Ensure that the predicate folds to constant TRUE
328EXPLAIN(COSTS OFF)
329SELECT id FROM dist_tab WHERE val_nn IS DISTINCT FROM NULL::INT;
330SELECT id FROM dist_tab WHERE val_nn IS DISTINCT FROM NULL::INT;
331
332-- Ensure that the predicate folds to constant FALSE
333EXPLAIN(COSTS OFF)
334SELECT id FROM dist_tab WHERE val_nn IS NOT DISTINCT FROM NULL::INT;
335SELECT id FROM dist_tab WHERE val_nn IS NOT DISTINCT FROM NULL::INT;
336
337-- Ensure that the predicate is converted to an inequality operator
338EXPLAIN (COSTS OFF)
339SELECT id FROM dist_tab WHERE val_nn IS DISTINCT FROM 10;
340SELECT id FROM dist_tab WHERE val_nn IS DISTINCT FROM 10;
341
342-- Ensure that the predicate is converted to an equality operator, and thus can
343-- use index scan
344SET enable_seqscan TO off;
345EXPLAIN (COSTS OFF)
346SELECT id FROM dist_tab WHERE val_nn IS NOT DISTINCT FROM 10;
347SELECT id FROM dist_tab WHERE val_nn IS NOT DISTINCT FROM 10;
348RESET enable_seqscan;
349
350-- Ensure that the predicate is preserved as "IS DISTINCT FROM"
351EXPLAIN (COSTS OFF)
352SELECT id FROM dist_tab WHERE val_null IS DISTINCT FROM 20;
353SELECT id FROM dist_tab WHERE val_null IS DISTINCT FROM 20;
354
355-- Safety check for rowtypes
356-- Ensure that the predicate is converted to an inequality operator
357EXPLAIN (COSTS OFF)
358SELECT id FROM dist_tab WHERE row_nn IS DISTINCT FROM ROW(1, 5)::dist_row_t;
359-- ... and that all 3 rows are returned
360SELECT id FROM dist_tab WHERE row_nn IS DISTINCT FROM ROW(1, 5)::dist_row_t;
361
362-- Ensure that the predicate is converted to an equality operator, and thus
363-- mergejoinable or hashjoinable
364SET enable_nestloop TO off;
365EXPLAIN (COSTS OFF)
366SELECT * FROM dist_tab t1 JOIN dist_tab t2 ON t1.val_nn IS NOT DISTINCT FROM t2.val_nn;
367SELECT * FROM dist_tab t1 JOIN dist_tab t2 ON t1.val_nn IS NOT DISTINCT FROM t2.val_nn;
368RESET enable_nestloop;
369
370-- Ensure that the predicate is converted to IS NOT NULL
371EXPLAIN (COSTS OFF)
372SELECT id FROM dist_tab WHERE val_null IS DISTINCT FROM NULL::INT;
373SELECT id FROM dist_tab WHERE val_null IS DISTINCT FROM NULL::INT;
374
375-- Ensure that the predicate is converted to IS NULL
376EXPLAIN (COSTS OFF)
377SELECT id FROM dist_tab WHERE val_null IS NOT DISTINCT FROM NULL::INT;
378SELECT id FROM dist_tab WHERE val_null IS NOT DISTINCT FROM NULL::INT;
379
380-- Safety check for rowtypes
381-- The predicate is converted to IS NOT NULL, and get_rule_expr prints it as IS
382-- DISTINCT FROM because argisrow is false, indicating that we're applying a
383-- scalar test
384EXPLAIN (COSTS OFF)
385SELECT id FROM dist_tab WHERE (val_null, val_null) IS DISTINCT FROM NULL::RECORD;
386SELECT id FROM dist_tab WHERE (val_null, val_null) IS DISTINCT FROM NULL::RECORD;
387
388-- The predicate is converted to IS NULL, and get_rule_expr prints it as IS NOT
389-- DISTINCT FROM because argisrow is false, indicating that we're applying a
390-- scalar test
391EXPLAIN (COSTS OFF)
392SELECT id FROM dist_tab WHERE (val_null, val_null) IS NOT DISTINCT FROM NULL::RECORD;
393SELECT id FROM dist_tab WHERE (val_null, val_null) IS NOT DISTINCT FROM NULL::RECORD;

395DROP TABLE dist_tab;
396DROP TYPE dist_row_t;
