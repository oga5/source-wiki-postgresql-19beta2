84--
85-- Memory contexts are logged and they are not returned to the function.
86-- Furthermore, their contents can vary depending on the timing. However,
87-- we can at least verify that the code doesn't fail, and that the
88-- permissions are set properly.

90
91SELECT pg_log_backend_memory_contexts(pg_backend_pid());
92
93SELECT pg_log_backend_memory_contexts(pid) FROM pg_stat_activity
94  WHERE backend_type = 'checkpointer';
95
96CREATE ROLE regress_log_memory;
97
98SELECT has_function_privilege('regress_log_memory',
99  'pg_log_backend_memory_contexts(integer)', 'EXECUTE'); -- no
100
101GRANT EXECUTE ON FUNCTION pg_log_backend_memory_contexts(integer)
102  TO regress_log_memory;
103
104SELECT has_function_privilege('regress_log_memory',
105  'pg_log_backend_memory_contexts(integer)', 'EXECUTE'); -- yes
106

108SELECT pg_log_backend_memory_contexts(pg_backend_pid());
109RESET ROLE;
110

112  FROM regress_log_memory;

203-- Test adding a support function to a subject function
204--
205
206CREATE FUNCTION my_int_eq(int, int) RETURNS bool
207  LANGUAGE internal STRICT IMMUTABLE PARALLEL SAFE
208  AS $$int4eq$$;
209
210-- By default, planner does not think that's selective
211EXPLAIN (COSTS OFF)
212SELECT * FROM tenk1 a JOIN tenk1 b ON a.unique1 = b.unique1
213WHERE my_int_eq(a.unique2, 42);
214
215-- With support function that knows it's int4eq, we get a different plan
216CREATE FUNCTION test_support_func(internal)
217    RETURNS internal
218    AS :'regresslib', 'test_support_func'
219    LANGUAGE C STRICT;
220
221ALTER FUNCTION my_int_eq(int, int) SUPPORT test_support_func;
222
223EXPLAIN (COSTS OFF)
224SELECT * FROM tenk1 a JOIN tenk1 b ON a.unique1 = b.unique1
225WHERE my_int_eq(a.unique2, 42);

248CREATE FUNCTION foo_from_bar(colname TEXT, tablename TEXT, filter TEXT)
249RETURNS SETOF TEXT
250LANGUAGE plpgsql
251AS $function$
252DECLARE
253  sql TEXT;
254BEGIN
255  sql := format('SELECT %I::text FROM %I', colname, tablename);
256  IF filter IS NOT NULL THEN
257    sql := CONCAT(sql, format(' WHERE %I::text = $1', colname));
258  END IF;
259  RETURN QUERY EXECUTE sql USING filter;
260END;
261$function$ STABLE;
262
263ALTER FUNCTION foo_from_bar(TEXT, TEXT, TEXT)
264  SUPPORT test_inline_in_from_support_func;
265
266SELECT * FROM foo_from_bar('f1', 'text_tbl', NULL);
267SELECT * FROM foo_from_bar('f1', 'text_tbl', 'doh!');
268EXPLAIN (COSTS OFF) SELECT * FROM foo_from_bar('f1', 'text_tbl', NULL);
269EXPLAIN (COSTS OFF) SELECT * FROM foo_from_bar('f1', 'text_tbl', 'doh!');
270
271DROP FUNCTION foo_from_bar;
