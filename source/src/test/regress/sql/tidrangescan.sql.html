1-- tests for tidrangescans
2
3SET enable_seqscan TO off;
4CREATE TABLE tidrangescan(id integer, data text);
5
6-- empty table
7EXPLAIN (COSTS OFF)
8SELECT ctid FROM tidrangescan WHERE ctid < '(1, 0)';
9SELECT ctid FROM tidrangescan WHERE ctid < '(1, 0)';
10
11EXPLAIN (COSTS OFF)
12SELECT ctid FROM tidrangescan WHERE ctid > '(9, 0)';
13SELECT ctid FROM tidrangescan WHERE ctid > '(9, 0)';
14
15-- insert enough tuples to fill at least two pages
16INSERT INTO tidrangescan SELECT i,repeat('x', 100) FROM generate_series(1,200) AS s(i);
17
18-- remove all tuples after the 10th tuple on each page.  Trying to ensure
19-- we get the same layout with all CPU architectures and smaller than standard
20-- page sizes.
21DELETE FROM tidrangescan
22WHERE substring(ctid::text FROM ',(\d+)\)')::integer > 10 OR substring(ctid::text FROM '\((\d+),')::integer > 2;
23VACUUM tidrangescan;
24

64-- extreme offsets
