5--
6-- box logic
7--	     o
8-- 3	  o--|X
9--	  |  o|
10-- 2	+-+-+ |
11--	| | | |
12-- 1	| o-+-o
13--	|   |
14-- 0	+---+
15--
16--	0 1 2 3
17--
18
19-- boxes are specified by two points, given by four floats x1,y1,x2,y2
20

54-- overlap
55SELECT b.f1
56   FROM BOX_TBL b
57   WHERE b.f1 && box '(2.5,2.5,1.0,1.0)';
58
59-- left-or-overlap (x only)
60SELECT b1.*
61   FROM BOX_TBL b1
62   WHERE b1.f1 &< box '(2.0,2.0,2.5,2.5)';
63
64-- right-or-overlap (x only)
65SELECT b1.*
66   FROM BOX_TBL b1
67   WHERE b1.f1 &> box '(2.0,2.0,2.5,2.5)';
68
69-- left of
70SELECT b.f1
71   FROM BOX_TBL b
72   WHERE b.f1 << box '(3.0,3.0,5.0,5.0)';
73
74-- area <=
75SELECT b.f1
76   FROM BOX_TBL b
77   WHERE b.f1 <= box '(3.0,3.0,5.0,5.0)';
78
79-- area <
80SELECT b.f1
81   FROM BOX_TBL b
82   WHERE b.f1 < box '(3.0,3.0,5.0,5.0)';
83
84-- area =
85SELECT b.f1
86   FROM BOX_TBL b
87   WHERE b.f1 = box '(3.0,3.0,5.0,5.0)';
88
89-- area >
90SELECT b.f1
91   FROM BOX_TBL b				-- zero area
92   WHERE b.f1 > box '(3.5,3.0,4.5,3.0)';
93
94-- area >=
95SELECT b.f1
96   FROM BOX_TBL b				-- zero area
97   WHERE b.f1 >= box '(3.5,3.0,4.5,3.0)';
98
99-- right of
100SELECT b.f1
101   FROM BOX_TBL b
102   WHERE box '(3.0,3.0,5.0,5.0)' >> b.f1;
103
104-- contained in
105SELECT b.f1
106   FROM BOX_TBL b
107   WHERE b.f1 <@ box '(0,0,3,3)';
108
109-- contains
110SELECT b.f1
111   FROM BOX_TBL b
112   WHERE box '(0,0,3,3)' @> b.f1;
113
114-- box equality
115SELECT b.f1
116   FROM BOX_TBL b
117   WHERE box '(1,1,3,3)' ~= b.f1;
118
119-- center of box, left unary operator
120SELECT @@(b1.f1) AS p
121   FROM BOX_TBL b1;
122
123-- wholly-contained
124SELECT b1.*, b2.*
125   FROM BOX_TBL b1, BOX_TBL b2
126   WHERE b1.f1 @> b2.f1 and not b1.f1 ~= b2.f1;
