368-- Test GUC categories and flag patterns
369SELECT pg_settings_get_flags(NULL);
370SELECT pg_settings_get_flags('does_not_exist');
371CREATE TABLE tab_settings_flags AS SELECT name, category,
372    'EXPLAIN'          = ANY(flags) AS explain,
373    'NO_RESET'         = ANY(flags) AS no_reset,
374    'NO_RESET_ALL'     = ANY(flags) AS no_reset_all,
375    'NOT_IN_SAMPLE'    = ANY(flags) AS not_in_sample,
376    'RUNTIME_COMPUTED' = ANY(flags) AS runtime_computed
377  FROM pg_show_all_settings() AS psas,
378    pg_settings_get_flags(psas.name) AS flags;
379
380-- Developer GUCs should be flagged with GUC_NOT_IN_SAMPLE:
381SELECT name FROM tab_settings_flags
382  WHERE category = 'Developer Options' AND NOT not_in_sample
383  ORDER BY 1;
384-- Most query-tuning GUCs are flagged as valid for EXPLAIN.
385-- default_statistics_target is an exception.
386SELECT name FROM tab_settings_flags
387  WHERE category ~ '^Query Tuning' AND NOT explain
388  ORDER BY 1;
389-- Runtime-computed GUCs should be part of the preset category.
390SELECT name FROM tab_settings_flags
391  WHERE NOT category = 'Preset Options' AND runtime_computed
392  ORDER BY 1;
393-- Preset GUCs are flagged as NOT_IN_SAMPLE.
394SELECT name FROM tab_settings_flags
395  WHERE category = 'Preset Options' AND NOT not_in_sample
396  ORDER BY 1;
397-- NO_RESET implies NO_RESET_ALL.
398SELECT name FROM tab_settings_flags
399  WHERE no_reset AND NOT no_reset_all
400  ORDER BY 1;
401DROP TABLE tab_settings_flags;
