src/test/regress/sql/compression_lz4.sql
1-- Tests for TOAST compression with lz4
2
3SELECT NOT(enumvals @> '{lz4}') AS skip_test FROM pg_settings WHERE
4  name = 'default_toast_compression' \gset
5\if :skip_test
6   \echo '*** skipping TOAST tests with lz4 (not supported) ***'
7   \quit
8\endif
9
10CREATE SCHEMA lz4;
11SET search_path TO lz4, public;
12
13\set HIDE_TOAST_COMPRESSION false
14
15-- Ensure we get stable results regardless of the installation's default.
16-- We rely on this GUC value for a few tests.
17SET default_toast_compression = 'pglz';
18
19-- test creating table with compression method
20CREATE TABLE cmdata_pglz(f1 text COMPRESSION pglz);
21CREATE INDEX idx ON cmdata_pglz(f1);
22INSERT INTO cmdata_pglz VALUES(repeat('1234567890', 1000));
23\d+ cmdata
24CREATE TABLE cmdata_lz4(f1 TEXT COMPRESSION lz4);
25INSERT INTO cmdata_lz4 VALUES(repeat('1234567890', 1004));
26\d+ cmdata1
27
28-- verify stored compression method in the data
29SELECT pg_column_compression(f1) FROM cmdata_lz4;
30
31-- decompress data slice
32SELECT SUBSTR(f1, 200, 5) FROM cmdata_pglz;
33SELECT SUBSTR(f1, 2000, 50) FROM cmdata_lz4;
34
35-- copy with table creation
36SELECT * INTO cmmove1 FROM cmdata_lz4;
37\d+ cmmove1
38SELECT pg_column_compression(f1) FROM cmmove1;
39
40-- test LIKE INCLUDING COMPRESSION.  The GUC default_toast_compression
41-- has no effect, the compression method from the table being copied.
42CREATE TABLE cmdata2 (LIKE cmdata_lz4 INCLUDING COMPRESSION);
43\d+ cmdata2
44DROP TABLE cmdata2;
45
46-- copy to existing table
47CREATE TABLE cmmove3(f1 text COMPRESSION pglz);
48INSERT INTO cmmove3 SELECT * FROM cmdata_pglz;
49INSERT INTO cmmove3 SELECT * FROM cmdata_lz4;
50SELECT pg_column_compression(f1) FROM cmmove3;
51
52-- update using datum from different table with LZ4 data.
53CREATE TABLE cmmove2(f1 text COMPRESSION pglz);
54INSERT INTO cmmove2 VALUES (repeat('1234567890', 1004));
55SELECT pg_column_compression(f1) FROM cmmove2;
56UPDATE cmmove2 SET f1 = cmdata_lz4.f1 FROM cmdata_lz4;
57SELECT pg_column_compression(f1) FROM cmmove2;
58
59-- test externally stored compressed data
60CREATE OR REPLACE FUNCTION large_val_lz4() RETURNS TEXT LANGUAGE SQL AS
61'select array_agg(fipshash(g::text))::text from generate_series(1, 256) g';
62CREATE TABLE cmdata2 (f1 text COMPRESSION lz4);
63INSERT INTO cmdata2 SELECT large_val_lz4() || repeat('a', 4000);
64SELECT pg_column_compression(f1) FROM cmdata2;
65SELECT SUBSTR(f1, 200, 5) FROM cmdata2;
66DROP TABLE cmdata2;
67DROP FUNCTION large_val_lz4;
68
69-- test compression with materialized view
70CREATE MATERIALIZED VIEW compressmv(x) AS SELECT * FROM cmdata_lz4;
71\d+ compressmv
72SELECT pg_column_compression(f1) FROM cmdata_lz4;
73SELECT pg_column_compression(x) FROM compressmv;
74
75-- test compression with partition
76CREATE TABLE cmpart(f1 text COMPRESSION lz4) PARTITION BY HASH(f1);
77CREATE TABLE cmpart1 PARTITION OF cmpart FOR VALUES WITH (MODULUS 2, REMAINDER 0);
78CREATE TABLE cmpart2(f1 text COMPRESSION pglz);
79
80ALTER TABLE cmpart ATTACH PARTITION cmpart2 FOR VALUES WITH (MODULUS 2, REMAINDER 1);
81INSERT INTO cmpart VALUES (repeat('123456789', 1004));
82INSERT INTO cmpart VALUES (repeat('123456789', 4004));
83SELECT pg_column_compression(f1) FROM cmpart1;
84SELECT pg_column_compression(f1) FROM cmpart2;
85
86-- test compression with inheritance
87CREATE TABLE cminh() INHERITS(cmdata_pglz, cmdata_lz4); -- error
88CREATE TABLE cminh(f1 TEXT COMPRESSION lz4) INHERITS(cmdata_pglz); -- error
89CREATE TABLE cmdata3(f1 text);
90CREATE TABLE cminh() INHERITS (cmdata_pglz, cmdata3);
91
92-- test default_toast_compression GUC
93SET default_toast_compression = 'lz4';
94
95-- test alter compression method
96ALTER TABLE cmdata_pglz ALTER COLUMN f1 SET COMPRESSION lz4;
97INSERT INTO cmdata_pglz VALUES (repeat('123456789', 4004));
98\d+ cmdata
99SELECT pg_column_compression(f1) FROM cmdata_pglz;
100ALTER TABLE cmdata_pglz ALTER COLUMN f1 SET COMPRESSION pglz;
101
102-- test alter compression method for materialized views
103ALTER MATERIALIZED VIEW compressmv ALTER COLUMN x SET COMPRESSION lz4;
104\d+ compressmv
105
106-- test alter compression method for partitioned tables
107ALTER TABLE cmpart1 ALTER COLUMN f1 SET COMPRESSION pglz;
108ALTER TABLE cmpart2 ALTER COLUMN f1 SET COMPRESSION lz4;
109
110-- new data should be compressed with the current compression method
111INSERT INTO cmpart VALUES (repeat('123456789', 1004));
112INSERT INTO cmpart VALUES (repeat('123456789', 4004));
113SELECT pg_column_compression(f1) FROM cmpart1;
114SELECT pg_column_compression(f1) FROM cmpart2;
115
116-- test expression index
117CREATE TABLE cmdata2 (f1 TEXT COMPRESSION pglz, f2 TEXT COMPRESSION lz4);
118CREATE UNIQUE INDEX idx1 ON cmdata2 ((f1 || f2));
119INSERT INTO cmdata2 VALUES((SELECT array_agg(fipshash(g::TEXT))::TEXT FROM
120generate_series(1, 50) g), VERSION());
121
122-- check data is ok
123SELECT length(f1) FROM cmdata_pglz;
124SELECT length(f1) FROM cmdata_lz4;
125SELECT length(f1) FROM cmmove1;
126SELECT length(f1) FROM cmmove2;
127SELECT length(f1) FROM cmmove3;

129\set HIDE_TOAST_COMPRESSION true
