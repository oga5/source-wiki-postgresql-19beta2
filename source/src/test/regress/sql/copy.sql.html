332-- Add a trigger to catch and print the contents of the catalog view
333-- pg_stat_progress_copy during data insertion.  This allows to test
334-- the validation of some progress reports for COPY FROM where the trigger
335-- would fire.
336create function notice_after_tab_progress_reporting() returns trigger AS
337$$
338declare report record;
339begin
340  -- The fields ignored here are the ones that may not remain
341  -- consistent across multiple runs.  The sizes reported may differ
342  -- across platforms, so just check if these are strictly positive.
343  with progress_data as (
344    select
345       relid::regclass::text as relname,
346       command,
347       type,
348       bytes_processed > 0 as has_bytes_processed,
349       bytes_total > 0 as has_bytes_total,
350       tuples_processed,
351       tuples_excluded,
352       tuples_skipped
353      from pg_stat_progress_copy
354      where pid = pg_backend_pid())
355  select into report (to_jsonb(r)) as value
356    from progress_data r;
357
358  raise info 'progress: %', report.value::text;
359  return new;
360end;
361$$ language plpgsql;
362
363create trigger check_after_tab_progress_reporting
364	after insert on tab_progress_reporting
365	for each statement
366	execute function notice_after_tab_progress_reporting();

479--
480-- Create partitioned table that does not allow bulk insertions, to test bugs
481-- related to the reuse of BulkInsertState across partitions (only done when
482-- not using bulk insert).  Switching between partitions often makes it more
483-- likely to encounter these bugs, so we just switch on roughly every insert
484-- by having an even/odd number partition and inserting evenly distributed
485-- data.
486--
487CREATE TABLE parted_si (
488  id int not null,
489  data text not null,
490  -- prevent use of bulk insert by having a volatile function
491  rand float8 not null default random()
492)
493PARTITION BY LIST((id % 2));
494
495CREATE TABLE parted_si_p_even PARTITION OF parted_si FOR VALUES IN (0);
496CREATE TABLE parted_si_p_odd PARTITION OF parted_si FOR VALUES IN (1);
497
498-- Test that bulk relation extension handles reusing a single BulkInsertState
499-- across partitions.  Without the fix applied, this reliably reproduces
500-- #18130 unless shared_buffers is extremely small (preventing any use of bulk
501-- relation extension). See
502-- https://postgr.es/m/18130-7a86a7356a75209d%40postgresql.org
503-- https://postgr.es/m/257696.1695670946%40sss.pgh.pa.us
504\set filename :abs_srcdir '/data/desc.data'
505COPY parted_si(id, data) FROM :'filename';
506
507-- An earlier bug (see commit b1ecb9b3fcf) could end up using a buffer from
508-- the wrong partition. This test is *not* guaranteed to trigger that bug, but
509-- does so when shared_buffers is small enough.  To test if we encountered the
510-- bug, check that the partition condition isn't violated.
511SELECT tableoid::regclass, id % 2 = 0 is_even, count(*) from parted_si GROUP BY 1, 2 ORDER BY 1;
512
513DROP TABLE parted_si;
