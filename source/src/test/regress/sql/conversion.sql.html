54--
55-- Test built-in conversion functions.
56--
57
58-- Helper function to test a conversion. Uses the test_enc_conversion function
59-- that was created in the create_function_0 test.
60create or replace function test_conv(
61  input IN bytea,
62  src_encoding IN text,
63  dst_encoding IN text,
64
65  result OUT bytea,
66  errorat OUT bytea,
67  error OUT text)
68language plpgsql as
69$$
70declare
71  validlen int;
72begin
73  -- First try to perform the conversion with noError = false. If that errors out,
74  -- capture the error message, and try again with noError = true. The second call
75  -- should succeed and return the position of the error, return that too.
76  begin
77    select * into validlen, result from test_enc_conversion(input, src_encoding, dst_encoding, false);
78    errorat = NULL;
79    error := NULL;
80  exception when others then
81    error := sqlerrm;
82    select * into validlen, result from test_enc_conversion(input, src_encoding, dst_encoding, true);
83    errorat = substr(input, validlen + 1);
84  end;
85  return;
86end;
87$$;

123-- Test UTF-8 verification with ASCII padding appended to provide
124-- coverage for algorithms that work on multiple bytes at a time.
125-- The error message for a sequence starting with a 4-byte lead
126-- will contain all 4 bytes if they are present, so various
127-- expressions below add 3 ASCII bytes to the end to ensure
128-- consistent error messages.
129-- The number 64 below needs to be at least the value of STRIDE_LENGTH in wchar.c.
130
131-- Test multibyte verification in fast path
132with test_bytes as (
133  select
134    inbytes,
135    description,
136    (test_conv(inbytes || repeat('.', 3)::bytea, 'utf8', 'utf8')).error
137  from utf8_verification_inputs
138), test_padded as (
139  select
140    description,
141    (test_conv(inbytes || repeat('.', 64)::bytea, 'utf8', 'utf8')).error
142  from test_bytes
143)
144select
145  description,
146  b.error as orig_error,
147  p.error as error_after_padding
148from test_padded p
149join test_bytes b
150using (description)
151where p.error is distinct from b.error
152order by description;
153
154-- Test ASCII verification in fast path where incomplete
155-- UTF-8 sequences fall at the end of the preceding chunk.
156with test_bytes as (
157  select
158    inbytes,
159    description,
160    (test_conv(inbytes || repeat('.', 3)::bytea, 'utf8', 'utf8')).error
161  from utf8_verification_inputs
162), test_padded as (
163  select
164    description,
165    (test_conv(repeat('.', 64 - length(inbytes))::bytea || inbytes || repeat('.', 64)::bytea, 'utf8', 'utf8')).error
166  from test_bytes
167)
168select
169  description,
170  b.error as orig_error,
171  p.error as error_after_padding
172from test_padded p
173join test_bytes b
174using (description)
175where p.error is distinct from b.error
176order by description;
177
178-- Test cases where UTF-8 sequences within short text
179-- come after the fast path returns.
180with test_bytes as (
181  select
182    inbytes,
183    description,
184    (test_conv(inbytes || repeat('.', 3)::bytea, 'utf8', 'utf8')).error
185  from utf8_verification_inputs
186), test_padded as (
187  select
188    description,
189    (test_conv(repeat('.', 64)::bytea || inbytes || repeat('.', 3)::bytea, 'utf8', 'utf8')).error
190  from test_bytes
191)
192select
193  description,
194  b.error as orig_error,
195  p.error as error_after_padding
196from test_padded p
197join test_bytes b
198using (description)
199where p.error is distinct from b.error
200order by description;
201
202-- Test cases where incomplete UTF-8 sequences fall at the
203-- end of the part checked by the fast path.
204with test_bytes as (
205  select
206    inbytes,
207    description,
208    (test_conv(inbytes || repeat('.', 3)::bytea, 'utf8', 'utf8')).error
209  from utf8_verification_inputs
210), test_padded as (
211  select
212    description,
213    (test_conv(repeat('.', 64 - length(inbytes))::bytea || inbytes || repeat('.', 3)::bytea, 'utf8', 'utf8')).error
214  from test_bytes
215)
216select
217  description,
218  b.error as orig_error,
219  p.error as error_after_padding
220from test_padded p
221join test_bytes b
222using (description)
223where p.error is distinct from b.error
224order by description;
