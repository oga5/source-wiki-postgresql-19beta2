589DECLARE
590    out_degree int;
591BEGIN
592    SELECT count(*) INTO out_degree FROM GRAPH_TABLE (g1 MATCH (src WHERE src.vname = sname)->() COLUMNS (src.vname));
593    RETURN out_degree;
594END;
595$$ LANGUAGE plpgsql;
