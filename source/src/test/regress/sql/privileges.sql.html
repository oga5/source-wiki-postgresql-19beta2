331  SELECT * FROM atest12 WHERE b <<< 5;
332CREATE VIEW atest12sbv WITH (security_barrier=true) AS
333  SELECT * FROM atest12 WHERE b <<< 5;

335-- This plan should use nestloop, knowing that few rows will be selected.
336EXPLAIN (COSTS OFF) SELECT * FROM atest12v x, atest12v y WHERE x.a = y.b;

343-- scan to be materialized
344EXPLAIN (COSTS OFF) SELECT * FROM atest12sbv x, atest12sbv y WHERE x.a = y.b;

349CREATE FUNCTION leak2(integer,integer) RETURNS boolean
350  AS $$begin raise notice 'leak % %', $1, $2; return $1 > $2; end$$
351  LANGUAGE plpgsql immutable;
352CREATE OPERATOR >>> (procedure = leak2, leftarg = integer, rightarg = integer,
353                     restrict = scalargtsel);
354
355-- These should not show any "leak" notices before failing.
356EXPLAIN (COSTS OFF) SELECT * FROM atest12 WHERE a >>> 0;
357EXPLAIN (COSTS OFF) SELECT * FROM atest12v WHERE a >>> 0;
358EXPLAIN (COSTS OFF) SELECT * FROM atest12sbv WHERE a >>> 0;
359
360-- Now regress_priv_user1 grants access to regress_priv_user2 via the views.
361SET SESSION AUTHORIZATION regress_priv_user1;
362GRANT SELECT ON atest12v TO PUBLIC;
363GRANT SELECT ON atest12sbv TO PUBLIC;
364SET SESSION AUTHORIZATION regress_priv_user2;
365
366-- These plans should continue to use a nestloop, since they execute with the
367-- privileges of the view owner.
368EXPLAIN (COSTS OFF) SELECT * FROM atest12v x, atest12v y WHERE x.a = y.b;
369EXPLAIN (COSTS OFF) SELECT * FROM atest12sbv x, atest12sbv y WHERE x.a = y.b;
370
371-- A non-security barrier view does not guard against information leakage.
372EXPLAIN (COSTS OFF) SELECT * FROM atest12v x, atest12v y
373  WHERE x.a = y.b and abs(y.a) <<< 5;
374
375-- But a security barrier view isolates the leaky operator.
376EXPLAIN (COSTS OFF) SELECT * FROM atest12sbv x, atest12sbv y
377  WHERE x.a = y.b and abs(y.a) <<< 5;
378
379-- Now regress_priv_user1 grants sufficient access to regress_priv_user2.

382SET SESSION AUTHORIZATION regress_priv_user2;

388-- to make use of the expression index's statistics.
