959-- Round-trip non-ASCII data through xpath().
960DO $$
961DECLARE
962  xml_declaration text := '<?xml version="1.0" encoding="ISO-8859-1"?>';
963  degree_symbol text;
964  res xml[];
965BEGIN
966  -- Per the documentation, except when the server encoding is UTF8, xpath()
967  -- may not work on non-ASCII data.  The untranslatable_character and
968  -- undefined_function traps below, currently dead code, will become relevant
969  -- if we remove this limitation.
970  IF current_setting('server_encoding') <> 'UTF8' THEN
971    RAISE LOG 'skip: encoding % unsupported for xpath',
972      current_setting('server_encoding');
973    RETURN;
974  END IF;
975
976  degree_symbol := convert_from('\xc2b0', 'UTF8');
977  res := xpath('text()', (xml_declaration ||
978    '<x>' || degree_symbol || '</x>')::xml);
979  IF degree_symbol <> res[1]::text THEN
980    RAISE 'expected % (%), got % (%)',
981      degree_symbol, convert_to(degree_symbol, 'UTF8'),
982      res[1], convert_to(res[1]::text, 'UTF8');
983  END IF;
984EXCEPTION
985  -- character with byte sequence 0xc2 0xb0 in encoding "UTF8" has no equivalent in encoding "LATIN8"
986  WHEN undefined_function
987  -- unsupported XML feature
988  OR feature_not_supported THEN
989    RAISE LOG 'skip: %', SQLERRM;
990END
991$$;
