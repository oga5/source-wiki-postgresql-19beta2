1--
2-- Tests for testing query planner selectivity and width estimates
3--
4-- Most selectivity and width estimations rely too heavily on statistics
5-- gathered by ANALYZE, or could vary depending on hardware.  However, there
6-- are a few cases where we can have more certainty about the expected number
7-- of rows, or width of rows.  This is a good home for such tests.
8--
9-- Function to assist with verifying EXPLAIN which includes costs.  A series
10-- of bool flags allows control over which portions are masked out
11CREATE FUNCTION explain_mask_costs(query text, do_analyze bool,
12    hide_costs bool, hide_row_est bool, hide_width bool) RETURNS setof text
13LANGUAGE plpgsql AS
14$$
15DECLARE
16    ln text;
17    analyze_str text;
18BEGIN
19    IF do_analyze = true THEN
20        analyze_str := 'on';
21    ELSE
22        analyze_str := 'off';
23    END IF;
24
25    -- avoid jit related output by disabling it
26    SET LOCAL jit = 0;
27
28    FOR ln IN
29        EXECUTE format('explain (analyze %s, costs on, summary off, timing off, buffers off) %s',
30            analyze_str, query)
31    LOOP
32        IF hide_costs = true THEN
33            ln := regexp_replace(ln, 'cost=\d+\.\d\d\.\.\d+\.\d\d', 'cost=N..N');
34        END IF;
35
36        IF hide_row_est = true THEN
37            -- don't use 'g' so that we leave the actual rows intact
38            ln := regexp_replace(ln, 'rows=\d+', 'rows=N');
39        END IF;
40
41        IF hide_width = true THEN
42            ln := regexp_replace(ln, 'width=\d+', 'width=N');
43        END IF;
44
45        RETURN NEXT ln;
46    END LOOP;
47END;
48$$;
