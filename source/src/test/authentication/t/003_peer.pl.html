4# Tests for peer authentication and user name map.
5# The test is skipped if the platform does not support peer authentication,
6# and is only able to run with Unix-domain sockets.
7
8use strict;
9use warnings FATAL => 'all';
10use PostgreSQL::Test::Cluster;
11use PostgreSQL::Test::Utils;
12use Test::More;
13if (!$use_unix_sockets)
14{
15	plan skip_all =>
16	  "authentication tests cannot run without Unix-domain sockets";
17}

19# Delete pg_hba.conf from the given node, add a new entry to it
20# and then execute a reload to refresh it.
21sub reset_pg_hba
22{
23	my $node = shift;
24	my $hba_method = shift;
25
26	unlink($node->data_dir . '/pg_hba.conf');
27	$node->append_conf('pg_hba.conf', "local all all $hba_method");
28	$node->reload;
29	return;
30}
31
32# Delete pg_ident.conf from the given node, add a new entry to it
33# and then execute a reload to refresh it.
34sub reset_pg_ident
35{
36	my $node = shift;
37	my $map_name = shift;
38	my $system_user = shift;
39	my $pg_user = shift;
40
41	unlink($node->data_dir . '/pg_ident.conf');
42	$node->append_conf('pg_ident.conf', "$map_name $system_user $pg_user");
43	$node->reload;
44	return;
45}

47# Test access for a single role, useful to wrap all tests into one.
48sub test_role
49{
50	local $Test::Builder::Level = $Test::Builder::Level + 1;
51
52	my ($node, $role, $method, $expected_res, $test_details, %params) = @_;
53	my $status_string = 'failed';
54	$status_string = 'success' if ($expected_res eq 0);
55
56	my $connstr = "user=$role";
57	my $testname =
58	  "authentication $status_string for method $method, role $role "
59	  . $test_details;
60
61	if ($expected_res eq 0)
62	{
63		$node->connect_ok($connstr, $testname, %params);
64	}
65	else
66	{
67		# No checks of the error message, only the status code.
68		$node->connect_fails($connstr, $testname, %params);
69	}
70}
71
72my $node = PostgreSQL::Test::Cluster->new('node');
73$node->init;
74$node->append_conf('postgresql.conf', "log_connections = authentication\n");
75# Needed to allow connect_fails to inspect postmaster log:
76$node->append_conf('postgresql.conf', "log_min_messages = debug2");
77$node->start;
78
79# Set pg_hba.conf with the peer authentication.
80reset_pg_hba($node, 'peer');
81
82# Check if peer authentication is supported on this platform.
83my $log_offset = -s $node->logfile;
84$node->psql('postgres');
85if ($node->log_contains(
86		qr/peer authentication is not supported on this platform/,
87		$log_offset))
88{
89	plan skip_all => 'peer authentication is not supported on this platform';
90}
91
92# Add a database role and a group, to use for the user name map.
93$node->safe_psql('postgres', qq{CREATE ROLE testmapuser LOGIN});
94$node->safe_psql('postgres', "CREATE ROLE testmapgroup NOLOGIN");
95$node->safe_psql('postgres', "GRANT testmapgroup TO testmapuser");
96# Note the double quotes here.
97$node->safe_psql('postgres', 'CREATE ROLE "testmapgroupliteral\\1" LOGIN');
98$node->safe_psql('postgres', 'GRANT "testmapgroupliteral\\1" TO testmapuser');
99
100# Extract as well the system user for the user name map.
101my $system_user =
102  $node->safe_psql('postgres',
103	q(select (string_to_array(SYSTEM_USER, ':'))[2]));
104
105# While on it, check the status of huge pages, that can be either on
106# or off, but never unknown.
107my $huge_pages_status =
108  $node->safe_psql('postgres', q(SHOW huge_pages_status;));
109isnt($huge_pages_status, 'unknown', "check huge_pages_status");
110
111# Tests without the user name map.
112# Failure as connection is attempted with a database role not mapping
113# to an authorized system user.
114test_role(
115	$node, qq{testmapuser}, 'peer', 2,
116	'without user name map',
117	log_like => [qr/Peer authentication failed for user "testmapuser"/]);
118
119# Tests with a user name map.
120reset_pg_ident($node, 'mypeermap', $system_user, 'testmapuser');
121reset_pg_hba($node, 'peer map=mypeermap');
122
123# Success as the database role matches with the system user in the map.
124test_role($node, qq{testmapuser}, 'peer', 0, 'with user name map',
125	log_like =>
126	  [qr/connection authenticated: identity="$system_user" method=peer/]);
127
128# Tests with the "all" keyword.
129reset_pg_ident($node, 'mypeermap', $system_user, 'all');
130
131test_role(
132	$node,
133	qq{testmapuser},
134	'peer',
135	0,
136	'with keyword "all" as database user in user name map',
137	log_like =>
138	  [qr/connection authenticated: identity="$system_user" method=peer/]);
139
140# Tests with the "all" keyword, but quoted (no effect here).
141reset_pg_ident($node, 'mypeermap', $system_user, '"all"');
142
143test_role(
144	$node,
145	qq{testmapuser},
146	'peer',
147	2,
148	'with quoted keyword "all" as database user in user name map',
149	log_like => [qr/no match in usermap "mypeermap" for user "testmapuser"/]);
150
151# Success as the regexp of the database user matches
152reset_pg_ident($node, 'mypeermap', $system_user, qq{/^testm.*\$});
153test_role(
154	$node,
155	qq{testmapuser},
156	'peer',
157	0,
158	'with regexp of database user in user name map',
159	log_like =>
160	  [qr/connection authenticated: identity="$system_user" method=peer/]);
161
162# Failure as the regexp of the database user does not match.
163reset_pg_ident($node, 'mypeermap', $system_user, qq{/^doesnotmatch.*\$});
164test_role(
165	$node,
166	qq{testmapuser},
167	'peer',
168	2,
169	'with bad regexp of database user in user name map',
170	log_like => [qr/no match in usermap "mypeermap" for user "testmapuser"/]);
171
172# Test with regular expression in user name map.
173# Extract the last 3 characters from the system_user
174# or the entire system_user name (if its length is <= 3).
175# We trust this will not include any regex metacharacters.
176my $regex_test_string = substr($system_user, -3);
177
178# Success as the system user regular expression matches.
179reset_pg_ident($node, 'mypeermap', qq{/^.*$regex_test_string\$},
180	'testmapuser');
181test_role(
182	$node,
183	qq{testmapuser},
184	'peer',
185	0,
186	'with regexp of system user in user name map',
187	log_like =>
188	  [qr/connection authenticated: identity="$system_user" method=peer/]);
189
190# Success as both regular expressions match.
191reset_pg_ident($node, 'mypeermap', qq{/^.*$regex_test_string\$},
192	qq{/^testm.*\$});
193test_role(
194	$node,
195	qq{testmapuser},
196	'peer',
197	0,
198	'with regexps for both system and database user in user name map',
199	log_like =>
200	  [qr/connection authenticated: identity="$system_user" method=peer/]);
201
202# Success as the regular expression matches and database role is the "all"
203# keyword.
204reset_pg_ident($node, 'mypeermap', qq{/^.*$regex_test_string\$}, 'all');
205test_role(
206	$node,
207	qq{testmapuser},
208	'peer',
209	0,
210	'with regexp of system user and keyword "all" in user name map',
211	log_like =>
212	  [qr/connection authenticated: identity="$system_user" method=peer/]);
213
214# Create target role for \1 tests.
215my $mapped_name = "test${regex_test_string}map${regex_test_string}user";
216$node->safe_psql('postgres', "CREATE ROLE $mapped_name LOGIN");
217
218# Success as the regular expression matches and \1 is replaced in the given
219# subexpression.
220reset_pg_ident($node, 'mypeermap', qq{/^.*($regex_test_string)\$},
221	'test\1map\1user');
222test_role(
223	$node,
224	$mapped_name,
225	'peer',
226	0,
227	'with regular expression in user name map with \1 replaced',
228	log_like =>
229	  [qr/connection authenticated: identity="$system_user" method=peer/]);
230
231# Success as the regular expression matches and \1 is replaced in the given
232# subexpression, even if quoted.
233reset_pg_ident($node, 'mypeermap', qq{/^.*($regex_test_string)\$},
234	'"test\1map\1user"');
235test_role(
236	$node,
237	$mapped_name,
238	'peer',
239	0,
240	'with regular expression in user name map with quoted \1 replaced',
241	log_like =>
242	  [qr/connection authenticated: identity="$system_user" method=peer/]);
243
244# Failure as the regular expression does not include a subexpression, but
245# the database user contains \1, requesting a replacement.
246reset_pg_ident($node, 'mypeermap', qq{/^$system_user\$}, '\1testmapuser');
247test_role(
248	$node,
249	qq{testmapuser},
250	'peer', 2,
251	'with regular expression in user name map with \1 not replaced',
252	log_like => [
253		qr/regular expression "\^$system_user\$" has no subexpressions as requested by backreference in "\\1testmapuser"/
254	]);
255
256# Concatenate system_user to system_user.
257my $bad_regex_test_string = $system_user . $system_user;
258
259# Failure as the regexp of system user does not match.
260reset_pg_ident($node, 'mypeermap', qq{/^.*$bad_regex_test_string\$},
261	'testmapuser');
262test_role(
263	$node,
264	qq{testmapuser},
265	'peer',
266	2,
267	'with regexp of system user in user name map',
268	log_like => [qr/no match in usermap "mypeermap" for user "testmapuser"/]);
269
270# Test using a group role match for the database user.
271reset_pg_ident($node, 'mypeermap', $system_user, '+testmapgroup');
272
273test_role($node, qq{testmapuser}, 'peer', 0, 'plain user with group',
274	log_like =>
275	  [qr/connection authenticated: identity="$system_user" method=peer/]);
276
277test_role(
278	$node, qq{testmapgroup}, 'peer', 2,

282# Now apply quotes to the group match, nullifying its effect.
283reset_pg_ident($node, 'mypeermap', $system_user, '"+testmapgroup"');
284test_role(
285	$node,
286	qq{testmapuser},
287	'peer',
288	2,
289	'plain user with quoted group name',
