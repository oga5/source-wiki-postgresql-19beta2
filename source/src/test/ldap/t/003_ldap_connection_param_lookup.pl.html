4use strict;
5use warnings FATAL => 'all';
6
7use FindBin;
8use lib "$FindBin::RealBin/..";
9
10use File::Copy;
11use LdapServer;
12use PostgreSQL::Test::Utils;
13use PostgreSQL::Test::Cluster;
14use Test::More;
15
16if ($ENV{with_ldap} ne 'yes')
17{
18	plan skip_all => 'LDAP not supported by this build';
19}
20elsif (!$ENV{PG_TEST_EXTRA} || $ENV{PG_TEST_EXTRA} !~ /\bldap\b/)
21{
22	plan skip_all =>
23	  'Potentially unsafe test LDAP not enabled in PG_TEST_EXTRA';
24}
25elsif (!$LdapServer::setup)
26{
27	plan skip_all => $LdapServer::setup_error;
28}
29
30# This tests scenarios related to the service name and the service file,
31# for the connection options and their environment variables.
32my $dummy_node = PostgreSQL::Test::Cluster->new('dummy_node');
33$dummy_node->init;
34
35my $node = PostgreSQL::Test::Cluster->new('node');
36$node->init;
37$node->start;
38
39note "setting up LDAP server";
40
41my $ldap_rootpw = 'secret';
42my $ldap = LdapServer->new($ldap_rootpw, 'anonymous');    # use anonymous auth
43$ldap->ldapadd_file('authdata.ldif');
44$ldap->ldapsetpw('uid=test1,dc=example,dc=net', 'secret1');
45$ldap->ldapsetpw('uid=test2,dc=example,dc=net', 'secret2');
46
47my $td = PostgreSQL::Test::Utils::tempdir;
48
49# create ldap file based on postgres connection info
50my $ldif_valid = "$td/connection_params.ldif";
51append_to_file(
52	$ldif_valid, qq{
53version:1
54dn:cn=mydatabase,dc=example,dc=net
55changetype:add
56objectclass:top
57objectclass:device
58cn:mydatabase
59description:host=} . $node->host . qq{
60description:port=} . $node->port . qq{
61});
62
63$ldap->ldapadd_file($ldif_valid);
64
65my ($ldap_server, $ldap_port, $ldaps_port, $ldap_url,
66	$ldaps_url, $ldap_basedn, $ldap_rootdn
67) = $ldap->prop(qw(server port s_port url s_url basedn rootdn));
68
69# don't bother to check the server's cert (though perhaps we should)
70$ENV{'LDAPTLS_REQCERT'} = "never";
71
72note "setting up PostgreSQL instance";
73
74# Create the set of service files used in the tests.
75
76# File that includes a valid service name, that uses a decomposed
77# connection string for its contents, split on spaces.
78my $srvfile_valid = "$td/pg_service_valid.conf";
79append_to_file(
80	$srvfile_valid, qq{
81[my_srv]
82ldap://localhost:$ldap_port/dc=example,dc=net?description?one?(cn=mydatabase)
83});
84
85# File defined with no contents, used as default value for
86# PGSERVICEFILE, so that no lookup is attempted in the user's home
87# directory.
88my $srvfile_empty = "$td/pg_service_empty.conf";
89append_to_file($srvfile_empty, '');
90
91# Default service file in PGSYSCONFDIR.
92my $srvfile_default = "$td/pg_service.conf";
93
94# Missing service file.
95my $srvfile_missing = "$td/pg_service_missing.conf";
96
97# Set the fallback directory lookup of the service file to the
98# temporary directory of this test.  PGSYSCONFDIR is used if the
99# service file defined in PGSERVICEFILE cannot be found, or when a
100# service file is found but not the service name.
101local $ENV{PGSYSCONFDIR} = $td;
102
103# Force PGSERVICEFILE to a default location, so as this test never
104# tries to look at a home directory.  This value needs to remain at
105# the top of this script before running any tests, and should never be
106# changed.
107local $ENV{PGSERVICEFILE} = "$srvfile_empty";
108
109# Checks combinations of service name and a valid service file.
110{
111	local $ENV{PGSERVICEFILE} = $srvfile_valid;
112
113	$dummy_node->connect_ok(
114		'service=my_srv',
115		'connection with correct "service" string and PGSERVICEFILE',
116		sql => "SELECT 'connect1_1'",
117		expected_stdout => qr/connect1_1/);
118
119	$dummy_node->connect_ok(
120		'postgres://?service=my_srv',
121		'connection with correct "service" URI and PGSERVICEFILE',
122		sql => "SELECT 'connect1_2'",
123		expected_stdout => qr/connect1_2/);
124
125	$dummy_node->connect_fails(
126		'service=undefined-service',
127		'connection with incorrect "service" string and PGSERVICEFILE',
128		expected_stderr =>
129		  qr/definition of service "undefined-service" not found/);
130
131	local $ENV{PGSERVICE} = 'my_srv';
132
133	$dummy_node->connect_ok(
134		'',
135		'connection with correct PGSERVICE and PGSERVICEFILE',
136		sql => "SELECT 'connect1_3'",
137		expected_stdout => qr/connect1_3/);
138
139	local $ENV{PGSERVICE} = 'undefined-service';
140
141	$dummy_node->connect_fails(
142		'',
143		'connection with incorrect PGSERVICE and PGSERVICEFILE',
144		expected_stdout =>
145		  qr/definition of service "undefined-service" not found/);
146}
147
148# Checks case of incorrect service file.
149{
150	local $ENV{PGSERVICEFILE} = $srvfile_missing;
151
152	$dummy_node->connect_fails(
153		'service=my_srv',
154		'connection with correct "service" string and incorrect PGSERVICEFILE',
155		expected_stderr =>
156		  qr/service file ".*pg_service_missing.conf" not found/);
157}
158
159# Checks case of service file named "pg_service.conf" in PGSYSCONFDIR.
160{
161	# Create copy of valid file
162	my $srvfile_default = "$td/pg_service.conf";
163	copy($srvfile_valid, $srvfile_default);
164
165	$dummy_node->connect_ok(
166		'service=my_srv',
167		'connection with correct "service" string and pg_service.conf',
168		sql => "SELECT 'connect2_1'",
169		expected_stdout => qr/connect2_1/);
170
171	$dummy_node->connect_ok(
172		'postgres://?service=my_srv',
173		'connection with correct "service" URI and default pg_service.conf',
174		sql => "SELECT 'connect2_2'",
175		expected_stdout => qr/connect2_2/);
176
177	$dummy_node->connect_fails(
178		'service=undefined-service',
179		'connection with incorrect "service" string and default pg_service.conf',
180		expected_stderr =>
181		  qr/definition of service "undefined-service" not found/);
182
183	local $ENV{PGSERVICE} = 'my_srv';
184
185	$dummy_node->connect_ok(
186		'',
187		'connection with correct PGSERVICE and default pg_service.conf',
188		sql => "SELECT 'connect2_3'",
189		expected_stdout => qr/connect2_3/);
190
191	local $ENV{PGSERVICE} = 'undefined-service';
192
193	$dummy_node->connect_fails(
194		'',
195		'connection with incorrect PGSERVICE and default pg_service.conf',
196		expected_stdout =>
197		  qr/definition of service "undefined-service" not found/);
198
199	# Remove default pg_service.conf.
200	unlink($srvfile_default);
201}
202
203$node->teardown_node;
204
205done_testing();
