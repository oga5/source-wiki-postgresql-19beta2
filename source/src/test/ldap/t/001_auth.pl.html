61sub test_access
62{
63	local $Test::Builder::Level = $Test::Builder::Level + 1;
64
65	my ($node, $role, $expected_res, $test_name, %params) = @_;
66	my $connstr = "user=$role";
67
68	if ($expected_res eq 0)
69	{
70		$node->connect_ok($connstr, $test_name, %params);
71	}
72	else
73	{
74		# No checks of the error message, only the status code.
75		$node->connect_fails($connstr, $test_name, %params);
76	}
77}

79note "simple bind";
80
81unlink($node->data_dir . '/pg_hba.conf');
82$node->append_conf('pg_hba.conf',
83	qq{local all all ldap ldapserver=$ldap_server ldapport=$ldap_port ldapprefix="uid=" ldapsuffix=",dc=example,dc=net"}
84);
85$node->restart;
86
87$ENV{"PGPASSWORD"} = 'wrong';
88test_access(
89	$node, 'test0', 2,
90	'simple bind authentication fails if user not found in LDAP',
91	log_unlike => [qr/connection authenticated:/]);
92test_access(
93	$node, 'test1', 2,
94	'simple bind authentication fails with wrong password',
95	log_unlike => [qr/connection authenticated:/]);
96
97$ENV{"PGPASSWORD"} = 'secret1';
98test_access(
99	$node, 'test1', 0,
100	'simple bind authentication succeeds',
101	log_like => [
102		qr/connection authenticated: identity="uid=test1,dc=example,dc=net" method=ldap/
103	],);

107	"password authentication required, works with ldap auth");

109	"SCRAM authentication required, fails with ldap auth");
