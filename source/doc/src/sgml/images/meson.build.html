1# doc/src/sgml/images/meson.build
2#
3# see README in this directory about image handling
4
5if not xsltproc_bin.found() or not dot.found() or not ditaa.found()
6  subdir_done()
7endif
8
9image_targets = []
10
11fixup_svg_xsl = files('fixup-svg.xsl')
12
13all_files = [
14  'genetic-algorithm.gv',
15  'gin.gv',
16  'pagelayout.txt',
17  'temporal-entities.txt',
18  'temporal-references.txt',
19  'temporal-update.txt',
20  'temporal-delete.txt',
21  'temporal-isolation.txt',
22]
23
24foreach file : all_files
25
26  str_split = file.split('.')
27  actual_file_name = str_split[0]
28  extension = str_split[1]
29  cur_file = files(file)
30  tmp_name = '@0@.svg.tmp'.format(file)
31  output_name = '@0@.svg'.format(actual_file_name)
32
33  command = []
34  if extension == 'gv'
35    command = [dot, '-T', 'svg', '-o', '@OUTPUT@', '@INPUT@']
36  elif extension == 'txt'
37    command = [ditaa, '-E', '-S', '--svg', '@INPUT@', '@OUTPUT@']
38  else
39    error('Unknown extension: ".@0@" while generating images'.format(extension))
40  endif
41
42  svg_tmp = custom_target(tmp_name,
43    input: cur_file,
44    output: tmp_name,
45    command: command,
46  )
47
48  current_svg = custom_target(output_name,
49    input: svg_tmp,
50    output: output_name,
51    command: [xsltproc_bin,
52      '--nonet',
53      # Use --novalid to avoid loading SVG DTD if a file specifies it, since
54      # it might not be available locally, and we don't need it.
55      '--novalid',
56      '-o', '@OUTPUT@',
57      fixup_svg_xsl,
58      '@INPUT@']
59  )
60
61  image_targets += current_svg
62endforeach
63
64alias_target('images', image_targets)
